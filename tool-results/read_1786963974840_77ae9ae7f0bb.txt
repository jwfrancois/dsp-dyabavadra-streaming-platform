   163→/** Convert Jellyfin ticks (100ns units) to seconds */
   164→export function ticksToSeconds(ticks: number | undefined): number {
   165→  if (ticks === undefined || ticks === null) return 0;
   166→  return Math.floor(ticks / TICKS_PER_SECOND);
   167→}
   168→
   169→/** Convert seconds to Jellyfin ticks */
   170→export function secondsToTicks(seconds: number): number {
   171→  return Math.floor(seconds * TICKS_PER_SECOND);
   172→}

   174→/** Strip trailing slash from a URL */
   175→function normalizeServerUrl(url: string): string {
   176→  return url.replace(/\/+$/, '');
   177→}
   178→
   179→/** Build a query string from a record, filtering out undefined/null/empty values */
   180→function buildQueryString(params: Record<string, string | number | boolean | undefined | null | string[] | number[]>): string {
   181→  const parts: string[] = [];
   182→  for (const [key, value] of Object.entries(params)) {
   183→    if (value === undefined || value === null || value === '') continue;
   184→    if (Array.isArray(value)) {
   185→      if (value.length > 0) {
   186→        parts.push(`${encodeURIComponent(key)}=${value.map(v => encodeURIComponent(String(v))).join(',')}`);
   187→      }
   188→    } else {
   189→      parts.push(`${encodeURIComponent(key)}=${encodeURIComponent(String(value))}`);
   190→    }
   191→  }
   192→  return parts.length > 0 ? `?${parts.join('&')}` : '';
   193→}

   201→  constructor() {

   206→        window.localStorage.setItem(testKey, '1');
   207→        window.localStorage.removeItem(testKey);
   208→      } else {
   209→        this.storageAvailable = false;
   210→      }
   211→    } catch {
   212→      this.storageAvailable = false;
   213→    }
   214→
   215→    // Auto-load persisted config
   216→    this.config = this.loadConfig();
   217→  }

   223→  private saveConfig(): void {
   224→    if (!this.config) return;
   225→    if (this.storageAvailable && typeof window !== 'undefined') {
   226→      try {
   227→        window.localStorage.setItem(STORAGE_KEY, JSON.stringify(this.config));
   228→      } catch {
   229→        // Storage full or unavailable — silently fail
   230→      }
   231→    }
   232→  }
   233→
   234→  private loadConfig(): JellyfinConfig | null {
   235→    if (!this.storageAvailable || typeof window === 'undefined') return null;
   236→    try {
   237→      const raw = window.localStorage.getItem(STORAGE_KEY);
   238→      if (!raw) return null;
   239→      const parsed = JSON.parse(raw) as JellyfinConfig;
   240→      // Basic validation
   241→      if (!parsed.serverUrl || !parsed.accessToken || !parsed.userId) return null;
   242→      return parsed;
   243→    } catch {
   244→      return null;
   245→    }
   246→  }

   257→  /**
   258→   * Make an API request through the Next.js server-side proxy.
   259→   * This avoids browser CORS issues when connecting to self-hosted Jellyfin servers.
   260→   * The proxy at /api/jellyfin/[...path] forwards requests server-to-server.
   261→   */
   262→  private async request<T>(path: string, options: RequestInit = {}, skipAuth = false): Promise<T> {
   263→    // Extract injected metadata from headers (used during connect flow)
   264→    const injectedHeaders = (options.headers as Record<string, string> | undefined);
   265→    const injectedBaseUrl = injectedHeaders?.__baseUrl;
   266→    const injectedAuth = injectedHeaders?.__authorization;
   267→
   268→    // Determine the target Jellyfin server URL
   269→    const serverUrl = skipAuth
   270→      ? normalizeServerUrl(injectedBaseUrl || '')
   271→      : this.getBaseUrl();
   272→
   273→    if (!serverUrl) {
   274→      throw new JellyfinApiError(
   275→        'No Jellyfin server URL configured. Please connect first.',
   276→        0,
   277→        path
   278→      );
   279→    }
   280→
   281→    // Build the proxy URL: /api/jellyfin/Users/AuthenticateByName
   282→    const proxyUrl = `/api/jellyfin${path}`;
   283→
   284→    // Build headers for the proxy — the proxy reads these to forward correctly
   285→    const proxyHeaders: Record<string, string> = {
   286→      'Content-Type': 'application/json',
   287→      'Accept': 'application/json',
   288→      'X-Jellyfin-Server-Url': serverUrl,
   289→    };
   290→
   291→    // Tell the proxy to skip forwarding the auth token for login endpoints
   292→    if (skipAuth) {
   293→      proxyHeaders['X-Jellyfin-Skip-Auth'] = 'true';
   294→    } else if (this.config) {
   295→      // Forward the auth token for authenticated requests
   296→      proxyHeaders['X-Jellyfin-Token'] = this.config.accessToken;
   297→    }
   298→
   299→    // Forward the MediaBrowser authorization header if present (for login)
   300→    if (injectedAuth) {
   301→      proxyHeaders['X-Jellyfin-Authorization'] = injectedAuth;
   302→    }
   303→
   304→    // Remove Content-Type for GET/DELETE if no body
   305→    if (!options.body && (options.method === 'GET' || options.method === 'DELETE')) {
   306→      delete proxyHeaders['Content-Type'];
   307→    }
   308→
   309→    let response: Response;
   310→    try {
   311→      response = await fetch(proxyUrl, {
   312→        method: options.method || 'GET',
   313→        headers: proxyHeaders,
   314→        body: options.body || undefined,
   315→      });
   316→    } catch (err) {
   317→      throw new JellyfinApiError(
   318→        `Network error: ${err instanceof Error ? err.message : 'Unknown error'}`,
   319→        0,
   320→        path,
   321→        err
   322→      );
   323→    }
   324→
   325→    if (!response.ok) {
   326→      let errorBody = '';
   327→      try {
   328→        errorBody = await response.text();
   329→      } catch {
   330→        // ignore
   331→      }
   332→      throw new JellyfinApiError(
   333→        `Jellyfin API error ${response.status}: ${response.statusText} — ${errorBody}`,
   334→        response.status,
   335→        path
   336→      );
   337→    }
   338→
   339→    // Some endpoints return empty responses (e.g., DELETE, favorite toggles)
   340→    const contentType = response.headers.get('content-type');
   341→    if (contentType && contentType.includes('application/json')) {
   342→      return response.json() as Promise<T>;
   343→    }
   344→
   345→    return undefined as unknown as T;
   346→  }

   352→  /**
   353→   * Connect to a Jellyfin server. Authenticates and fetches server info.
   354→   * Persists the configuration to localStorage.
   355→   */
   356→  async connect(serverUrl: string, username: string, password: string): Promise<JellyfinConfig> {
   357→    const baseUrl = normalizeServerUrl(serverUrl);
   358→
   359→    // Step 1: Authenticate (skip auth headers — we don't have a token yet)
   360→    const deviceId = typeof window !== 'undefined'
   361→      ? window.navigator.userAgent.slice(0, 32).replace(/[^a-zA-Z0-9]/g, '')
   362→      : 'ssr';
   363→    const mediaBrowserAuth = `MediaBrowser Client="DSP", Device="Browser", DeviceId="dsp-jellyfin-${deviceId}", Version="1.0.0"`;
   364→
   365→    const authResponse = await this.request<JellyfinAuthResponse>(
   366→      '/Users/AuthenticateByName',
   367→      {
   368→        method: 'POST',
   369→        body: JSON.stringify({ Username: username, Pw: password }),
   370→        headers: {
   371→          '__baseUrl': baseUrl,
   372→          '__authorization': mediaBrowserAuth,
   373→        } as Record<string, string>,
   374→      },
   375→      true // skipAuth
   376→    );
   377→
   378→    const token = authResponse.AccessToken;
   379→    const userId = authResponse.User.Id;
   380→    const serverId = authResponse.ServerId;
   381→
   382→    // Step 2: Fetch server info
   383→    let serverName = 'Jellyfin';
   384→    let serverVersion = 'unknown';
   385→    try {
   386→      // Temporarily set config for the getSystemInfo call
   387→      this.config = {
   388→        serverUrl: baseUrl,
   389→        username,
   390→        accessToken: token,
   391→        userId,
   392→        serverId,
   393→        serverName,
   394→        serverVersion,
   395→        connectedAt: new Date().toISOString(),
   396→        lastHeartbeat: new Date().toISOString(),
   397→        musicLibraryId: '',
   398→        podcastLibraryId: '',
   399→      };
   400→      const sysInfo = await this.getSystemInfo();
   401→      serverName = sysInfo.ServerName;
   402→      serverVersion = sysInfo.Version;
   403→    } catch {
   404→      // If system info fails, continue with defaults
   405→    }
   406→
   407→    // Step 3: Auto-detect music and podcast libraries
   408→    let musicLibraryId = '';
   409→    let podcastLibraryId = '';
   410→    try {
   411→      const views = await this.getViews();
   412→      for (const item of views.Items) {
   413→        if (item.Type !== 'CollectionFolder') continue;
   414→        const name = item.Name.toLowerCase();
   415→        if (name.includes('music') && !name.includes('podcast') && !musicLibraryId) {
   416→          musicLibraryId = item.Id;
   417→        }
   418→        if (name.includes('podcast') && !podcastLibraryId) {
   419→          podcastLibraryId = item.Id;
   420→        }
   421→      }
   422→    } catch {
   423→      // Continue without auto-detected library
   424→    }
   425→
   426→    // Step 4: Build and save config
   427→    const config: JellyfinConfig = {
   428→      serverUrl: baseUrl,
   429→      username,
   430→      accessToken: token,
   431→      userId,
   432→      serverId,
   433→      serverName,
   434→      serverVersion,
   435→      connectedAt: new Date().toISOString(),
   436→      lastHeartbeat: new Date().toISOString(),
   437→      musicLibraryId,
   438→      podcastLibraryId,
   439→    };
   440→
   441→    this.config = config;
   442→    this.saveConfig();
   443→
   444→    return config;
   445→  }

   447→  /**
   448→   * Disconnect from the Jellyfin server. Revokes the token and clears local state.
   449→   */
   450→  async disconnect(): Promise<void> {
   451→    if (this.config) {
   452→      try {
   453→        await this.request<void>(`/Users/${this.config.userId}/Tokens`, {
   454→          method: 'DELETE',
   455→        });
   456→      } catch {
   457→        // Best-effort — continue clearing local state
   458→      }
   459→    }
   460→
   461→    this.config = null;
   462→    if (this.storageAvailable && typeof window !== 'undefined') {
   463→      try {
   464→        window.localStorage.removeItem(STORAGE_KEY);
   465→      } catch {
   466→        // ignore
   467→      }
   468→    }
   469→  }

   888→  /**
   889→   * Get a playable audio stream URL for an item.
   890→   * Prefers direct play for best quality; falls back to transcoded stream.
   891→   */
   892→  async getPlaybackStreamUrl(
   893→    itemId: string,
   894→    maxBitrate: number = DEFAULT_MAX_BITRATE
   895→  ): Promise<string> {
   896→    const baseUrl = this.getBaseUrl();
   897→    const token = this.config!.accessToken;
   898→    const userId = this.config!.userId;
   899→
   900→    try {
   901→      // Request playback info to determine best playback method
   902→      const playbackInfo = await this.request<JellyfinPlaybackInfo>(
   903→        `/Items/${itemId}/PlaybackInfo`,
   904→        {
   905→          method: 'POST',
   906→          body: JSON.stringify({
   907→            UserId: userId,
   908→            MaxStreamingBitrate: maxBitrate,
   909→            AutoOpenLiveStreams: true,
   910→          }),
   911→        }
   912→      );
   913→
   914→      if (playbackInfo.MediaSources && playbackInfo.MediaSources.length > 0) {
   915→        const source = playbackInfo.MediaSources[0];
   916→
   917→        // Prefer direct play — best quality, no server transcoding
   918→        if (source.SupportsDirectPlay) {
   919→          return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${token}&UserId=${userId}`;
   920→        }
   921→
   922→        // Fallback to direct stream
   923→        if (source.SupportsDirectStream && source.DirectStreamUrl) {
   924→          return `${baseUrl}${source.DirectStreamUrl}`;
   925→        }
   926→
   927→        // Fallback to transcoded stream (HLS)
   928→        if (source.SupportsTranscoding && source.TranscodingUrl) {
   929→          return `${baseUrl}${source.TranscodingUrl}`;
   930→        }
   931→
   932→        // Last resort: construct a universal audio transcoding URL
   933→        const container = source.Container || 'mp3';
   934→        return `${baseUrl}/Audio/${itemId}/universal?api_key=${token}&UserId=${userId}&MaxStreamingBitrate=${maxBitrate}&Container=${container}&TranscodingMaxAudioChannels=2&SegmentContainer=ts&AudioCodec=aac&Protocol=hls`;
   935→      }
   936→    } catch {
   937→      // If playback info fails, fall back to direct stream URL
   938→    }
   939→
   940→    // Fallback: try static stream directly
   941→    return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${token}&UserId=${userId}`;
   942→  }

   952→  getImageUrl(
   953→    itemId: string,

   961→    const token = this.config.accessToken;
   962→

   973→
   974→    // Route through the existing audio proxy (it handles any content type)
