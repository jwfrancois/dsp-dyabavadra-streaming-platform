   267→   306→
   268→   307→  // Filter only browser-imported tracks (isLocal === true)
   269→   308→  const allLocalTracks = localStore.tracks.filter((t: any) => t.isLocal);
   270→   309→
   271→   310→  // Derive artists / albums only from local tracks
   272→   311→  const localArtists = React.useMemo(() => {
   273→   312→    const set = new Set<string>();
   274→   313→    for (const t of allLocalTracks) {
   275→   314→      if (t.artist) set.add(t.artist);
   276→   315→      if (t.albumArtist && t.albumArtist !== t.artist) set.add(t.albumArtist);
   277→   316→    }
   278→   317→    return [...set].sort();
   279→   318→  }, [allLocalTracks]);
   280→   319→
   281→   320→  const localAlbums = React.useMemo(() => {
   282→   321→    const map = new Map<string, { name: string; artist: string; albumArtist: string; trackCount: number; year: number; coverArt: string | null }>();
   283→   322→    for (const t of allLocalTracks) {
   284→   323→      const key = `${t.albumArtist || t.artist}|||${t.album}`;
   285→   324→      const existing = map.get(key);
   286→   325→      if (existing) {
   287→   326→        existing.trackCount++;
   288→   327→        if (t.year && (!existing.year || t.year > existing.year)) existing.year = t.year;
   289→   328→        if (!existing.coverArt && t.coverArt) existing.coverArt = t.coverArt;
   290→   329→      } else {
   291→   330→        map.set(key, {
   292→   331→          name: t.album,
   293→   332→          artist: t.artist,
   294→   333→          albumArtist: t.albumArtist || t.artist,
   295→   334→          trackCount: 1,
   296→   335→          year: t.year,
   297→   336→          coverArt: t.coverArt,
   298→   337→        });
   299→   338→      }
   300→   339→    }
   301→   340→    return Array.from(map.values());

   305→   344→  const allAlbums = localAlbums;

   321→   360→        t.artist.toLowerCase().includes(q) ||
   322→   361→        t.album.toLowerCase().includes(q) ||
   323→   362→        t.genre.toLowerCase().includes(q) ||
   324→   363→        (t.composer && t.composer.toLowerCase().includes(q)) ||
   325→   364→        (t.albumArtist && t.albumArtist.toLowerCase().includes(q))

   327→   366→  }, [allLocalTracks]);

   333→   372→    );
