   165→/** Convert Jellyfin ticks (100ns units) to seconds */
   166→export function ticksToSeconds(ticks: number | undefined): number {
   167→  if (ticks === undefined || ticks === null) return 0;
   168→  return Math.floor(ticks / TICKS_PER_SECOND);
   169→}
   170→
   171→/** Convert seconds to Jellyfin ticks */
   172→export function secondsToTicks(seconds: number): number {
   173→  return Math.floor(seconds * TICKS_PER_SECOND);
   174→}

   176→/** Strip trailing slash from a URL */
   177→function normalizeServerUrl(url: string): string {
   178→  return url.replace(/\/+$/, '');

   181→/** Build a query string from a record, filtering out undefined/null/empty values */
   182→function buildQueryString(params: Record<string, string | number | boolean | undefined | null | string[] | number[]>): string {
   183→  const parts: string[] = [];
   184→  for (const [key, value] of Object.entries(params)) {
   185→    if (value === undefined || value === null || value === '') continue;
   186→    if (Array.isArray(value)) {
   187→      if (value.length > 0) {
   188→        parts.push(`${encodeURIComponent(key)}=${value.map(v => encodeURIComponent(String(v))).join(',')}`);
   189→      }
   190→    } else {
   191→      parts.push(`${encodeURIComponent(key)}=${encodeURIComponent(String(value))}`);
   192→    }
   193→  }
   194→  return parts.length > 0 ? `?${parts.join('&')}` : '';
   195→}

   225→  private saveConfig(): void {
   226→    if (!this.config) return;
   227→    if (this.storageAvailable && typeof window !== 'undefined') {
   228→      try {
   229→        window.localStorage.setItem(STORAGE_KEY, JSON.stringify(this.config));

   231→        // Storage full or unavailable — silently fail
   232→      }
   233→    }

   236→  private loadConfig(): JellyfinConfig | null {
   237→    if (!this.storageAvailable || typeof window === 'undefined') return null;
   238→    try {
   239→      const raw = window.localStorage.getItem(STORAGE_KEY);
   240→      if (!raw) return null;
   241→      const parsed = JSON.parse(raw) as JellyfinConfig;
   242→      // Basic validation
   243→      if (!parsed.serverUrl || !parsed.accessToken || !parsed.userId) return null;
   244→      return parsed;
   245→    } catch {
   246→      return null;
   247→    }
   248→  }

   259→  /**
   260→   * Make an API request through the Next.js server-side proxy.
   261→   * This avoids browser CORS issues when connecting to self-hosted Jellyfin servers.
   262→   * The proxy at /api/jellyfin/[...path] forwards requests server-to-server.
   263→   */
   264→  private async request<T>(path: string, options: RequestInit = {}, skipAuth = false): Promise<T> {
   265→    // Extract injected metadata from headers (used during connect flow)
   266→    const injectedHeaders = (options.headers as Record<string, string> | undefined);
   267→    const injectedBaseUrl = injectedHeaders?.__baseUrl;
   268→    const injectedAuth = injectedHeaders?.__authorization;
   269→
   270→    // Determine the target Jellyfin server URL
   271→    const serverUrl = skipAuth
   272→      ? normalizeServerUrl(injectedBaseUrl || '')
   273→      : this.getBaseUrl();
   274→
   275→    if (!serverUrl) {
   276→      throw new JellyfinApiError(
   277→        'No Jellyfin server URL configured. Please connect first.',
   278→        0,
   279→        path
   280→      );
   281→    }
   282→
   283→    // Build the proxy URL: /api/jellyfin/Users/AuthenticateByName
   284→    const proxyUrl = `/api/jellyfin${path}`;
   285→
   286→    // Build headers for the proxy — the proxy reads these to forward correctly
   287→    const proxyHeaders: Record<string, string> = {
   288→      'Content-Type': 'application/json',
   289→      'Accept': 'application/json',
   290→      'X-Jellyfin-Server-Url': serverUrl,
   291→    };
   292→
   293→    // Tell the proxy to skip forwarding the auth token for login endpoints
   294→    if (skipAuth) {
   295→      proxyHeaders['X-Jellyfin-Skip-Auth'] = 'true';
   296→    } else if (this.config) {
   297→      // Forward the auth token for authenticated requests
   298→      proxyHeaders['X-Jellyfin-Token'] = this.config.accessToken;
   299→    }
   300→
   301→    // Forward the MediaBrowser authorization header if present (for login)
   302→    if (injectedAuth) {
   303→      proxyHeaders['X-Jellyfin-Authorization'] = injectedAuth;
   304→    }
   305→
   306→    // Remove Content-Type for GET/DELETE if no body
   307→    if (!options.body && (options.method === 'GET' || options.method === 'DELETE')) {
   308→      delete proxyHeaders['Content-Type'];
   309→    }
   310→
   311→    let response: Response;
   312→    try {
   313→      response = await fetch(proxyUrl, {
   314→        method: options.method || 'GET',
   315→        headers: proxyHeaders,
   316→        body: options.body || undefined,
   317→      });
   318→    } catch (err) {
   319→      throw new JellyfinApiError(
   320→        `Network error: ${err instanceof Error ? err.message : 'Unknown error'}`,
   321→        0,
   322→        path,
   323→        err
   324→      );
   325→    }
   326→
   327→    if (!response.ok) {
   328→      let errorBody = '';
   329→      try {
   330→        errorBody = await response.text();
   331→      } catch {
   332→        // ignore
   333→      }
   334→      throw new JellyfinApiError(
   335→        `Jellyfin API error ${response.status}: ${response.statusText} — ${errorBody}`,
   336→        response.status,
   337→        path
   338→      );
   339→    }
   340→
   341→    // Some endpoints return empty responses (e.g., DELETE, favorite toggles)
   342→    const contentType = response.headers.get('content-type');
   343→    if (contentType && contentType.includes('application/json')) {
   344→      return response.json() as Promise<T>;
   345→    }
   346→
   347→    return undefined as unknown as T;
   348→  }

   354→  /**
   355→   * Connect to a Jellyfin server. Authenticates and fetches server info.
   356→   * Persists the configuration to localStorage.
   357→   */
   358→  async connect(serverUrl: string, username: string, password: string): Promise<JellyfinConfig> {
   359→    const baseUrl = normalizeServerUrl(serverUrl);
   360→
   361→    // Step 1: Authenticate (skip auth headers — we don't have a token yet)
   362→    const deviceId = typeof window !== 'undefined'
   363→      ? window.navigator.userAgent.slice(0, 32).replace(/[^a-zA-Z0-9]/g, '')
   364→      : 'ssr';
   365→    const mediaBrowserAuth = `MediaBrowser Client="DSP", Device="Browser", DeviceId="dsp-jellyfin-${deviceId}", Version="1.0.0"`;
   366→
   367→    const authResponse = await this.request<JellyfinAuthResponse>(
   368→      '/Users/AuthenticateByName',
   369→      {
   370→        method: 'POST',
   371→        body: JSON.stringify({ Username: username, Pw: password }),
   372→        headers: {
   373→          '__baseUrl': baseUrl,
   374→          '__authorization': mediaBrowserAuth,
   375→        } as Record<string, string>,
   376→      },
   377→      true // skipAuth
   378→    );
   379→
   380→    const token = authResponse.AccessToken;
   381→    const userId = authResponse.User.Id;
   382→    const serverId = authResponse.ServerId;
   383→
   384→    // Step 2: Fetch server info
   385→    let serverName = 'Jellyfin';
   386→    let serverVersion = 'unknown';
   387→    try {
   388→      // Temporarily set config for the getSystemInfo call
   389→      this.config = {
   390→        serverUrl: baseUrl,
   391→        username,
   392→        accessToken: token,
   393→        userId,
   394→        serverId,
   395→        serverName,
   396→        serverVersion,
   397→        connectedAt: new Date().toISOString(),
   398→        lastHeartbeat: new Date().toISOString(),
   399→        musicLibraryId: '',
   400→        podcastLibraryId: '',
   401→      };
   402→      const sysInfo = await this.getSystemInfo();
   403→      serverName = sysInfo.ServerName;
   404→      serverVersion = sysInfo.Version;
   405→    } catch {
   406→      // If system info fails, continue with defaults
   407→    }
   408→
   409→    // Step 3: Auto-detect music and podcast libraries
   410→    // Jellyfin's /Library/VirtualFolders returns an ARRAY of folder objects.
   411→    // Each has CollectionType ("music", "podcasts", etc.) and ItemId (the view ID for ParentId queries).
   412→    let musicLibraryId = '';
   413→    let podcastLibraryId = '';
   414→    try {
   415→      const vfData = await this.request<Array<{ Name: string; CollectionType?: string; ItemId?: string }> | { Items?: Array<{ Name: string; CollectionType?: string; ItemId?: string }> }>('/Library/VirtualFolders');
   416→      // The API returns either a plain array or { Items: [...] } depending on Jellyfin version
   417→      const folders = Array.isArray(vfData) ? vfData : (vfData.Items ?? []);
   418→      for (const folder of folders) {
   419→        const ct = (folder.CollectionType || '').toLowerCase();
   420→        const folderId = folder.ItemId || '';
   421→        if (ct === 'music' && !musicLibraryId) {
   422→          musicLibraryId = folderId;
   423→        }
   424→        if ((ct === 'podcasts' || ct === 'podcast') && !podcastLibraryId) {
   425→          podcastLibraryId = folderId;
   426→        }
   427→      }
   428→    } catch {
   429→ // VirtualFolders may fail; fall through to Views-based detection
   430→    }
   431→    // Fallback: use /Users/{userId}/Views to find libraries by name
   432→    if (!musicLibraryId || !podcastLibraryId) {
   433→      try {
   434→        const views = await this.getViews();
   435→        for (const item of views.Items) {
   436→          if (item.Type !== 'CollectionFolder') continue;
   437→          const name = item.Name.toLowerCase();
   438→          if (!musicLibraryId && name.includes('music') && !name.includes('podcast')) {
   439→            musicLibraryId = item.Id;
   440→          }
   441→          if (!podcastLibraryId && (name.includes('podcast'))) {
   442→            podcastLibraryId = item.Id;
   443→          }
   444→        }
   445→      } catch {
   446→        // Continue without auto-detected library
   447→      }
   448→    }
   449→
   450→    console.log(`[Jellyfin] Library detection: musicLibraryId=${musicLibraryId || '(none)'}, podcastLibraryId=${podcastLibraryId || '(none)'}`);
   451→
   452→    // Step 4: Build and save config
   453→    const config: JellyfinConfig = {
   454→      serverUrl: baseUrl,
   455→      username,
   456→      accessToken: token,
   457→      userId,
   458→      serverId,
   459→      serverName,
   460→      serverVersion,
   461→      connectedAt: new Date().toISOString(),
   462→      lastHeartbeat: new Date().toISOString(),
   463→      musicLibraryId,
   464→      podcastLibraryId,
   465→    };
   466→
   467→    this.config = config;
   468→    this.saveConfig();
   469→
   470→    return config;
   471→  }

   479→        await this.request<void>(`/Users/${this.config.userId}/Tokens`, {

   482→      } catch {
   483→        // Best-effort — continue clearing local state
   484→      }
   485→    }
   486→
   487→    this.config = null;
   488→    if (this.storageAvailable && typeof window !== 'undefined') {
   489→      try {
   490→        window.localStorage.removeItem(STORAGE_KEY);
   491→      } catch {
   492→        // ignore
   493→      }
   494→    }

   528→        }
   529→        if ((ct === 'podcasts' || ct === 'podcast') && !this.config.podcastLibraryId) {
   530→          this.config.podcastLibraryId = folderId;
   531→        }
   532→      }
   533→      // Fallback to Views-based matching
   534→      if (!this.config.musicLibraryId || !this.config.podcastLibraryId) {
   535→        const views = await this.getViews();
   536→        for (const item of views.Items) {

   540→            this.config.musicLibraryId = item.Id;
   541→          }
   542→          if (!this.config.podcastLibraryId && name.includes('podcast')) {
   543→            this.config.podcastLibraryId = item.Id;
   544→          }
   545→        }
   546→      }
   547→      this.saveConfig();
   548→    } catch {
   549→      // Non-critical — queries will still work globally
   550→    }
   551→  }

   613→      const folders = Array.isArray(vfData) ? vfData : (vfData.Items ?? []);
   614→      for (const folder of folders) {
   615→        const ct = (folder.CollectionType || '').toLowerCase();
   616→        const folderId = folder.ItemId || '';
   617→        if (ct === 'music' && !musicLibraryId) {
   618→          musicLibraryId = folderId;
   619→        }
   620→        if ((ct === 'podcasts' || ct === 'podcast') && !podcastLibraryId) {
   621→          podcastLibraryId = folderId;
   622→        }
   623→      }
   624→      if (!musicLibraryId || !podcastLibraryId) {
   625→        const views = await this.getViews();
   626→        for (const item of views.Items) {
   627→          if (item.Type !== 'CollectionFolder') continue;
   628→          const name = item.Name.toLowerCase();
   629→          if (!musicLibraryId && name.includes('music') && !name.includes('podcast')) {
   630→            musicLibraryId = item.Id;
   631→          }
   632→          if (!podcastLibraryId && name.includes('podcast')) {
   633→            podcastLibraryId = item.Id;
   634→          }
   635→        }
   636→      }
   637→    } catch {
   638→      // Continue without auto-detected library
   639→    }
   640→
   641→    const config: JellyfinConfig = {
   642→      serverUrl: baseUrl,
   643→      username: userInfo.Name || 'API Key User',
   644→      accessToken: token,
   645→      userId,
   646→      serverId,
   647→      serverName,
   648→      serverVersion,
   649→      connectedAt: new Date().toISOString(),
   650→      lastHeartbeat: new Date().toISOString(),
   651→      musicLibraryId,
   652→      podcastLibraryId,
   653→    };
   654→
   655→    this.config = config;
   656→    this.saveConfig();
   657→    return config;
   658→  }

  1067→  /**
  1068→   * Get a playable audio stream URL for an item.
  1069→   * Prefers direct play for best quality; falls back to transcoded stream.
  1070→   */
  1071→  async getPlaybackStreamUrl(
  1072→    itemId: string,
  1073→    maxBitrate: number = DEFAULT_MAX_BITRATE
  1074→  ): Promise<string> {
  1075→    const baseUrl = this.getBaseUrl();
  1076→    const token = this.config!.accessToken;
  1077→    const userId = this.config!.userId;
  1078→
  1079→    try {
  1080→      // Request playback info to determine best playback method
  1081→      const playbackInfo = await this.request<JellyfinPlaybackInfo>(
  1082→        `/Items/${itemId}/PlaybackInfo`,
  1083→        {
  1084→          method: 'POST',
  1085→          body: JSON.stringify({
  1086→            UserId: userId,
  1087→            MaxStreamingBitrate: maxBitrate,
  1088→            AutoOpenLiveStreams: true,
  1089→          }),
  1090→        }
  1091→      );
  1092→
  1093→      if (playbackInfo.MediaSources && playbackInfo.MediaSources.length > 0) {
  1094→        const source = playbackInfo.MediaSources[0];
  1095→
  1096→        // Prefer direct play — best quality, no server transcoding
  1097→        if (source.SupportsDirectPlay) {
  1098→          return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${token}&UserId=${userId}`;
  1099→        }
  1100→
  1101→        // Fallback to direct stream
  1102→        if (source.SupportsDirectStream && source.DirectStreamUrl) {
  1103→          return `${baseUrl}${source.DirectStreamUrl}`;
  1104→        }
  1105→
  1106→        // Fallback to transcoded stream (HLS)
  1107→        if (source.SupportsTranscoding && source.TranscodingUrl) {
  1108→          return `${baseUrl}${source.TranscodingUrl}`;
  1109→        }
  1110→
  1111→        // Last resort: construct a universal audio transcoding URL
  1112→        const container = source.Container || 'mp3';
  1113→        return `${baseUrl}/Audio/${itemId}/universal?api_key=${token}&UserId=${userId}&MaxStreamingBitrate=${maxBitrate}&Container=${container}&TranscodingMaxAudioChannels=2&SegmentContainer=ts&AudioCodec=aac&Protocol=hls`;
  1114→      }
  1115→    } catch {
  1116→      // If playback info fails, fall back to direct stream URL
  1117→    }
  1118→
  1119→    // Fallback: try static stream directly
  1120→    return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${token}&UserId=${userId}`;
  1121→  }
