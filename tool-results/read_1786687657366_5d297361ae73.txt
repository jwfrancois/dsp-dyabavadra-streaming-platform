    97→    97→
    98→    98→function countryCodeToFlag(code: string): string {
    99→    99→  if (!code || code.length !== 2) return '🌐';
   100→   100→  const c = code.toUpperCase();
   101→   101→  // Convert country code to flag emoji using regional indicator symbols
   102→   102→  const base = 0x1F1E6 - 65; // 'A' = 65, regional indicator for A = U+1F1E6
   103→   103→  const char1 = String.fromCodePoint(base + c.charCodeAt(0));
   104→   104→  const char2 = String.fromCodePoint(base + c.charCodeAt(1));
   105→   105→  return char1 + char2;
   106→   106→}

   112→   112→type SortOption = 'az' | 'za' | 'country' | 'genre';
   113→   113→
   114→   114→function sortStations(stations: RadioStation[], sort: SortOption): RadioStation[] {
   115→   115→  const sorted = [...stations];
   116→   116→  switch (sort) {
   117→   117→    case 'az':
   118→   118→      return sorted.sort((a, b) => a.name.localeCompare(b.name));
   119→   119→    case 'za':
   120→   120→      return sorted.sort((a, b) => b.name.localeCompare(a.name));
   121→   121→    case 'country':
   122→   122→      return sorted.sort((a, b) => a.country.localeCompare(b.country) || a.name.localeCompare(b.name));
   123→   123→    case 'genre':
   124→   124→      return sorted.sort((a, b) => a.genre.localeCompare(b.genre) || a.name.localeCompare(b.name));
   125→   125→    default:
   126→   126→      return sorted;
   127→   127→  }
   128→   128→}

   133→   133→
   134→   134→function useDebouncedValue<T>(value: T, delay: number): T {
   135→   135→  const [debounced, setDebounced] = useState(value);
   136→   136→  const timerRef = useRef<ReturnType<typeof setTimeout>>();
   137→   137→
   138→   138→  useEffect(() => {
   139→   139→    if (timerRef.current) clearTimeout(timerRef.current);
   140→   140→    timerRef.current = setTimeout(() => setDebounced(value), delay);
   141→   141→    return () => {
   142→   142→      if (timerRef.current) clearTimeout(timerRef.current);
   143→   143→    };
   144→   144→  }, [value, delay]);
   145→   145→
   146→   146→  return debounced;
   147→   147→}
