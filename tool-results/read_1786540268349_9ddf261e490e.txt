     3→import React, { useState, useMemo } from 'react';
     4→import { usePlayerStore } from '@/store/player';
     5→import { useDiscoveryStore } from '@/store/discovery';
     6→import { tracks, artists, genres, playlists, getCoverGradient, formatDuration, getTrackById, getArtistById } from '@/lib/data';
     7→import { radioStations as allRadioStations, generateRadio } from '@/lib/metadata';
     8→import type { RadioSeed } from '@/lib/metadata';
     9→import { Card, CardContent } from '@/components/ui/card';
    10→import { Button } from '@/components/ui/button';

    22→      {[1, 2, 3, 4, 5, 6, 7, 8].map((i) => (

    40→
    41→export function RadioView() {
    42→  const { currentTrack, setQueue } = usePlayerStore();
    43→  const {
    44→    isRadioPlaying,
    45→    currentRadioStation,
    46→    currentGeneratedRadio,
    47→    favoriteStationIds,
    48→    playRadioStation,
    49→    stopRadio,
    50→    startRadioFrom,
    51→    toggleStationFavorite,
    52→  } = useDiscoveryStore();
    53→
    54→  const [searchQuery, setSearchQuery] = useState('');
    55→  const [genreFilter, setGenreFilter] = useState('all');
    56→  const [seedType, setSeedType] = useState<'track' | 'artist' | 'genre' | 'playlist'>('track');
    57→  const [selectedSeedId, setSelectedSeedId] = useState('');
    58→
    59→  const allGenres = useMemo(() => {
    60→    const g = new Set(allRadioStations.map(s => s.genre));
    61→    return Array.from(g).sort();
    62→  }, []);
    63→
    64→  const favoriteStations = useMemo(
    65→    () => allRadioStations.filter(s => favoriteStationIds.includes(s.id)),
    66→    [favoriteStationIds]
    67→  );
    68→
    69→  const filteredStations = useMemo(() => {
    70→    let result = allRadioStations.filter(s => !favoriteStationIds.includes(s.id));
    71→    if (genreFilter !== 'all') {
    72→      result = result.filter(s => s.genre === genreFilter);
    73→    }
    74→    if (searchQuery.trim()) {
    75→      const q = searchQuery.toLowerCase();
    76→      result = result.filter(
    77→        s =>
    78→          s.name.toLowerCase().includes(q) ||
    79→          s.description.toLowerCase().includes(q) ||
    80→          s.genre.toLowerCase().includes(q) ||
    81→          s.country.toLowerCase().includes(q) ||
    82→          s.tags.some(t => t.toLowerCase().includes(q))
    83→      );
    84→    }
    85→    return result;
    86→  }, [searchQuery, genreFilter, favoriteStationIds]);
    87→
    88→  const seedOptions = useMemo(() => {
    89→    if (seedType === 'track') {
    90→      return tracks.slice(0, 30).map(t => ({ id: t.id, name: `${t.title} — ${t.artistName}` }));
    91→    }
    92→    if (seedType === 'artist') {
    93→      return artists.map(a => ({ id: a.id, name: a.name }));
    94→    }
    95→    if (seedType === 'genre') {
    96→      return genres.map(g => ({ id: g.id, name: g.name }));
    97→    }
    98→    if (seedType === 'playlist') {
    99→      return playlists.map(p => ({ id: p.id, name: p.name }));
   100→    }
   101→    return [];
   102→  }, [seedType]);
   103→
   104→  const handlePlayStation = (station: typeof allRadioStations[0]) => {
   105→    if (isRadioPlaying && currentRadioStation?.id === station.id) {
   106→      stopRadio();
   107→    } else {
   108→      playRadioStation(station);
   109→    }
   110→  };
   111→
   112→  const handleStartFromCurrentTrack = () => {
   113→    if (currentTrack) {
   114→      startRadioFrom({ type: 'track', id: currentTrack.id, name: currentTrack.title });
   115→    }
   116→  };
   117→
   118→  const handleStartFromArtist = () => {
   119→    if (currentTrack) {
   120→      const artist = getArtistById(currentTrack.artistId);
   121→      if (artist) {
   122→        startRadioFrom({ type: 'artist', id: artist.id, name: artist.name });
   123→      }
   124→    }
   125→  };
   126→
   127→  const handleCreateMix = () => {
   128→    if (!selectedSeedId) return;
   129→    const opt = seedOptions.find(o => o.id === selectedSeedId);
   130→    if (!opt) return;
   131→    startRadioFrom({ type: seedType, id: selectedSeedId, name: opt.name });
   132→  };
   133→
   134→  const generatedTracks = useMemo(() => {
   135→    if (!currentGeneratedRadio) return [];
   136→    return currentGeneratedRadio.trackIds
   137→      .map(id => getTrackById(id))
   138→      .filter(Boolean) as typeof tracks;
   139→  }, [currentGeneratedRadio]);
   140→
   141→  const handlePlayGeneratedTrack = (index: number) => {
   142→    if (generatedTracks.length > 0) {
   143→      setQueue(generatedTracks, index);
   144→    }
   145→  };
   146→
   147→  return (
   148→    <ScrollArea className="h-full">
   149→      <div className="max-w-5xl mx-auto p-6">
   150→        <div className="flex items-center gap-3 mb-6">
   151→          <div className="w-10 h-10 rounded-lg bg-gradient-to-br from-rose-900 to-orange-900 flex items-center justify-center">
   152→            <Radio className="w-5 h-5 text-rose-200" />
   153→          </div>
   154→          <div>
   155→            <h1 className="text-2xl font-bold">Radio</h1>
   156→            <p className="text-sm text-muted-foreground">Internet radio & algorithmic mixes</p>
   157→          </div>
   158→        </div>
   159→
   160→        <Tabs defaultValue="stations" className="w-full">
   161→          <TabsList className="mb-6">
   162→            <TabsTrigger value="stations">
   163→              <Radio className="w-4 h-4 mr-2" />
   164→              Stations
   165→            </TabsTrigger>
   166→            <TabsTrigger value="mix">
   167→              <Shuffle className="w-4 h-4 mr-2" />
   168→              Mix Radio
   169→            </TabsTrigger>
   170→            <TabsTrigger value="now-playing">
   171→              <Signal className="w-4 h-4 mr-2" />
   172→              Now Playing
   173→            </TabsTrigger>
   174→          </TabsList>
   175→
   176→          {/* ═══ STATIONS TAB ═══ */}
   177→          <TabsContent value="stations">
   178→            {/* Search & Filter */}
   179→            <div className="flex flex-col sm:flex-row gap-3 mb-6">
   180→              <div className="relative flex-1">
   181→                <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   182→                <Input
   183→                  placeholder="Search stations..."
   184→                  value={searchQuery}
   185→                  onChange={(e) => setSearchQuery(e.target.value)}
   186→                  className="pl-9"
   187→                />
   188→              </div>
   189→              <Select value={genreFilter} onValueChange={setGenreFilter}>
   190→                <SelectTrigger className="w-[180px]">
   191→                  <SelectValue placeholder="Filter by genre" />
   192→                </SelectTrigger>
   193→                <SelectContent>
   194→                  <SelectItem value="all">All Genres</SelectItem>
   195→                  {allGenres.map(g => (
   196→                    <SelectItem key={g} value={g}>{g}</SelectItem>
   197→                  ))}
   198→                </SelectContent>
   199→              </Select>
   200→            </div>
   201→
   202→            {/* Favorites Section */}
   203→            {favoriteStations.length > 0 && (
   204→              <div className="mb-8">
   205→                <h2 className="text-lg font-semibold mb-3 flex items-center gap-2">
   206→                  <Star className="w-4 h-4 text-amber-400" />
   207→                  Favorites
   208→                </h2>
   209→                <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   210→                  {favoriteStations.map(station => (
   211→                    <StationCard
   212→                      key={station.id}
   213→                      station={station}
   214→                      isPlaying={isRadioPlaying && currentRadioStation?.id === station.id}
   215→                      isFavorite
   216→                      onPlay={() => handlePlayStation(station)}
   217→                      onToggleFavorite={() => toggleStationFavorite(station.id)}
   218→                    />
   219→                  ))}
   220→                </div>
   221→                <Separator className="mt-6" />
   222→              </div>
   223→            )}
   224→
   225→            {/* All Stations */}
   226→            <h2 className="text-lg font-semibold mb-3">
   227→              All Stations{genreFilter !== 'all' ? ` — ${genreFilter}` : ''}
   228→              <span className="text-sm font-normal text-muted-foreground ml-2">
   229→                ({filteredStations.length})
   230→              </span>
   231→            </h2>
   232→
   233→            {filteredStations.length === 0 ? (
   234→              <div className="text-center py-12 text-muted-foreground">
   235→                <Radio className="w-10 h-10 mx-auto mb-3 opacity-40" />
   236→                <p>No stations found matching your criteria</p>
   237→              </div>
   238→            ) : (
   239→              <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   240→                {filteredStations.map(station => (
   241→                  <StationCard
   242→                    key={station.id}
   243→                    station={station}
   244→                    isPlaying={isRadioPlaying && currentRadioStation?.id === station.id}
   245→                    isFavorite={favoriteStationIds.includes(station.id)}
   246→                    onPlay={() => handlePlayStation(station)}
   247→                    onToggleFavorite={() => toggleStationFavorite(station.id)}
   248→                  />
   249→                ))}
   250→              </div>
   251→            )}
   252→          </TabsContent>
   253→
   254→          {/* ═══ MIX RADIO TAB ═══ */}
   255→          <TabsContent value="mix">
   256→            {/* Quick Start Buttons */}
   257→            <Card className="mb-6">
   258→              <CardContent className="p-5">
   259→                <h2 className="text-lg font-semibold mb-4 flex items-center gap-2">
   260→                  <Zap className="w-5 h-5 text-amber-400" />
   261→                  Quick Start
   262→                </h2>
   263→                <p className="text-sm text-muted-foreground mb-4">
   264→                  Generate an algorithmic radio mix from your current listening context.
   265→                </p>
   266→                <div className="flex flex-wrap gap-3">
   267→                  <Button
   268→                    variant="outline"
   269→                    onClick={handleStartFromCurrentTrack}
   270→                    disabled={!currentTrack}
   271→                  >
   272→                    <Music className="w-4 h-4 mr-2" />
   273→                    Start from Current Track
   274→                  </Button>
   275→                  <Button
   276→                    variant="outline"
   277→                    onClick={handleStartFromArtist}
   278→                    disabled={!currentTrack}
   279→                  >
   280→                    <User className="w-4 h-4 mr-2" />
   281→                    Start from Current Artist
   282→                  </Button>
   283→                </div>
   284→                {!currentTrack && (
   285→                  <p className="text-xs text-muted-foreground mt-2">
   286→                    Play a track first to use quick start options.
   287→                  </p>
   288→                )}
   289→              </CardContent>
   290→            </Card>
   291→
   292→            {/* Create Mix Section */}
   293→            <Card className="mb-6">
   294→              <CardContent className="p-5">
   295→                <h2 className="text-lg font-semibold mb-4 flex items-center gap-2">
   296→                  <ListMusic className="w-5 h-5 text-emerald-400" />
   297→                  Create Mix
   298→                </h2>
   299→                <p className="text-sm text-muted-foreground mb-4">
   300→                  Pick a seed to generate a personalized radio mix from your library.
   301→                </p>
   302→                <div className="flex flex-col sm:flex-row gap-3 mb-4">
   303→                  <Select
   304→                    value={seedType}
   305→                    onValueChange={(v) => {
   306→                      setSeedType(v as RadioSeed['type']);
   307→                      setSelectedSeedId('');
   308→                    }}
   309→                  >
   310→                    <SelectTrigger className="w-[160px]">
   311→                      <SelectValue />
   312→                    </SelectTrigger>
   313→                    <SelectContent>
   314→                      <SelectItem value="track">
   315→                        <span className="flex items-center gap-2"><Music className="w-3 h-3" /> Track</span>
   316→                      </SelectItem>
   317→                      <SelectItem value="artist">
   318→                        <span className="flex items-center gap-2"><User className="w-3 h-3" /> Artist</span>
   319→                      </SelectItem>
   320→                      <SelectItem value="genre">
   321→                        <span className="flex items-center gap-2"><Globe className="w-3 h-3" /> Genre</span>
   322→                      </SelectItem>
   323→                      <SelectItem value="playlist">
   324→                        <span className="flex items-center gap-2"><ListMusic className="w-3 h-3" /> Playlist</span>
   325→                      </SelectItem>
   326→                    </SelectContent>
   327→                  </Select>
   328→
   329→                  <Select value={selectedSeedId} onValueChange={setSelectedSeedId}>
   330→                    <SelectTrigger className="flex-1">
   331→                      <SelectValue placeholder={`Select a ${seedType}...`} />
   332→                    </SelectTrigger>
   333→                    <SelectContent>
   334→                      {seedOptions.map(opt => (
   335→                        <SelectItem key={opt.id} value={opt.id}>
   336→                          {opt.name}
   337→                        </SelectItem>
   338→                      ))}
   339→                    </SelectContent>
   340→                  </Select>
   341→
   342→                  <Button onClick={handleCreateMix} disabled={!selectedSeedId}>
   343→                    <Shuffle className="w-4 h-4 mr-2" />
   344→                    Generate
   345→                  </Button>
   346→                </div>
   347→              </CardContent>
   348→            </Card>
   349→
   350→            {/* Generated Radio Preview */}
   351→            {currentGeneratedRadio && (
   352→              <Card>
   353→                <CardContent className="p-5">
   354→                  <div className="flex items-center gap-3 mb-4">
   355→                    <div className={`w-12 h-12 rounded-lg bg-gradient-to-br ${getCoverGradient(currentGeneratedRadio.id)} flex items-center justify-center`}>
   356→                      <Shuffle className="w-5 h-5 text-white" />
   357→                    </div>
   358→                    <div className="flex-1 min-w-0">
   359→                      <h3 className="font-semibold truncate">{currentGeneratedRadio.name}</h3>
   360→                      <p className="text-xs text-muted-foreground">
   361→                        Seeded from {currentGeneratedRadio.seed.name} ({currentGeneratedRadio.seed.type})
   362→                      </p>
   363→                    </div>
   364→                    <AnimatedWaveform />
   365→                  </div>
   366→                  <p className="text-sm text-muted-foreground mb-4">
   367→                    {currentGeneratedRadio.description}
   368→                  </p>
   369→                  <Separator className="mb-4" />
   370→                  <h4 className="text-sm font-semibold mb-3 flex items-center gap-2">
   371→                    <ListMusic className="w-4 h-4" />
   372→                    Track Preview
   373→                    <Badge variant="secondary" className="text-xs">
   374→                      {generatedTracks.length} tracks
   375→                    </Badge>
   376→                  </h4>
   377→                  <div className="space-y-1 max-h-96 overflow-y-auto">
   378→                    {generatedTracks.map((track, idx) => (
   379→                      <button
   380→                        key={track.id}
   381→                        className="w-full flex items-center gap-3 p-2 rounded-md hover:bg-accent/30 text-left group"
   382→                        onClick={() => handlePlayGeneratedTrack(idx)}
   383→                      >
   384→                        <span className="text-xs text-muted-foreground w-6 text-right">
   385→                          {idx + 1}
   386→                        </span>
   387→                        <div className={`w-8 h-8 rounded bg-gradient-to-br ${getCoverGradient(track.albumId)} flex-shrink-0`} />
   388→                        <div className="flex-1 min-w-0">
   389→                          <p className="text-sm font-medium truncate group-hover:text-primary transition-colors">
   390→                            {track.title}
   391→                          </p>
   392→                          <p className="text-xs text-muted-foreground truncate">
   393→                            {track.artistName} — {track.albumName}
   394→                          </p>
   395→                        </div>
   396→                        <span className="text-xs text-muted-foreground">
   397→                          {formatDuration(track.duration)}
   398→                        </span>
   399→                        <Play className="w-4 h-4 text-muted-foreground opacity-0 group-hover:opacity-100 transition-opacity" />
   400→                      </button>
   401→                    ))}
   402→                  </div>
   403→                </CardContent>
   404→              </Card>
   405→            )}
   406→          </TabsContent>
   407→
   408→          {/* ═══ NOW PLAYING TAB ═══ */}
   409→          <TabsContent value="now-playing">
   410→            {!isRadioPlaying ? (
   411→              <div className="text-center py-16 text-muted-foreground">
   412→                <Radio className="w-14 h-14 mx-auto mb-4 opacity-30" />
   413→                <h2 className="text-lg font-semibold mb-2">Nothing Playing</h2>
   414→                <p className="text-sm">Tune into a station or start a mix to begin listening.</p>
   415→              </div>
   416→            ) : currentRadioStation ? (
   417→              <Card>
   418→                <CardContent className="p-6">
   419→                  <div className="flex flex-col sm:flex-row gap-6">
   420→                    <div className={`w-40 h-40 rounded-xl bg-gradient-to-br ${getCoverGradient(currentRadioStation.id)} flex-shrink-0 flex items-center justify-center shadow-lg`}>
   421→                      <Radio className="w-14 h-14 text-white/80" />
   422→                    </div>
   423→                    <div className="flex-1">
   424→                      <div className="flex items-center gap-3 mb-2">
   425→                        <AnimatedWaveform />
   426→                        <Badge variant="destructive" className="text-xs">LIVE</Badge>
   427→                      </div>
   428→                      <h2 className="text-2xl font-bold mb-1">{currentRadioStation.name}</h2>
   429→                      <p className="text-sm text-muted-foreground mb-4">
   430→                        {currentRadioStation.description}
   431→                      </p>
   432→                      <div className="flex flex-wrap gap-2 mb-4">
   433→                        <Badge variant="secondary">{currentRadioStation.genre}</Badge>
   434→                        <Badge variant="outline" className="gap-1">
   435→                          <Globe className="w-3 h-3" />
   436→                          {currentRadioStation.country}
   437→                        </Badge>
   438→                        <Badge variant="outline" className="gap-1">
   439→                          <Signal className="w-3 h-3" />
   440→                          {currentRadioStation.codec} / {currentRadioStation.bitrate}kbps

   444→                        {currentRadioStation.tags.map(tag => (
   445→                          <Badge key={tag} variant="outline" className="text-xs">
   446→                            {tag}

   451→                        <Square className="w-4 h-4 mr-2" />

   455→                  </div>

   457→              </Card>
   458→            ) : currentGeneratedRadio ? (
   459→              <Card>
   460→                <CardContent className="p-6">
   461→                  <div className="flex flex-col sm:flex-row gap-6 mb-6">
   462→                    <div className={`w-40 h-40 rounded-xl bg-gradient-to-br ${getCoverGradient(currentGeneratedRadio.id)} flex-shrink-0 flex items-center justify-center shadow-lg`}>
   463→                      <Shuffle className="w-14 h-14 text-white/80" />
   464→                    </div>
   465→                    <div className="flex-1">
   466→                      <div className="flex items-center gap-3 mb-2">
   467→                        <AnimatedWaveform />
   468→                        <Badge variant="secondary" className="text-xs">MIX RADIO</Badge>
   469→                      </div>
   470→                      <h2 className="text-2xl font-bold mb-1">{currentGeneratedRadio.name}</h2>
   471→                      <p className="text-sm text-muted-foreground mb-2">
   472→                        {currentGeneratedRadio.description}
   473→                      </p>
   474→                      <p className="text-xs text-muted-foreground">
   475→                        Seed: {currentGeneratedRadio.seed.name} ({currentGeneratedRadio.seed.type})
   476→                      </p>
   477→                      <Button variant="destructive" size="sm" className="mt-4" onClick={stopRadio}>
   478→                        <Square className="w-4 h-4 mr-2" />
   479→                        Stop
   480→                      </Button>
   481→                    </div>
   482→                  </div>
   483→
   484→                  {generatedTracks.length > 0 && (
   485→                    <>
   486→                      <Separator className="mb-4" />
   487→                      <h3 className="text-sm font-semibold mb-3 flex items-center gap-2">
   488→                        <Clock className="w-4 h-4" />
   489→                        Up Next
   490→                      </h3>
   491→                      <div className="space-y-1 max-h-96 overflow-y-auto">
   492→                        {generatedTracks.map((track, idx) => (

   494→                            key={track.id}
   495→                            className="w-full flex items-center gap-3 p-2 rounded-md hover:bg-accent/30 text-left group"
   496→                            onClick={() => handlePlayGeneratedTrack(idx)}
   497→                          >
   498→                            <span className="text-xs text-muted-foreground w-6 text-right">

   500→                            </span>

   502→                            <div className="flex-1 min-w-0">
   503→                              <p className="text-sm font-medium truncate group-hover:text-primary transition-colors">
   504→                                {track.title}
   505→                              </p>
   506→                              <p className="text-xs text-muted-foreground truncate">
   507→                                {track.artistName} — {track.albumName}
   508→                              </p>
   509→                            </div>
   510→                            <span className="text-xs text-muted-foreground">

   514→                          </button>

   516→                      </div>
   517→                    </>

   519→                </CardContent>

   522→          </TabsContent>

   524→      </div>

   530→
   531→function StationCard({
   532→  station,
   533→  isPlaying,
   534→  isFavorite,
   535→  onPlay,
   536→  onToggleFavorite,
