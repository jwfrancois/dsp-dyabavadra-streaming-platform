   791→  const m = Math.floor((seconds % 3600) / 60);
   792→  const s = seconds % 60;
   793→  if (h > 0) {
   794→    return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
   795→  }
