   162→   162→/** Convert Jellyfin ticks (100ns units) to seconds */
   163→   163→export function ticksToSeconds(ticks: number | undefined): number {
   164→   164→  if (ticks === undefined || ticks === null) return 0;
   165→   165→  return Math.floor(ticks / TICKS_PER_SECOND);
   166→   166→}

   168→   168→/** Convert seconds to Jellyfin ticks */
   169→   169→export function secondsToTicks(seconds: number): number {

   178→   178→/** Build a query string from a record, filtering out undefined/null/empty values */
   179→   179→function buildQueryString(params: Record<string, string | number | boolean | undefined | null | string[] | number[]>): string {
   180→   180→  const parts: string[] = [];
   181→   181→  for (const [key, value] of Object.entries(params)) {
   182→   182→    if (value === undefined || value === null || value === '') continue;
   183→   183→    if (Array.isArray(value)) {
   184→   184→      if (value.length > 0) {
   185→   185→        parts.push(`${encodeURIComponent(key)}=${value.map(v => encodeURIComponent(String(v))).join(',')}`);
   186→   186→      }
   187→   187→    } else {
   188→   188→      parts.push(`${encodeURIComponent(key)}=${encodeURIComponent(String(value))}`);
   189→   189→    }
   190→   190→  }
   191→   191→  return parts.length > 0 ? `?${parts.join('&')}` : '';
   192→   192→}

   237→   237→      if (!raw) return null;

   239→   239→      // Basic validation
   240→   240→      if (!parsed.serverUrl || !parsed.accessToken || !parsed.userId) return null;

   251→   251→  private getBaseUrl(): string {
   252→   252→    if (!this.config) throw new JellyfinNotAuthenticatedError();
   253→   253→    return normalizeServerUrl(this.config.serverUrl);
   254→   254→  }
   255→   255→
   256→   256→  private getHeaders(): Record<string, string> {
   257→   257→    if (!this.config) throw new JellyfinNotAuthenticatedError();
   258→   258→    return {
   259→   259→      'X-Emby-Token': this.config.accessToken,
   260→   260→      'Content-Type': 'application/json',
   261→   261→      'Accept': 'application/json',
   262→   262→    };

   264→   264→
   265→   265→  private async request<T>(path: string, options: RequestInit = {}): Promise<T> {
   266→   266→    const baseUrl = this.getBaseUrl();
   267→   267→    const url = `${baseUrl}${path}`;
   268→   268→    const headers: Record<string, string> = {
   269→   269→      ...this.getHeaders(),
   270→   270→      ...(options.headers as Record<string, string> | undefined),
   271→   271→    };
   272→   272→
   273→   273→    // Remove Content-Type for GET/DELETE if no body
   274→   274→    if (!options.body && (options.method === 'GET' || options.method === 'DELETE')) {
   275→   275→      delete headers['Content-Type'];
   276→   276→    }
   277→   277→
   278→   278→    let response: Response;
   279→   279→    try {
   280→   280→      response = await fetch(url, {
   281→   281→        ...options,
   282→   282→        headers,
   283→   283→      });
   284→   284→    } catch (err) {
   285→   285→      throw new JellyfinApiError(
   286→   286→        `Network error: ${err instanceof Error ? err.message : 'Unknown error'}`,
   287→   287→        0,
   288→   288→        path,
   289→   289→        err
   290→   290→      );
   291→   291→    }
   292→   292→
   293→   293→    if (!response.ok) {
   294→   294→      let errorBody = '';
   295→   295→      try {
   296→   296→        errorBody = await response.text();
   297→   297→      } catch {
   298→   298→        // ignore
   299→   299→      }
   300→   300→      throw new JellyfinApiError(
   301→   301→        `Jellyfin API error ${response.status}: ${response.statusText} — ${errorBody}`,
   302→   302→        response.status,
   303→   303→        path
   304→   304→      );
   305→   305→    }
   306→   306→
   307→   307→    // Some endpoints return empty responses (e.g., DELETE, favorite toggles)
   308→   308→    const contentType = response.headers.get('content-type');
   309→   309→    if (contentType && contentType.includes('application/json')) {
   310→   310→      return response.json() as Promise<T>;
   311→   311→    }
   312→   312→
   313→   313→    return undefined as unknown as T;
   314→   314→  }

   320→   320→  /**
   321→   321→   * Connect to a Jellyfin server. Authenticates and fetches server info.
   322→   322→   * Persists the configuration to localStorage.
   323→   323→   */
   324→   324→  async connect(serverUrl: string, username: string, password: string): Promise<JellyfinConfig> {
   325→   325→    const baseUrl = normalizeServerUrl(serverUrl);
   326→   326→
   327→   327→    // Step 1: Authenticate
   328→   328→    const authResponse = await this.request<JellyfinAuthResponse>(
   329→   329→      '/Users/AuthenticateByName',
   330→   330→      {
   331→   331→        method: 'POST',
   332→   332→        body: JSON.stringify({ Username: username, Pw: password }),
   333→   333→        headers: {
   334→   334→          'Content-Type': 'application/json',
   335→   335→          'Authorization': `MediaBrowser Client="DSP", Device="Browser", DeviceId="dsp-jellyfin-${typeof window !== 'undefined' ? window.navigator.userAgent.slice(0, 32).replace(/[^a-zA-Z0-9]/g, '') : 'ssr'}", Version="1.0.0"`,
   336→   336→        },
   337→   337→      }
   338→   338→    );
   339→   339→
   340→   340→    const token = authResponse.AccessToken;
   341→   341→    const userId = authResponse.User.Id;
   342→   342→    const serverId = authResponse.ServerId;
   343→   343→
   344→   344→    // Step 2: Fetch server info
   345→   345→    let serverName = 'Jellyfin';
   346→   346→    let serverVersion = 'unknown';
   347→   347→    try {
   348→   348→      // Temporarily set config for the getSystemInfo call
   349→   349→      this.config = {
   350→   350→        serverUrl: baseUrl,
   351→   351→        username,
   352→   352→        accessToken: token,
   353→   353→        userId,
   354→   354→        serverId,
   355→   355→        serverName,
   356→   356→        serverVersion,
   357→   357→        connectedAt: new Date().toISOString(),
   358→   358→        lastHeartbeat: new Date().toISOString(),
   359→   359→        musicLibraryId: '',
   360→   360→      };
   361→   361→      const sysInfo = await this.getSystemInfo();
   362→   362→      serverName = sysInfo.ServerName;
   363→   363→      serverVersion = sysInfo.Version;
   364→   364→    } catch {
   365→   365→      // If system info fails, continue with defaults
   366→   366→    }
   367→   367→
   368→   368→    // Step 3: Auto-detect music library
   369→   369→    let musicLibraryId = '';
   370→   370→    try {
   371→   371→      const views = await this.getViews();
   372→   372→      const musicFolder = views.Items.find(
   373→   373→        item => item.Type === 'CollectionFolder' && item.Name.toLowerCase().includes('music')
   374→   374→      );
   375→   375→      if (musicFolder) {
   376→   376→        musicLibraryId = musicFolder.Id;
   377→   377→      }
   378→   378→    } catch {
   379→   379→      // Continue without auto-detected library
   380→   380→    }
   381→   381→
   382→   382→    // Step 4: Build and save config
   383→   383→    const config: JellyfinConfig = {
   384→   384→      serverUrl: baseUrl,
   385→   385→      username,
   386→   386→      accessToken: token,
   387→   387→      userId,
   388→   388→      serverId,
   389→   389→      serverName,
   390→   390→      serverVersion,
   391→   391→      connectedAt: new Date().toISOString(),
   392→   392→      lastHeartbeat: new Date().toISOString(),
   393→   393→      musicLibraryId,
   394→   394→    };
   395→   395→
   396→   396→    this.config = config;
   397→   397→    this.saveConfig();
   398→   398→
   399→   399→    return config;
   400→   400→  }

   478→   478→   */
   479→   479→  async getItems(params: {
   480→   480→    parentId?: string;
   481→   481→    includeItemTypes?: string[];
   482→   482→    sortBy?: string;
   483→   483→    sortOrder?: 'Ascending' | 'Descending';
   484→   484→    limit?: number;
   485→   485→    startIndex?: number;
   486→   486→    searchTerm?: string;
   487→   487→    fields?: string[];
   488→   488→    recursive?: boolean;
   489→   489→    genres?: string[];
   490→   490→    artistIds?: string[];
   491→   491→    albumArtistIds?: string[];
   492→   492→    years?: number[];
   493→   493→    ids?: string[];
   494→   494→    filters?: string[];
   495→   495→  }): Promise<JellyfinItemsResponse> {
   496→   496→    const defaultFields = [
   497→   497→      'PrimaryImageAspectRatio',
   498→   498→      'BasicSyncInfo',
   499→   499→      'Genres',
   500→   500→      'Overview',
   501→   501→      'People',
   502→   502→      'ProductionYear',
   503→   503→      'CommunityRating',
   504→   504→    ];
   505→   505→
   506→   506→    const qs = buildQueryString({
   507→   507→      UserId: this.config!.userId,
   508→   508→      ParentId: params.parentId,
   509→   509→      IncludeItemTypes: params.includeItemTypes,
   510→   510→      SortBy: params.sortBy,
   511→   511→      SortOrder: params.sortOrder,
   512→   512→      Limit: params.limit,
   513→   513→      StartIndex: params.startIndex,
   514→   514→      SearchTerm: params.searchTerm,
   515→   515→      Fields: params.fields ?? defaultFields,
   516→   516→      Recursive: params.recursive,
   517→   517→      Genres: params.genres,
   518→   518→      ArtistIds: params.artistIds,
   519→   519→      AlbumArtistIds: params.albumArtistIds,
   520→   520→      Years: params.years,
   521→   521→      Ids: params.ids,
   522→   522→      Filters: params.filters,
   523→   523→    });

   779→   779→  /**
   780→   780→   * Get a playable audio stream URL for an item.
   781→   781→   * Prefers direct play for best quality; falls back to transcoded stream.
   782→   782→   */
   783→   783→  async getPlaybackStreamUrl(
   784→   784→    itemId: string,
   785→   785→    maxBitrate: number = DEFAULT_MAX_BITRATE
   786→   786→  ): Promise<string> {
   787→   787→    const baseUrl = this.getBaseUrl();
   788→   788→    const token = this.config!.accessToken;
   789→   789→    const userId = this.config!.userId;
   790→   790→
   791→   791→    try {
   792→   792→      // Request playback info to determine best playback method
   793→   793→      const playbackInfo = await this.request<JellyfinPlaybackInfo>(
   794→   794→        `/Items/${itemId}/PlaybackInfo`,
   795→   795→        {
   796→   796→          method: 'POST',
   797→   797→          body: JSON.stringify({
   798→   798→            UserId: userId,
   799→   799→            MaxStreamingBitrate: maxBitrate,
   800→   800→            AutoOpenLiveStreams: true,
   801→   801→          }),
   802→   802→        }
   803→   803→      );
   804→   804→
   805→   805→      if (playbackInfo.MediaSources && playbackInfo.MediaSources.length > 0) {
   806→   806→        const source = playbackInfo.MediaSources[0];
   807→   807→
   808→   808→        // Prefer direct play — best quality, no server transcoding
   809→   809→        if (source.SupportsDirectPlay) {
   810→   810→          return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${token}&UserId=${userId}`;
   811→   811→        }
   812→   812→
   813→   813→        // Fallback to direct stream
   814→   814→        if (source.SupportsDirectStream && source.DirectStreamUrl) {
   815→   815→          return `${baseUrl}${source.DirectStreamUrl}`;
   816→   816→        }
   817→   817→
   818→   818→        // Fallback to transcoded stream (HLS)
   819→   819→        if (source.SupportsTranscoding && source.TranscodingUrl) {
   820→   820→          return `${baseUrl}${source.TranscodingUrl}`;
   821→   821→        }
   822→   822→
   823→   823→        // Last resort: construct a universal audio transcoding URL
   824→   824→        const container = source.Container || 'mp3';
   825→   825→        return `${baseUrl}/Audio/${itemId}/universal?api_key=${token}&UserId=${userId}&MaxStreamingBitrate=${maxBitrate}&Container=${container}&TranscodingMaxAudioChannels=2&SegmentContainer=ts&AudioCodec=aac&Protocol=hls`;
   826→   826→      }
   827→   827→    } catch {
   828→   828→      // If playback info fails, fall back to direct stream URL
   829→   829→    }
   830→   830→
   831→   831→    // Fallback: try static stream directly
   832→   832→    return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${token}&UserId=${userId}`;
   833→   833→  }
