    23→    23→    23→function formatBytes(bytes: number): string {
    24→    24→    24→  if (bytes >= 1099511627776) return (bytes / 1099511627776).toFixed(1) + ' TB';
    25→    25→    25→  if (bytes >= 1073741824) return (bytes / 1073741824).toFixed(1) + ' GB';
    26→    26→    26→  if (bytes >= 1048576) return (bytes / 1048576).toFixed(1) + ' MB';
    27→    27→    27→  return bytes.toLocaleString() + ' B';
    28→    28→    28→}
    29→    29→    29→
    30→    30→    30→function formatUptime(s: number): string {
    31→    31→    31→  return Math.floor(s / 86400) + 'd ' + Math.floor((s % 86400) / 3600) + 'h ' + Math.floor((s % 3600) / 60) + 'm';
    32→    32→    32→}

    34→    34→    34→function usageColor(pct: number): string {
    35→    35→    35→  if (pct < 60) return 'bg-signal-green';
    36→    36→    36→  if (pct < 80) return 'bg-signal-amber';
    37→    37→    37→  return 'bg-signal-red';
    38→    38→    38→}

    83→    83→    83→function formatTimeAgo(iso?: string): string {
    84→    84→    84→  if (!iso) return '—';
    85→    85→    85→  const diff = Date.now() - new Date(iso).getTime();
    86→    86→    86→  const mins = Math.floor(diff / 60000);
    87→    87→    87→  if (mins < 1) return 'just now';
    88→    88→    88→  if (mins < 60) return `${mins}m ago`;
    89→    89→    89→  const hrs = Math.floor(mins / 60);
    90→    90→    90→  if (hrs < 24) return `${hrs}h ago`;
    91→    91→    91→  return `${Math.floor(hrs / 24)}d ago`;
    92→    92→    92→}
