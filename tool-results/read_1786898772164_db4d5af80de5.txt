     1→     1→'use client';

     4→     4→import { useUIStore } from '@/store/ui';
     5→     5→import { usePlayerStore } from '@/store/player';
     6→     6→import { useLocalLibraryStore } from '@/store/local-library';

    10→    10→import { Button } from '@/components/ui/button';

    12→    12→import { ScrollArea } from '@/components/ui/scroll-area';

    14→    14→import { Tabs, TabsList, TabsTrigger } from '@/components/ui/tabs';
    15→    15→import {
    16→    16→  ArrowLeft, Play, Clock, Disc3, Calendar, User, Music,

    18→    18→  Users, CalendarDays, ChevronRight,

    20→    20→
    21→    21→const typeConfig: Record<EditorialCollection['type'], { label: string; color: string; bgClass: string; icon: typeof Star }> = {
    22→    22→  'new-releases': { label: 'New Release', color: 'text-primary', bgClass: 'bg-primary/15 border-primary/30 text-primary', icon: Sparkles },
    23→    23→  'genre-primer': { label: 'Genre Primer', color: 'text-blue-400', bgClass: 'bg-blue-500/15 border-blue-500/30 text-blue-400', icon: BookOpen },
    24→    24→  'best-of': { label: 'Best Of', color: 'text-yellow-400', bgClass: 'bg-yellow-500/15 border-yellow-500/30 text-yellow-400', icon: Trophy },
    25→    25→  'curated': { label: 'Curated', color: 'text-purple-400', bgClass: 'bg-purple-500/15 border-purple-500/30 text-purple-400', icon: ListMusic },
    26→    26→  'staff-picks': { label: 'Staff Picks', color: 'text-green-400', bgClass: 'bg-green-500/15 border-green-500/30 text-green-400', icon: Users },
    27→    27→  'on-this-day': { label: 'On This Day', color: 'text-amber-400', bgClass: 'bg-amber-500/15 border-amber-500/30 text-amber-400', icon: CalendarDays },

    29→    29→
    30→    30→type TabKey = 'all' | 'featured' | 'new-releases' | 'genre-primer' | 'best-of' | 'staff-picks';
    31→    31→
    32→    32→export function EditorialView() {
    33→    33→  const { viewParams, navigate } = useUIStore();
    34→    34→  const { play, setQueue, currentTrack, isPlaying } = usePlayerStore();
    35→    35→  const [selectedTab, setSelectedTab] = useState<TabKey>('all');
    36→    36→  const [selectedCollection, setSelectedCollection] = useState<string | null>(viewParams.collectionId || null);
    37→    37→
    38→    38→  const localStore = useLocalLibraryStore();
    39→    39→  const localTracks = localStore.tracks;
    40→    40→
    41→    41→  // Helper to convert local track to playable Track
    42→    42→  function localTrackToTrack(lt: any): import('@/lib/data').Track {
    43→    43→    return {
    44→    44→      id: lt.id,
    45→    45→      title: lt.title,
    46→    46→      albumId: lt.album,
    47→    47→      albumName: lt.album,
    48→    48→      artistId: lt.artist,
    49→    49→      artistName: lt.artist,
    50→    50→      trackNumber: lt.trackNumber,
    51→    51→      discNumber: lt.discNumber,
    52→    52→      duration: lt.duration,
    53→    53→      format: lt.format,
    54→    54→      bitDepth: lt.bitDepth,
    55→    55→      sampleRate: lt.sampleRate,
    56→    56→      channels: lt.channels,
    57→    57→      bitrate: lt.bitrate,
    58→    58→      filePath: lt.filePath,
    59→    59→      fileSize: lt.fileSize,
    60→    60→      composers: lt.composer ? [lt.composer] : [],
    61→    61→      performers: [],
    62→    62→      genre: lt.genre,
    63→    63→      loved: false,
    64→    64→      playCount: 0,
    65→    65→      source: 'local',
    66→    66→      isAvailable: true,
    67→    67→      blobUrl: lt.blobUrl,
    68→    68→    };
    69→    69→  }
    70→    70→
    71→    71→  const editorialCollections = useMemo(() => {
    72→    72→    const albums = localStore.getAlbums();
    73→    73→    const artists = localStore.getArtists();
    74→    74→    const allTrackIds = localTracks.map(t => t.id);
    75→    75→    const allAlbumIds = albums.map(a => a.id);
    76→    76→
    77→    77→    const today = new Date().toISOString().slice(0, 10);
    78→    78→
    79→    79→    const collections: EditorialCollection[] = [];
    80→    80→
    81→    81→    // New Releases - tracks added recently (by index as proxy)
    82→    82→    if (localTracks.length > 0) {
    83→    83→      const recentTracks = localTracks.slice(0, Math.min(20, localTracks.length));
    84→    84→      const recentAlbumIds = [...new Set(recentTracks.map(t => t.album))];
    85→    85→      collections.push({
    86→    86→        id: 'ed-recent',
    87→    87→        title: 'Recently Added',
    88→    88→        subtitle: 'Your latest library additions',
    89→    89→        description: 'The most recently imported tracks and albums in your library. Refresh your memory with what you just added — these tracks are sorted by import order, capturing your latest musical discoveries and acquisitions across all genres and formats.',
    90→    90→        type: 'new-releases',
    91→    91→        coverUrl: '',
    92→    92→        curator: 'DSP Library',
    93→    93→        trackIds: recentTracks.map(t => t.id),
    94→    94→        albumIds: recentAlbumIds,
    95→    95→        tags: ['recent', 'new', 'library'],
    96→    96→        publishedAt: today,
    97→    97→        featured: true,
    98→    98→      });
    99→    99→    }
   100→   100→
   101→   101→    // Genre Primers
   102→   102→    const genres = [...new Set(localTracks.map(t => t.genre).filter(Boolean))];
   103→   103→    for (const genre of genres.slice(0, 5)) {
   104→   104→      const genreTracks = localTracks.filter(t => t.genre === genre);
   105→   105→      const genreAlbumIds = [...new Set(genreTracks.map(t => t.album))];
   106→   106→      collections.push({
   107→   107→        id: `ed-genre-${genre.toLowerCase().replace(/[^a-z0-9]/g, '-')}`,
   108→   108→        title: `${genre} Primer`,
   109→   109→        subtitle: `Explore ${genre} in your library`,
   110→   110→        description: `A curated introduction to the ${genre} genre, featuring tracks from your library. This collection showcases the breadth and depth of your ${genre} collection, from essential classics to hidden gems that define the genre's evolution and character.`,
   111→   111→        type: 'genre-primer',
   112→   112→        coverUrl: '',
   113→   113→        curator: 'DSP Editorial',
   114→   114→        trackIds: genreTracks.slice(0, 30).map(t => t.id),
   115→   115→        albumIds: genreAlbumIds.slice(0, 10),
   116→   116→        tags: [genre.toLowerCase(), 'genre-primer', 'exploration'],
   117→   117→        publishedAt: today,
   118→   118→        featured: false,
   119→   119→      });
   120→   120→    }
   121→   121→
   122→   122→    // Best Of - by artist track count
   123→   123→    if (localTracks.length > 0) {
   124→   124→      const artistCounts = new Map<string, number>();
   125→   125→      for (const t of localTracks) {
   126→   126→        artistCounts.set(t.artist, (artistCounts.get(t.artist) || 0) + 1);
   127→   127→      }
   128→   128→      const topArtists = [...artistCounts.entries()].sort((a, b) => b[1] - a[1]).slice(0, 5);
   129→   129→      for (const [artist, count] of topArtists) {
   130→   130→        const artistTracks = localTracks.filter(t => t.artist === artist);
   131→   131→        const artistAlbumIds = [...new Set(artistTracks.map(t => t.album))];
   132→   132→        collections.push({
   133→   133→          id: `ed-best-${artist.toLowerCase().replace(/[^a-z0-9]/g, '-')}`,
   134→   134→          title: `Best of ${artist}`,
   135→   135→          subtitle: `${count} tracks in your library`,
   136→   136→          description: `The essential ${artist} tracks from your collection. This compilation brings together the defining works that make ${artist} a cornerstone of your music library, spanning multiple albums and eras of their artistic career.`,
   137→   137→          type: 'best-of',
   138→   138→          coverUrl: '',
   139→   139→          curator: 'DSP Editorial',
   140→   140→          trackIds: artistTracks.map(t => t.id),
   141→   141→          albumIds: artistAlbumIds,
   142→   142→          tags: ['best-of', artist.toLowerCase(), 'essential'],
   143→   143→          publishedAt: today,
   144→   144→          featured: count > 5,
   145→   145→        });
   146→   146→      }
   147→   147→    }
   148→   148→
   149→   149→    // Staff Picks - random selection
   150→   150→    if (localTracks.length > 3) {
   151→   151→      const shuffled = [...localTracks].sort(() => Math.random() - 0.5);
   152→   152→      const picks = shuffled.slice(0, Math.min(15, shuffled.length));
   153→   153→      const pickAlbumIds = [...new Set(picks.map(t => t.album))];
   154→   154→      collections.push({
   155→   155→        id: 'ed-staff-picks',
   156→   156→        title: 'Staff Picks',
   157→   157→        subtitle: 'Handpicked tracks from your library',
   158→   158→        description: 'A handpicked selection of outstanding tracks from your library. These picks span genres and eras, chosen for their exceptional recording quality, musical merit, and lasting listenability. Perfect for discovering forgotten gems in your collection.',
   159→   159→        type: 'staff-picks',
   160→   160→        coverUrl: '',
   161→   161→        curator: 'DSP Staff',
   162→   162→        trackIds: picks.map(t => t.id),
   163→   163→        albumIds: pickAlbumIds,
   164→   164→        tags: ['picks', 'curated', 'discovery'],
   165→   165→        publishedAt: today,
   166→   166→        featured: true,
   167→   167→      });
   168→   168→    }
   169→   169→
   170→   170→    // Full Library collection
   171→   171→    if (localTracks.length > 0) {
   172→   172→      collections.push({
   173→   173→        id: 'ed-full-library',
   174→   174→        title: 'Complete Library',
   175→   175→        subtitle: `${localTracks.length} tracks across ${albums.length} albums`,
   176→   176→        description: 'Your entire music library in one collection. Browse through every track and album you have imported, spanning all genres, artists, and formats. This is your personal music universe — explore it freely.',
   177→   177→        type: 'curated',
   178→   178→        coverUrl: '',
   179→   179→        curator: 'DSP System',
   180→   180→        trackIds: allTrackIds,
   181→   181→        albumIds: allAlbumIds,
   182→   182→        tags: ['library', 'complete', 'all'],
   183→   183→        publishedAt: today,
   184→   184→        featured: false,
   185→   185→      });
   186→   186→    }
   187→   187→
   188→   188→    // HiFi Favorites - highest quality tracks
   189→   189→    const hifiTracks = localTracks.filter(t => t.bitDepth >= 24 || t.sampleRate >= 88200);
   190→   190→    if (hifiTracks.length > 0) {
   191→   191→      const hifiAlbumIds = [...new Set(hifiTracks.map(t => t.album))];
   192→   192→      collections.push({
   193→   193→        id: 'ed-hifi',
   194→   194→        title: 'HiFi Collection',
   195→   195→        subtitle: `${hifiTracks.length} high-resolution tracks`,
   196→   196→        description: 'The finest audio quality in your library. These tracks feature high-resolution formats (24-bit or above, 88.2kHz or above), delivering the full fidelity of the original recording. Best enjoyed with critical listening on your HiFi system.',
   197→   197→        type: 'curated',
   198→   198→        coverUrl: '',
   199→   199→        curator: 'DSP Audio Lab',
   200→   200→        trackIds: hifiTracks.map(t => t.id),
   201→   201→        albumIds: hifiAlbumIds,
   202→   202→        tags: ['hifi', 'high-res', 'audiophile', 'lossless'],
   203→   203→        publishedAt: today,
   204→   204→        featured: true,
   205→   205→      });
   206→   206→    }
   207→   207→
   208→   208→    return collections;
   209→   209→  }, [localTracks, localStore]);
   210→   210→
   211→   211→  // Filter by tab
   212→   212→  const filteredCollections = useMemo(() => {
   213→   213→    if (selectedTab === 'all') return editorialCollections;
   214→   214→    if (selectedTab === 'featured') return editorialCollections.filter(c => c.featured);
   215→   215→    return editorialCollections.filter(c => c.type === selectedTab);
   216→   216→  }, [editorialCollections, selectedTab]);
   217→   217→
   218→   218→  const featuredCollections = useMemo(
   219→   219→    () => editorialCollections.filter(c => c.featured),
   220→   220→    [editorialCollections]
   221→   221→  );
   222→   222→
   223→   223→  const activeCollection = selectedCollection
   224→   224→    ? editorialCollections.find(c => c.id === selectedCollection) ?? null
   225→   225→    : null;
   226→   226→
   227→   227→  const collectionTracks = useMemo((): import('@/lib/data').Track[] => {
   228→   228→    if (!activeCollection) return [];
   229→   229→    return activeCollection.trackIds
   230→   230→      .map(id => localTracks.find(t => t.id === id))
   231→   231→      .filter(Boolean)
   232→   232→      .map(t => localTrackToTrack(t!));
   233→   233→  }, [activeCollection, localTracks]);
   234→   234→
   235→   235→  const collectionAlbums = useMemo((): { id: string; title: string; artistName: string; year: number }[] => {
   236→   236→    if (!activeCollection) return [];
   237→   237→    const albumMap = new Map<string, { id: string; title: string; artistName: string; year: number }>();
   238→   238→    for (const trackId of activeCollection.albumIds) {
   239→   239→      const lt = localTracks.find(t => t.id === trackId || t.album === trackId);
   240→   240→      if (lt) {
   241→   241→        const key = lt.album;
   242→   242→        if (!albumMap.has(key)) {
   243→   243→          albumMap.set(key, { id: key, title: lt.album, artistName: lt.albumArtist || lt.artist, year: lt.year });
   244→   244→        }
   245→   245→      }
   246→   246→    }
   247→   247→    return [...albumMap.values()];
   248→   248→  }, [activeCollection, localTracks]);
   249→   249→
   250→   250→  const playCollection = () => {
   251→   251→    if (collectionTracks.length > 0) setQueue(collectionTracks, 0);
   252→   252→  };
   253→   253→
   254→   254→  const playTrack = (trackId: string) => {
   255→   255→    const lt = localTracks.find(t => t.id === trackId);
   256→   256→    if (lt) play(localTrackToTrack(lt));
   257→   257→  };
   258→   258→
   259→   259→  const playAlbum = (albumId: string) => {
   260→   260→    const albumTracks = localTracks.filter(t => t.album === albumId);
   261→   261→    if (albumTracks.length > 0) setQueue(albumTracks.map(t => localTrackToTrack(t)), 0);
   262→   262→  };
   263→   263→
   264→   264→  const openCollection = (id: string) => {
   265→   265→    setSelectedCollection(id);
   266→   266→  };
   267→   267→
   268→   268→  const closeCollection = () => {
   269→   269→    setSelectedCollection(null);
   270→   270→  };
   271→   271→
   272→   272→  const formatDate = (iso: string) => {
   273→   273→    const d = new Date(iso);
   274→   274→    return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
   275→   275→  };
   276→   276→
   277→   277→  // ─── Detail View ───
   278→   278→  if (activeCollection) {
   279→   279→    const cfg = typeConfig[activeCollection.type];
   280→   280→    const TypeIcon = cfg.icon;
   281→   281→
   282→   282→    return (
   283→   283→      <ScrollArea className="h-full">
   284→   284→        <div className="max-w-5xl mx-auto p-6">
   285→   285→          <Button variant="ghost" size="sm" className="mb-4 text-muted-foreground" onClick={closeCollection}>
   286→   286→            <ArrowLeft className="w-4 h-4 mr-2" /> Back to Collections
   287→   287→          </Button>
   288→   288→
   289→   289→          {/* Hero */}
   290→   290→          <div className="flex flex-col md:flex-row gap-6 mb-8">
   291→   291→            <div className={`w-48 h-48 md:w-56 md:h-56 rounded-xl bg-gradient-to-br ${getCoverGradient(activeCollection.id)} shadow-2xl flex-shrink-0`} />
   292→   292→            <div className="flex-1">
   293→   293→              <div className="flex items-center gap-2 mb-3">
   294→   294→                <Badge variant="outline" className={`text-xs ${cfg.bgClass}`}>
   295→   295→                  <TypeIcon className="w-3 h-3 mr-1" />
   296→   296→                  {cfg.label}
   297→   297→                </Badge>
   298→   298→                {activeCollection.featured && (
   299→   299→                  <Badge variant="outline" className="text-xs bg-yellow-500/15 border-yellow-500/30 text-yellow-400">
   300→   300→                    <Star className="w-3 h-3 mr-1" /> Featured
   301→   301→                  </Badge>
   302→   302→                )}
   303→   303→              </div>
   304→   304→              <h1 className="text-3xl font-bold mb-2">{activeCollection.title}</h1>
   305→   305→              <p className="text-muted-foreground mb-3">{activeCollection.subtitle}</p>
   306→   306→
   307→   307→              <div className="flex flex-wrap items-center gap-4 text-sm text-muted-foreground mb-4">
   308→   308→                {activeCollection.curator && (
   309→   309→                  <span className="flex items-center gap-1.5">
   310→   310→                    <User className="w-4 h-4" /> {activeCollection.curator}
   311→   311→                  </span>
   312→   312→                )}
   313→   313→                <span className="flex items-center gap-1.5">
   314→   314→                  <Calendar className="w-4 h-4" /> {formatDate(activeCollection.publishedAt)}
   315→   315→                </span>
   316→   316→                <span className="flex items-center gap-1.5">
   317→   317→                  <Music className="w-4 h-4" /> {activeCollection.trackIds.length} tracks
   318→   318→                </span>
   319→   319→                <span className="flex items-center gap-1.5">
   320→   320→                  <Disc3 className="w-4 h-4" /> {activeCollection.albumIds.length} albums
   321→   321→                </span>
   322→   322→              </div>
   323→   323→
   324→   324→              {collectionTracks.length > 0 && (
   325→   325→                <Button onClick={playCollection}>
   326→   326→                  <Play className="w-4 h-4 mr-2" /> Play All Tracks
   327→   327→                </Button>
   328→   328→              )}
   329→   329→            </div>
   330→   330→          </div>
   331→   331→
   332→   332→          {/* Description */}
   333→   333→          <section className="mb-8">
   334→   334→            <h2 className="text-lg font-semibold mb-3">About This Collection</h2>
   335→   335→            <p className="text-sm text-muted-foreground leading-relaxed max-w-3xl">
   336→   336→              {activeCollection.description}
   337→   337→            </p>
   338→   338→          </section>
   339→   339→
   340→   340→          {/* Tags */}
   341→   341→          {activeCollection.tags.length > 0 && (
   342→   342→            <section className="mb-8">
   343→   343→              <div className="flex flex-wrap gap-2">
   344→   344→                {activeCollection.tags.map(tag => (
   345→   345→                  <Badge key={tag} variant="secondary" className="text-xs">{tag}</Badge>
   346→   346→                ))}
   347→   347→              </div>
   348→   348→            </section>
   349→   349→          )}
   350→   350→
   351→   351→          <Separator className="my-6" />
   352→   352→
   353→   353→          {/* Albums in Collection */}
   354→   354→          {collectionAlbums.length > 0 && (
   355→   355→            <section className="mb-8">
   356→   356→              <div className="flex items-center justify-between mb-4">
   357→   357→                <h2 className="text-lg font-semibold flex items-center gap-2">
   358→   358→                  <Disc3 className="w-5 h-5 text-primary" /> Albums
   359→   359→                </h2>
   360→   360→                <Badge variant="secondary" className="text-xs">{collectionAlbums.length}</Badge>
   361→   361→              </div>
   362→   362→              <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 gap-4">
   363→   363→                {collectionAlbums.map(album => album && (
   364→   364→                  <div
   365→   365→                    key={album.id}
   366→   366→                    className="group cursor-pointer"
   367→   367→                    onClick={() => navigate('album-detail', { albumId: album.id })}
   368→   368→                  >
   369→   369→                    <div className="relative mb-2">
   370→   370→                      <div className={`w-full aspect-square rounded-lg bg-gradient-to-br ${getCoverGradient(album.id)} cover-art-hover shadow-lg`} />
   371→   371→                      <Button
   372→   372→                        variant="default"
   373→   373→                        size="icon"
   374→   374→                        className="absolute bottom-2 right-2 h-8 w-8 rounded-full opacity-0 group-hover:opacity-100 transition-all translate-y-1 group-hover:translate-y-0"
   375→   375→                        onClick={(e) => { e.stopPropagation(); playAlbum(album.id); }}
   376→   376→                      >
   377→   377→                        <Play className="w-3.5 h-3.5 ml-0.5" />
   378→   378→                      </Button>
   379→   379→                    </div>
   380→   380→                    <p className="text-sm font-medium truncate group-hover:text-primary transition-colors">{album.title}</p>
   381→   381→                    <p className="text-xs text-muted-foreground">{album.artistName} · {album.year}</p>
   382→   382→                  </div>
   383→   383→                ))}
   384→   384→              </div>
   385→   385→            </section>
   386→   386→          )}
   387→   387→
   388→   388→          {/* Tracks in Collection */}
   389→   389→          {collectionTracks.length > 0 && (
   390→   390→            <section className="mb-8">
   391→   391→              <div className="flex items-center justify-between mb-4">
   392→   392→                <h2 className="text-lg font-semibold flex items-center gap-2">
   393→   393→                  <Music className="w-5 h-5 text-primary" /> Tracks
   394→   394→                </h2>
   395→   395→                <Badge variant="secondary" className="text-xs">{collectionTracks.length}</Badge>
   396→   396→              </div>
   397→   397→              <div className="space-y-1">
   398→   398→                {collectionTracks.map((track, i) => {
   399→   399→                  const isCurrentTrack = currentTrack?.id === track.id;
   400→   400→                  return (
   401→   401→                    <div
   402→   402→                      key={track.id}
   403→   403→                      className={`flex items-center gap-3 p-2.5 rounded-lg cursor-pointer transition-colors group ${
   404→   404→                        isCurrentTrack ? 'bg-primary/10 border border-primary/20' : 'hover:bg-accent/30'
   405→   405→                      }`}
   406→   406→                      onClick={() => playTrack(track.id)}
