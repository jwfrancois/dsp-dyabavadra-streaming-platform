    38→    full: 'w-full aspect-square',

   415→// MAIN JELLYFIN VIEW
   416→// ═══════════════════════════════════════════════════════════
   417→
   418→export function JellyfinView() {
   419→  const { navigate } = useUIStore();
   420→  const { setQueue, play, currentTrack, isPlaying } = usePlayerStore();
   421→  const store = useJellyfinStore();
   422→
   423→  const [activeTab, setActiveTab] = useState<JellyfinTab>('recent');
   424→  const [viewMode, setViewMode] = useState<ViewMode>('grid');
   425→  const [searchQuery, setSearchQuery] = useState('');
   426→
   427→  const isConnected = store.connectionStatus === 'connected';
   428→  const { isLoadingAlbums, isLoadingArtists, isLoadingTracks, isLoadingRecent } = store;
   429→
   430→  // ── Data Fetching ──
   431→  useEffect(() => {
   432→    if (!isConnected) return;
   433→    switch (activeTab) {
   434→      case 'albums': store.fetchAlbums(true); break;
   435→      case 'artists': store.fetchArtists(true); break;
   436→      case 'tracks': store.fetchAlbums(true); break; // tracks shown within albums
   437→      case 'playlists': store.fetchPlaylists(); break;
   438→      case 'recent': store.fetchRecentAlbums(); break;
   439→    }
   440→  }, [activeTab, isConnected]);
   441→
   442→  // ── Player Actions ──
   443→  const handlePlayAlbum = useCallback(async (album: JellyfinAlbum) => {
   444→    if (!isConnected) return;
   445→    await store.fetchAlbumTracks(album.id);
   446→    const tracks = useJellyfinStore.getState().currentAlbumTracks;
   447→    if (tracks.length === 0) return;
   448→    const dspTracks = tracks.map(t => store.convertToTrack(t));
   449→    setQueue(dspTracks, 0);
   450→  }, [isConnected, setQueue]);
   451→
   452→  const handlePlayTrack = useCallback((track: JellyfinTrack) => {
   453→    const dspTrack = store.convertToTrack(track);
   454→    play(dspTrack);
   455→  }, [play]);
   456→
   457→  const handleAlbumClick = useCallback((album: JellyfinAlbum) => {
   458→    navigate('jellyfin-album', { albumId: album.id });
   459→  }, [navigate]);
   460→
   461→  const handleArtistClick = useCallback((artist: JellyfinArtist) => {
   462→    navigate('jellyfin-artist', { artistId: artist.id });
   463→  }, [navigate]);
   464→
   465→  const handlePlaylistClick = useCallback((playlist: { id: string }) => {
   466→    navigate('jellyfin-playlist', { playlistId: playlist.id });
   467→  }, [navigate]);
   468→
   469→  const handleLoadMore = useCallback(() => {
   470→    if (activeTab === 'albums') store.loadMoreAlbums();
   471→    else if (activeTab === 'artists') store.loadMoreArtists();
   472→  }, [activeTab]);
   473→
   474→  // ── Filter ──
   475→  const filteredAlbums = useMemo(() => {
   476→    if (!searchQuery) return store.albums;
   477→    const q = searchQuery.toLowerCase();
   478→    return store.albums.filter(a =>
   479→      a.name.toLowerCase().includes(q) ||
   480→      a.artistName.toLowerCase().includes(q) ||
   481→      (a.genre && a.genre.toLowerCase().includes(q))
   482→    );
   483→  }, [store.albums, searchQuery]);
   484→
   485→  const filteredArtists = useMemo(() => {
   486→    if (!searchQuery) return store.artists;
   487→    const q = searchQuery.toLowerCase();
   488→    return store.artists.filter(a => a.name.toLowerCase().includes(q));
   489→  }, [store.artists, searchQuery]);
   490→
   491→  // ── Current track check ──
   492→  const isTrackPlaying = useCallback((track: JellyfinTrack) => {
   493→    return currentTrack?.id === `jf-${track.id}`;
   494→  }, [currentTrack]);
   495→
   496→  // ── Render Tab Content ──
   497→  const renderContent = () => {
   498→    const loading = isLoadingAlbums || isLoadingArtists || isLoadingTracks || isLoadingRecent;
   499→
   500→    if (loading && store.albums.length === 0 && store.artists.length === 0 && store.recentAlbums.length === 0) {
   501→      return (
   502→        <div className="flex items-center justify-center py-20">
   503→          <Loader2 className="w-8 h-8 text-primary animate-spin" />
   504→          <span className="ml-3 text-sm text-muted-foreground">Loading library...</span>
   505→        </div>
   506→      );
   507→    }
   508→
   509→    switch (activeTab) {
   510→      case 'recent':
   511→        return (
   512→          <>
   513→            {store.recentAlbums.length === 0 ? (
   514→              <EmptyState icon={Clock} title="No recent additions" description="Recently added albums from your Jellyfin library will appear here" />
   515→            ) : (
   516→              <AlbumsGrid albums={store.recentAlbums} onAlbumClick={handleAlbumClick} onAlbumPlay={handlePlayAlbum} />
   517→            )}
   518→          </>
   519→        );
   520→
   521→      case 'albums':
   522→        return (
   523→          <>
   524→            {filteredAlbums.length === 0 ? (
   525→              <EmptyState icon={Disc3} title="No albums" description="Browse your Jellyfin music library by album" />
   526→            ) : (
   527→              <AlbumsGrid albums={filteredAlbums} onAlbumClick={handleAlbumClick} onAlbumPlay={handlePlayAlbum} />
   528→            )}
   529→            {store.albums.length > filteredAlbums.length && (
   530→              <div className="text-center mt-6">
   531→                <Button variant="outline" size="sm" onClick={store.loadMoreAlbums} disabled={isLoadingAlbums}>
   532→                  {isLoadingAlbums ? <Loader2 className="w-4 h-4 animate-spin mr-2" /> : null}
   533→                  Load More Albums
   534→                </Button>
   535→              </div>
   536→            )}
   537→          </>
   538→        );
   539→
   540→      case 'artists':
   541→        return (
   542→          <>
   543→            {filteredArtists.length === 0 ? (
   544→              <EmptyState icon={User} title="No artists" description="Album artists from your Jellyfin library will appear here" />
   545→            ) : (
   546→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-1">
   547→                {filteredArtists.map(artist => (
   548→                  <ArtistRow key={artist.id} artist={artist} onClick={() => handleArtistClick(artist)} />
   549→                ))}
   550→              </div>
   551→            )}
   552→            {store.artists.length > filteredArtists.length && (
   553→              <div className="text-center mt-6">
   554→                <Button variant="outline" size="sm" onClick={store.loadMoreArtists} disabled={isLoadingArtists}>
   555→                  {isLoadingArtists ? <Loader2 className="w-4 h-4 animate-spin mr-2" /> : null}
   556→                  Load More Artists
   557→                </Button>
   558→              </div>
   559→            )}
   560→          </>
   561→        );
   562→
   563→      case 'tracks':
   564→        return (
   565→          <>
   566→            {filteredAlbums.length === 0 ? (
   567→              <EmptyState icon={Music} title="No tracks" description="Browse your Jellyfin tracks by album" />
   568→            ) : (
   569→              <div className="space-y-6">
   570→                <p className="text-xs text-muted-foreground -mt-2">Tracks are organized by album. Click an album to view its tracks.</p>
   571→                <AlbumsGrid albums={filteredAlbums} onAlbumClick={handleAlbumClick} onAlbumPlay={handlePlayAlbum} />
   572→              </div>
   573→            )}
   574→          </>
   575→        );
   576→
   577→      case 'playlists':
   578→        return (
   579→          <>
   580→            {store.playlists.length === 0 ? (
   581→              <EmptyState icon={ListMusic} title="No playlists" description="Playlists from your Jellyfin library will appear here" />
   582→            ) : (
   583→              <div className="space-y-1">
   584→                {store.playlists.map(pl => (
   585→                  <PlaylistRow key={pl.id} playlist={pl} onClick={() => handlePlaylistClick(pl)} />
   586→                ))}
   587→              </div>
   588→            )}
   589→          </>
   590→        );
   591→    }
   592→  };
   593→
   594→  return (
   595→    <ScrollArea className="h-full">
   596→      <div className="p-6 max-w-7xl mx-auto">
   597→        {/* ── Header ── */}
   598→        <div className="flex items-center justify-between mb-6">
   599→          <div>
   600→            <h1 className="text-2xl font-bold flex items-center gap-2">
   601→              <Server className="w-6 h-6 text-purple-400" />
   602→              Jellyfin
   603→            </h1>
   604→            <p className="text-sm text-muted-foreground mt-1">
   605→              {isConnected
   606→                ? `Browsing ${store.config?.serverName || 'Jellyfin'} library`
   607→                : 'Connect to your Jellyfin media server'}
   608→            </p>
   609→          </div>
   610→          {isConnected && (
   611→            <div className="flex items-center gap-2">
   612→              <Button
   613→                variant={viewMode === 'grid' ? 'default' : 'outline'}
   614→                size="sm"
   615→                className="h-7 text-xs"
   616→                onClick={() => setViewMode('grid')}
   617→              >
   618→                <Grid3X3 className="w-3.5 h-3.5" />
   619→              </Button>
   620→              <Button
   621→                variant={viewMode === 'list' ? 'default' : 'outline'}
   622→                size="sm"
   623→                className="h-7 text-xs"
   624→                onClick={() => setViewMode('list')}
   625→              >
   626→                <List className="w-3.5 h-3.5" />
   627→              </Button>
   628→            </div>
   629→          )}
   630→        </div>
   631→
   632→        {/* ── Connection Panel ── */}
   633→        <ConnectionPanel />
   634→
   635→        {/* ── Library Content (only when connected) ── */}
   636→        {isConnected && (
   637→          <>
   638→            {/* Search */}
   639→            <div className="relative mb-4">
   640→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   641→              <Input
   642→                placeholder="Search albums, artists..."
   643→                value={searchQuery}
   644→                onChange={(e) => setSearchQuery(e.target.value)}
   645→                className="pl-9 bg-card border-border max-w-md"
   646→              />
   647→            </div>
   648→
   649→            {/* Tabs */}
   650→            <TabBar active={activeTab} onChange={setActiveTab} />
   651→
   652→            {/* Content */}
   653→            {renderContent()}
   654→          </>
   655→        )}
   656→
   657→        {/* ── Info Card ── */}
   658→        <Separator className="my-8 bg-border" />
   659→        <Card className="bg-card border-border">
   660→          <CardContent className="p-4">
   661→            <h2 className="text-sm font-semibold mb-3 flex items-center gap-2">
   662→              <Radio className="w-4 h-4 text-purple-400" />
   663→              About Jellyfin Integration
   664→            </h2>
   665→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 text-xs text-muted-foreground">
   666→              <div className="flex items-start gap-2 p-2.5 rounded-lg bg-surface/50">
   667→                <Shield className="w-4 h-4 text-green-400 mt-0.5 flex-shrink-0" />
   668→                <div>
   669→                  <p className="font-medium text-foreground text-xs">Open Source &amp; Private</p>
   670→                  <p className="mt-0.5">Jellyfin is the free software media system. No tracking, no ads, no subscriptions. Your library stays on your server.</p>
   671→                </div>
   672→              </div>
   673→              <div className="flex items-start gap-2 p-2.5 rounded-lg bg-surface/50">
   674→                <Gauge className="w-4 h-4 text-blue-400 mt-0.5 flex-shrink-0" />
   675→                <div>
   676→                  <p className="font-medium text-foreground text-xs">Hi-Res Audio Support</p>
   677→                  <p className="mt-0.5">Stream FLAC, WAV, DSD, and other lossless formats from Jellyfin through DSP with full signal path visualization.</p>
   678→                </div>
   679→              </div>
   680→              <div className="flex items-start gap-2 p-2.5 rounded-lg bg-surface/50">
   681→                <Zap className="w-4 h-4 text-amber-400 mt-0.5 flex-shrink-0" />
   682→                <div>
   683→                  <p className="font-medium text-foreground text-xs">Direct Play</p>
   684→                  <p className="mt-0.5">Audio is streamed directly from your server without transcoding, preserving the original quality and format.</p>
   685→                </div>
   686→              </div>
   687→              <div className="flex items-start gap-2 p-2.5 rounded-lg bg-surface/50">
   688→                <Signal className="w-4 h-4 text-purple-400 mt-0.5 flex-shrink-0" />
   689→                <div>
   690→                  <p className="font-medium text-foreground text-xs">DSP Processing</p>
   691→                  <p className="mt-0.5">Jellyfin audio flows through your DSP pipeline — parametric EQ, room correction, upsampling, and all other modules.</p>
   692→                </div>
   693→              </div>
   694→            </div>
   695→          </CardContent>
   696→        </Card>
   697→      </div>
   698→    </ScrollArea>
   699→  );
   700→}
