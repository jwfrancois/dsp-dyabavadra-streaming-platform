    82→function localTrackToTrack(lt: LocalTrack): Track {
    83→  return {
    84→    id: lt.id,
    85→    title: lt.title,
    86→    albumId: lt.album,
    87→    albumName: lt.album,
    88→    artistId: lt.artist,
    89→    artistName: lt.artist,
    90→    trackNumber: lt.trackNumber,
    91→    discNumber: lt.discNumber,
    92→    duration: lt.duration,
    93→    format: lt.format,
    94→    bitDepth: lt.bitDepth,
    95→    sampleRate: lt.sampleRate,
    96→    channels: lt.channels,
    97→    bitrate: lt.bitrate,
    98→    filePath: lt.filePath,
    99→    fileSize: lt.fileSize,
   100→    composers: lt.composer ? [lt.composer] : [],
   101→    performers: [],
   102→    genre: lt.genre,
   103→    loved: false,
   104→    playCount: 0,
   105→    source: 'local',
   106→    isAvailable: true,
   107→    blobUrl: (lt as any).blobUrl as string | undefined,

   111→function playLocalTrack(lt: LocalTrack) {
   112→  const track = localTrackToTrack(lt);

   116→function playLocalTracks(lts: LocalTrack[]) {

   298→function BrowseLocalPanel() {
   299→  const localStore = useLocalLibraryStore();
   300→  const [view, setView] = React.useState<BrowseLocalView>('artists');
   301→  const [searchQuery, setSearchQuery] = React.useState('');
   302→
   303→  // Breadcrumb navigation state
   304→  const [selectedArtist, setSelectedArtist] = React.useState<string | null>(null);
   305→  const [selectedAlbum, setSelectedAlbum] = React.useState<string | null>(null);
   306→
   307→  // Filter only browser-imported tracks (isLocal === true)
   308→  const allLocalTracks = localStore.tracks.filter((t: any) => t.isLocal);
   309→
   310→  // Derive artists / albums only from local tracks
   311→  const localArtists = React.useMemo(() => {
   312→    const set = new Set<string>();
   313→    for (const t of allLocalTracks) {
   314→      if (t.artist) set.add(t.artist);
   315→      if (t.albumArtist && t.albumArtist !== t.artist) set.add(t.albumArtist);
   316→    }
   317→    return [...set].sort();
   318→  }, [allLocalTracks]);
   319→
   320→  const localAlbums = React.useMemo(() => {
   321→    const map = new Map<string, { name: string; artist: string; albumArtist: string; trackCount: number; year: number; coverArt: string | null }>();
   322→    for (const t of allLocalTracks) {
   323→      const key = `${t.albumArtist || t.artist}|||${t.album}`;
   324→      const existing = map.get(key);
   325→      if (existing) {
   326→        existing.trackCount++;
   327→        if (t.year && (!existing.year || t.year > existing.year)) existing.year = t.year;
   328→        if (!existing.coverArt && t.coverArt) existing.coverArt = t.coverArt;
   329→      } else {
   330→        map.set(key, {
   331→          name: t.album,
   332→          artist: t.artist,
   333→          albumArtist: t.albumArtist || t.artist,
   334→          trackCount: 1,
   335→          year: t.year,
   336→          coverArt: t.coverArt,
   337→        });
   338→      }
   339→    }
   340→    return Array.from(map.values());
   341→  }, [allLocalTracks]);
   342→
   343→  const allArtists = localArtists;
   344→  const allAlbums = localAlbums;
   345→  const allTracks = allLocalTracks;
   346→
   347→  // Derived data based on navigation state
   348→  const breadcrumbs: BreadcrumbItem[] = [{ label: 'Artists', onClick: () => { setSelectedArtist(null); setSelectedAlbum(null); setView('artists'); } }];
   349→
   350→  let displayedArtists = allArtists;
   351→  let displayedAlbums = allAlbums;
   352→  let displayedTracks = allTracks;
   353→
   354→  // Local-only search helper
   355→  const searchLocalTracks = React.useCallback((query: string) => {
   356→    const q = query.toLowerCase();
   357→    return allLocalTracks.filter(
   358→      t =>
   359→        t.title.toLowerCase().includes(q) ||
   360→        t.artist.toLowerCase().includes(q) ||
   361→        t.album.toLowerCase().includes(q) ||
   362→        t.genre.toLowerCase().includes(q) ||
   363→        (t.composer && t.composer.toLowerCase().includes(q)) ||
   364→        (t.albumArtist && t.albumArtist.toLowerCase().includes(q))
   365→    );
   366→  }, [allLocalTracks]);
   367→
   368→  if (searchQuery.trim()) {
   369→    displayedTracks = searchLocalTracks(searchQuery.trim());
   370→    displayedAlbums = allAlbums.filter(a =>
   371→      displayedTracks.some(t => t.album === a.name && (t.artist === a.artist || t.albumArtist === a.albumArtist))
   372→    );
   373→    displayedArtists = allArtists.filter(a =>
   374→      displayedTracks.some(t => t.artist === a || t.albumArtist === a)
   375→    );
   376→  } else if (selectedArtist && selectedAlbum) {
   377→    // Artist > Album > Tracks
   378→    breadcrumbs.push(
   379→      { label: selectedArtist, onClick: () => { setSelectedAlbum(null); } },
   380→      { label: selectedAlbum, onClick: () => {} },
   381→    );
   382→    displayedTracks = allLocalTracks.filter(t => t.album === selectedAlbum && (t.artist === selectedArtist || t.albumArtist === selectedArtist));
   383→  } else if (selectedArtist) {
   384→    // Artist > Albums
   385→    breadcrumbs.push({ label: selectedArtist, onClick: () => {} });
   386→    displayedAlbums = allAlbums.filter(a => a.artist === selectedArtist || a.albumArtist === selectedArtist);
   387→    displayedTracks = allLocalTracks.filter(t => t.artist === selectedArtist || t.albumArtist === selectedArtist);
   388→  } else if (selectedAlbum) {
   389→    // Albums > Tracks
   390→    breadcrumbs.push({ label: selectedAlbum, onClick: () => {} });
   391→    displayedTracks = allLocalTracks.filter(t => t.album === selectedAlbum);
   392→  }
   393→
   394→  const showArtistsView = !selectedArtist && !selectedAlbum && view === 'artists' && !searchQuery.trim();
   395→  const showAlbumsView = (selectedArtist || view === 'albums') && !selectedAlbum && !searchQuery.trim();
   396→  const showTracksView = selectedAlbum || view === 'tracks' || searchQuery.trim();
   397→
   398→  return (
   399→    <div className="space-y-4">
   400→      {/* Search and View Switcher */}
   401→      <div className="flex items-center gap-3">
   402→        <div className="relative flex-1">
   403→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   404→          <Input
   405→            placeholder="Search local library..."
   406→            value={searchQuery}
   407→            onChange={e => { setSearchQuery(e.target.value); if (e.target.value) { setSelectedArtist(null); setSelectedAlbum(null); } }}
   408→            className="pl-9 h-9 text-sm"
   409→          />
   410→        </div>
   411→        <div className="flex items-center bg-surface rounded-lg p-0.5">
   412→          {([['artists', User, 'Artists'], ['albums', Disc3, 'Albums'], ['tracks', Music2, 'Tracks']] as const).map(([key, Icon, label]) => (
   413→            <Button
   414→              key={key}
   415→              variant={view === key ? 'default' : 'ghost'}
   416→              size="sm"
   417→              className="h-8 text-xs gap-1.5 rounded-md"
   418→              onClick={() => { setView(key as BrowseLocalView); setSelectedArtist(null); setSelectedAlbum(null); setSearchQuery(''); }}
   419→            >
   420→              <Icon className="w-3.5 h-3.5" /> {label}
   421→            </Button>
   422→          ))}
   423→        </div>
   424→      </div>
   425→
   426→      {/* Breadcrumb */}
   427→      {(selectedArtist || selectedAlbum) && !searchQuery.trim() && (
   428→        <div className="flex items-center gap-1 text-xs">
   429→          {breadcrumbs.map((crumb, i) => (
   430→            <React.Fragment key={i}>
   431→              {i > 0 && <ChevronRight className="w-3 h-3 text-muted-foreground" />}
   432→              <button
   433→                className={`${i === breadcrumbs.length - 1 ? 'text-foreground font-medium' : 'text-muted-foreground hover:text-foreground'} transition-colors`}
   434→                onClick={crumb.onClick}
   435→              >
   436→                {crumb.label}
   437→              </button>
   438→            </React.Fragment>
   439→          ))}
   440→        </div>
   441→      )}
   442→
   443→      {/* Empty State */}
   444→      {allTracks.length === 0 && (
   445→        <Card className="bg-card border-border">
   446→          <CardContent className="p-12 text-center">
   447→            <LibraryIcon className="w-12 h-12 mx-auto mb-4 text-muted-foreground/30" />
   448→            <h3 className="text-sm font-semibold mb-1">No Local Tracks</h3>
   449→            <p className="text-xs text-muted-foreground max-w-md mx-auto">
   450→              Scan your music folders to build your local library. Go to the &quot;Scan Folders&quot; tab to add directories and start scanning.
   451→            </p>
   452→          </CardContent>
   453→        </Card>
   454→      )}
   455→
   456→      {/* Artists Grid */}
   457→      {showArtistsView && allTracks.length > 0 && (
   458→        <div className="space-y-3">
   459→          <div className="flex items-center justify-between">
   460→            <h2 className="text-sm font-semibold">Artists ({displayedArtists.length})</h2>
   461→          </div>
   462→          {displayedArtists.length === 0 ? (
   463→            <p className="text-xs text-muted-foreground py-8 text-center">No artists found.</p>
   464→          ) : (
   465→            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-6 gap-3">
   466→              {displayedArtists.sort().map(artist => {
   467→                const artistAlbums = allAlbums.filter(a => a.artist === artist);
   468→                const artistTrackCount = artistAlbums.reduce((s, a) => s + a.trackCount, 0);
   469→                return (
   470→                  <Card
   471→                    key={artist}
   472→                    className="bg-card border-border hover:border-primary/30 cursor-pointer transition-all group"

   475→                    <CardContent className="p-4 text-center">

   497→            <p className="text-xs text-muted-foreground py-8 text-center">No albums found.</p>

  1479→function NetworkSharesPanel() {
  1480→  const localStore = useLocalLibraryStore();

  1482→  const [uncInput, setUncInput] = React.useState('');

  1485→  const [mounting, setMounting] = React.useState<string | null>(null);
  1486→  const [loading, setLoading] = React.useState(true);
  1487→  const [error, setError] = React.useState<string | null>(null);
  1488→
  1489→  React.useEffect(() => { fetchShares(); }, []);
  1490→
  1491→  const fetchShares = async () => {
  1492→    setLoading(true);
  1493→    try {
  1494→      const res = await fetch('/api/library/mount');
  1495→      const data = await res.json();
  1496→      if (data.success) { setShares(data.shares || []); setError(null); }
  1497→      else { setError(data.error || 'Failed to load network shares'); }
  1498→    } catch { setError('Failed to connect to server'); }
  1499→    setLoading(false);
  1500→  };
  1501→
  1502→  const handleMount = async () => {
  1503→    const unc = uncInput.trim();
  1504→    if (!unc) return;
  1505→    setMounting(unc); setError(null);
  1506→    try {
  1507→      const res = await fetch('/api/library/mount', {
  1508→        method: 'POST', headers: { 'Content-Type': 'application/json' },
  1509→        body: JSON.stringify({ uncPath: unc, username: usernameInput.trim() || undefined, password: passwordInput || undefined }),
  1510→      });
  1511→      const data = await res.json();
  1512→      if (data.success && data.share) {
  1513→        if (data.scanPath) { localStore.addDirectory(data.scanPath); localStore.startScan(data.scanPath); }
  1514→        setUncInput(''); setUsernameInput(''); setPasswordInput('');
  1515→        await fetchShares();
  1516→      } else { setError(data.error || 'Mount failed'); }
  1517→    } catch { setError('Failed to mount share'); }
  1518→    setMounting(null);
  1519→  };
  1520→
  1521→  const handleUnmount = async (shareId: string) => {
  1522→    try { await fetch(`/api/library/mount?id=${encodeURIComponent(shareId)}`, { method: 'DELETE' }); await fetchShares(); }
  1523→    catch { setError('Failed to unmount'); }
  1524→  };
  1525→
  1526→  const handleRemount = async (share: NetworkShareInfo) => {
  1527→    setMounting(share.id); setError(null);
  1528→    try {
  1529→      const res = await fetch('/api/library/mount', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ uncPath: share.uncPath }) });
  1530→      const data = await res.json();
  1531→      if (data.success) { if (data.scanPath) localStore.addDirectory(data.scanPath); await fetchShares(); }
  1532→      else { setError(data.error || 'Remount failed'); }
  1533→    } catch { setError('Failed to remount'); }
  1534→    setMounting(null);
  1535→  };
  1536→
  1537→  const handleScanMounted = (share: NetworkShareInfo) => {
  1538→    const scanPath = share.mountPoint + (share.subPath ? `/${share.subPath}` : '');
  1539→    localStore.addDirectory(scanPath); localStore.startScan(scanPath);
  1540→  };

  1551→            </div>
  1552→          </div>
  1553→        </CardContent>
  1554→      </Card>
  1555→
  1556→      <Card className="bg-card border-border">
  1557→        <CardHeader className="pb-3"><CardTitle className="text-sm flex items-center gap-2"><Globe className="w-4 h-4 text-primary" /> Add Network Share</CardTitle></CardHeader>

  1561→            <Input placeholder="\\10.0.0.80\iguey\Media\Music" value={uncInput} onChange={e => setUncInput(e.target.value)} className="h-9 text-sm font-mono" disabled={!!mounting} />
