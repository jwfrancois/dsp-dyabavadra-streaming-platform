   109→/** Map a Jellyfin Container string to a human-readable format name */
   110→function getFormatName(container?: string): string {
   111→  if (!container) return '';
   112→  const lower = container.toLowerCase();
   113→  const map: Record<string, string> = {
   114→    flac: 'FLAC',
   115→    mp3: 'MP3',
   116→    wav: 'WAV',
   117→    m4a: 'AAC',
   118→    aac: 'AAC',
   119→    ogg: 'OGG',
   120→    opus: 'OPUS',
   121→    wma: 'WMA',
   122→    aiff: 'AIFF',
   123→    dsd: 'DSD',
   124→    dsf: 'DSD',
   125→   dff: 'DSD',
   126→  };
   127→  return map[lower] || upperFirst(lower);
   128→}
   129→
   130→function upperFirst(s: string): string {
   131→  return s.charAt(0).toUpperCase() + s.slice(1);
   132→}

   134→/** Safely get an image URL from the Jellyfin client, returning an empty string on failure */
   135→function safeGetImageUrl(
   136→  itemId: string,
   137→  maxWidth?: number,
   138→  maxHeight?: number
   139→): string {
   140→  try {
   141→    return jellyfinClient.getImageUrl(itemId, 'Primary', maxWidth, maxHeight);
   142→  } catch {
   143→    return '';
   144→  }
   145→}

   147→/** Map a JellyfinItem (MusicArtist) → JellyfinArtist */
   148→function mapArtist(item: JellyfinItem): JellyfinArtist {
   149→  return {
   150→    id: item.Id,
   151→    name: item.Name,
   152→    imageUrl: safeGetImageUrl(item.Id, 300, 300),
   153→    albumCount: item.ChildCount ?? 0,
   154→  };
   155→}
   156→
   157→/** Map a JellyfinItem (MusicAlbum) → JellyfinAlbum */
   158→function mapAlbum(item: JellyfinItem): JellyfinAlbum {
   159→  const artistObj = item.AlbumArtists?.[0];
   160→  return {
   161→    id: item.Id,
   162→    name: item.Name,
   163→    artistName: artistObj?.Name ?? item.AlbumArtist ?? 'Unknown Artist',
   164→    artistId: artistObj?.Id ?? '',
   165→    year: item.ProductionYear ?? 0,
   166→    genre: item.Genres?.[0] ?? '',
   167→    imageUrl: safeGetImageUrl(item.Id, 300, 300),
   168→    duration: ticksToSeconds(item.RunTimeTicks),
   169→    trackCount: item.ChildCount ?? 0,
   170→    communityRating: item.CommunityRating ?? 0,
   171→    isFavorite: item.UserData?.IsFavorite ?? false,
   172→    overview: item.Overview,
   173→  };
   174→}
   175→
   176→/** Map a JellyfinItem (Audio) → JellyfinTrack */
   177→function mapTrack(item: JellyfinItem): JellyfinTrack {
   178→  const artistObj = item.AlbumArtists?.[0];
   179→  const albumArtist = artistObj?.Name ?? item.Artists?.[0] ?? 'Unknown Artist';
   180→  const albumArtistId = artistObj?.Id ?? '';
   181→
   182→  return {
   183→    id: item.Id,
   184→    name: item.Name,
   185→    albumId: item.AlbumId ?? item.ParentId ?? '',
   186→    albumName: '', // populated from parent context when available
   187→    artistName: albumArtist,
   188→    artistId: albumArtistId,
   189→    trackNumber: 0, // Jellyfin SortName is typically "disc-track name"; use index from list
   190→    duration: ticksToSeconds(item.RunTimeTicks),
   191→    format: getFormatName(item.Container),
   192→    bitrate: item.BitRate ?? 0,
   193→    sampleRate: item.SampleRate ?? 0,
   194→    bitDepth: item.BitsPerSample ?? 0,
   195→    channels: item.Channels ?? 0,
   196→    imageUrl: safeGetImageUrl(item.AlbumId ?? item.ParentId ?? item.Id, 300, 300),
   197→    isFavorite: item.UserData?.IsFavorite ?? false,
   198→    playCount: item.UserData?.PlayCount ?? 0,
   199→  };
   200→}

   217→/** Build a static Jellyfin stream URL for a given item ID (synchronous) */
   218→function buildStreamUrl(itemId: string, config: JellyfinConfig): string {
   219→  const baseUrl = config.serverUrl.replace(/\/+$/, '');
   220→  return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${config.accessToken}&UserId=${config.userId}`;
   221→}

   252→/** Fetch the total track count and update the store (fire-and-forget helper) */
   253→async function refreshTotalTrackCount(): Promise<void> {
   254→  try {
   255→    const state = useJellyfinStore.getState();
   256→    if (state.connectionStatus !== 'connected') return;
   257→    const response = await jellyfinClient.getItems({
   258→      includeItemTypes: ['Audio'],
   259→      recursive: true,
   260→      limit: 1,
   261→    });
   262→    useJellyfinStore.setState({ totalTracks: response.TotalRecordCount });
   263→  } catch {
   264→    // Non-critical; ignore
   265→  }
   266→}

   343→export const useJellyfinStore = create<JellyfinState>((set, get) => ({
   344→  // ── Initial State ──
   345→
   346→  connectionStatus:
   347→    typeof window !== 'undefined' && jellyfinClient.isAuthenticated()
   348→      ? 'connected'
   349→      : 'disconnected',
   350→  config: typeof window !== 'undefined' ? jellyfinClient.getConfig() : null,
   351→  error: null,
   352→
   353→  // Browse cache
   354→  artists: [],
   355→  albums: [],
   356→  recentAlbums: [],
   357→  playlists: [],
   358→  podcastShows: [],
   359→  podcastEpisodes: [],
   360→  totalArtists: 0,
   361→  totalAlbums: 0,
   362→  totalTracks: 0,
   363→  totalPodcastShows: 0,
   364→
   365→  // Detail cache
   366→  currentArtistAlbums: [],
   367→  currentArtistTracks: [],
   368→  currentAlbumTracks: [],
   369→  currentPlaylistTracks: [],
   370→
   371→  // Loading states
   372→  isLoadingArtists: false,
   373→  isLoadingAlbums: false,
   374→  isLoadingTracks: false,
   375→  isLoadingRecent: false,
   376→  isLoadingPodcasts: false,
   377→  isLoadingAllAlbums: false,
   378→  isLoadingAllArtists: false,
   379→
   380→  // Pagination
   381→  artistsPage: 0,
   382→  albumsPage: 0,
   383→
   384→  // ═══════════════════════════════════════════════════════
   385→  // Connection
   386→  // ═══════════════════════════════════════════════════════
   387→
   388→  connect: async (serverUrl, username, password) => {
   389→    set({ connectionStatus: 'connecting', error: null });
   390→    try {
   391→      const config = await jellyfinClient.connect(serverUrl, username, password);
   392→      set({
   393→        connectionStatus: 'connected',
   394→        config,
   395→        error: null,
   396→      });
   397→      // Auto-fetch recent albums after successful connection
   398→      get().fetchRecentAlbums();
   399→      return true;
   400→    } catch (err) {
   401→      let message: string;
   402→      if (err instanceof JellyfinApiError) {
   403→        if (err.statusCode === 0) {
   404→          message = 'Could not reach the Jellyfin server. Check the server URL and ensure the server is running and accessible.';
   405→        } else if (err.statusCode === 401) {
   406→          message = 'Authentication failed. Please check your username and password.';
   407→        } else if (err.statusCode === 404) {
   408→          message = 'Jellyfin server found but the API endpoint was not available. Ensure you are running Jellyfin 10.7+.';
   409→        } else {
   410→          message = err.message;
   411→        }
   412→      } else if (err instanceof Error) {
   413→        message = err.message;
   414→      } else {
   415→        message = 'Connection failed';
   416→      }
   417→      set({ connectionStatus: 'error', error: message });
   418→      return false;
   419→    }
   420→  },
   421→
   422→  connectWithApiKey: async (serverUrl, apiKey) => {
   423→    set({ connectionStatus: 'connecting', error: null });
   424→    try {
   425→      const config = await jellyfinClient.connectWithApiKey(serverUrl, apiKey);
   426→      set({
   427→        connectionStatus: 'connected',
   428→        config,
   429→        error: null,
   430→      });
   431→      get().fetchRecentAlbums();
   432→      return true;
   433→    } catch (err) {
   434→      let message: string;
   435→      if (err instanceof JellyfinApiError) {
   436→        if (err.statusCode === 0) {
   437→          message = 'Could not reach the Jellyfin server. Check the server URL and ensure the server is running and accessible.';
   438→        } else if (err.statusCode === 401) {
   439→          message = 'API key authentication failed. Please check your API key.';
   440→        } else {
   441→          message = err.message;
   442→        }
   443→      } else if (err instanceof Error) {
   444→        message = err.message;
   445→      } else {
   446→        message = 'Connection failed';
   447→      }
   448→      set({ connectionStatus: 'error', error: message });
   449→      return false;
   450→    }
   451→  },
   452→
   453→  disconnect: async () => {
   454→    try {
   455→      await jellyfinClient.disconnect();
   456→    } catch {
   457→      // Best-effort disconnect
   458→    }
   459→    set({
   460→      connectionStatus: 'disconnected',
   461→      config: null,
   462→      error: null,
   463→      artists: [],
   464→      albums: [],
   465→      recentAlbums: [],
   466→      playlists: [],
   467→      podcastShows: [],
   468→      podcastEpisodes: [],
   469→      totalArtists: 0,
   470→      totalAlbums: 0,
   471→      totalTracks: 0,
   472→      totalPodcastShows: 0,
   473→      currentArtistAlbums: [],
   474→      currentArtistTracks: [],
   475→      currentAlbumTracks: [],
   476→      currentPlaylistTracks: [],
   477→      isLoadingArtists: false,
   478→      isLoadingAlbums: false,
   479→      isLoadingTracks: false,
   480→      isLoadingRecent: false,
   481→      isLoadingPodcasts: false,
   482→      isLoadingAllAlbums: false,
   483→      isLoadingAllArtists: false,
   484→      artistsPage: 0,
   485→      albumsPage: 0,
   486→    });
   487→  },
   488→
   489→  clearError: () => set({ error: null }),
   490→
   491→  // ═══════════════════════════════════════════════════════
   492→  // Browse
   493→  // ═══════════════════════════════════════════════════════
   494→
   495→  fetchArtists: async (reset = false) => {
   496→    const { connectionStatus, artistsPage } = get();
   497→    if (connectionStatus !== 'connected') return;
   498→
   499→    const page = reset ? 0 : artistsPage;
   500→    set({ isLoadingArtists: true });
   501→
   502→    try {
   503→      const response = await jellyfinClient.getArtists({
   504→        limit: PAGE_SIZE,
   505→        startIndex: page * PAGE_SIZE,
   506→      });
   507→
   508→      const mapped = response.Items.map(mapArtist);
   509→      set({
   510→        artists: reset ? mapped : [...get().artists, ...mapped],
   511→        totalArtists: response.TotalRecordCount,
   512→        artistsPage: page + 1,
   513→        isLoadingArtists: false,
   514→      });
   515→
   516→      // Also fetch total track count in the background
   517→      if (get().totalTracks === 0) {
   518→        refreshTotalTrackCount();
   519→      }
   520→    } catch (err) {
   521→      const message =
   522→        err instanceof JellyfinApiError
   523→          ? err.message
   524→          : err instanceof Error
   525→            ? err.message
   526→            : 'Failed to fetch artists';
   527→      set({ isLoadingArtists: false, error: message });
   528→    }
   529→  },
   530→
   531→  fetchAlbums: async (reset = false) => {
   532→    const { connectionStatus, albumsPage } = get();
   533→    if (connectionStatus !== 'connected') return;
   534→
   535→    const page = reset ? 0 : albumsPage;
   536→    set({ isLoadingAlbums: true });
   537→
   538→    try {
   539→      const response = await jellyfinClient.getAlbums({
   540→        limit: PAGE_SIZE,
   541→        startIndex: page * PAGE_SIZE,
   542→      });
   543→
   544→      const mapped = response.Items.map(mapAlbum);
   545→      set({
   546→        albums: reset ? mapped : [...get().albums, ...mapped],
   547→        totalAlbums: response.TotalRecordCount,
   548→        albumsPage: page + 1,
   549→        isLoadingAlbums: false,
   550→      });
   551→
   552→      // Also fetch total track count in the background
   553→      if (get().totalTracks === 0) {
   554→        refreshTotalTrackCount();
   555→      }
   556→    } catch (err) {
   557→      const message =
   558→        err instanceof JellyfinApiError
   559→          ? err.message
   560→          : err instanceof Error
   561→            ? err.message
   562→            : 'Failed to fetch albums';
   563→      set({ isLoadingAlbums: false, error: message });
   564→    }
   565→  },
   566→
   567→  fetchRecentAlbums: async () => {
   568→    const { connectionStatus } = get();
   569→    if (connectionStatus !== 'connected') return;
   570→
   571→    set({ isLoadingRecent: true });
   572→
   573→    try {
   574→      const response = await jellyfinClient.getRecentItems(20);
   575→      // Recent items may include non-album types; filter to albums only
   576→      const albumItems = response.Items.filter(
   577→        item => item.Type === 'MusicAlbum'
   578→      );
   579→      const mapped = albumItems.map(mapAlbum);
   580→      set({ recentAlbums: mapped, isLoadingRecent: false });
   581→    } catch (err) {
   582→      const message =
   583→        err instanceof JellyfinApiError
   584→          ? err.message
   585→          : err instanceof Error
   586→            ? err.message
   587→            : 'Failed to fetch recent albums';
   588→      set({ isLoadingRecent: false, error: message });
   589→    }
   590→  },
   591→
   592→  fetchPlaylists: async () => {
   593→    const { connectionStatus } = get();
   594→    if (connectionStatus !== 'connected') return;
   595→
   596→    try {
   597→      const response = await jellyfinClient.getPlaylists();
   598→      const mapped = response.Items.map(mapPlaylist);
   599→      set({ playlists: mapped });
   600→    } catch (err) {
   601→      const message =
   602→        err instanceof JellyfinApiError
   603→          ? err.message
   604→          : err instanceof Error
   605→            ? err.message
   606→            : 'Failed to fetch playlists';
   607→      set({ error: message });
   608→    }
   609→  },
   610→
   611→  loadMoreArtists: async () => {
   612→    const { isLoadingArtists, artists, totalArtists } = get();
   613→    if (isLoadingArtists || artists.length >= totalArtists) return;
   614→    get().fetchArtists(false);
   615→  },
   616→
   617→  loadMoreAlbums: async () => {
   618→    const { isLoadingAlbums, albums, totalAlbums } = get();
   619→    if (isLoadingAlbums || albums.length >= totalAlbums) return;
   620→    get().fetchAlbums(false);
   621→  },
   622→
   623→  // ── Bulk Fetch All (auto-pagination) ──
   624→
   625→  fetchAllAlbums: async () => {
   626→    const { connectionStatus } = get();
   627→    if (connectionStatus !== 'connected') return;
   628→
   629→    set({ isLoadingAllAlbums: true });
   630→
   631→    try {
   632→      // First fetch to get total count
   633→      const first = await jellyfinClient.getAlbums({ limit: PAGE_SIZE, startIndex: 0 });
   634→      const allAlbums = [...first.Items.map(mapAlbum)];
   635→      const totalCount = first.TotalRecordCount;
   636→      set({ albums: allAlbums, totalAlbums: totalCount, albumsPage: 1 });
   637→
   638→      // If there are more pages, fetch them concurrently
   639→      if (allAlbums.length < totalCount) {
   640→        const totalPages = Math.ceil(totalCount / PAGE_SIZE);
   641→        const remainingPages: number[] = [];
   642→
   643→        for (let p = 1; p < totalPages; p++) {
   644→          remainingPages.push(p);
   645→        }
   646→
   647→        // Fetch in batches of AUTO_FETCH_CONCURRENCY
   648→        for (let i = 0; i < remainingPages.length; i += AUTO_FETCH_CONCURRENCY) {
   649→          const batch = remainingPages.slice(i, i + AUTO_FETCH_CONCURRENCY);
   650→          const results = await Promise.all(
   651→            batch.map(page =>
   652→              jellyfinClient.getAlbums({ limit: PAGE_SIZE, startIndex: page * PAGE_SIZE })
   653→                .then(r => r.Items.map(mapAlbum))
   654→                .catch(() => [] as JellyfinAlbum[])
   655→            )
   656→          );
   657→
   658→          const current = useJellyfinStore.getState();
   659→          // Check if we got disconnected or tab changed mid-fetch
   660→          if (current.connectionStatus !== 'connected') break;
   661→
   662→          const newAlbums = results.flat();
   663→          set(prev => ({
   664→            albums: [...prev.albums, ...newAlbums],
   665→            albumsPage: Math.max(batch[batch.length - 1] + 1, prev.albumsPage),
   666→          }));
   667→        }
   668→      }
   669→
   670→      // Also fetch total track count in the background
   671→      if (get().totalTracks === 0) {
   672→        refreshTotalTrackCount();
   673→      }
   674→
   675→      set({ isLoadingAllAlbums: false });
   676→    } catch (err) {
   677→      const message =
   678→        err instanceof JellyfinApiError
   679→          ? err.message
   680→          : err instanceof Error
   681→            ? err.message
   682→            : 'Failed to fetch all albums';
   683→      set({ isLoadingAllAlbums: false, error: message });
   684→    }
   685→  },

   948→  search: async (query: string) => {
   949→    const { connectionStatus } = get();
   950→    if (connectionStatus !== 'connected') {
   951→      return { artists: [], albums: [], tracks: [], playlists: [], podcasts: [], podcastEpisodes: [] };
   952→    }
   953→
   954→    try {
   955→      const response = await jellyfinClient.search(query);
   956→
   957→      const artists: JellyfinArtist[] = [];
   958→      const albums: JellyfinAlbum[] = [];
   959→      const tracks: JellyfinTrack[] = [];
   960→      const playlists: JellyfinPlaylist[] = [];
   961→      const podcasts: JellyfinPodcastShow[] = [];
   962→      const podcastEpisodes: JellyfinPodcastEpisode[] = [];
   963→
   964→      for (const item of response.Items) {
   965→        switch (item.Type) {
   966→          case 'MusicArtist':
   967→            artists.push(mapArtist(item));
   968→            break;
   969→          case 'MusicAlbum':
   970→            albums.push(mapAlbum(item));
   971→            break;
   972→          case 'Audio':
   973→            tracks.push(mapTrack(item));
   974→            break;
   975→          case 'Playlist':
   976→            playlists.push(mapPlaylist(item));
   977→            break;
   978→          case 'BoxSet':
   979→            // Jellyfin stores some podcasts as BoxSet type
   980→            if (item.Tags?.some(t => t.toLowerCase().includes('podcast')) ||
   981→                item.Genres?.some(g => g.toLowerCase().includes('podcast'))) {
   982→              podcasts.push(mapPodcastShow(item));
   983→            }
   984→            break;
   985→        }
   986→      }
   987→
   988→      return { artists, albums, tracks, playlists, podcasts, podcastEpisodes };
   989→    } catch (err) {
   990→      const message =
   991→        err instanceof JellyfinApiError
   992→          ? err.message
   993→          : err instanceof Error
   994→            ? err.message
   995→            : 'Search failed';
   996→      set({ error: message });
   997→      return { artists: [], albums: [], tracks: [], playlists: [], podcasts: [], podcastEpisodes: [] };
   998→    }
   999→  },
