    95→// COUNTRY FLAG EMOJIS
    96→// ═══════════════════════════════════════════════════════
    97→
    98→function countryCodeToFlag(code: string): string {
    99→  if (!code || code.length !== 2) return '🌐';
   100→  const c = code.toUpperCase();
   101→  // Convert country code to flag emoji using regional indicator symbols
   102→  const base = 0x1F1E6 - 65; // 'A' = 65, regional indicator for A = U+1F1E6
   103→  const char1 = String.fromCodePoint(base + c.charCodeAt(0));
   104→  const char2 = String.fromCodePoint(base + c.charCodeAt(1));
   105→  return char1 + char2;
   106→}

   108→// ═══════════════════════════════════════════════════════
   109→// SORT OPTIONS
   110→// ═══════════════════════════════════════════════════════
   111→
   112→type SortOption = 'az' | 'za' | 'country' | 'genre';
   113→
   114→function sortStations(stations: RadioStation[], sort: SortOption): RadioStation[] {
   115→  const sorted = [...stations];
   116→  switch (sort) {
   117→    case 'az':
   118→      return sorted.sort((a, b) => a.name.localeCompare(b.name));
   119→    case 'za':
   120→      return sorted.sort((a, b) => b.name.localeCompare(a.name));
   121→    case 'country':
   122→      return sorted.sort((a, b) => a.country.localeCompare(b.country) || a.name.localeCompare(b.name));
   123→    case 'genre':
   124→      return sorted.sort((a, b) => a.genre.localeCompare(b.genre) || a.name.localeCompare(b.name));
   125→    default:
   126→      return sorted;
   127→  }
   128→}

   131→// DEBOUNCED HOOK
   132→// ═══════════════════════════════════════════════════════
   133→
   134→function useDebouncedValue<T>(value: T, delay: number): T {
   135→  const [debounced, setDebounced] = useState(value);
   136→  const timerRef = useRef<ReturnType<typeof setTimeout>>();
   137→
   138→  useEffect(() => {
   139→    if (timerRef.current) clearTimeout(timerRef.current);
   140→    timerRef.current = setTimeout(() => setDebounced(value), delay);
   141→    return () => {
   142→      if (timerRef.current) clearTimeout(timerRef.current);
   143→    };
   144→  }, [value, delay]);
   145→
   146→  return debounced;
   147→}

   542→          <Search className="absolute left-4 top-1/2 -translate-y-1/2 w-5 h-5 text-muted-foreground" />
   543→          <Input
   544→            placeholder="Search stations by name, genre, country, or tags..."

   551→        {/* ═══ GENRE CHIPS ROW ═══ */}

   555→              onClick={() => handleGenreChipClick('all')}

   557→                'flex-shrink-0 px-3 py-1.5 rounded-full text-xs font-medium transition-colors border',

   567→                key={genre}
   568→                onClick={() => handleGenreChipClick(genre)}

   570→                  'flex-shrink-0 px-3 py-1.5 rounded-full text-xs font-medium transition-colors border',
   571→                  genreFilter === genre
   572→                    ? 'bg-primary text-primary-foreground border-primary'

   574→                )}
   575→              >
   576→                {genre}
   577→              </button>

   579→          </div>
   580→          {/* Fade edges */}
   581→          <div className="absolute top-0 right-0 bottom-2 w-12 bg-gradient-to-l from-background to-transparent pointer-events-none" />

   583→
   584→        {/* ═══ FILTER BAR ═══ */}

   587→          <Select value={genreFilter} onValueChange={handleGenreFilterChange}>

   597→          </Select>
   598→
   599→          {/* Country Dropdown */}
   600→          <Select value={countryFilter} onValueChange={handleCountryFilterChange}>

   602→              <SelectValue placeholder="All Countries" />
   603→            </SelectTrigger>

   605→              <SelectItem value="all">All Countries</SelectItem>

   608→                  <span className="flex items-center gap-1.5">

   611→                  </span>

   615→          </Select>

   617→          {/* Source Dropdown */}
   618→          <Select value={sourceFilter} onValueChange={handleSourceFilterChange}>

   621→            </SelectTrigger>

   625→                <SelectItem key={src} value={src}>{src}</SelectItem>

   627→            </SelectContent>
   628→          </Select>
   629→
   630→          {/* Sort Dropdown */}
   631→          <Select value={sortOption} onValueChange={handleSortChange}>
   632→            <SelectTrigger className="w-[150px] h-9">
   633→              <SelectValue />
   634→            </SelectTrigger>
   635→            <SelectContent>
   636→              <SelectItem value="az">
   637→                <span className="flex items-center gap-1.5">
   638→                  <ArrowDownAZ className="w-3 h-3" /> A → Z

   642→                <span className="flex items-center gap-1.5">
   643→                  <ArrowUpAZ className="w-3 h-3" /> Z → A

   645→              </SelectItem>

   647→                <span className="flex items-center gap-1.5">
   648→                  <Globe className="w-3 h-3" /> Country
   649→                </span>
   650→              </SelectItem>
   651→              <SelectItem value="genre">
   652→                <span className="flex items-center gap-1.5">
   653→                  <Filter className="w-3 h-3" /> Genre
   654→                </span>
   655→              </SelectItem>

   665→              setShowFavoritesOnly(prev => !prev);
   666→              setVisibleCount(PAGE_SIZE);
   667→            }}
   668→          >
   669→            <Heart
   670→              className={cn(
   671→                'w-4 h-4',
   672→                showFavoritesOnly && 'fill-current'

   675→            Favorites

   677→        </div>
   678→
   679→        {/* ═══ SOMAFM FEATURED SECTION ═══ */}
   680→        {!showFavoritesOnly && !activeSearch.trim() && genreFilter === 'all' && countryFilter === 'all' && sourceFilter === 'all' && (

   683→              <Waves className="w-4 h-4 text-orange-400" />

   686→                {somaFMStations.length} channels
   687→              </Badge>
   688→            </div>
   689→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">

   691→                <SomaFMCard
   692→                  key={station.id}
   693→                  station={station}
   694→                  isPlaying={playbackMode === 'radio' && currentRadioStationId === station.id && isPlaying}

   698→              ))}
   699→            </div>
   700→            <Separator />
   701→          </div>
   702→        )}
   703→
   704→        {/* ═══ RESULTS COUNT ═══ */}
   705→        <div className="flex items-center justify-between">
   706→          <p className="text-sm text-muted-foreground">
   707→            Showing{' '}
   708→            <span className="text-foreground font-medium">{Math.min(visibleCount, filteredStations.length)}</span>
   709→            {' '}of{' '}
   710→            <span className="text-foreground font-medium">{filteredStations.length}</span>
   711→            {' '}stations
   712→          </p>
   713→          {(activeSearch.trim() || genreFilter !== 'all' || countryFilter !== 'all' || showFavoritesOnly) && (
   714→            <Button
   715→              variant="ghost"
   716→              size="sm"
   717→              className="h-7 text-xs"
   718→              onClick={() => {
   719→                setLocalSearch('');
   720→                setGenreFilter('all');
   721→                setCountryFilter('all');
   722→                setShowFavoritesOnly(false);
   723→                setSortOption('az');
   724→                setVisibleCount(PAGE_SIZE);
   725→              }}
   726→            >
   727→              Clear filters
   728→            </Button>
   729→          )}
   730→        </div>
   731→
   732→        {/* ═══ STATION LIST ═══ */}
   733→        {filteredStations.length === 0 ? (
   734→          <div className="text-center py-16 text-muted-foreground">
   735→            <Radio className="w-12 h-12 mx-auto mb-4 opacity-30" />
   736→            <h3 className="text-lg font-medium mb-1">No stations found</h3>
   737→            <p className="text-sm">Try adjusting your search or filters</p>
   738→          </div>
   739→        ) : (
   740→          <div className="space-y-1">
   741→            {visibleStations.map((station) => (
   742→              <StationListItem
   743→                key={station.id}
   744→                station={station}
   745→                isPlaying={playbackMode === 'radio' && currentRadioStationId === station.id && isPlaying}
   746→                onPlay={() => handlePlayStation(station)}
   747→                onToggleFavorite={() => handleToggleFavorite(station.id)}
   748→              />
   749→            ))}
   750→
   751→            {/* Load More */}
   752→            {hasMore && (
   753→              <div className="flex justify-center pt-4 pb-2">
   754→                <Button
   755→                  variant="outline"
   756→                  onClick={handleLoadMore}
   757→                  className="gap-2"
   758→                >
   759→                  <ChevronDown className="w-4 h-4" />
   760→                  Load More
   761→                  <span className="text-muted-foreground text-xs">
   762→                    ({filteredStations.length - visibleCount} remaining)
   763→                  </span>
   764→                </Button>
   765→              </div>
   766→            )}
   767→          </div>
   768→        )}
   769→      </div>
   770→    </ScrollArea>
   771→  );
