    62→function sfmUrl(channel: string): string {
    63→  return `https://ice1.somafm.com/${SOMAFM[channel] || 'groovesalad-128-mp3'}`;

    65→
    66→function sfm(
    67→  id: number, name: string, desc: string, genre: string,
    68→  country: string, cc: string, channel: string,
    69→  tags: string[], lang = 'English',
    70→): RadioStation {
    71→  return {
    72→    id: `radio-${id}`, name, description: desc, genre,
    73→    country, countryCode: cc, streamUrl: sfmUrl(channel),
    74→    codec: 'MP3', bitrate: 128, sampleRate: 44100,
    75→    isFavorite: false, tags, source: 'SomaFM', website: 'https://somafm.com', language: lang,
    76→  };
    77→}
    78→
    79→function station(
    80→  id: number, name: string, desc: string, genre: string,
    81→  country: string, cc: string, url: string,
    82→  tags: string[], source: RadioSource = 'Community',
    83→  lang = 'English', website?: string, bitrate = 128,
    84→): RadioStation {
    85→  return {
    86→    id: `radio-${id}`, name, description: desc, genre,
    87→    country, countryCode: cc, streamUrl: url,
    88→    codec: bitrate >= 192 ? 'AAC' : 'MP3', bitrate, sampleRate: 44100,
    89→    isFavorite: false, tags, source, website, language: lang,
    90→  };
    91→}

   535→export function searchRadioStations(query: string): RadioStation[] {
   536→  const q = query.toLowerCase().trim();
   537→  if (!q) return _stations;
   538→  return _stations.filter(s =>
   539→    s.name.toLowerCase().includes(q) ||
   540→    s.description.toLowerCase().includes(q) ||
   541→    s.genre.toLowerCase().includes(q) ||
   542→    s.country.toLowerCase().includes(q) ||
   543→    s.tags.some(t => t.includes(q)),
   544→  );
   545→}
   546→
   547→export function getRadioStationsByGenre(genre: string): RadioStation[] {
   548→  return _stations.filter(s =>
   549→    s.genre.toLowerCase() === genre.toLowerCase() ||
   550→    s.tags.includes(genre.toLowerCase()),
   551→  );
   552→}
   553→
   554→export function getRadioStationsByCountry(country: string): RadioStation[] {
   555→  const c = country.toLowerCase();
   556→  return _stations.filter(s =>
   557→    s.country.toLowerCase() === c || s.countryCode.toLowerCase() === c,
   558→  );
   559→}

   561→export function toggleRadioFavorite(stationId: string): void {
   562→  if (_favorites.has(stationId)) {
   563→    _favorites.delete(stationId);
   564→  } else {
   565→    _favorites.add(stationId);
   566→  }
   567→}

   569→export function getRadioFavorites(): RadioStation[] {
   570→  return _stations.filter(s => _favorites.has(s.id));
   571→}

   573→export function getAllGenres(): string[] {
   574→  const genres = new Set(_stations.map(s => s.genre));
   575→  return Array.from(genres).sort();
   576→}

   578→export function getAllSources(): RadioSource[] {
   579→  const sources = new Set<RadioSource>(_stations.map(s => s.source));
   580→  return Array.from(sources).sort();
   581→}
   582→
   583→export interface RadioCountry {
   584→  code: string;
   585→  name: string;
   586→}
   587→
   588→export function getAllCountries(): RadioCountry[] {
   589→  const seen = new Map<string, string>();
   590→  for (const s of _stations) {
   591→    if (!seen.has(s.countryCode)) {
   592→      seen.set(s.countryCode, s.country);
   593→    }
   594→  }
   595→  return Array.from(seen.entries())
   596→    .sort((a, b) => a[0].localeCompare(b[0]))
   597→    .map(([code, name]) => ({ code, name }));
   598→}
   599→
   600→export function getRadioStationsBySource(source: RadioSource): RadioStation[] {
   601→  return _stations.filter(s => s.source === source);
   602→}
