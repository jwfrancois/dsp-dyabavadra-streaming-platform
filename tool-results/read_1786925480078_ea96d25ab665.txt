    44→// ── Audio quality badge color logic ──
    45→function getQualityColor(format: string, bitDepth: number, sampleRate: number): { label: string; className: string } {
    46→  const fmt = format.toUpperCase();
    47→  const rateKHz = sampleRate / 1000;
    48→  if (fmt === 'DSD') return { label: 'DSD', className: 'bg-purple-500/15 text-purple-400 border-purple-500/30' };
    49→  if (fmt === 'FLAC' && bitDepth >= 24 && rateKHz >= 88.2) return { label: 'Hi-Res', className: 'bg-amber-500/15 text-amber-400 border-amber-500/30' };
    50→  if (fmt === 'FLAC' && bitDepth >= 24) return { label: 'Hi-Res', className: 'bg-amber-500/15 text-amber-400 border-amber-500/30' };
    51→  if (fmt === 'FLAC' && rateKHz >= 44.1) return { label: 'CD Quality', className: 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30' };
    52→  if (['ALAC', 'WAV', 'AIFF', 'APE'].includes(fmt)) return { label: 'Lossless', className: 'bg-sky-500/15 text-sky-400 border-sky-500/30' };
    53→  return { label: 'Lossy', className: 'bg-zinc-500/15 text-zinc-400 border-zinc-500/30' };
    54→}
