    27→    27→// ── Quality badge helper ──
    28→    28→function QualityBadge({ format, sampleRate, bitDepth }: { format: string; sampleRate: number; bitDepth: number }) {
    29→    29→  const isHiRes = sampleRate > 48000 || bitDepth > 16;
    30→    30→  const isLossless = ['FLAC', 'WAV', 'AIFF', 'ALAC', 'DSF', 'DFF', 'WavPack', 'APE', 'TAK'].includes(format.toUpperCase());
    31→    31→  const isDSD = ['DSF', 'DFF', 'DSD'].includes(format.toUpperCase());
    32→    32→
    33→    33→  if (isDSD) return <Badge className="text-[9px] bg-purple-500/20 text-purple-400 border-purple-500/30 h-4 px-1">DSD</Badge>;
    34→    34→  if (isHiRes && isLossless) return <Badge className="text-[9px] bg-primary/20 text-primary border-primary/30 h-4 px-1">Hi-Res</Badge>;
    35→    35→  if (isLossless) return <Badge className="text-[9px] bg-signal-green/20 text-signal-green border-signal-green/30 h-4 px-1">Lossless</Badge>;
    36→    36→  return null;
    37→    37→}

    40→    40→function getGreeting(): { text: string; icon: React.ReactNode; emoji: string } {
    41→    41→  const hour = new Date().getHours();
    42→    42→  if (hour < 6) return { text: 'Night Owl Mode', icon: <Moon className="w-5 h-5" />, emoji: '' };
    43→    43→  if (hour < 12) return { text: 'Good Morning', icon: <Sunrise className="w-5 h-5" />, emoji: '' };
    44→    44→  if (hour < 14) return { text: 'Good Afternoon', icon: <Sun className="w-5 h-5" />, emoji: '' };
    45→    45→  if (hour < 18) return { text: 'Good Afternoon', icon: <Sun className="w-5 h-5" />, emoji: '' };
    46→    46→  if (hour < 22) return { text: 'Good Evening', icon: <Moon className="w-5 h-5" />, emoji: '' };
    47→    47→  return { text: 'Night Owl Mode', icon: <Moon className="w-5 h-5" />, emoji: '' };

    63→    63→// ── Time ago helper ──
    64→    64→function timeAgo(isoString: string): string {
    65→    65→  const diffMin = Math.floor((Date.now() - new Date(isoString).getTime()) / 60000);
    66→    66→  if (diffMin < 1) return 'Just now';
    67→    67→  if (diffMin < 60) return `${diffMin}m ago`;
    68→    68→  const diffHr = Math.floor(diffMin / 60);
    69→    69→  if (diffHr < 24) return `${diffHr}h ago`;
    70→    70→  return `${Math.floor(diffHr / 24)}d ago`;
    71→    71→}

   210→   210→function InteractiveCalendar({ className }: { className?: string }) {
   211→   211→  const today = new Date();
   212→   212→  const [viewDate, setViewDate] = useState(() => new Date());
   213→   213→  const [selectedDate, setSelectedDate] = useState<Date | null>(null);
   214→   214→
   215→   215→  const year = viewDate.getFullYear();
   216→   216→  const month = viewDate.getMonth();
   217→   217→
   218→   218→  const monthName = viewDate.toLocaleDateString('en-US', { month: 'long', year: 'numeric' });
   219→   219→
   220→   220→  const firstDayOfMonth = new Date(year, month, 1);
   221→   221→  const lastDayOfMonth = new Date(year, month + 1, 0);
   222→   222→  const daysInMonth = lastDayOfMonth.getDate();
   223→   223→  const startingDay = firstDayOfMonth.getDay();
   224→   224→
   225→   225→  const days = useMemo(() => {
   226→   226→    const result: (number | null)[] = [];
   227→   227→    for (let i = 0; i < startingDay; i++) result.push(null);
   228→   228→    for (let d = 1; d <= daysInMonth; d++) result.push(d);
   229→   229→    return result;
   230→   230→  }, [startingDay, daysInMonth]);
   231→   231→
   232→   232→  const prevMonth = useCallback(() => {
   233→   233→    setViewDate(d => new Date(d.getFullYear(), d.getMonth() - 1, 1));
   234→   234→  }, []);
   235→   235→
   236→   236→  const nextMonth = useCallback(() => {
   237→   237→    setViewDate(d => new Date(d.getFullYear(), d.getMonth() + 1, 1));
   238→   238→  }, []);
   239→   239→
   240→   240→  const isToday = (day: number) =>
   241→   241→    day === today.getDate() && month === today.getMonth() && year === today.getFullYear();
   242→   242→
   243→   243→  const isSelected = (day: number) =>
   244→   244→    selectedDate &&
   245→   245→    day === selectedDate.getDate() &&
   246→   246→    month === selectedDate.getMonth() &&
   247→   247→    year === selectedDate.getFullYear();
   248→   248→
   249→   249→  const weekDays = ['Su', 'Mo', 'Tu', 'We', 'Th', 'Fr', 'Sa'];
   250→   250→
   251→   251→  return (
   252→   252→    <div className={cn('rounded-xl border border-border bg-card overflow-hidden', className)}>
   253→   253→      <div className="flex items-center justify-between px-4 py-3 bg-gradient-to-r from-primary/5 to-accent/5 border-b border-border/50">
   254→   254→        <button onClick={prevMonth} className="p-1 rounded-md hover:bg-accent/50 transition-colors text-muted-foreground hover:text-foreground">
   255→   255→          <ChevronLeft className="w-4 h-4" />
   256→   256→        </button>
   257→   257→        <h3 className="text-sm font-semibold tracking-wide">{monthName}</h3>
   258→   258→        <button onClick={nextMonth} className="p-1 rounded-md hover:bg-accent/50 transition-colors text-muted-foreground hover:text-foreground">
   259→   259→          <ChevronRight className="w-4 h-4" />
   260→   260→        </button>
   261→   261→      </div>
   262→   262→
   263→   263→      <div className="grid grid-cols-7 border-b border-border/30">
   264→   264→        {weekDays.map(d => (
   265→   265→          <div key={d} className="py-2 text-center text-[10px] font-medium text-muted-foreground uppercase tracking-wider">
   266→   266→            {d}
   267→   267→          </div>
   268→   268→        ))}
   269→   269→      </div>
   270→   270→
   271→   271→      <div className="grid grid-cols-7">
   272→   272→        {days.map((day, idx) => {
   273→   273→          if (day === null) return <div key={`empty-${idx}`} className="h-9" />;
   274→   274→          const todayFlag = isToday(day);
   275→   275→          const selectedFlag = isSelected(day);
   276→   276→          return (
   277→   277→            <button key={day}
   278→   278→              onClick={() => setSelectedDate(new Date(year, month, day))}
   279→   279→              className={cn(
   280→   280→                'h-9 flex items-center justify-center text-xs font-medium transition-all duration-200 relative',
   281→   281→                'hover:bg-accent/40 rounded-md m-0.5',
   282→   282→                todayFlag && !selectedFlag && 'text-primary font-bold',
   283→   283→                selectedFlag && 'bg-primary text-primary-foreground rounded-md shadow-sm shadow-primary/20',
   284→   284→                !todayFlag && !selectedFlag && 'text-foreground/70'
   285→   285→              )}
   286→   286→            >
   287→   287→              {day}
   288→   288→              {todayFlag && !selectedFlag && (
   289→   289→                <span className="absolute bottom-0.5 left-1/2 -translate-x-1/2 w-1 h-1 rounded-full bg-primary" />
   290→   290→              )}
   291→   291→            </button>
   292→   292→          );
   293→   293→        })}
   294→   294→      </div>
   295→   295→
   296→   296→      {selectedDate && (
   297→   297→        <div className="px-4 py-2.5 border-t border-border/30 bg-primary/[0.03]">
   298→   298→          <p className="text-xs text-muted-foreground">
   299→   299→            Selected:{' '}
   300→   300→            <span className="text-foreground font-medium">
   301→   301→              {selectedDate.toLocaleDateString('en-US', { weekday: 'long', month: 'long', day: 'numeric', year: 'numeric' })}
   302→   302→            </span>
   303→   303→          </p>
   304→   304→        </div>
   305→   305→      )}
   306→   306→    </div>
   307→   307→  );
   308→   308→}
