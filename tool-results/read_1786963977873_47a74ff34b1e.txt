   163→   163→/** Convert Jellyfin ticks (100ns units) to seconds */
   164→   164→export function ticksToSeconds(ticks: number | undefined): number {
   165→   165→  if (ticks === undefined || ticks === null) return 0;
   166→   166→  return Math.floor(ticks / TICKS_PER_SECOND);
   167→   167→}

   169→   169→/** Convert seconds to Jellyfin ticks */
   170→   170→export function secondsToTicks(seconds: number): number {
   171→   171→  return Math.floor(seconds * TICKS_PER_SECOND);
   172→   172→}

   179→   179→/** Build a query string from a record, filtering out undefined/null/empty values */
   180→   180→function buildQueryString(params: Record<string, string | number | boolean | undefined | null | string[] | number[]>): string {
   181→   181→  const parts: string[] = [];
   182→   182→  for (const [key, value] of Object.entries(params)) {
   183→   183→    if (value === undefined || value === null || value === '') continue;
   184→   184→    if (Array.isArray(value)) {
   185→   185→      if (value.length > 0) {
   186→   186→        parts.push(`${encodeURIComponent(key)}=${value.map(v => encodeURIComponent(String(v))).join(',')}`);
   187→   187→      }
   188→   188→    } else {
   189→   189→      parts.push(`${encodeURIComponent(key)}=${encodeURIComponent(String(value))}`);
   190→   190→    }
   191→   191→  }
   192→   192→  return parts.length > 0 ? `?${parts.join('&')}` : '';
   193→   193→}

   223→   223→  private saveConfig(): void {
   224→   224→    if (!this.config) return;

   235→   235→    if (!this.storageAvailable || typeof window === 'undefined') return null;

   238→   238→      if (!raw) return null;
   239→   239→      const parsed = JSON.parse(raw) as JellyfinConfig;
   240→   240→      // Basic validation
   241→   241→      if (!parsed.serverUrl || !parsed.accessToken || !parsed.userId) return null;
   242→   242→      return parsed;
   243→   243→    } catch {
   244→   244→      return null;

   252→   252→  private getBaseUrl(): string {

   257→   257→  /**
   258→   258→   * Make an API request through the Next.js server-side proxy.
   259→   259→   * This avoids browser CORS issues when connecting to self-hosted Jellyfin servers.
   260→   260→   * The proxy at /api/jellyfin/[...path] forwards requests server-to-server.
   261→   261→   */
   262→   262→  private async request<T>(path: string, options: RequestInit = {}, skipAuth = false): Promise<T> {
   263→   263→    // Extract injected metadata from headers (used during connect flow)
   264→   264→    const injectedHeaders = (options.headers as Record<string, string> | undefined);
   265→   265→    const injectedBaseUrl = injectedHeaders?.__baseUrl;
   266→   266→    const injectedAuth = injectedHeaders?.__authorization;
   267→   267→
   268→   268→    // Determine the target Jellyfin server URL
   269→   269→    const serverUrl = skipAuth
   270→   270→      ? normalizeServerUrl(injectedBaseUrl || '')
   271→   271→      : this.getBaseUrl();
   272→   272→
   273→   273→    if (!serverUrl) {
   274→   274→      throw new JellyfinApiError(
   275→   275→        'No Jellyfin server URL configured. Please connect first.',
   276→   276→        0,
   277→   277→        path
   278→   278→      );
   279→   279→    }
   280→   280→
   281→   281→    // Build the proxy URL: /api/jellyfin/Users/AuthenticateByName
   282→   282→    const proxyUrl = `/api/jellyfin${path}`;
   283→   283→
   284→   284→    // Build headers for the proxy — the proxy reads these to forward correctly
   285→   285→    const proxyHeaders: Record<string, string> = {
   286→   286→      'Content-Type': 'application/json',
   287→   287→      'Accept': 'application/json',
   288→   288→      'X-Jellyfin-Server-Url': serverUrl,
   289→   289→    };
   290→   290→
   291→   291→    // Tell the proxy to skip forwarding the auth token for login endpoints
   292→   292→    if (skipAuth) {
   293→   293→      proxyHeaders['X-Jellyfin-Skip-Auth'] = 'true';
   294→   294→    } else if (this.config) {
   295→   295→      // Forward the auth token for authenticated requests
   296→   296→      proxyHeaders['X-Jellyfin-Token'] = this.config.accessToken;
   297→   297→    }
   298→   298→
   299→   299→    // Forward the MediaBrowser authorization header if present (for login)
   300→   300→    if (injectedAuth) {
   301→   301→      proxyHeaders['X-Jellyfin-Authorization'] = injectedAuth;
   302→   302→    }
   303→   303→
   304→   304→    // Remove Content-Type for GET/DELETE if no body
   305→   305→    if (!options.body && (options.method === 'GET' || options.method === 'DELETE')) {
   306→   306→      delete proxyHeaders['Content-Type'];
   307→   307→    }
   308→   308→
   309→   309→    let response: Response;
   310→   310→    try {
   311→   311→      response = await fetch(proxyUrl, {
   312→   312→        method: options.method || 'GET',
   313→   313→        headers: proxyHeaders,
   314→   314→        body: options.body || undefined,
   315→   315→      });
   316→   316→    } catch (err) {
   317→   317→      throw new JellyfinApiError(
   318→   318→        `Network error: ${err instanceof Error ? err.message : 'Unknown error'}`,
   319→   319→        0,
   320→   320→        path,
   321→   321→        err
   322→   322→      );
   323→   323→    }
   324→   324→
   325→   325→    if (!response.ok) {
   326→   326→      let errorBody = '';
   327→   327→      try {
   328→   328→        errorBody = await response.text();
   329→   329→      } catch {
   330→   330→        // ignore
   331→   331→      }
   332→   332→      throw new JellyfinApiError(
   333→   333→        `Jellyfin API error ${response.status}: ${response.statusText} — ${errorBody}`,
   334→   334→        response.status,
   335→   335→        path
   336→   336→      );
   337→   337→    }
   338→   338→
   339→   339→    // Some endpoints return empty responses (e.g., DELETE, favorite toggles)
   340→   340→    const contentType = response.headers.get('content-type');
   341→   341→    if (contentType && contentType.includes('application/json')) {
   342→   342→      return response.json() as Promise<T>;
   343→   343→    }
   344→   344→
   345→   345→    return undefined as unknown as T;
   346→   346→  }

   352→   352→  /**
   353→   353→   * Connect to a Jellyfin server. Authenticates and fetches server info.
   354→   354→   * Persists the configuration to localStorage.
   355→   355→   */
   356→   356→  async connect(serverUrl: string, username: string, password: string): Promise<JellyfinConfig> {
   357→   357→    const baseUrl = normalizeServerUrl(serverUrl);
   358→   358→
   359→   359→    // Step 1: Authenticate (skip auth headers — we don't have a token yet)
   360→   360→    const deviceId = typeof window !== 'undefined'
   361→   361→      ? window.navigator.userAgent.slice(0, 32).replace(/[^a-zA-Z0-9]/g, '')
   362→   362→      : 'ssr';
   363→   363→    const mediaBrowserAuth = `MediaBrowser Client="DSP", Device="Browser", DeviceId="dsp-jellyfin-${deviceId}", Version="1.0.0"`;
   364→   364→
   365→   365→    const authResponse = await this.request<JellyfinAuthResponse>(
   366→   366→      '/Users/AuthenticateByName',
   367→   367→      {
   368→   368→        method: 'POST',
   369→   369→        body: JSON.stringify({ Username: username, Pw: password }),
   370→   370→        headers: {
   371→   371→          '__baseUrl': baseUrl,
   372→   372→          '__authorization': mediaBrowserAuth,
   373→   373→        } as Record<string, string>,
   374→   374→      },
   375→   375→      true // skipAuth
   376→   376→    );
   377→   377→
   378→   378→    const token = authResponse.AccessToken;
   379→   379→    const userId = authResponse.User.Id;
   380→   380→    const serverId = authResponse.ServerId;
   381→   381→
   382→   382→    // Step 2: Fetch server info
   383→   383→    let serverName = 'Jellyfin';
   384→   384→    let serverVersion = 'unknown';
   385→   385→    try {
   386→   386→      // Temporarily set config for the getSystemInfo call
   387→   387→      this.config = {
   388→   388→        serverUrl: baseUrl,
   389→   389→        username,
   390→   390→        accessToken: token,
   391→   391→        userId,
   392→   392→        serverId,
   393→   393→        serverName,
   394→   394→        serverVersion,
   395→   395→        connectedAt: new Date().toISOString(),
   396→   396→        lastHeartbeat: new Date().toISOString(),
   397→   397→        musicLibraryId: '',
   398→   398→        podcastLibraryId: '',
   399→   399→      };
   400→   400→      const sysInfo = await this.getSystemInfo();
   401→   401→      serverName = sysInfo.ServerName;
   402→   402→      serverVersion = sysInfo.Version;
   403→   403→    } catch {
   404→   404→      // If system info fails, continue with defaults
   405→   405→    }
   406→   406→
   407→   407→    // Step 3: Auto-detect music and podcast libraries
   408→   408→    let musicLibraryId = '';
   409→   409→    let podcastLibraryId = '';
   410→   410→    try {
   411→   411→      const views = await this.getViews();
   412→   412→      for (const item of views.Items) {
   413→   413→        if (item.Type !== 'CollectionFolder') continue;
   414→   414→        const name = item.Name.toLowerCase();
   415→   415→        if (name.includes('music') && !name.includes('podcast') && !musicLibraryId) {
   416→   416→          musicLibraryId = item.Id;
   417→   417→        }
   418→   418→        if (name.includes('podcast') && !podcastLibraryId) {
   419→   419→          podcastLibraryId = item.Id;
   420→   420→        }
   421→   421→      }
   422→   422→    } catch {
   423→   423→      // Continue without auto-detected library
   424→   424→    }
   425→   425→
   426→   426→    // Step 4: Build and save config
   427→   427→    const config: JellyfinConfig = {
   428→   428→      serverUrl: baseUrl,
   429→   429→      username,
   430→   430→      accessToken: token,
   431→   431→      userId,
   432→   432→      serverId,
   433→   433→      serverName,
   434→   434→      serverVersion,
   435→   435→      connectedAt: new Date().toISOString(),
   436→   436→      lastHeartbeat: new Date().toISOString(),
   437→   437→      musicLibraryId,
   438→   438→      podcastLibraryId,
   439→   439→    };
   440→   440→
   441→   441→    this.config = config;
   442→   442→    this.saveConfig();
   443→   443→
   444→   444→    return config;
   445→   445→  }

   448→   448→   * Disconnect from the Jellyfin server. Revokes the token and clears local state.

   450→   450→  async disconnect(): Promise<void> {
   451→   451→    if (this.config) {
   452→   452→      try {
   453→   453→        await this.request<void>(`/Users/${this.config.userId}/Tokens`, {
   454→   454→          method: 'DELETE',
   455→   455→        });
   456→   456→      } catch {
   457→   457→        // Best-effort — continue clearing local state
   458→   458→      }
   459→   459→    }
   460→   460→
   461→   461→    this.config = null;
   462→   462→    if (this.storageAvailable && typeof window !== 'undefined') {
   463→   463→      try {
   464→   464→        window.localStorage.removeItem(STORAGE_KEY);
   465→   465→      } catch {
   466→   466→        // ignore
   467→   467→      }
   468→   468→    }
   469→   469→  }

   888→   888→  /**
   889→   889→   * Get a playable audio stream URL for an item.
   890→   890→   * Prefers direct play for best quality; falls back to transcoded stream.
   891→   891→   */
   892→   892→  async getPlaybackStreamUrl(
   893→   893→    itemId: string,
   894→   894→    maxBitrate: number = DEFAULT_MAX_BITRATE
   895→   895→  ): Promise<string> {
   896→   896→    const baseUrl = this.getBaseUrl();
   897→   897→    const token = this.config!.accessToken;
   898→   898→    const userId = this.config!.userId;
   899→   899→
   900→   900→    try {
   901→   901→      // Request playback info to determine best playback method
   902→   902→      const playbackInfo = await this.request<JellyfinPlaybackInfo>(
   903→   903→        `/Items/${itemId}/PlaybackInfo`,
   904→   904→        {
   905→   905→          method: 'POST',
   906→   906→          body: JSON.stringify({
   907→   907→            UserId: userId,
   908→   908→            MaxStreamingBitrate: maxBitrate,
   909→   909→            AutoOpenLiveStreams: true,
   910→   910→          }),
   911→   911→        }
   912→   912→      );
   913→   913→
   914→   914→      if (playbackInfo.MediaSources && playbackInfo.MediaSources.length > 0) {
   915→   915→        const source = playbackInfo.MediaSources[0];
   916→   916→
   917→   917→        // Prefer direct play — best quality, no server transcoding
   918→   918→        if (source.SupportsDirectPlay) {
   919→   919→          return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${token}&UserId=${userId}`;
   920→   920→        }
   921→   921→
   922→   922→        // Fallback to direct stream
   923→   923→        if (source.SupportsDirectStream && source.DirectStreamUrl) {
   924→   924→          return `${baseUrl}${source.DirectStreamUrl}`;
   925→   925→        }
   926→   926→
   927→   927→        // Fallback to transcoded stream (HLS)
   928→   928→        if (source.SupportsTranscoding && source.TranscodingUrl) {
   929→   929→          return `${baseUrl}${source.TranscodingUrl}`;
   930→   930→        }
   931→   931→
   932→   932→        // Last resort: construct a universal audio transcoding URL
   933→   933→        const container = source.Container || 'mp3';
   934→   934→        return `${baseUrl}/Audio/${itemId}/universal?api_key=${token}&UserId=${userId}&MaxStreamingBitrate=${maxBitrate}&Container=${container}&TranscodingMaxAudioChannels=2&SegmentContainer=ts&AudioCodec=aac&Protocol=hls`;
   935→   935→      }
   936→   936→    } catch {
   937→   937→      // If playback info fails, fall back to direct stream URL
   938→   938→    }
   939→   939→
   940→   940→    // Fallback: try static stream directly
   941→   941→    return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${token}&UserId=${userId}`;
   942→   942→  }
