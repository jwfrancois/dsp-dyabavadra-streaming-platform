   297→   297→
   298→   298→function BrowseLocalPanel() {
   299→   299→  const localStore = useLocalLibraryStore();
   300→   300→  const [view, setView] = React.useState<BrowseLocalView>('artists');
   301→   301→  const [searchQuery, setSearchQuery] = React.useState('');
   302→   302→
   303→   303→  // Breadcrumb navigation state
   304→   304→  const [selectedArtist, setSelectedArtist] = React.useState<string | null>(null);
   305→   305→  const [selectedAlbum, setSelectedAlbum] = React.useState<string | null>(null);
   306→   306→
   307→   307→  // Filter only browser-imported tracks (isLocal === true)
   308→   308→  const allLocalTracks = localStore.tracks.filter((t: any) => t.isLocal);
   309→   309→
   310→   310→  // Derive artists / albums only from local tracks
   311→   311→  const localArtists = React.useMemo(() => {
   312→   312→    const set = new Set<string>();
   313→   313→    for (const t of allLocalTracks) {
   314→   314→      if (t.artist) set.add(t.artist);
   315→   315→      if (t.albumArtist && t.albumArtist !== t.artist) set.add(t.albumArtist);
   316→   316→    }
   317→   317→    return [...set].sort();
   318→   318→  }, [allLocalTracks]);
   319→   319→
   320→   320→  const localAlbums = React.useMemo(() => {
   321→   321→    const map = new Map<string, { name: string; artist: string; albumArtist: string; trackCount: number; year: number; coverArt: string | null }>();
   322→   322→    for (const t of allLocalTracks) {
   323→   323→      const key = `${t.albumArtist || t.artist}|||${t.album}`;
   324→   324→      const existing = map.get(key);
   325→   325→      if (existing) {
   326→   326→        existing.trackCount++;
   327→   327→        if (t.year && (!existing.year || t.year > existing.year)) existing.year = t.year;
   328→   328→        if (!existing.coverArt && t.coverArt) existing.coverArt = t.coverArt;
   329→   329→      } else {
   330→   330→        map.set(key, {
   331→   331→          name: t.album,
   332→   332→          artist: t.artist,
   333→   333→          albumArtist: t.albumArtist || t.artist,
   334→   334→          trackCount: 1,
   335→   335→          year: t.year,
   336→   336→          coverArt: t.coverArt,
   337→   337→        });
   338→   338→      }
   339→   339→    }
   340→   340→    return Array.from(map.values());
   341→   341→  }, [allLocalTracks]);
   342→   342→
   343→   343→  const allArtists = localArtists;
   344→   344→  const allAlbums = localAlbums;
   345→   345→  const allTracks = allLocalTracks;
   346→   346→
   347→   347→  // Derived data based on navigation state
   348→   348→  const breadcrumbs: BreadcrumbItem[] = [{ label: 'Artists', onClick: () => { setSelectedArtist(null); setSelectedAlbum(null); setView('artists'); } }];
   349→   349→
   350→   350→  let displayedArtists = allArtists;
   351→   351→  let displayedAlbums = allAlbums;
   352→   352→  let displayedTracks = allTracks;
   353→   353→
   354→   354→  // Local-only search helper
   355→   355→  const searchLocalTracks = React.useCallback((query: string) => {
   356→   356→    const q = query.toLowerCase();
   357→   357→    return allLocalTracks.filter(
   358→   358→      t =>
   359→   359→        t.title.toLowerCase().includes(q) ||
   360→   360→        t.artist.toLowerCase().includes(q) ||
   361→   361→        t.album.toLowerCase().includes(q) ||
   362→   362→        t.genre.toLowerCase().includes(q) ||
   363→   363→        (t.composer && t.composer.toLowerCase().includes(q)) ||
   364→   364→        (t.albumArtist && t.albumArtist.toLowerCase().includes(q))
   365→   365→    );
   366→   366→  }, [allLocalTracks]);
   367→   367→
   368→   368→  if (searchQuery.trim()) {
   369→   369→    displayedTracks = searchLocalTracks(searchQuery.trim());
   370→   370→    displayedAlbums = allAlbums.filter(a =>
   371→   371→      displayedTracks.some(t => t.album === a.name && (t.artist === a.artist || t.albumArtist === a.albumArtist))
   372→   372→    );
   373→   373→    displayedArtists = allArtists.filter(a =>
   374→   374→      displayedTracks.some(t => t.artist === a || t.albumArtist === a)
   375→   375→    );
   376→   376→  } else if (selectedArtist && selectedAlbum) {
   377→   377→    // Artist > Album > Tracks
   378→   378→    breadcrumbs.push(
   379→   379→      { label: selectedArtist, onClick: () => { setSelectedAlbum(null); } },
   380→   380→      { label: selectedAlbum, onClick: () => {} },
   381→   381→    );
   382→   382→    displayedTracks = allLocalTracks.filter(t => t.album === selectedAlbum && (t.artist === selectedArtist || t.albumArtist === selectedArtist));
   383→   383→  } else if (selectedArtist) {
   384→   384→    // Artist > Albums
   385→   385→    breadcrumbs.push({ label: selectedArtist, onClick: () => {} });
   386→   386→    displayedAlbums = allAlbums.filter(a => a.artist === selectedArtist || a.albumArtist === selectedArtist);
   387→   387→    displayedTracks = allLocalTracks.filter(t => t.artist === selectedArtist || t.albumArtist === selectedArtist);
   388→   388→  } else if (selectedAlbum) {
   389→   389→    // Albums > Tracks
   390→   390→    breadcrumbs.push({ label: selectedAlbum, onClick: () => {} });
   391→   391→    displayedTracks = allLocalTracks.filter(t => t.album === selectedAlbum);
   392→   392→  }
   393→   393→
   394→   394→  const showArtistsView = !selectedArtist && !selectedAlbum && view === 'artists' && !searchQuery.trim();
   395→   395→  const showAlbumsView = (selectedArtist || view === 'albums') && !selectedAlbum && !searchQuery.trim();
   396→   396→  const showTracksView = selectedAlbum || view === 'tracks' || searchQuery.trim();
   397→   397→
   398→   398→  return (
   399→   399→    <div className="space-y-4">
   400→   400→      {/* Search and View Switcher */}
   401→   401→      <div className="flex items-center gap-3">
   402→   402→        <div className="relative flex-1">
   403→   403→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   404→   404→          <Input
   405→   405→            placeholder="Search local library..."
