    83→function localTrackToTrack(lt: LocalTrack): Track {
    84→  return {
    85→    id: lt.id,
    86→    title: lt.title,
    87→    albumId: lt.album,
    88→    albumName: lt.album,
    89→    artistId: lt.artist,
    90→    artistName: lt.artist,
    91→    trackNumber: lt.trackNumber,
    92→    discNumber: lt.discNumber,
    93→    duration: lt.duration,
    94→    format: lt.format,
    95→    bitDepth: lt.bitDepth,
    96→    sampleRate: lt.sampleRate,
    97→    channels: lt.channels,
    98→    bitrate: lt.bitrate,
    99→    filePath: lt.filePath,
   100→    fileSize: lt.fileSize,
   101→    composers: lt.composer ? [lt.composer] : [],
   102→    performers: [],
   103→    genre: lt.genre,
   104→    loved: false,
   105→    playCount: 0,
   106→    source: 'local',
   107→    isAvailable: true,
   108→    blobUrl: (lt as any).blobUrl as string | undefined,
   109→  };
   110→}
   111→
   112→function playLocalTrack(lt: LocalTrack) {
   113→  const track = localTrackToTrack(lt);
   114→  usePlayerStore.getState().play(track);
   115→}
   116→
   117→function playLocalTracks(lts: LocalTrack[]) {
   118→  if (lts.length === 0) return;
   119→  const queue = lts.map(localTrackToTrack);
   120→  usePlayerStore.getState().setQueue(queue, 0);
   121→}

   293→
   294→function BrowseLocalPanel() {
   295→  const localStore = useLocalLibraryStore();
   296→  const [view, setView] = React.useState<BrowseLocalView>('artists');
   297→  const [searchQuery, setSearchQuery] = React.useState('');
   298→
   299→  // Breadcrumb navigation state
   300→  const [selectedArtist, setSelectedArtist] = React.useState<string | null>(null);
   301→  const [selectedAlbum, setSelectedAlbum] = React.useState<string | null>(null);
   302→
   303→  const allArtists = localStore.getArtists();
   304→  const allAlbums = localStore.getAlbums();
   305→  const allTracks = localStore.tracks;
   306→
   307→  // Derived data based on navigation state
   308→  const breadcrumbs: BreadcrumbItem[] = [{ label: 'Artists', onClick: () => { setSelectedArtist(null); setSelectedAlbum(null); setView('artists'); } }];
   309→
   310→  let displayedArtists = allArtists;
   311→  let displayedAlbums = allAlbums;
   312→  let displayedTracks = allTracks;
   313→
   314→  if (searchQuery.trim()) {
   315→    displayedTracks = localStore.searchTracks(searchQuery.trim());
   316→    displayedAlbums = allAlbums.filter(a =>
   317→      displayedTracks.some(t => t.album === a.name && t.artist === a.artist)
   318→    );
   319→    displayedArtists = allArtists.filter(a =>
   320→      displayedTracks.some(t => t.artist === a)
   321→    );
   322→  } else if (selectedArtist && selectedAlbum) {
   323→    // Artist > Album > Tracks
   324→    breadcrumbs.push(
   325→      { label: selectedArtist, onClick: () => { setSelectedAlbum(null); } },
   326→      { label: selectedAlbum, onClick: () => {} },
   327→    );
   328→    displayedTracks = localStore.getTracksByAlbum(selectedAlbum).filter(t => t.artist === selectedArtist);
   329→  } else if (selectedArtist) {
   330→    // Artist > Albums
   331→    breadcrumbs.push({ label: selectedArtist, onClick: () => {} });
   332→    displayedAlbums = allAlbums.filter(a => a.artist === selectedArtist);
   333→    displayedTracks = localStore.getTracksByArtist(selectedArtist);
   334→  } else if (selectedAlbum) {
   335→    // Albums > Tracks
   336→    breadcrumbs.push({ label: selectedAlbum, onClick: () => {} });
   337→    displayedTracks = localStore.getTracksByAlbum(selectedAlbum);
   338→  }
   339→
   340→  const showArtistsView = !selectedArtist && !selectedAlbum && view === 'artists' && !searchQuery.trim();
   341→  const showAlbumsView = (selectedArtist || view === 'albums') && !selectedAlbum && !searchQuery.trim();
   342→  const showTracksView = selectedAlbum || view === 'tracks' || searchQuery.trim();
   343→
   344→  return (
   345→    <div className="space-y-4">
   346→      {/* Search and View Switcher */}
   347→      <div className="flex items-center gap-3">
   348→        <div className="relative flex-1">
   349→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   350→          <Input
   351→            placeholder="Search local library..."
   352→            value={searchQuery}
   353→            onChange={e => { setSearchQuery(e.target.value); if (e.target.value) { setSelectedArtist(null); setSelectedAlbum(null); } }}
   354→            className="pl-9 h-9 text-sm"
   355→          />
   356→        </div>
   357→        <div className="flex items-center bg-surface rounded-lg p-0.5">
   358→          {([['artists', User, 'Artists'], ['albums', Disc3, 'Albums'], ['tracks', Music2, 'Tracks']] as const).map(([key, Icon, label]) => (
   359→            <Button
   360→              key={key}
   361→              variant={view === key ? 'default' : 'ghost'}
   362→              size="sm"
   363→              className="h-8 text-xs gap-1.5 rounded-md"
   364→              onClick={() => { setView(key as BrowseLocalView); setSelectedArtist(null); setSelectedAlbum(null); setSearchQuery(''); }}
   365→            >
   366→              <Icon className="w-3.5 h-3.5" /> {label}
   367→            </Button>
   368→          ))}
   369→        </div>
   370→      </div>
   371→
   372→      {/* Breadcrumb */}
   373→      {(selectedArtist || selectedAlbum) && !searchQuery.trim() && (
   374→        <div className="flex items-center gap-1 text-xs">
   375→          {breadcrumbs.map((crumb, i) => (
   376→            <React.Fragment key={i}>
   377→              {i > 0 && <ChevronRight className="w-3 h-3 text-muted-foreground" />}
   378→              <button
   379→                className={`${i === breadcrumbs.length - 1 ? 'text-foreground font-medium' : 'text-muted-foreground hover:text-foreground'} transition-colors`}
   380→                onClick={crumb.onClick}
   381→              >
   382→                {crumb.label}
   383→              </button>
   384→            </React.Fragment>
   385→          ))}
   386→        </div>
   387→      )}
   388→
   389→      {/* Empty State */}
   390→      {allTracks.length === 0 && (
   391→        <Card className="bg-card border-border">
   392→          <CardContent className="p-12 text-center">
   393→            <LibraryIcon className="w-12 h-12 mx-auto mb-4 text-muted-foreground/30" />
   394→            <h3 className="text-sm font-semibold mb-1">No Local Tracks</h3>
   395→            <p className="text-xs text-muted-foreground max-w-md mx-auto">
   396→              Scan your music folders to build your local library. Go to the &quot;Scan Folders&quot; tab to add directories and start scanning.
   397→            </p>
   398→          </CardContent>
   399→        </Card>
   400→      )}
   401→
   402→      {/* Artists Grid */}
   403→      {showArtistsView && allTracks.length > 0 && (
   404→        <div className="space-y-3">
   405→          <div className="flex items-center justify-between">
   406→            <h2 className="text-sm font-semibold">Artists ({displayedArtists.length})</h2>
   407→          </div>
   408→          {displayedArtists.length === 0 ? (
   409→            <p className="text-xs text-muted-foreground py-8 text-center">No artists found.</p>
   410→          ) : (
   411→            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-6 gap-3">
   412→              {displayedArtists.sort().map(artist => {
   413→                const artistAlbums = allAlbums.filter(a => a.artist === artist);
   414→                const artistTrackCount = artistAlbums.reduce((s, a) => s + a.trackCount, 0);
   415→                return (
   416→                  <Card
   417→                    key={artist}
   418→                    className="bg-card border-border hover:border-primary/30 cursor-pointer transition-all group"
   419→                    onClick={() => { setSelectedArtist(artist); setSelectedAlbum(null); setView('albums'); }}
   420→                  >
   421→                    <CardContent className="p-4 text-center">
   422→                      <div className="w-16 h-16 mx-auto rounded-full bg-gradient-to-br from-primary/20 to-primary/5 flex items-center justify-center mb-3 group-hover:from-primary/30 group-hover:to-primary/10 transition-all">
   423→                        <User className="w-7 h-7 text-primary/60" />
   424→                      </div>
   425→                      <p className="text-xs font-medium truncate">{artist}</p>
   426→                      <p className="text-[10px] text-muted-foreground mt-0.5">{artistAlbums.length} album{artistAlbums.length !== 1 ? 's' : ''} · {artistTrackCount} tracks</p>
   427→                    </CardContent>
   428→                  </Card>
   429→                );
   430→              })}
   431→            </div>
   432→          )}
   433→        </div>
   434→      )}
   435→
   436→      {/* Albums Grid */}
   437→      {showAlbumsView && allTracks.length > 0 && (
   438→        <div className="space-y-3">
   439→          <div className="flex items-center justify-between">
   440→            <h2 className="text-sm font-semibold">{selectedArtist ? `${selectedArtist} — ` : ''}Albums ({displayedAlbums.length})</h2>
   441→          </div>
   442→          {displayedAlbums.length === 0 ? (
   443→            <p className="text-xs text-muted-foreground py-8 text-center">No albums found.</p>
   444→          ) : (
   445→            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 gap-4">
   446→              {displayedAlbums.map(album => {
   447→                const albumTracks = localStore.getTracksByAlbum(album.name).filter(t => !selectedArtist || t.artist === selectedArtist);
   448→                const firstTrack = albumTracks[0];
   449→                return (
   450→                  <Card
   451→                    key={`${album.artist}-${album.name}`}
   452→                    className="bg-card border-border hover:border-primary/30 cursor-pointer transition-all group"
   453→                    onDoubleClick={() => playLocalTracks(albumTracks.sort((a, b) => a.trackNumber - b.trackNumber))}
   454→                    onClick={() => { setSelectedAlbum(album.name); if (selectedArtist) { /* keep artist */ } else { setSelectedArtist(album.artist); } }}
   455→                  >
   456→                    <CardContent className="p-3">
   457→                      {/* Cover Art */}
   458→                      <div className="relative aspect-square rounded-lg overflow-hidden mb-3">
   459→                        {firstTrack?.coverArt ? (
   460→                          <img
   461→                            src={`data:image/jpeg;base64,${firstTrack.coverArt}`}
   462→                            alt={album.name}
   463→                            className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-300"
   464→                          />
