    83→function localTrackToTrack(lt: LocalTrack): Track {
    84→  return {
    85→    id: lt.id,
    86→    title: lt.title,
    87→    albumId: lt.album,
    88→    albumName: lt.album,
    89→    artistId: lt.artist,
    90→    artistName: lt.artist,
    91→    trackNumber: lt.trackNumber,
    92→    discNumber: lt.discNumber,
    93→    duration: lt.duration,
    94→    format: lt.format,
    95→    bitDepth: lt.bitDepth,
    96→    sampleRate: lt.sampleRate,
    97→    channels: lt.channels,
    98→    bitrate: lt.bitrate,
    99→    filePath: lt.filePath,
   100→    fileSize: lt.fileSize,
   101→    composers: lt.composer ? [lt.composer] : [],
   102→    performers: [],
   103→    genre: lt.genre,
   104→    loved: false,
   105→    playCount: 0,
   106→    source: 'local',
   107→    isAvailable: true,
   108→  };
   109→}
   110→
   111→function playLocalTrack(lt: LocalTrack) {
   112→  const track = localTrackToTrack(lt);
   113→  usePlayerStore.getState().play(track);
   114→}
   115→
   116→function playLocalTracks(lts: LocalTrack[]) {
   117→  if (lts.length === 0) return;
   118→  const first = localTrackToTrack(lts[0]);
   119→  const queue = lts.slice(1).map(localTrackToTrack);
   120→  usePlayerStore.getState().play(first, queue);
   121→}

   285→function BrowseLocalPanel() {
   286→  const localStore = useLocalLibraryStore();
   287→  const [view, setView] = React.useState<BrowseLocalView>('artists');
   288→  const [searchQuery, setSearchQuery] = React.useState('');
   289→
   290→  // Breadcrumb navigation state
   291→  const [selectedArtist, setSelectedArtist] = React.useState<string | null>(null);
   292→  const [selectedAlbum, setSelectedAlbum] = React.useState<string | null>(null);
   293→
   294→  const allArtists = localStore.getArtists();
   295→  const allAlbums = localStore.getAlbums();
   296→  const allTracks = localStore.tracks;
   297→
   298→  // Derived data based on navigation state
   299→  const breadcrumbs: BreadcrumbItem[] = [{ label: 'Artists', onClick: () => { setSelectedArtist(null); setSelectedAlbum(null); setView('artists'); } }];
   300→
   301→  let displayedArtists = allArtists;
   302→  let displayedAlbums = allAlbums;
   303→  let displayedTracks = allTracks;
   304→
   305→  if (searchQuery.trim()) {
   306→    displayedTracks = localStore.searchTracks(searchQuery.trim());
   307→    displayedAlbums = allAlbums.filter(a =>
   308→      displayedTracks.some(t => t.album === a.name && t.artist === a.artist)
   309→    );
   310→    displayedArtists = allArtists.filter(a =>
   311→      displayedTracks.some(t => t.artist === a)
   312→    );
   313→  } else if (selectedArtist && selectedAlbum) {
   314→    // Artist > Album > Tracks
   315→    breadcrumbs.push(
   316→      { label: selectedArtist, onClick: () => { setSelectedAlbum(null); } },
   317→      { label: selectedAlbum, onClick: () => {} },
   318→    );
   319→    displayedTracks = localStore.getTracksByAlbum(selectedAlbum).filter(t => t.artist === selectedArtist);
   320→  } else if (selectedArtist) {
   321→    // Artist > Albums
   322→    breadcrumbs.push({ label: selectedArtist, onClick: () => {} });
   323→    displayedAlbums = allAlbums.filter(a => a.artist === selectedArtist);
   324→    displayedTracks = localStore.getTracksByArtist(selectedArtist);
   325→  } else if (selectedAlbum) {
   326→    // Albums > Tracks
   327→    breadcrumbs.push({ label: selectedAlbum, onClick: () => {} });
   328→    displayedTracks = localStore.getTracksByAlbum(selectedAlbum);
   329→  }
   330→
   331→  const showArtistsView = !selectedArtist && !selectedAlbum && view === 'artists' && !searchQuery.trim();
   332→  const showAlbumsView = (selectedArtist || view === 'albums') && !selectedAlbum && !searchQuery.trim();
   333→  const showTracksView = selectedAlbum || view === 'tracks' || searchQuery.trim();
   334→
   335→  return (
   336→    <div className="space-y-4">
   337→      {/* Search and View Switcher */}
   338→      <div className="flex items-center gap-3">
   339→        <div className="relative flex-1">
   340→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   341→          <Input
   342→            placeholder="Search local library..."
   343→            value={searchQuery}
   344→            onChange={e => { setSearchQuery(e.target.value); if (e.target.value) { setSelectedArtist(null); setSelectedAlbum(null); } }}
   345→            className="pl-9 h-9 text-sm"
   346→          />
   347→        </div>
   348→        <div className="flex items-center bg-surface rounded-lg p-0.5">
   349→          {([['artists', User, 'Artists'], ['albums', Disc3, 'Albums'], ['tracks', Music2, 'Tracks']] as const).map(([key, Icon, label]) => (
   350→            <Button
   351→              key={key}
   352→              variant={view === key ? 'default' : 'ghost'}
   353→              size="sm"
   354→              className="h-8 text-xs gap-1.5 rounded-md"
   355→              onClick={() => { setView(key as BrowseLocalView); setSelectedArtist(null); setSelectedAlbum(null); setSearchQuery(''); }}
   356→            >
   357→              <Icon className="w-3.5 h-3.5" /> {label}
   358→            </Button>
   359→          ))}
   360→        </div>
   361→      </div>
   362→
   363→      {/* Breadcrumb */}
   364→      {(selectedArtist || selectedAlbum) && !searchQuery.trim() && (
   365→        <div className="flex items-center gap-1 text-xs">
   366→          {breadcrumbs.map((crumb, i) => (
   367→            <React.Fragment key={i}>
   368→              {i > 0 && <ChevronRight className="w-3 h-3 text-muted-foreground" />}
   369→              <button
   370→                className={`${i === breadcrumbs.length - 1 ? 'text-foreground font-medium' : 'text-muted-foreground hover:text-foreground'} transition-colors`}
   371→                onClick={crumb.onClick}
   372→              >
   373→                {crumb.label}
   374→              </button>
   375→            </React.Fragment>
   376→          ))}
   377→        </div>
   378→      )}
   379→
   380→      {/* Empty State */}
   381→      {allTracks.length === 0 && (
   382→        <Card className="bg-card border-border">
   383→          <CardContent className="p-12 text-center">
   384→            <LibraryIcon className="w-12 h-12 mx-auto mb-4 text-muted-foreground/30" />
   385→            <h3 className="text-sm font-semibold mb-1">No Local Tracks</h3>
   386→            <p className="text-xs text-muted-foreground max-w-md mx-auto">
   387→              Scan your music folders to build your local library. Go to the &quot;Scan Folders&quot; tab to add directories and start scanning.
   388→            </p>
   389→          </CardContent>
   390→        </Card>
   391→      )}
   392→
   393→      {/* Artists Grid */}
   394→      {showArtistsView && allTracks.length > 0 && (
   395→        <div className="space-y-3">
   396→          <div className="flex items-center justify-between">
   397→            <h2 className="text-sm font-semibold">Artists ({displayedArtists.length})</h2>
   398→          </div>
   399→          {displayedArtists.length === 0 ? (
   400→            <p className="text-xs text-muted-foreground py-8 text-center">No artists found.</p>
   401→          ) : (
   402→            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-6 gap-3">
   403→              {displayedArtists.sort().map(artist => {
   404→                const artistAlbums = allAlbums.filter(a => a.artist === artist);
   405→                const artistTrackCount = artistAlbums.reduce((s, a) => s + a.trackCount, 0);
   406→                return (
   407→                  <Card
   408→                    key={artist}
   409→                    className="bg-card border-border hover:border-primary/30 cursor-pointer transition-all group"
   410→                    onClick={() => { setSelectedArtist(artist); setSelectedAlbum(null); setView('albums'); }}
   411→                  >
   412→                    <CardContent className="p-4 text-center">
   413→                      <div className="w-16 h-16 mx-auto rounded-full bg-gradient-to-br from-primary/20 to-primary/5 flex items-center justify-center mb-3 group-hover:from-primary/30 group-hover:to-primary/10 transition-all">
   414→                        <User className="w-7 h-7 text-primary/60" />
   415→                      </div>
   416→                      <p className="text-xs font-medium truncate">{artist}</p>
   417→                      <p className="text-[10px] text-muted-foreground mt-0.5">{artistAlbums.length} album{artistAlbums.length !== 1 ? 's' : ''} · {artistTrackCount} tracks</p>
   418→                    </CardContent>
   419→                  </Card>
   420→                );
   421→              })}
   422→            </div>
   423→          )}
   424→        </div>
   425→      )}
   426→
   427→      {/* Albums Grid */}
   428→      {showAlbumsView && allTracks.length > 0 && (
   429→        <div className="space-y-3">
   430→          <div className="flex items-center justify-between">
   431→            <h2 className="text-sm font-semibold">{selectedArtist ? `${selectedArtist} — ` : ''}Albums ({displayedAlbums.length})</h2>
   432→          </div>
   433→          {displayedAlbums.length === 0 ? (
   434→            <p className="text-xs text-muted-foreground py-8 text-center">No albums found.</p>
   435→          ) : (
   436→            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 gap-4">
   437→              {displayedAlbums.map(album => {
   438→                const albumTracks = localStore.getTracksByAlbum(album.name).filter(t => !selectedArtist || t.artist === selectedArtist);
   439→                const firstTrack = albumTracks[0];
   440→                return (
   441→                  <Card
   442→                    key={`${album.artist}-${album.name}`}
   443→                    className="bg-card border-border hover:border-primary/30 cursor-pointer transition-all group"
   444→                    onDoubleClick={() => playLocalTracks(albumTracks.sort((a, b) => a.trackNumber - b.trackNumber))}
   445→                    onClick={() => { setSelectedAlbum(album.name); if (selectedArtist) { /* keep artist */ } else { setSelectedArtist(album.artist); } }}
   446→                  >
   447→                    <CardContent className="p-3">
   448→                      {/* Cover Art */}
   449→                      <div className="relative aspect-square rounded-lg overflow-hidden mb-3">
   450→                        {firstTrack?.coverArt ? (
   451→                          <img
   452→                            src={`data:image/jpeg;base64,${firstTrack.coverArt}`}
   453→                            alt={album.name}
   454→                            className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-300"
   455→                          />
   456→                        ) : (
   457→                          <div className={`w-full h-full bg-gradient-to-br ${getCoverGradient(album.name + album.artist)} flex items-center justify-center`}>
   458→                            <Disc3 className="w-10 h-10 text-white/20" />
   459→                          </div>
   460→                        )}
   461→                        <div className="absolute inset-0 bg-black/0 group-hover:bg-black/20 transition-colors flex items-center justify-center opacity-0 group-hover:opacity-100">
   462→                          <Button size="icon" className="h-10 w-10 rounded-full bg-primary/90 hover:bg-primary text-primary-foreground shadow-lg">
   463→                            <Play className="w-5 h-5 ml-0.5" />

   606→  const handleScan = React.useCallback(async () => {
   607→    const dir = directoryInput.trim();

   619→      const fmt = t.format.toUpperCase();
   620→      counts[fmt] = (counts[fmt] || 0) + 1;

   627→    : 0;

   629→    ? (tracks.reduce((s, t) => s + t.bitDepth, 0) / tracks.length).toFixed(1)
