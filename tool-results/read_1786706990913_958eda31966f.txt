  1479→function NetworkSharesPanel() {
  1480→  const localStore = useLocalLibraryStore();
  1481→  const [shares, setShares] = React.useState<NetworkShareInfo[]>([]);
  1482→  const [uncInput, setUncInput] = React.useState('');
  1483→  const [usernameInput, setUsernameInput] = React.useState('');
  1484→  const [passwordInput, setPasswordInput] = React.useState('');
  1485→  const [mounting, setMounting] = React.useState<string | null>(null);
  1486→  const [loading, setLoading] = React.useState(true);
  1487→  const [error, setError] = React.useState<string | null>(null);
  1488→
  1489→  React.useEffect(() => { fetchShares(); }, []);
  1490→
  1491→  const fetchShares = async () => {
  1492→    setLoading(true);
  1493→    try {
  1494→      const res = await fetch('/api/library/mount');
  1495→      const data = await res.json();
  1496→      if (data.success) { setShares(data.shares || []); setError(null); }
  1497→      else { setError(data.error || 'Failed to load network shares'); }

  1500→  };
  1501→
  1502→  const handleMount = async () => {
  1503→    const unc = uncInput.trim();

  1505→    setMounting(unc); setError(null);
  1506→    try {
  1507→      const res = await fetch('/api/library/mount', {
  1508→        method: 'POST', headers: { 'Content-Type': 'application/json' },
  1509→        body: JSON.stringify({ uncPath: unc, username: usernameInput.trim() || undefined, password: passwordInput || undefined }),
  1510→      });
  1511→      const data = await res.json();
  1512→      if (data.success && data.share) {
  1513→        if (data.scanPath) { localStore.addDirectory(data.scanPath); localStore.startScan(data.scanPath); }
  1514→        setUncInput(''); setUsernameInput(''); setPasswordInput('');
  1515→        await fetchShares();
  1516→      } else { setError(data.error || 'Mount failed'); }
  1517→    } catch { setError('Failed to mount share'); }
  1518→    setMounting(null);
  1519→  };
  1520→
  1521→  const handleUnmount = async (shareId: string) => {
  1522→    try { await fetch(`/api/library/mount?id=${encodeURIComponent(shareId)}`, { method: 'DELETE' }); await fetchShares(); }
  1523→    catch { setError('Failed to unmount'); }
  1524→  };
  1525→
  1526→  const handleRemount = async (share: NetworkShareInfo) => {
  1527→    setMounting(share.id); setError(null);
  1528→    try {
  1529→      const res = await fetch('/api/library/mount', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ uncPath: share.uncPath }) });
  1530→      const data = await res.json();
  1531→      if (data.success) { if (data.scanPath) localStore.addDirectory(data.scanPath); await fetchShares(); }
  1532→      else { setError(data.error || 'Remount failed'); }
  1533→    } catch { setError('Failed to remount'); }
  1534→    setMounting(null);
  1535→  };
  1536→
  1537→  const handleScanMounted = (share: NetworkShareInfo) => {
  1538→    const scanPath = share.mountPoint + (share.subPath ? `/${share.subPath}` : '');
  1539→    localStore.addDirectory(scanPath); localStore.startScan(scanPath);
  1540→  };
  1541→
  1542→  return (
  1543→    <div className="space-y-6">
  1544→      <Card className="bg-card border-border">
  1545→        <CardContent className="p-4">
  1546→          <div className="flex items-start gap-3">
  1547→            <Server className="w-5 h-5 text-primary flex-shrink-0 mt-0.5" />
  1548→            <div className="text-xs text-muted-foreground space-y-1">
  1549→              <p className="font-medium text-foreground">Access your NAS or network storage</p>
  1550→              <p>Mount SMB/CIFS network shares (TrueNAS, Synology, Windows shares, etc.) to scan and play music from network-attached storage.</p>
  1551→            </div>
  1552→          </div>
  1553→        </CardContent>
  1554→      </Card>
  1555→
  1556→      <Card className="bg-card border-border">
  1557→        <CardHeader className="pb-3"><CardTitle className="text-sm flex items-center gap-2"><Globe className="w-4 h-4 text-primary" /> Add Network Share</CardTitle></CardHeader>
  1558→        <CardContent className="space-y-4">
  1559→          <div className="space-y-2">
  1560→            <label className="text-xs font-medium text-muted-foreground">Network Path (UNC or SMB URL)</label>
  1561→            <Input placeholder="\\10.0.0.80\iguey\Media\Music" value={uncInput} onChange={e => setUncInput(e.target.value)} className="h-9 text-sm font-mono" disabled={!!mounting} />
  1562→          </div>
  1563→          <div className="grid grid-cols-2 gap-3">
  1564→            <div className="space-y-2">
  1565→              <label className="text-xs font-medium text-muted-foreground">Username (optional)</label>
  1566→              <Input placeholder="guest" value={usernameInput} onChange={e => setUsernameInput(e.target.value)} className="h-9 text-sm" disabled={!!mounting} />
