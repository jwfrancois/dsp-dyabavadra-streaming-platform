   162→/** Convert Jellyfin ticks (100ns units) to seconds */
   163→export function ticksToSeconds(ticks: number | undefined): number {
   164→  if (ticks === undefined || ticks === null) return 0;
   165→  return Math.floor(ticks / TICKS_PER_SECOND);
   166→}
   167→
   168→/** Convert seconds to Jellyfin ticks */
   169→export function secondsToTicks(seconds: number): number {
   170→  return Math.floor(seconds * TICKS_PER_SECOND);
   171→}

   173→/** Strip trailing slash from a URL */
   174→function normalizeServerUrl(url: string): string {
   175→  return url.replace(/\/+$/, '');
   176→}
   177→
   178→/** Build a query string from a record, filtering out undefined/null/empty values */
   179→function buildQueryString(params: Record<string, string | number | boolean | undefined | null | string[] | number[]>): string {
   180→  const parts: string[] = [];
   181→  for (const [key, value] of Object.entries(params)) {
   182→    if (value === undefined || value === null || value === '') continue;
   183→    if (Array.isArray(value)) {
   184→      if (value.length > 0) {
   185→        parts.push(`${encodeURIComponent(key)}=${value.map(v => encodeURIComponent(String(v))).join(',')}`);
   186→      }
   187→    } else {
   188→      parts.push(`${encodeURIComponent(key)}=${encodeURIComponent(String(value))}`);
   189→    }
   190→  }
   191→  return parts.length > 0 ? `?${parts.join('&')}` : '';
   192→}

   196→export class JellyfinClient {
   197→  private config: JellyfinConfig | null = null;
   198→  private storageAvailable = true;
   199→
   200→  constructor() {
   201→    // Check if localStorage is available (SSR-safe)
   202→    try {
   203→      const testKey = '__dsp_jellyfin_test__';
   204→      if (typeof window !== 'undefined') {
   205→        window.localStorage.setItem(testKey, '1');
   206→        window.localStorage.removeItem(testKey);
   207→      } else {
   208→        this.storageAvailable = false;
   209→      }
   210→    } catch {
   211→      this.storageAvailable = false;
   212→    }
   213→
   214→    // Auto-load persisted config
   215→    this.config = this.loadConfig();
   216→  }
   217→
   218→  // ═══════════════════════════════════════════════════════
   219→  // Persistence
   220→  // ═══════════════════════════════════════════════════════
   221→
   222→  private saveConfig(): void {
   223→    if (!this.config) return;
   224→    if (this.storageAvailable && typeof window !== 'undefined') {
   225→      try {
   226→        window.localStorage.setItem(STORAGE_KEY, JSON.stringify(this.config));
   227→      } catch {
   228→        // Storage full or unavailable — silently fail
   229→      }
   230→    }
   231→  }
   232→
   233→  private loadConfig(): JellyfinConfig | null {
   234→    if (!this.storageAvailable || typeof window === 'undefined') return null;
   235→    try {
   236→      const raw = window.localStorage.getItem(STORAGE_KEY);
   237→      if (!raw) return null;
   238→      const parsed = JSON.parse(raw) as JellyfinConfig;
   239→      // Basic validation
   240→      if (!parsed.serverUrl || !parsed.accessToken || !parsed.userId) return null;
   241→      return parsed;
   242→    } catch {
   243→      return null;
   244→    }
   245→  }
   246→
   247→  // ═══════════════════════════════════════════════════════
   248→  // Core HTTP
   249→  // ═══════════════════════════════════════════════════════
   250→
   251→  private getBaseUrl(): string {
   252→    if (!this.config) throw new JellyfinNotAuthenticatedError();
   253→    return normalizeServerUrl(this.config.serverUrl);
   254→  }
   255→
   256→  private getHeaders(): Record<string, string> {
   257→    if (!this.config) throw new JellyfinNotAuthenticatedError();
   258→    return {
   259→      'X-Emby-Token': this.config.accessToken,
   260→      'Content-Type': 'application/json',
   261→      'Accept': 'application/json',
   262→    };
   263→  }
   264→
   265→  private async request<T>(path: string, options: RequestInit = {}): Promise<T> {
   266→    const baseUrl = this.getBaseUrl();
   267→    const url = `${baseUrl}${path}`;
   268→    const headers: Record<string, string> = {
   269→      ...this.getHeaders(),
   270→      ...(options.headers as Record<string, string> | undefined),
   271→    };
   272→
   273→    // Remove Content-Type for GET/DELETE if no body
   274→    if (!options.body && (options.method === 'GET' || options.method === 'DELETE')) {
   275→      delete headers['Content-Type'];
   276→    }
   277→
   278→    let response: Response;
   279→    try {
   280→      response = await fetch(url, {
   281→        ...options,
   282→        headers,
   283→      });
   284→    } catch (err) {
   285→      throw new JellyfinApiError(
   286→        `Network error: ${err instanceof Error ? err.message : 'Unknown error'}`,
   287→        0,
   288→        path,
   289→        err
   290→      );
   291→    }
   292→
   293→    if (!response.ok) {
   294→      let errorBody = '';
   295→      try {
   296→        errorBody = await response.text();
   297→      } catch {
   298→        // ignore
   299→      }
   300→      throw new JellyfinApiError(
   301→        `Jellyfin API error ${response.status}: ${response.statusText} — ${errorBody}`,
   302→        response.status,
   303→        path
   304→      );
   305→    }
   306→
   307→    // Some endpoints return empty responses (e.g., DELETE, favorite toggles)
   308→    const contentType = response.headers.get('content-type');
   309→    if (contentType && contentType.includes('application/json')) {
   310→      return response.json() as Promise<T>;
   311→    }
   312→
   313→    return undefined as unknown as T;
   314→  }

   320→  /**
   321→   * Connect to a Jellyfin server. Authenticates and fetches server info.
   322→   * Persists the configuration to localStorage.
   323→   */
   324→  async connect(serverUrl: string, username: string, password: string): Promise<JellyfinConfig> {
   325→    const baseUrl = normalizeServerUrl(serverUrl);
   326→
   327→    // Step 1: Authenticate
   328→    const authResponse = await this.request<JellyfinAuthResponse>(
   329→      '/Users/AuthenticateByName',
   330→      {
   331→        method: 'POST',
   332→        body: JSON.stringify({ Username: username, Pw: password }),
   333→        headers: {
   334→          'Content-Type': 'application/json',
   335→          'Authorization': `MediaBrowser Client="DSP", Device="Browser", DeviceId="dsp-jellyfin-${typeof window !== 'undefined' ? window.navigator.userAgent.slice(0, 32).replace(/[^a-zA-Z0-9]/g, '') : 'ssr'}", Version="1.0.0"`,
   336→        },
   337→      }
   338→    );
   339→
   340→    const token = authResponse.AccessToken;
   341→    const userId = authResponse.User.Id;
   342→    const serverId = authResponse.ServerId;
   343→
   344→    // Step 2: Fetch server info
   345→    let serverName = 'Jellyfin';
   346→    let serverVersion = 'unknown';
   347→    try {
   348→      // Temporarily set config for the getSystemInfo call
   349→      this.config = {
   350→        serverUrl: baseUrl,
   351→        username,
   352→        accessToken: token,
   353→        userId,
   354→        serverId,
   355→        serverName,
   356→        serverVersion,
   357→        connectedAt: new Date().toISOString(),
   358→        lastHeartbeat: new Date().toISOString(),
   359→        musicLibraryId: '',
   360→      };
   361→      const sysInfo = await this.getSystemInfo();
   362→      serverName = sysInfo.ServerName;
   363→      serverVersion = sysInfo.Version;
   364→    } catch {
   365→      // If system info fails, continue with defaults
   366→    }
   367→
   368→    // Step 3: Auto-detect music library
   369→    let musicLibraryId = '';
   370→    try {
   371→      const views = await this.getViews();
   372→      const musicFolder = views.Items.find(
   373→        item => item.Type === 'CollectionFolder' && item.Name.toLowerCase().includes('music')
   374→      );
   375→      if (musicFolder) {
   376→        musicLibraryId = musicFolder.Id;
   377→      }
   378→    } catch {
   379→      // Continue without auto-detected library
   380→    }
   381→
   382→    // Step 4: Build and save config
   383→    const config: JellyfinConfig = {
   384→      serverUrl: baseUrl,
   385→      username,
   386→      accessToken: token,
   387→      userId,
   388→      serverId,
   389→      serverName,
   390→      serverVersion,
   391→      connectedAt: new Date().toISOString(),
   392→      lastHeartbeat: new Date().toISOString(),
   393→      musicLibraryId,
   394→    };
   395→
   396→    this.config = config;
   397→    this.saveConfig();
   398→
   399→    return config;
   400→  }

   402→  /**
   403→   * Disconnect from the Jellyfin server. Revokes the token and clears local state.
   404→   */
   405→  async disconnect(): Promise<void> {
   406→    if (this.config) {
   407→      try {
   408→        await this.request<void>(`/Users/${this.config.userId}/Tokens`, {
   409→          method: 'DELETE',
   410→        });
   411→      } catch {
   412→        // Best-effort — continue clearing local state
   413→      }
   414→    }
   415→
   416→    this.config = null;
   417→    if (this.storageAvailable && typeof window !== 'undefined') {
   418→      try {
   419→        window.localStorage.removeItem(STORAGE_KEY);
   420→      } catch {
   421→        // ignore
   422→      }
   423→    }
   424→  }

   429→      this.config &&
   430→      this.config.accessToken &&
   431→      this.config.userId &&
   432→      this.config.serverUrl
   433→    );
   434→  }

   436→  /** Get the current configuration, or null if not authenticated */

   777→  // ═══════════════════════════════════════════════════════

   779→  /**
   780→   * Get a playable audio stream URL for an item.
   781→   * Prefers direct play for best quality; falls back to transcoded stream.
   782→   */
   783→  async getPlaybackStreamUrl(
   784→    itemId: string,
   785→    maxBitrate: number = DEFAULT_MAX_BITRATE
   786→  ): Promise<string> {
   787→    const baseUrl = this.getBaseUrl();
   788→    const token = this.config!.accessToken;
   789→    const userId = this.config!.userId;
   790→
   791→    try {
   792→      // Request playback info to determine best playback method
   793→      const playbackInfo = await this.request<JellyfinPlaybackInfo>(
   794→        `/Items/${itemId}/PlaybackInfo`,
   795→        {
   796→          method: 'POST',
   797→          body: JSON.stringify({
   798→            UserId: userId,
   799→            MaxStreamingBitrate: maxBitrate,
   800→            AutoOpenLiveStreams: true,
   801→          }),
   802→        }
   803→      );
   804→
   805→      if (playbackInfo.MediaSources && playbackInfo.MediaSources.length > 0) {
   806→        const source = playbackInfo.MediaSources[0];
   807→
   808→        // Prefer direct play — best quality, no server transcoding
   809→        if (source.SupportsDirectPlay) {
   810→          return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${token}&UserId=${userId}`;
   811→        }
   812→
   813→        // Fallback to direct stream
   814→        if (source.SupportsDirectStream && source.DirectStreamUrl) {
   815→          return `${baseUrl}${source.DirectStreamUrl}`;
   816→        }
   817→
   818→        // Fallback to transcoded stream (HLS)
   819→        if (source.SupportsTranscoding && source.TranscodingUrl) {
   820→          return `${baseUrl}${source.TranscodingUrl}`;
   821→        }
   822→
   823→        // Last resort: construct a universal audio transcoding URL
   824→        const container = source.Container || 'mp3';
   825→        return `${baseUrl}/Audio/${itemId}/universal?api_key=${token}&UserId=${userId}&MaxStreamingBitrate=${maxBitrate}&Container=${container}&TranscodingMaxAudioChannels=2&SegmentContainer=ts&AudioCodec=aac&Protocol=hls`;
   826→      }
   827→    } catch {
   828→      // If playback info fails, fall back to direct stream URL
   829→    }
   830→
   831→    // Fallback: try static stream directly
   832→    return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${token}&UserId=${userId}`;
   833→  }
