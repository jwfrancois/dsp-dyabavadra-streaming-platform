  1106→
  1107→        // readEntries() may return at most 100 entries per call (Chrome).

  1109→        async function readAllEntries(reader: any): Promise<any[]> {

  1113→            batch = await new Promise<any[]>((resolve, reject) => {
  1114→              reader.readEntries(resolve, reject);

  1116→            all.push(...batch);
  1117→          } while (batch.length > 0);
  1118→          return all;
  1119→        }
  1120→
  1121→        async function readEntry(entry: any) {
  1122→          if (entry.isFile) {
  1123→            const file = await new Promise<File>((resolve) => entry.file(resolve));
  1124→            files.push(file);
  1125→          } else if (entry.isDirectory) {
  1126→            const reader = entry.createReader();

  1128→            for (const sub of subEntries) {
  1129→              await readEntry(sub);
  1130→            }
  1131→          }
  1132→        }
  1133→
  1134→        for (const entry of entries) {
  1135→          await readEntry(entry);
  1136→        }

  1138→        files.push(...Array.from(e.dataTransfer.files));
  1139→      }

  1150→        onProgress: setProgress,

  1488→
  1489→  React.useEffect(() => { fetchShares(); }, []);

  1492→    setLoading(true);

  1494→      const res = await fetch('/api/library/mount');
  1495→      const data = await res.json();
  1496→      if (data.success) { setShares(data.shares || []); setError(null); }

  1500→  };
  1501→
  1502→  const handleMount = async () => {
  1503→    const unc = uncInput.trim();

  1505→    setMounting(unc); setError(null);
  1506→    try {
  1507→      const res = await fetch('/api/library/mount', {
  1508→        method: 'POST', headers: { 'Content-Type': 'application/json' },
  1509→        body: JSON.stringify({ uncPath: unc, username: usernameInput.trim() || undefined, password: passwordInput || undefined }),
  1510→      });
  1511→      const data = await res.json();
  1512→      if (data.success && data.share) {
  1513→        if (data.scanPath) { localStore.addDirectory(data.scanPath); localStore.startScan(data.scanPath); }
  1514→        setUncInput(''); setUsernameInput(''); setPasswordInput('');
  1515→        await fetchShares();
  1516→      } else { setError(data.error || 'Mount failed'); }
  1517→    } catch { setError('Failed to mount share'); }
  1518→    setMounting(null);
  1519→  };
  1520→
  1521→  const handleUnmount = async (shareId: string) => {
  1522→    try { await fetch(`/api/library/mount?id=${encodeURIComponent(shareId)}`, { method: 'DELETE' }); await fetchShares(); }

  1526→  const handleRemount = async (share: NetworkShareInfo) => {
  1527→    setMounting(share.id); setError(null);
  1528→    try {
  1529→      const res = await fetch('/api/library/mount', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ uncPath: share.uncPath }) });
  1530→      const data = await res.json();
  1531→      if (data.success) { if (data.scanPath) localStore.addDirectory(data.scanPath); await fetchShares(); }

  1552→          </div>
  1553→        </CardContent>
  1554→      </Card>

  1556→      <Card className="bg-card border-border">
