   106→   106→      stopRadio();

   123→   123→      }
   124→   124→    }

   132→   132→  };

   152→   152→            <Radio className="w-5 h-5 text-rose-200" />
   153→   153→          </div>

   198→   198→                </SelectContent>
   199→   199→              </Select>

   269→   269→                    onClick={handleStartFromCurrentTrack}

   279→   279→                  >
   280→   280→                    <User className="w-4 h-4 mr-2" />
   281→   281→                    Start from Current Artist
   282→   282→                  </Button>

   284→   284→                {!currentTrack && (
   285→   285→                  <p className="text-xs text-muted-foreground mt-2">
   286→   286→                    Play a track first to use quick start options.

   289→   289→              </CardContent>
   290→   290→            </Card>
   291→   291→

   296→   296→                  <ListMusic className="w-5 h-5 text-emerald-400" />
   297→   297→                  Create Mix
   298→   298→                </h2>
   299→   299→                <p className="text-sm text-muted-foreground mb-4">
   300→   300→                  Pick a seed to generate a personalized radio mix from your library.
   301→   301→                </p>
   302→   302→                <div className="flex flex-col sm:flex-row gap-3 mb-4">
   303→   303→                  <Select
   304→   304→                    value={seedType}
   305→   305→                    onValueChange={(v) => {
   306→   306→                      setSeedType(v as RadioSeed['type']);
   307→   307→                      setSelectedSeedId('');
   308→   308→                    }}
   309→   309→                  >
   310→   310→                    <SelectTrigger className="w-[160px]">
   311→   311→                      <SelectValue />
   312→   312→                    </SelectTrigger>
   313→   313→                    <SelectContent>

   315→   315→                        <span className="flex items-center gap-2"><Music className="w-3 h-3" /> Track</span>

   332→   332→                    </SelectTrigger>

   336→   336→                          {opt.name}

   350→   350→            {/* Generated Radio Preview */}
