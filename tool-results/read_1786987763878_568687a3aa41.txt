    32→function CoverArt({ url, name, size = 'md', rounded = 'lg' }: { url: string; name: string; size?: 'sm' | 'md' | 'lg' | 'xl'; rounded?: 'sm' | 'md' | 'lg' | 'xl' | 'full' }) {
    33→  const sizeClasses = {
    34→    sm: 'w-8 h-8',
    35→    md: 'w-10 h-10',
    36→    lg: 'w-14 h-14',
    37→    xl: 'w-full aspect-square',
    38→    full: 'w-full aspect-square',
    39→  };
    40→  const roundedClasses = {
    41→    sm: 'rounded-sm',
    42→    md: 'rounded-md',
    43→    lg: 'rounded-lg',
    44→    xl: 'rounded-xl',
    45→    full: 'rounded-xl',
    46→  };
    47→  const [error, setError] = useState(false);
    48→
    49→  return (
    50→    <div className={`${sizeClasses[size]} ${roundedClasses[rounded]} overflow-hidden bg-surface flex-shrink-0 relative`}>
    51→      {url && !error ? (
    52→        <img
    53→          src={url}
    54→          alt={name}
    55→          className="w-full h-full object-cover"
    56→          onError={() => setError(true)}
    57→          draggable={false}
    58→          loading="lazy"
    59→        />
    60→      ) : (
    61→        <div className="w-full h-full bg-gradient-to-br from-purple-900/60 to-blue-900/60 flex items-center justify-center">
    62→          <Music className="w-1/3 h-1/3 text-muted-foreground/40" />
    63→        </div>
    64→      )}
    65→    </div>
    66→  );
    67→}

    69→// ─── Connection Panel ───
    70→
    71→function ConnectionPanel() {
    72→  const [serverUrl, setServerUrl] = useState('');
    73→  const [username, setUsername] = useState('');
    74→  const [password, setPassword] = useState('');
    75→  const [apiKey, setApiKey] = useState('');
    76→  const [showPassword, setShowPassword] = useState(false);
    77→  const [authMode, setAuthMode] = useState<'password' | 'apikey'>('password');
    78→  const { connectionStatus, error, config, connect, connectWithApiKey, disconnect, clearError } = useJellyfinStore();
    79→
    80→  useEffect(() => {
    81→    if (config?.serverUrl) setServerUrl(config.serverUrl);
    82→    if (config?.username) setUsername(config.username);
    83→  }, [config]);
    84→
    85→  const handleConnect = useCallback(async () => {
    86→    clearError();
    87→    const url = serverUrl.trim().replace(/\/+$/, '');
    88→    if (!url) return;
    89→    if (authMode === 'apikey') {
    90→      if (!apiKey.trim()) return;
    91→      await connectWithApiKey(url, apiKey.trim());
    92→    } else {
    93→      if (!username.trim()) return;
    94→      await connect(url, username.trim(), password);
    95→    }
    96→  }, [serverUrl, username, password, apiKey, authMode, connect, connectWithApiKey, clearError]);
    97→
    98→  const handleDisconnect = useCallback(async () => {
    99→    await disconnect();
   100→  }, [disconnect]);
   101→
   102→  if (connectionStatus === 'connected' && config) {
   103→    return (
   104→      <Card className="border-border bg-card mb-6">
   105→        <CardContent className="p-4">
   106→          <div className="flex items-start gap-4">
   107→            <div className="w-12 h-12 rounded-xl bg-green-500/15 flex items-center justify-center flex-shrink-0">
   108→              <Server className="w-6 h-6 text-green-400" />
   109→            </div>
   110→            <div className="flex-1 min-w-0">
   111→              <div className="flex items-center gap-2 mb-1">
   112→                <h3 className="font-semibold text-sm">{config.serverName}</h3>
   113→                <Badge variant="outline" className="text-[10px] border-green-500/30 text-green-400 bg-green-500/10">
   114→                  <CheckCircle2 className="w-3 h-3 mr-1" /> Connected
   115→                </Badge>
   116→              </div>
   117→              <div className="flex flex-wrap items-center gap-x-3 gap-y-1 text-xs text-muted-foreground">
   118→                <span className="flex items-center gap-1">
   119→                  <Globe className="w-3 h-3" /> {config.serverUrl}
   120→                </span>
   121→                <span className="flex items-center gap-1">
   122→                  <User className="w-3 h-3" /> {config.username}
   123→                </span>
   124→                <span className="flex items-center gap-1">
   125→                  <Box className="w-3 h-3" /> v{config.serverVersion}
   126→                </span>
   127→              </div>
   128→            </div>
   129→            <Button
   130→              variant="outline"
   131→              size="sm"
   132→              className="text-xs gap-1.5 text-red-400 hover:text-red-300 hover:border-red-500/30 hover:bg-red-950/30 flex-shrink-0"
   133→              onClick={handleDisconnect}
   134→            >
   135→              <Unplug className="w-3 h-3" /> Disconnect
   136→            </Button>
   137→          </div>
   138→        </CardContent>
   139→      </Card>
   140→    );
   141→  }
   142→
   143→  return (
   144→    <Card className="border-border bg-card mb-6">
   145→      <CardContent className="p-4">
   146→        <div className="flex items-center gap-3 mb-4">
   147→          <div className="w-10 h-10 rounded-xl bg-purple-500/15 flex items-center justify-center flex-shrink-0">
   148→            <Server className="w-5 h-5 text-purple-400" />
   149→          </div>
   150→          <div>
   151→            <h3 className="font-semibold text-sm">Connect to Jellyfin</h3>
   152→            <p className="text-xs text-muted-foreground">Enter your Jellyfin server details to browse your music library</p>
   153→          </div>
   154→        </div>
   155→
   156→        <div className="space-y-3">
   157→          <div className="flex flex-col sm:flex-row gap-3">
   158→            <div className="flex-1">
   159→              <label className="text-xs text-muted-foreground mb-1 block">Server URL</label>
   160→              <Input
   161→                placeholder="https://jellyfin.example.com"
   162→                value={serverUrl}
   163→                onChange={(e) => setServerUrl(e.target.value)}
   164→                className="bg-surface/50 border-border"
   165→                disabled={connectionStatus === 'connecting'}
   166→              />
   167→            </div>
   168→          </div>
   169→
   170→          {/* Auth mode toggle */}
   171→          <div className="flex items-center gap-1 p-1 rounded-lg bg-surface/50 border border-border w-fit">
   172→            <button
   173→              type="button"
   174→              className={`px-3 py-1.5 text-xs rounded-md transition-all ${
   175→                authMode === 'password'
   176→                  ? 'bg-purple-500/20 text-purple-300 shadow-sm'
   177→                  : 'text-muted-foreground hover:text-foreground'
   178→              }`}
   179→              onClick={() => setAuthMode('password')}
   180→            >
   181→              <User className="w-3 h-3 inline mr-1" />Username / Password
   182→            </button>
   183→            <button
   184→              type="button"
   185→              className={`px-3 py-1.5 text-xs rounded-md transition-all ${
   186→                authMode === 'apikey'
   187→                  ? 'bg-purple-500/20 text-purple-300 shadow-sm'
   188→                  : 'text-muted-foreground hover:text-foreground'
   189→              }`}
   190→              onClick={() => setAuthMode('apikey')}
   191→            >
   192→              <Key className="w-3 h-3 inline mr-1" />API Key
   193→            </button>
   194→          </div>
   195→
   196→          {authMode === 'password' ? (
   197→            <>
   198→              <div className="flex flex-col sm:flex-row gap-3">
   199→                <div className="flex-1">
   200→                  <label className="text-xs text-muted-foreground mb-1 block">Username</label>
   201→                  <Input
   202→                    placeholder="admin"
   203→                    value={username}
   204→                    onChange={(e) => setUsername(e.target.value)}
   205→                    className="bg-surface/50 border-border"
   206→                    disabled={connectionStatus === 'connecting'}
   207→                  />
   208→                </div>
   209→              </div>
   210→              <div className="relative">
   211→                <label className="text-xs text-muted-foreground mb-1 block">Password</label>
   212→                <Input
   213→                  type={showPassword ? 'text' : 'password'}
   214→                  placeholder="••••••••"
   215→                  value={password}
   216→                  onChange={(e) => setPassword(e.target.value)}
   217→                  className="bg-surface/50 border-border pr-10"
   218→                  disabled={connectionStatus === 'connecting'}
   219→                  onKeyDown={(e) => e.key === 'Enter' && handleConnect()}
   220→                />
   221→                <button
   222→                  className="absolute right-3 top-[38px] text-muted-foreground hover:text-foreground transition-colors"
   223→                  onClick={() => setShowPassword(!showPassword)}
   224→                  type="button"
   225→                  tabIndex={-1}
   226→                >
   227→                  {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   228→                </button>
   229→              </div>
   230→            </>
   231→          ) : (
   232→            <div>
   233→              <label className="text-xs text-muted-foreground mb-1 block">API Key</label>
   234→              <Input
   235→                type="text"
   236→                placeholder="Your Jellyfin API key"
   237→                value={apiKey}
   238→                onChange={(e) => setApiKey(e.target.value)}
   239→                className="bg-surface/50 border-border font-mono text-xs"
   240→                disabled={connectionStatus === 'connecting'}
   241→                onKeyDown={(e) => e.key === 'Enter' && handleConnect()}
   242→              />
   243→              <p className="text-[10px] text-muted-foreground mt-1">
   244→                Find your API key in Jellyfin under Dashboard → API Keys
   245→              </p>

   248→
   249→          {error && (

   251→              <AlertTriangle className="w-4 h-4 text-red-400 mt-0.5 flex-shrink-0" />
   252→              <p className="text-xs text-red-300">{error}</p>
   253→            </div>
   254→          )}
   255→
   256→          <Button
   257→            className="w-full gap-2"
   258→            onClick={handleConnect}
   259→            disabled={connectionStatus === 'connecting' || !serverUrl.trim() || (authMode === 'password' ? !username.trim() : !apiKey.trim())}
   260→          >
   261→            {connectionStatus === 'connecting' ? (
   262→              <>
   263→                <Loader2 className="w-4 h-4 animate-spin" /> Connecting...
   264→              </>
   265→            ) : (
   266→              <>
   267→                <Plug className="w-4 h-4" /> Connect to Server
   268→              </>
   269→            )}
   270→          </Button>
   271→        </div>
   272→
   273→        <div className="mt-3 flex items-start gap-2 p-2.5 rounded-lg bg-blue-950/20 border border-blue-800/20">
   274→          <Shield className="w-3.5 h-3.5 text-blue-400 mt-0.5 flex-shrink-0" />
   275→          <p className="text-[10px] text-muted-foreground leading-relaxed">
   276→            Your credentials are stored locally in this browser. DSP connects directly to your Jellyfin server — no data is sent to any third party.
   277→          </p>
   278→        </div>
   279→      </CardContent>
   280→    </Card>
   281→  );
   282→}

   286→function TabBar({ active, onChange }: { active: JellyfinTab; onChange: (tab: JellyfinTab) => void }) {
   287→  const { totalArtists, totalAlbums, totalTracks, playlists, totalPodcastShows } = useJellyfinStore();
   288→  const tabs: { key: JellyfinTab; label: string; icon: typeof Album; count?: number }[] = [
   289→    { key: 'recent', label: 'Recent', icon: Clock },
   290→    { key: 'albums', label: 'Albums', icon: Disc3, count: totalAlbums },
   291→    { key: 'artists', label: 'Artists', icon: User, count: totalArtists },
   292→    { key: 'tracks', label: 'Tracks', icon: Music, count: totalTracks },
   293→    { key: 'podcasts', label: 'Podcasts', icon: Podcast, count: totalPodcastShows },
   294→    { key: 'playlists', label: 'Playlists', icon: ListMusic, count: playlists.length },
   295→  ];
   296→
   297→  return (
   298→    <div className="flex gap-1 mb-6 bg-surface/30 p-1 rounded-lg">
   299→      {tabs.map(tab => (
   300→        <button
   301→          key={tab.key}
   302→          onClick={() => onChange(tab.key)}
   303→          className={`flex items-center gap-1.5 px-3 py-2 rounded-md text-xs font-medium transition-all flex-1 justify-center ${
   304→            active === tab.key
   305→              ? 'bg-primary text-primary-foreground shadow-sm'
   306→              : 'text-muted-foreground hover:text-foreground hover:bg-surface/60'
   307→          }`}
   308→        >
   309→          <tab.icon className="w-3.5 h-3.5" />
   310→          <span className="hidden sm:inline">{tab.label}</span>
   311→          {tab.count !== undefined && tab.count > 0 && (
   312→            <span className={`${active === tab.key ? 'text-primary-foreground/70' : 'text-muted-foreground/60'}`}>
   313→              ({tab.count > 999 ? `${(tab.count / 1000).toFixed(1)}k` : tab.count})
   314→            </span>
   315→          )}
   316→        </button>
   317→      ))}
   318→    </div>
   319→  );
   320→}

   374→function TrackRow({ track, trackNumber, onPlay, isPlaying, isCurrent }: { track: JellyfinTrack; trackNumber: number; onPlay: () => void; isPlaying: boolean; isCurrent: boolean }) {

   376→    <div

   384→        ) : isCurrent ? (
   385→          <Play className="w-3.5 h-3.5 ml-auto text-primary" fill="currentColor" />
   386→        ) : (
   387→          trackNumber
   388→        )}
   389→      </span>
   390→      <CoverArt url={track.imageUrl} name={track.name} size="sm" rounded="sm" />
   391→      <div className="flex-1 min-w-0">
   392→        <p className={`text-sm truncate ${isCurrent ? 'font-medium text-primary' : ''}`}>{track.name}</p>
   393→        <p className="text-xs text-muted-foreground truncate">
   394→          {track.artistName} · {track.albumName}
   395→        </p>
   396→      </div>
   397→      <div className="hidden sm:flex items-center gap-3 text-xs text-muted-foreground/70 flex-shrink-0">
   398→        {track.format && <Badge variant="secondary" className="text-[9px] px-1.5 py-0 font-mono uppercase">{track.format}</Badge>}
   399→        {track.sampleRate > 0 && <span>{(track.sampleRate / 1000).toFixed(0)}kHz</span>}
   400→        {track.bitDepth > 0 && <span>{track.bitDepth}bit</span>}
   401→      </div>
   402→      <span className="text-xs text-muted-foreground/70 tabular-nums w-10 text-right flex-shrink-0">
   403→        {track.duration > 0 ? formatDuration(track.duration) : '--:--'}
   404→      </span>
   405→      <button className="opacity-0 group-hover:opacity-100 transition-opacity flex-shrink-0" onClick={onPlay}>
   406→        {isCurrent && isPlaying ? (
   407→          <Pause className="w-4 h-4 text-primary" />
   408→        ) : (
   409→          <Play className="w-4 h-4 text-primary" fill="currentColor" />
   410→        )}
   411→      </button>
   412→    </div>
   413→  );
   414→}

   471→      }`}
   472→    >
   473→      <span className={`w-8 text-xs text-right tabular-nums flex-shrink-0 ${isCurrent ? 'text-primary font-semibold' : 'text-muted-foreground'}`}>
   474→        {isCurrent && isPlaying ? (
   475→          <Volume2 className="w-3.5 h-3.5 ml-auto text-primary animate-pulse" />
   476→        ) : isCurrent ? (
   477→          <Play className="w-3.5 h-3.5 ml-auto text-primary" fill="currentColor" />
   478→        ) : (
   479→          <span>{index}</span>
   480→        )}
   481→      </span>
   482→      <CoverArt url={episode.imageUrl} name={episode.showName} size="sm" rounded="sm" />
   483→      <div className="flex-1 min-w-0">
   484→        <p className={`text-sm truncate ${isCurrent ? 'font-medium text-primary' : ''}`}>{episode.name}</p>
   485→        <p className="text-xs text-muted-foreground truncate">{episode.showName}</p>
   486→      </div>
   487→      <div className="hidden sm:flex items-center gap-3 text-xs text-muted-foreground/70 flex-shrink-0">
   488→        {episode.releaseDate && (
   489→          <span>{new Date(episode.releaseDate).toLocaleDateString()}</span>
   490→        )}
   491→        <span className="tabular-nums">
   492→          {episode.duration > 0 ? formatDuration(episode.duration) : '--:--'}
   493→        </span>
   494→      </div>
   495→      <button className="opacity-0 group-hover:opacity-100 transition-opacity flex-shrink-0" onClick={onPlay}>
   496→        {isCurrent && isPlaying ? (
   497→          <Pause className="w-4 h-4 text-primary" />
   498→        ) : (
   499→          <Play className="w-4 h-4 text-primary" fill="currentColor" />
   500→        )}
   501→      </button>
   502→    </div>
   503→  );
   504→}

   508→function AlbumsGrid({ albums, onAlbumClick, onAlbumPlay }: { albums: JellyfinAlbum[]; onAlbumClick: (album: JellyfinAlbum) => void; onAlbumPlay: (album: JellyfinAlbum) => void }) {
   509→  if (albums.length === 0) {
   510→    return (
   511→      <div className="text-center py-12">
   512→        <Disc3 className="w-10 h-10 text-muted-foreground/30 mx-auto mb-3" />
   513→        <p className="text-sm text-muted-foreground">No albums found</p>
   514→      </div>
   515→    );
   516→  }
   517→
   518→  return (
   519→    <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6 gap-4">
   520→      {albums.map(album => (
   521→        <AlbumCard
   522→          key={album.id}
   523→          album={album}
   524→          onPlay={() => onAlbumPlay(album)}
   525→          onAlbumClick={() => onAlbumClick(album)}
   526→        />
   527→      ))}
   528→    </div>
   529→  );
   530→}

   544→// ═══════════════════════════════════════════════════════════
   545→// MAIN JELLYFIN VIEW
   546→// ═══════════════════════════════════════════════════════════
   547→
   548→export function JellyfinView() {
   549→  const { navigate } = useUIStore();
   550→  const { setQueue, play, currentTrack, isPlaying } = usePlayerStore();
   551→  const store = useJellyfinStore();
   552→
   553→  const [activeTab, setActiveTab] = useState<JellyfinTab>('recent');
   554→  const [viewMode, setViewMode] = useState<ViewMode>('grid');
   555→  const [searchQuery, setSearchQuery] = useState('');
   556→  const [selectedPodcastShow, setSelectedPodcastShow] = useState<string | null>(null);
   557→
   558→  const isConnected = store.connectionStatus === 'connected';
   559→  const { isLoadingAlbums, isLoadingArtists, isLoadingTracks, isLoadingRecent, isLoadingPodcasts, isLoadingAllAlbums, isLoadingAllArtists } = store;
   560→
   561→  // ── Data Fetching ──
   562→  useEffect(() => {
   563→    if (!isConnected) return;
   564→    switch (activeTab) {
   565→      case 'albums': store.fetchAllAlbums(); break;
   566→      case 'artists': store.fetchAllArtists(); break;
   567→      case 'tracks': store.fetchAllAlbums(); break; // tracks shown within albums
   568→      case 'playlists': store.fetchPlaylists(); break;
   569→      case 'podcasts': store.fetchPodcasts(); break;
   570→      case 'recent': store.fetchRecentAlbums(); break;
   571→    }
   572→  }, [activeTab, isConnected]);
   573→
   574→  // ── Player Actions ──
   575→  const handlePlayAlbum = useCallback(async (album: JellyfinAlbum) => {
   576→    if (!isConnected) return;
   577→    await store.fetchAlbumTracks(album.id);
   578→    const tracks = useJellyfinStore.getState().currentAlbumTracks;
   579→    if (tracks.length === 0) return;
   580→    const dspTracks = tracks.map(t => store.convertToTrack(t));
   581→    setQueue(dspTracks, 0);
   582→  }, [isConnected, setQueue]);
   583→
   584→  const handlePlayTrack = useCallback((track: JellyfinTrack) => {
   585→    const dspTrack = store.convertToTrack(track);
   586→    play(dspTrack);
   587→  }, [play]);
   588→
   589→  const handleAlbumClick = useCallback((album: JellyfinAlbum) => {
   590→    navigate('jellyfin-album', { albumId: album.id });
   591→  }, [navigate]);
   592→
   593→  const handleArtistClick = useCallback((artist: JellyfinArtist) => {
   594→    navigate('jellyfin-artist', { artistId: artist.id });
   595→  }, [navigate]);
   596→
   597→  const handlePlaylistClick = useCallback((playlist: { id: string }) => {
   598→    navigate('jellyfin-playlist', { playlistId: playlist.id });
   599→  }, [navigate]);
   600→
   601→  const handlePodcastShowClick = useCallback(async (showId: string) => {
   602→    setSelectedPodcastShow(showId);
   603→    await store.fetchPodcastEpisodes(showId);
   604→  }, [store]);
   605→
   606→  const handlePodcastBack = useCallback(() => {
   607→    setSelectedPodcastShow(null);
   608→  }, []);
   609→
   610→  const handlePlayPodcastEpisode = useCallback((episode: JellyfinPodcastEpisode) => {
   611→    const config = store.config;
   612→    if (!config) return;
   613→    const baseUrl = config.serverUrl.replace(/\/+$/, '');
   614→    const streamUrl = `${baseUrl}/Audio/${episode.id}/stream?static=true&api_key=${config.accessToken}&UserId=${config.userId}`;
   615→
   616→    const dspTrack: Track = {
   617→      id: `jf-${episode.id}`,
   618→      title: episode.name,
   619→      albumId: `jf-${episode.showId}`,
   620→      albumName: episode.showName,
   621→      artistId: '',
   622→      artistName: '',
   623→      trackNumber: 0,
   624→      discNumber: 1,
   625→      duration: episode.duration,
   626→      format: '',
   627→      bitDepth: 0,
   628→      sampleRate: 0,
   629→      channels: 0,
   630→      bitrate: 0,
   631→      filePath: streamUrl,
   632→      fileSize: 0,
   633→      composers: [],
   634→      performers: [],
   635→      genre: '',
   636→      loved: episode.isFavorite,
   637→      playCount: episode.playCount,
   638→      source: 'local',
   639→      isAvailable: true,
   640→    };
   641→    play(dspTrack);
   642→  }, [store.config, play]);
   643→
   644→  const handlePlayAllPodcastEpisodes = useCallback((showId: string) => {
   645→    const episodes = store.podcastEpisodes;
   646→    if (episodes.length === 0) return;
   647→    const config = store.config;
   648→    if (!config) return;
   649→    const baseUrl = config.serverUrl.replace(/\/+$/, '');
   650→
   651→    const dspTracks = episodes.map((episode, index) => {
   652→      const streamUrl = `${baseUrl}/Audio/${episode.id}/stream?static=true&api_key=${config.accessToken}&UserId=${config.userId}`;
   653→      return {
   654→        id: `jf-${episode.id}`,
   655→        title: episode.name,
   656→        albumId: `jf-${episode.showId}`,
   657→        albumName: episode.showName,
   658→        artistId: '',
   659→        artistName: '',
   660→        trackNumber: index + 1,
   661→        discNumber: 1,
   662→        duration: episode.duration,
   663→        format: '',
   664→        bitDepth: 0,
   665→        sampleRate: 0,
   666→        channels: 0,
   667→        bitrate: 0,
   668→        filePath: streamUrl,
   669→        fileSize: 0,
   670→        composers: [],
   671→        performers: [],
   672→        genre: '',
   673→        loved: episode.isFavorite,
   674→        playCount: episode.playCount,
   675→        source: 'local',
   676→        isAvailable: true,
   677→      } as Track;
   678→    });
   679→    setQueue(dspTracks, 0);
   680→  }, [store.config, store.podcastEpisodes, setQueue]);
   681→
   682→  const handleLoadMore = useCallback(() => {
   683→    if (activeTab === 'albums') store.loadMoreAlbums();
   684→    else if (activeTab === 'artists') store.loadMoreArtists();
   685→  }, [activeTab]);
   686→
   687→  // ── Filter ──
   688→  const filteredAlbums = useMemo(() => {
   689→    if (!searchQuery) return store.albums;
   690→    const q = searchQuery.toLowerCase();
   691→    return store.albums.filter(a =>
   692→      a.name.toLowerCase().includes(q) ||
   693→      a.artistName.toLowerCase().includes(q) ||
   694→      (a.genre && a.genre.toLowerCase().includes(q))
   695→    );
   696→  }, [store.albums, searchQuery]);
   697→
   698→  const filteredArtists = useMemo(() => {
   699→    if (!searchQuery) return store.artists;
   700→    const q = searchQuery.toLowerCase();
   701→    return store.artists.filter(a => a.name.toLowerCase().includes(q));
   702→  }, [store.artists, searchQuery]);
   703→
   704→  // ── Current track check ──
   705→  const isTrackPlaying = useCallback((track: JellyfinTrack) => {
   706→    return currentTrack?.id === `jf-${track.id}`;
   707→  }, [currentTrack]);
   708→
   709→  const isEpisodePlaying = useCallback((episode: JellyfinPodcastEpisode) => {
   710→    return currentTrack?.id === `jf-${episode.id}`;
   711→  }, [currentTrack]);
   712→
   713→  // ── Loading indicator ──
   714→  const isBulkLoading = isLoadingAllAlbums || isLoadingAllArtists;
   715→
   716→  // ── Render Tab Content ──
   717→  const renderContent = () => {
   718→    const loading = isLoadingAlbums || isLoadingArtists || isLoadingTracks || isLoadingRecent || isLoadingPodcasts;
   719→
   720→    if (loading && store.albums.length === 0 && store.artists.length === 0 && store.recentAlbums.length === 0 && store.podcastShows.length === 0) {
   721→      return (
   722→        <div className="flex items-center justify-center py-20">
   723→          <Loader2 className="w-8 h-8 text-primary animate-spin" />
   724→          <span className="ml-3 text-sm text-muted-foreground">Loading library...</span>
   725→        </div>
   726→      );
   727→    }
   728→
   729→    switch (activeTab) {
   730→      case 'recent':
   731→        return (
   732→          <>
   733→            {store.recentAlbums.length === 0 ? (
   734→              <EmptyState icon={Clock} title="No recent additions" description="Recently added albums from your Jellyfin library will appear here" />
   735→            ) : (
   736→              <AlbumsGrid albums={store.recentAlbums} onAlbumClick={handleAlbumClick} onAlbumPlay={handlePlayAlbum} />
   737→            )}
   738→          </>
   739→        );
   740→
   741→      case 'albums':
   742→        return (
   743→          <>
   744→            {isBulkLoading && store.albums.length > 0 && (
   745→              <div className="flex items-center gap-2 mb-4 px-1">
   746→                <Loader2 className="w-4 h-4 text-primary animate-spin" />
   747→                <span className="text-xs text-muted-foreground">
   748→                  Loading albums... {store.albums.length.toLocaleString()} of {store.totalAlbums.toLocaleString()}
   749→                </span>
   750→              </div>
   751→            )}
   752→            {filteredAlbums.length === 0 && !isBulkLoading ? (
   753→              <EmptyState icon={Disc3} title="No albums" description="Browse your Jellyfin music library by album" />
   754→            ) : filteredAlbums.length > 0 ? (
   755→              <AlbumsGrid albums={filteredAlbums} onAlbumClick={handleAlbumClick} onAlbumPlay={handlePlayAlbum} />
   756→            ) : null}
   757→          </>
   758→        );
   759→
   760→      case 'artists':
   761→        return (
   762→          <>
   763→            {isBulkLoading && store.artists.length > 0 && (
   764→              <div className="flex items-center gap-2 mb-4 px-1">
   765→                <Loader2 className="w-4 h-4 text-primary animate-spin" />
   766→                <span className="text-xs text-muted-foreground">
   767→                  Loading artists... {store.artists.length.toLocaleString()} of {store.totalArtists.toLocaleString()}
   768→                </span>
   769→              </div>
   770→            )}
   771→            {filteredArtists.length === 0 && !isBulkLoading ? (
   772→              <EmptyState icon={User} title="No artists" description="Album artists from your Jellyfin library will appear here" />
   773→            ) : filteredArtists.length > 0 ? (
   774→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-1">
   775→                {filteredArtists.map(artist => (
   776→                  <ArtistRow key={artist.id} artist={artist} onClick={() => handleArtistClick(artist)} />
   777→                ))}
   778→              </div>
   779→            ) : null}
   780→          </>
   781→        );
   782→
   783→      case 'tracks':
   784→        return (
   785→          <>
   786→            {filteredAlbums.length === 0 ? (
   787→              <EmptyState icon={Music} title="No tracks" description="Browse your Jellyfin tracks by album" />
   788→            ) : (
   789→              <div className="space-y-6">
   790→                <p className="text-xs text-muted-foreground -mt-2">Tracks are organized by album. Click an album to view its tracks.</p>
   791→                <AlbumsGrid albums={filteredAlbums} onAlbumClick={handleAlbumClick} onAlbumPlay={handlePlayAlbum} />
   792→              </div>
   793→            )}
   794→          </>
   795→        );
   796→
   797→      case 'podcasts':
   798→        if (selectedPodcastShow) {
   799→          const show = store.podcastShows.find(s => s.id === selectedPodcastShow);
   800→          return (
   801→            <div>
   802→              <button
   803→                onClick={handlePodcastBack}
   804→                className="flex items-center gap-1.5 text-sm text-muted-foreground hover:text-foreground transition-colors mb-4"
   805→              >
   806→                <ArrowLeft className="w-4 h-4" /> Back to Podcasts
   807→              </button>
   808→              {show && (
   809→                <div className="flex items-center gap-3 mb-4">
   810→                  <CoverArt url={show.imageUrl} name={show.name} size="lg" rounded="lg" />
   811→                  <div className="flex-1 min-w-0">
   812→                    <h2 className="text-lg font-semibold truncate">{show.name}</h2>
   813→                    <p className="text-xs text-muted-foreground mt-0.5">{store.podcastEpisodes.length} episode{store.podcastEpisodes.length !== 1 ? 's' : ''}</p>
   814→                    {show.genres.length > 0 && (
   815→                      <div className="flex gap-1.5 mt-1.5">
   816→                        {show.genres.slice(0, 3).map(g => (
   817→                          <Badge key={g} variant="secondary" className="text-[9px] px-1.5 py-0">{g}</Badge>
   818→                        ))}
   819→                      </div>
   820→                    )}
   821→                  </div>
   822→                  {store.podcastEpisodes.length > 0 && (
   823→                    <Button
   824→                      size="sm"
   825→                      className="h-8 gap-1.5 flex-shrink-0"
   826→                      onClick={() => handlePlayAllPodcastEpisodes(selectedPodcastShow)}
   827→                    >
   828→                      <Play className="w-3.5 h-3.5" fill="currentColor" /> Play All
   829→                    </Button>
   830→                  )}
   831→                </div>
   832→              )}
   833→              {isLoadingTracks ? (
   834→                <div className="flex items-center justify-center py-12">
   835→                  <Loader2 className="w-6 h-6 text-primary animate-spin" />
   836→                  <span className="ml-2 text-sm text-muted-foreground">Loading episodes...</span>
   837→                </div>
   838→              ) : store.podcastEpisodes.length === 0 ? (
   839→                <EmptyState icon={Podcast} title="No episodes" description="No episodes found for this podcast" />
   840→              ) : (
   841→                <div className="space-y-1">
   842→                  {store.podcastEpisodes.map((episode, index) => (
   843→                    <PodcastEpisodeRow
   844→                      key={episode.id}
   845→                      episode={episode}
   846→                      index={index + 1}
   847→                      onPlay={() => handlePlayPodcastEpisode(episode)}
   848→                      isPlaying={isPlaying}
   849→                      isCurrent={isEpisodePlaying(episode)}
   850→                    />
   851→                  ))}
   852→                </div>
   853→              )}
   854→            </div>
   855→          );
   856→        }
   857→        return (
   858→          <>
   859→            {store.podcastShows.length === 0 && !isLoadingPodcasts ? (
   860→              <EmptyState icon={Podcast} title="No podcasts" description="Podcasts from your Jellyfin library will appear here" />
   861→            ) : isLoadingPodcasts ? (
   862→              <div className="flex items-center justify-center py-12">
   863→                <Loader2 className="w-6 h-6 text-primary animate-spin" />
   864→                <span className="ml-2 text-sm text-muted-foreground">Loading podcasts...</span>
   865→              </div>
   866→            ) : (
   867→              <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6 gap-4">
   868→                {store.podcastShows.map(show => (
   869→                  <PodcastShowCard key={show.id} show={show} onClick={() => handlePodcastShowClick(show.id)} />
   870→                ))}
   871→              </div>
   872→            )}
   873→          </>
   874→        );
   875→
   876→      case 'playlists':
   877→        return (
   878→          <>
   879→            {store.playlists.length === 0 ? (
   880→              <EmptyState icon={ListMusic} title="No playlists" description="Playlists from your Jellyfin library will appear here" />
   881→            ) : (
   882→              <div className="space-y-1">
   883→                {store.playlists.map(pl => (
   884→                  <PlaylistRow key={pl.id} playlist={pl} onClick={() => handlePlaylistClick(pl)} />
   885→                ))}
   886→              </div>
   887→            )}
   888→          </>
   889→        );
   890→    }
   891→  };
   892→
   893→  return (
   894→    <ScrollArea className="h-full">
   895→      <div className="p-6 max-w-7xl mx-auto">
   896→        {/* ── Header ── */}
   897→        <div className="flex items-center justify-between mb-6">
   898→          <div>
   899→            <h1 className="text-2xl font-bold flex items-center gap-2">
   900→              <Server className="w-6 h-6 text-purple-400" />
   901→              Jellyfin
   902→            </h1>
   903→            <p className="text-sm text-muted-foreground mt-1">
   904→              {isConnected
   905→                ? `Browsing ${store.config?.serverName || 'Jellyfin'} library`
   906→                : 'Connect to your Jellyfin media server'}
   907→            </p>
   908→          </div>
   909→          {isConnected && (
   910→            <div className="flex items-center gap-2">
   911→              <Button
   912→                variant={viewMode === 'grid' ? 'default' : 'outline'}
   913→                size="sm"
   914→                className="h-7 text-xs"
   915→                onClick={() => setViewMode('grid')}
   916→              >
   917→                <Grid3X3 className="w-3.5 h-3.5" />
   918→              </Button>
   919→              <Button
   920→                variant={viewMode === 'list' ? 'default' : 'outline'}
   921→                size="sm"
   922→                className="h-7 text-xs"
   923→                onClick={() => setViewMode('list')}
   924→              >
   925→                <List className="w-3.5 h-3.5" />
   926→              </Button>
   927→            </div>
   928→          )}
   929→        </div>
   930→
   931→        {/* ── Connection Panel ── */}
   932→        <ConnectionPanel />
   933→
   934→        {/* ── Library Content (only when connected) ── */}
   935→        {isConnected && (
   936→          <>
   937→            {/* Search */}
   938→            <div className="relative mb-4">
   939→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   940→              <Input
   941→                placeholder="Search albums, artists..."
   942→                value={searchQuery}
   943→                onChange={(e) => setSearchQuery(e.target.value)}
   944→                className="pl-9 bg-card border-border max-w-md"
   945→              />
   946→            </div>
   947→
   948→            {/* Tabs */}
   949→            <TabBar active={activeTab} onChange={setActiveTab} />
   950→
   951→            {/* Content */}
   952→            {renderContent()}
   953→          </>
   954→        )}
   955→
   956→        {/* ── Info Card ── */}
   957→        <Separator className="my-8 bg-border" />
   958→        <Card className="bg-card border-border">
   959→          <CardContent className="p-4">
   960→            <h2 className="text-sm font-semibold mb-3 flex items-center gap-2">
   961→              <Radio className="w-4 h-4 text-purple-400" />
   962→              About Jellyfin Integration
   963→            </h2>
   964→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 text-xs text-muted-foreground">
   965→              <div className="flex items-start gap-2 p-2.5 rounded-lg bg-surface/50">
   966→                <Shield className="w-4 h-4 text-green-400 mt-0.5 flex-shrink-0" />
   967→                <div>
   968→                  <p className="font-medium text-foreground text-xs">Open Source &amp; Private</p>
   969→                  <p className="mt-0.5">Jellyfin is the free software media system. No tracking, no ads, no subscriptions. Your library stays on your server.</p>
   970→                </div>
   971→              </div>
   972→              <div className="flex items-start gap-2 p-2.5 rounded-lg bg-surface/50">
   973→                <Gauge className="w-4 h-4 text-blue-400 mt-0.5 flex-shrink-0" />
