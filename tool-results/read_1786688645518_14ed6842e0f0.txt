   148→   148→  const EQCurveSVG = () => {
   149→   149→    const bands = dspConfig.eq ?? [];
   150→   150→    const points = 48;
   151→   151→    const w = 480;
   152→   152→    const h = 100;
   153→   153→    const mid = h / 2;
   154→   154→
   155→   155→    const computeGain = (freq: number) => {
   156→   156→      let total = 0;
   157→   157→      for (const band of bands) {
   158→   158→        if (!band.enabled) continue;
   159→   159→        const diff = Math.log2(freq / band.frequency);
   160→   160→        const bw = band.q * diff;
   161→   161→        const response = 1 / (1 + bw * bw);
   162→   162→        if (band.type === 'high-shelf' && freq > band.frequency) {
   163→   163→          total += band.gain * response;
   164→   164→        } else if (band.type === 'low-shelf' && freq < band.frequency) {
   165→   165→          total += band.gain * response;
   166→   166→        } else if (band.type === 'parametric' || band.type === 'notch' || band.type === 'band-pass') {
   167→   167→          total += band.gain * response;
   168→   168→        }
   169→   169→      }
   170→   170→      return total;
   171→   171→    };
   172→   172→
   173→   173→    const pathParts: string[] = [];

   176→   176→      const freq = 20 * Math.pow(1000, t);
   177→   177→      const gain = computeGain(freq);
   178→   178→      const x = (i / points) * w;
   179→   179→      const y = mid - (gain / 24) * (mid - 8);
   180→   180→      pathParts.push(`${i === 0 ? 'M' : 'L'} ${x.toFixed(1)} ${y.toFixed(1)}`);
   181→   181→    }
   182→   182→
   183→   183→    return (
   184→   184→      <svg viewBox={`0 0 ${w} ${h}`} className="w-full h-24 opacity-40" preserveAspectRatio="none">

   186→   186→          <linearGradient id="eqGrad" x1="0" y1="0" x2="0" y2="1">
   187→   187→            <stop offset="0%" stopColor="currentColor" stopOpacity="0.3" />
   188→   188→            <stop offset="100%" stopColor="currentColor" stopOpacity="0" />
   189→   189→          </linearGradient>
   190→   190→        </defs>
   191→   191→        {/* Grid lines */}
   192→   192→        <line x1={0} y1={mid} x2={w} y2={mid} stroke="currentColor" strokeOpacity="0.15" strokeWidth="0.5" />
   193→   193→        <line x1={0} y1={mid - (mid - 8)} x2={w} y2={mid - (mid - 8)} stroke="currentColor" strokeOpacity="0.08" strokeWidth="0.5" />

   195→   195→        {/* Filled area */}
   196→   196→        <path d={`${pathParts.join(' ')} L ${w} ${mid} L 0 ${mid} Z`} fill="url(#eqGrad)" className="text-primary" />
   197→   197→        {/* Curve line */}
   198→   198→        <path d={pathParts.join(' ')} fill="none" stroke="currentColor" strokeWidth="1.5" className="text-primary" />
   199→   199→        {/* Frequency labels */}
   200→   200→        <text x={0} y={h - 2} fill="currentColor" className="text-muted-foreground" fontSize="8" fontFamily="monospace">20</text>
   201→   201→        <text x={w * 0.21} y={h - 2} fill="currentColor" className="text-muted-foreground" fontSize="8" fontFamily="monospace">100</text>
   202→   202→        <text x={w * 0.43} y={h - 2} fill="currentColor" className="text-muted-foreground" fontSize="8" fontFamily="monospace">1k</text>
   203→   203→        <text x={w * 0.64} y={h - 2} fill="currentColor" className="text-muted-foreground" fontSize="8" fontFamily="monospace">10k</text>
   204→   204→        <text x={w * 0.88} y={h - 2} fill="currentColor" className="text-muted-foreground" fontSize="8" fontFamily="monospace">20k</text>
   205→   205→      </svg>
   206→   206→    );

   380→   380→            </button>
   381→   381→
   382→   382→            {expandedSections.eq && (
   383→   383→              <div className="mt-4 space-y-4">
   384→   384→                {/* EQ Curve Visualization */}
   385→   385→                <div className="rounded-lg bg-surface/50 p-3">

   390→   390→
   391→   391→                {/* EQ Bands */}
   392→   392→                <div className="space-y-3">
   393→   393→                  {(dspConfig.eq ?? []).map((band, idx) => (
