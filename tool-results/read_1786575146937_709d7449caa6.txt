   617→    s.genre.toLowerCase().includes(q) ||
   618→    s.country.toLowerCase().includes(q) ||
   619→    s.tags.some(t => t.includes(q))
   620→  );

   629→  const map = new Map<string, string>();

   640→export function toggleRadioFavorite(stationId: string): void {

   648→  if (station) station.isFavorite = _favorites.has(stationId);
