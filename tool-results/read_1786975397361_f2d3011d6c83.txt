   109→/** Map a Jellyfin Container string to a human-readable format name */
   110→function getFormatName(container?: string): string {
   111→  if (!container) return '';
   112→  const lower = container.toLowerCase();
   113→  const map: Record<string, string> = {
   114→    flac: 'FLAC',
   115→    mp3: 'MP3',
   116→    wav: 'WAV',
   117→    m4a: 'AAC',
   118→    aac: 'AAC',
   119→    ogg: 'OGG',
   120→    opus: 'OPUS',
   121→    wma: 'WMA',
   122→    aiff: 'AIFF',
   123→    dsd: 'DSD',
   124→    dsf: 'DSD',
   125→   dff: 'DSD',
   126→  };
   127→  return map[lower] || upperFirst(lower);
   128→}
   129→
   130→function upperFirst(s: string): string {
   131→  return s.charAt(0).toUpperCase() + s.slice(1);
   132→}

   134→/** Safely get an image URL from the Jellyfin client, returning an empty string on failure */
   135→function safeGetImageUrl(
   136→  itemId: string,
   137→  maxWidth?: number,
   138→  maxHeight?: number
   139→): string {
   140→  try {
   141→    return jellyfinClient.getImageUrl(itemId, 'Primary', maxWidth, maxHeight);
   142→  } catch {
   143→    return '';
   144→  }
   145→}

   147→/** Map a JellyfinItem (MusicArtist) → JellyfinArtist */
   148→function mapArtist(item: JellyfinItem): JellyfinArtist {
   149→  return {
   150→    id: item.Id,
   151→    name: item.Name,
   152→    imageUrl: safeGetImageUrl(item.Id, 300, 300),
   153→    albumCount: item.ChildCount ?? 0,
   154→  };
   155→}
   156→
   157→/** Map a JellyfinItem (MusicAlbum) → JellyfinAlbum */
   158→function mapAlbum(item: JellyfinItem): JellyfinAlbum {
   159→  const artistObj = item.AlbumArtists?.[0];
   160→  return {
   161→    id: item.Id,
   162→    name: item.Name,
   163→    artistName: artistObj?.Name ?? item.AlbumArtist ?? 'Unknown Artist',
   164→    artistId: artistObj?.Id ?? '',
   165→    year: item.ProductionYear ?? 0,
   166→    genre: item.Genres?.[0] ?? '',
   167→    imageUrl: safeGetImageUrl(item.Id, 300, 300),
   168→    duration: ticksToSeconds(item.RunTimeTicks),
   169→    trackCount: item.ChildCount ?? 0,
   170→    communityRating: item.CommunityRating ?? 0,
   171→    isFavorite: item.UserData?.IsFavorite ?? false,
   172→    overview: item.Overview,
   173→  };
   174→}
   175→
   176→/** Map a JellyfinItem (Audio) → JellyfinTrack */
   177→function mapTrack(item: JellyfinItem): JellyfinTrack {
   178→  const artistObj = item.AlbumArtists?.[0];
   179→  const albumArtist = artistObj?.Name ?? item.Artists?.[0] ?? 'Unknown Artist';
   180→  const albumArtistId = artistObj?.Id ?? '';
   181→
   182→  return {
   183→    id: item.Id,
   184→    name: item.Name,
   185→    albumId: item.AlbumId ?? item.ParentId ?? '',
   186→    albumName: '', // populated from parent context when available
   187→    artistName: albumArtist,
   188→    artistId: albumArtistId,
   189→    trackNumber: 0, // Jellyfin SortName is typically "disc-track name"; use index from list
   190→    duration: ticksToSeconds(item.RunTimeTicks),
   191→    format: getFormatName(item.Container),
   192→    bitrate: item.BitRate ?? 0,
   193→    sampleRate: item.SampleRate ?? 0,
   194→    bitDepth: item.BitsPerSample ?? 0,
   195→    channels: item.Channels ?? 0,
   196→    imageUrl: safeGetImageUrl(item.AlbumId ?? item.ParentId ?? item.Id, 300, 300),
   197→    isFavorite: item.UserData?.IsFavorite ?? false,
   198→    playCount: item.UserData?.PlayCount ?? 0,
   199→  };
   200→}

   217→/** Build a static Jellyfin stream URL for a given item ID (synchronous) */
   218→function buildStreamUrl(itemId: string, config: JellyfinConfig): string {
   219→  const baseUrl = config.serverUrl.replace(/\/+$/, '');
   220→  return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${config.accessToken}&UserId=${config.userId}`;
   221→}

   252→/** Fetch the total track count and update the store (fire-and-forget helper) */
   253→async function refreshTotalTrackCount(): Promise<void> {
   254→  try {
   255→    const state = useJellyfinStore.getState();
   256→    if (state.connectionStatus !== 'connected') return;
   257→    const response = await jellyfinClient.getItems({
   258→      includeItemTypes: ['Audio'],
   259→      recursive: true,
   260→      limit: 1,
   261→    });
   262→    useJellyfinStore.setState({ totalTracks: response.TotalRecordCount });
   263→  } catch {
   264→    // Non-critical; ignore
   265→  }
   266→}

   344→export const useJellyfinStore = create<JellyfinState>((set, get) => ({
   345→  // ── Initial State ──
   346→
   347→  connectionStatus:
   348→    typeof window !== 'undefined' && jellyfinClient.isAuthenticated()
   349→      ? 'connected'
   350→      : 'disconnected',
   351→  config: typeof window !== 'undefined' ? jellyfinClient.getConfig() : null,
   352→  error: null,
   353→
   354→  // Browse cache
   355→  artists: [],
   356→  albums: [],
   357→  recentAlbums: [],
   358→  playlists: [],
   359→  podcastShows: [],
   360→  podcastEpisodes: [],
   361→  totalArtists: 0,
   362→  totalAlbums: 0,
   363→  totalTracks: 0,
   364→  totalPodcastShows: 0,
   365→
   366→  // Detail cache
   367→  currentArtistAlbums: [],
   368→  currentArtistTracks: [],
   369→  currentAlbumTracks: [],
   370→  currentPlaylistTracks: [],
   371→
   372→  // Loading states
   373→  isLoadingArtists: false,
   374→  isLoadingAlbums: false,
   375→  isLoadingTracks: false,
   376→  isLoadingRecent: false,
   377→  isLoadingPodcasts: false,
   378→  isLoadingAllAlbums: false,
   379→  isLoadingAllArtists: false,
   380→
   381→  // Pagination
   382→  artistsPage: 0,
   383→  albumsPage: 0,
   384→
   385→  // ═══════════════════════════════════════════════════════
   386→  // Connection
   387→  // ═══════════════════════════════════════════════════════
   388→
   389→  connect: async (serverUrl, username, password) => {
   390→    set({ connectionStatus: 'connecting', error: null });
   391→    try {
   392→      const config = await jellyfinClient.connect(serverUrl, username, password);
   393→      set({
   394→        connectionStatus: 'connected',
   395→        config,
   396→        error: null,
   397→      });
   398→      // Auto-fetch recent albums after successful connection
   399→      get().fetchRecentAlbums();
   400→      return true;
   401→    } catch (err) {
   402→      let message: string;
   403→      if (err instanceof JellyfinApiError) {
   404→        if (err.statusCode === 0) {
   405→          message = 'Could not reach the Jellyfin server. Check the server URL and ensure the server is running and accessible.';
   406→        } else if (err.statusCode === 401) {
   407→          message = 'Authentication failed. Please check your username and password.';
   408→        } else if (err.statusCode === 404) {
   409→          message = 'Jellyfin server found but the API endpoint was not available. Ensure you are running Jellyfin 10.7+.';
   410→        } else {
   411→          message = err.message;
   412→        }
   413→      } else if (err instanceof Error) {
   414→        message = err.message;
   415→      } else {
   416→        message = 'Connection failed';
   417→      }
   418→      set({ connectionStatus: 'error', error: message });
   419→      return false;
   420→    }
   421→  },
   422→
   423→  connectWithApiKey: async (serverUrl, apiKey) => {
   424→    set({ connectionStatus: 'connecting', error: null });
   425→    try {
   426→      const config = await jellyfinClient.connectWithApiKey(serverUrl, apiKey);
   427→      set({
   428→        connectionStatus: 'connected',
   429→        config,
   430→        error: null,
   431→      });
   432→      get().fetchRecentAlbums();
   433→      return true;
   434→    } catch (err) {
   435→      let message: string;
   436→      if (err instanceof JellyfinApiError) {
   437→        if (err.statusCode === 0) {
   438→          message = 'Could not reach the Jellyfin server. Check the server URL and ensure the server is running and accessible.';
   439→        } else if (err.statusCode === 401) {
   440→          message = 'API key authentication failed. Please check your API key.';
   441→        } else {
   442→          message = err.message;
   443→        }
   444→      } else if (err instanceof Error) {
   445→        message = err.message;
   446→      } else {
   447→        message = 'Connection failed';
   448→      }
   449→      set({ connectionStatus: 'error', error: message });
   450→      return false;
   451→    }
   452→  },
   453→
   454→  disconnect: async () => {
   455→    try {
   456→      await jellyfinClient.disconnect();
   457→    } catch {
   458→      // Best-effort disconnect
   459→    }
   460→    set({
   461→      connectionStatus: 'disconnected',
   462→      config: null,
   463→      error: null,
   464→      artists: [],
   465→      albums: [],
   466→      recentAlbums: [],
   467→      playlists: [],
   468→      podcastShows: [],
   469→      podcastEpisodes: [],
   470→      totalArtists: 0,
   471→      totalAlbums: 0,
   472→      totalTracks: 0,
   473→      totalPodcastShows: 0,
   474→      currentArtistAlbums: [],
   475→      currentArtistTracks: [],
   476→      currentAlbumTracks: [],
   477→      currentPlaylistTracks: [],
   478→      isLoadingArtists: false,
   479→      isLoadingAlbums: false,
   480→      isLoadingTracks: false,
   481→      isLoadingRecent: false,
   482→      isLoadingPodcasts: false,
   483→      isLoadingAllAlbums: false,
   484→      isLoadingAllArtists: false,
   485→      artistsPage: 0,
   486→      albumsPage: 0,
   487→    });
   488→  },
   489→
   490→  clearError: () => set({ error: null }),
   491→
   492→  // ═══════════════════════════════════════════════════════
   493→  // Browse
   494→  // ═══════════════════════════════════════════════════════
   495→
   496→  fetchArtists: async (reset = false) => {
   497→    const { connectionStatus, artistsPage } = get();
   498→    if (connectionStatus !== 'connected') return;
   499→
   500→    const page = reset ? 0 : artistsPage;
   501→    set({ isLoadingArtists: true });
   502→
   503→    try {
   504→      const response = await jellyfinClient.getArtists({
   505→        limit: PAGE_SIZE,
   506→        startIndex: page * PAGE_SIZE,
   507→      });
   508→
   509→      const mapped = response.Items.map(mapArtist);
   510→      set({
   511→        artists: reset ? mapped : [...get().artists, ...mapped],
   512→        totalArtists: response.TotalRecordCount,
   513→        artistsPage: page + 1,
   514→        isLoadingArtists: false,
   515→      });
   516→
   517→      // Also fetch total track count in the background
   518→      if (get().totalTracks === 0) {
   519→        refreshTotalTrackCount();
   520→      }
   521→    } catch (err) {
   522→      const message =
   523→        err instanceof JellyfinApiError
   524→          ? err.message
   525→          : err instanceof Error
   526→            ? err.message
   527→            : 'Failed to fetch artists';
   528→      set({ isLoadingArtists: false, error: message });
   529→    }
   530→  },
   531→
   532→  fetchAlbums: async (reset = false) => {
   533→    const { connectionStatus, albumsPage } = get();
   534→    if (connectionStatus !== 'connected') return;
   535→
   536→    const page = reset ? 0 : albumsPage;
   537→    set({ isLoadingAlbums: true });
   538→
   539→    try {
   540→      const response = await jellyfinClient.getAlbums({
   541→        limit: PAGE_SIZE,
   542→        startIndex: page * PAGE_SIZE,
   543→      });
   544→
   545→      const mapped = response.Items.map(mapAlbum);
   546→      set({
   547→        albums: reset ? mapped : [...get().albums, ...mapped],
   548→        totalAlbums: response.TotalRecordCount,
   549→        albumsPage: page + 1,
   550→        isLoadingAlbums: false,
   551→      });
   552→
   553→      // Also fetch total track count in the background
   554→      if (get().totalTracks === 0) {
   555→        refreshTotalTrackCount();
   556→      }
   557→    } catch (err) {
   558→      const message =
   559→        err instanceof JellyfinApiError
   560→          ? err.message
   561→          : err instanceof Error
   562→            ? err.message
   563→            : 'Failed to fetch albums';
   564→      set({ isLoadingAlbums: false, error: message });
   565→    }
   566→  },
   567→
   568→  fetchRecentAlbums: async () => {
   569→    const { connectionStatus } = get();
   570→    if (connectionStatus !== 'connected') return;
   571→
   572→    set({ isLoadingRecent: true });
   573→
   574→    try {
   575→      const response = await jellyfinClient.getRecentItems(20);
   576→      // Recent items may include non-album types; filter to albums only
   577→      const albumItems = response.Items.filter(
   578→        item => item.Type === 'MusicAlbum'
   579→      );
   580→      const mapped = albumItems.map(mapAlbum);
   581→      set({ recentAlbums: mapped, isLoadingRecent: false });
   582→    } catch (err) {
   583→      const message =
   584→        err instanceof JellyfinApiError
   585→          ? err.message
   586→          : err instanceof Error
   587→            ? err.message
   588→            : 'Failed to fetch recent albums';
   589→      set({ isLoadingRecent: false, error: message });
   590→    }
   591→  },
   592→
   593→  fetchPlaylists: async () => {
   594→    const { connectionStatus } = get();
   595→    if (connectionStatus !== 'connected') return;
   596→
   597→    try {
   598→      const response = await jellyfinClient.getPlaylists();
   599→      const mapped = response.Items.map(mapPlaylist);
   600→      set({ playlists: mapped });
   601→    } catch (err) {
   602→      const message =
   603→        err instanceof JellyfinApiError
   604→          ? err.message
   605→          : err instanceof Error
   606→            ? err.message
   607→            : 'Failed to fetch playlists';
   608→      set({ error: message });
   609→    }
   610→  },
   611→
   612→  loadMoreArtists: async () => {
   613→    const { isLoadingArtists, artists, totalArtists } = get();
   614→    if (isLoadingArtists || artists.length >= totalArtists) return;
   615→    get().fetchArtists(false);
   616→  },
   617→
   618→  loadMoreAlbums: async () => {
   619→    const { isLoadingAlbums, albums, totalAlbums } = get();
   620→    if (isLoadingAlbums || albums.length >= totalAlbums) return;
   621→    get().fetchAlbums(false);
   622→  },
   623→
   624→  // ── Bulk Fetch All (auto-pagination) ──
   625→
   626→  fetchAllAlbums: async () => {
   627→    const { connectionStatus } = get();
   628→    if (connectionStatus !== 'connected') return;
   629→
   630→    set({ isLoadingAllAlbums: true });
   631→
   632→    try {

   652→            batch.map(page =>
   653→              jellyfinClient.getAlbums({ limit: PAGE_SIZE, startIndex: page * PAGE_SIZE })

   663→          const newAlbums = results.flat();
   664→          set(prev => ({
   665→            albums: [...prev.albums, ...newAlbums],
   666→            albumsPage: Math.max(batch[batch.length - 1] + 1, prev.albumsPage),
   667→          }));
   668→        }
   669→      }
   670→
   671→      // Also fetch total track count in the background
   672→      if (get().totalTracks === 0) {
   673→        refreshTotalTrackCount();
   674→      }
   675→

   679→        err instanceof JellyfinApiError

   963→  search: async (query: string) => {
   964→    const { connectionStatus } = get();
   965→    if (connectionStatus !== 'connected') {
   966→      return { artists: [], albums: [], tracks: [], playlists: [], podcasts: [], podcastEpisodes: [] };
   967→    }
   968→
   969→    try {
   970→      const response = await jellyfinClient.search(query);
   971→
   972→      const artists: JellyfinArtist[] = [];
   973→      const albums: JellyfinAlbum[] = [];
   974→      const tracks: JellyfinTrack[] = [];
   975→      const playlists: JellyfinPlaylist[] = [];
   976→      const podcasts: JellyfinPodcastShow[] = [];
   977→      const podcastEpisodes: JellyfinPodcastEpisode[] = [];
   978→
   979→      for (const item of response.Items) {
   980→        switch (item.Type) {
   981→          case 'MusicArtist':
   982→            artists.push(mapArtist(item));
   983→            break;
   984→          case 'MusicAlbum':
   985→            albums.push(mapAlbum(item));
   986→            break;
   987→          case 'Audio':
   988→            tracks.push(mapTrack(item));
   989→            break;
   990→          case 'Playlist':
   991→            playlists.push(mapPlaylist(item));
   992→            break;
   993→          case 'BoxSet':
   994→            // Jellyfin stores some podcasts as BoxSet type
   995→            if (item.Tags?.some(t => t.toLowerCase().includes('podcast')) ||
   996→                item.Genres?.some(g => g.toLowerCase().includes('podcast'))) {
   997→              podcasts.push(mapPodcastShow(item));
   998→            }
   999→            break;
  1000→        }
  1001→      }
  1002→
  1003→      return { artists, albums, tracks, playlists, podcasts, podcastEpisodes };
  1004→    } catch (err) {
  1005→      const message =
  1006→        err instanceof JellyfinApiError
  1007→          ? err.message
  1008→          : err instanceof Error
  1009→            ? err.message
  1010→            : 'Search failed';
  1011→      set({ error: message });
  1012→      return { artists: [], albums: [], tracks: [], playlists: [], podcasts: [], podcastEpisodes: [] };
  1013→    }
  1014→  },
