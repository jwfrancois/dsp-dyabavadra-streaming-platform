   178→   178→/** Build a query string from a record, filtering out undefined/null/empty values */
   179→   179→function buildQueryString(params: Record<string, string | number | boolean | undefined | null | string[] | number[]>): string {
   180→   180→  const parts: string[] = [];
   181→   181→  for (const [key, value] of Object.entries(params)) {
   182→   182→    if (value === undefined || value === null || value === '') continue;
   183→   183→    if (Array.isArray(value)) {
   184→   184→      if (value.length > 0) {
   185→   185→        parts.push(`${encodeURIComponent(key)}=${value.map(v => encodeURIComponent(String(v))).join(',')}`);
   186→   186→      }
   187→   187→    } else {
   188→   188→      parts.push(`${encodeURIComponent(key)}=${encodeURIComponent(String(value))}`);
   189→   189→    }
   190→   190→  }
   191→   191→  return parts.length > 0 ? `?${parts.join('&')}` : '';
   192→   192→}

   258→   258→    return {
   259→   259→      'X-Emby-Token': this.config.accessToken,

   264→   264→
   265→   265→  private async request<T>(path: string, options: RequestInit = {}, skipAuth = false): Promise<T> {
   266→   266→    const injectedBaseUrl = (options.headers as Record<string, string> | undefined)?.__baseUrl;
   267→   267→    const baseUrl = skipAuth
   268→   268→      ? normalizeServerUrl(injectedBaseUrl || '')
   269→   269→      : this.getBaseUrl();
   270→   270→    const url = `${baseUrl}${path}`;
   271→   271→
   272→   272→    // Build headers — extract __baseUrl before passing to fetch
   273→   273→    const externalHeaders = { ...(options.headers as Record<string, string> | undefined) };
   274→   274→    delete externalHeaders['__baseUrl'];
   275→   275→
   276→   276→    const headers: Record<string, string> = skipAuth
   277→   277→      ? externalHeaders
   278→   278→      : {
   279→   279→          ...this.getHeaders(),
   280→   280→          ...externalHeaders,
   281→   281→        };
   282→   282→
   283→   283→    // Remove Content-Type for GET/DELETE if no body
   284→   284→    if (!options.body && (options.method === 'GET' || options.method === 'DELETE')) {
   285→   285→      delete headers['Content-Type'];
   286→   286→    }
   287→   287→
   288→   288→    let response: Response;
   289→   289→    try {
   290→   290→      response = await fetch(url, {
   291→   291→        ...options,
   292→   292→        headers,
   293→   293→      });
   294→   294→    } catch (err) {
   295→   295→      throw new JellyfinApiError(
   296→   296→        `Network error: ${err instanceof Error ? err.message : 'Unknown error'}`,
   297→   297→        0,
   298→   298→        path,
   299→   299→        err
   300→   300→      );
   301→   301→    }
   302→   302→
   303→   303→    if (!response.ok) {
   304→   304→      let errorBody = '';
   305→   305→      try {
   306→   306→        errorBody = await response.text();
   307→   307→      } catch {
   308→   308→        // ignore
   309→   309→      }
   310→   310→      throw new JellyfinApiError(
   311→   311→        `Jellyfin API error ${response.status}: ${response.statusText} — ${errorBody}`,
   312→   312→        response.status,
   313→   313→        path
   314→   314→      );
   315→   315→    }
   316→   316→
   317→   317→    // Some endpoints return empty responses (e.g., DELETE, favorite toggles)
   318→   318→    const contentType = response.headers.get('content-type');
   319→   319→    if (contentType && contentType.includes('application/json')) {
   320→   320→      return response.json() as Promise<T>;
   321→   321→    }
   322→   322→
   323→   323→    return undefined as unknown as T;
   324→   324→  }

   330→   330→  /**
   331→   331→   * Connect to a Jellyfin server. Authenticates and fetches server info.
   332→   332→   * Persists the configuration to localStorage.
   333→   333→   */
   334→   334→  async connect(serverUrl: string, username: string, password: string): Promise<JellyfinConfig> {
   335→   335→    const baseUrl = normalizeServerUrl(serverUrl);
   336→   336→
   337→   337→    // Step 1: Authenticate (skip auth headers — we don't have a token yet)
   338→   338→    const authResponse = await this.request<JellyfinAuthResponse>(
   339→   339→      '/Users/AuthenticateByName',
   340→   340→      {
   341→   341→        method: 'POST',
   342→   342→        body: JSON.stringify({ Username: username, Pw: password }),
   343→   343→        headers: {
   344→   344→          'Content-Type': 'application/json',
   345→   345→          'Accept': 'application/json',
   346→   346→          'Authorization': `MediaBrowser Client="DSP", Device="Browser", DeviceId="dsp-jellyfin-${typeof window !== 'undefined' ? window.navigator.userAgent.slice(0, 32).replace(/[^a-zA-Z0-9]/g, '') : 'ssr'}", Version="1.0.0"`,
   347→   347→          '__baseUrl': baseUrl,
   348→   348→        } as Record<string, string>,
   349→   349→      },
   350→   350→      true // skipAuth
   351→   351→    );
   352→   352→
   353→   353→    const token = authResponse.AccessToken;
   354→   354→    const userId = authResponse.User.Id;
   355→   355→    const serverId = authResponse.ServerId;
   356→   356→
   357→   357→    // Step 2: Fetch server info
   358→   358→    let serverName = 'Jellyfin';
   359→   359→    let serverVersion = 'unknown';
   360→   360→    try {
   361→   361→      // Temporarily set config for the getSystemInfo call
   362→   362→      this.config = {
   363→   363→        serverUrl: baseUrl,
   364→   364→        username,
   365→   365→        accessToken: token,
   366→   366→        userId,
   367→   367→        serverId,
   368→   368→        serverName,
   369→   369→        serverVersion,
   370→   370→        connectedAt: new Date().toISOString(),
   371→   371→        lastHeartbeat: new Date().toISOString(),
   372→   372→        musicLibraryId: '',
   373→   373→      };
   374→   374→      const sysInfo = await this.getSystemInfo();
   375→   375→      serverName = sysInfo.ServerName;
   376→   376→      serverVersion = sysInfo.Version;
   377→   377→    } catch {
   378→   378→      // If system info fails, continue with defaults
   379→   379→    }
   380→   380→
   381→   381→    // Step 3: Auto-detect music library
   382→   382→    let musicLibraryId = '';
   383→   383→    try {
   384→   384→      const views = await this.getViews();
   385→   385→      const musicFolder = views.Items.find(
   386→   386→        item => item.Type === 'CollectionFolder' && item.Name.toLowerCase().includes('music')
   387→   387→      );
   388→   388→      if (musicFolder) {
   389→   389→        musicLibraryId = musicFolder.Id;
   390→   390→      }
   391→   391→    } catch {
   392→   392→      // Continue without auto-detected library
   393→   393→    }
   394→   394→
   395→   395→    // Step 4: Build and save config
   396→   396→    const config: JellyfinConfig = {
   397→   397→      serverUrl: baseUrl,
   398→   398→      username,
   399→   399→      accessToken: token,
   400→   400→      userId,
   401→   401→      serverId,
   402→   402→      serverName,
   403→   403→      serverVersion,
   404→   404→      connectedAt: new Date().toISOString(),
   405→   405→      lastHeartbeat: new Date().toISOString(),
   406→   406→      musicLibraryId,
   407→   407→    };
   408→   408→
   409→   409→    this.config = config;
   410→   410→    this.saveConfig();
   411→   411→
   412→   412→    return config;
   413→   413→  }

   425→   425→        // Best-effort — continue clearing local state
   426→   426→      }
   427→   427→    }
   428→   428→
   429→   429→    this.config = null;
   430→   430→    if (this.storageAvailable && typeof window !== 'undefined') {
   431→   431→      try {
   432→   432→        window.localStorage.removeItem(STORAGE_KEY);
   433→   433→      } catch {
   434→   434→        // ignore
   435→   435→      }
   436→   436→    }

   489→   489→  /**
   490→   490→   * Generic items query. Supports all major Jellyfin /Users/{userId}/Items parameters.
   491→   491→   */
   492→   492→  async getItems(params: {
   493→   493→    parentId?: string;
   494→   494→    includeItemTypes?: string[];
   495→   495→    sortBy?: string;
   496→   496→    sortOrder?: 'Ascending' | 'Descending';
   497→   497→    limit?: number;
   498→   498→    startIndex?: number;
   499→   499→    searchTerm?: string;
   500→   500→    fields?: string[];
   501→   501→    recursive?: boolean;
   502→   502→    genres?: string[];
   503→   503→    artistIds?: string[];
   504→   504→    albumArtistIds?: string[];
   505→   505→    years?: number[];
   506→   506→    ids?: string[];
   507→   507→    filters?: string[];
   508→   508→  }): Promise<JellyfinItemsResponse> {
   509→   509→    const defaultFields = [
   510→   510→      'PrimaryImageAspectRatio',
   511→   511→      'BasicSyncInfo',
   512→   512→      'Genres',
   513→   513→      'Overview',
   514→   514→      'People',
   515→   515→      'ProductionYear',
   516→   516→      'CommunityRating',
   517→   517→    ];
   518→   518→
   519→   519→    const qs = buildQueryString({
   520→   520→      UserId: this.config!.userId,
   521→   521→      ParentId: params.parentId,

   792→   792→  /**
   793→   793→   * Get a playable audio stream URL for an item.
   794→   794→   * Prefers direct play for best quality; falls back to transcoded stream.
   795→   795→   */
   796→   796→  async getPlaybackStreamUrl(
   797→   797→    itemId: string,
   798→   798→    maxBitrate: number = DEFAULT_MAX_BITRATE
   799→   799→  ): Promise<string> {
   800→   800→    const baseUrl = this.getBaseUrl();
   801→   801→    const token = this.config!.accessToken;
   802→   802→    const userId = this.config!.userId;
   803→   803→
   804→   804→    try {
   805→   805→      // Request playback info to determine best playback method
   806→   806→      const playbackInfo = await this.request<JellyfinPlaybackInfo>(
   807→   807→        `/Items/${itemId}/PlaybackInfo`,
   808→   808→        {
   809→   809→          method: 'POST',
   810→   810→          body: JSON.stringify({
   811→   811→            UserId: userId,
   812→   812→            MaxStreamingBitrate: maxBitrate,
   813→   813→            AutoOpenLiveStreams: true,
   814→   814→          }),
   815→   815→        }
   816→   816→      );
   817→   817→
   818→   818→      if (playbackInfo.MediaSources && playbackInfo.MediaSources.length > 0) {
   819→   819→        const source = playbackInfo.MediaSources[0];
   820→   820→
   821→   821→        // Prefer direct play — best quality, no server transcoding
   822→   822→        if (source.SupportsDirectPlay) {
   823→   823→          return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${token}&UserId=${userId}`;
   824→   824→        }
   825→   825→
   826→   826→        // Fallback to direct stream
   827→   827→        if (source.SupportsDirectStream && source.DirectStreamUrl) {
   828→   828→          return `${baseUrl}${source.DirectStreamUrl}`;
   829→   829→        }
   830→   830→
   831→   831→        // Fallback to transcoded stream (HLS)
   832→   832→        if (source.SupportsTranscoding && source.TranscodingUrl) {
   833→   833→          return `${baseUrl}${source.TranscodingUrl}`;
   834→   834→        }
   835→   835→
   836→   836→        // Last resort: construct a universal audio transcoding URL
   837→   837→        const container = source.Container || 'mp3';
   838→   838→        return `${baseUrl}/Audio/${itemId}/universal?api_key=${token}&UserId=${userId}&MaxStreamingBitrate=${maxBitrate}&Container=${container}&TranscodingMaxAudioChannels=2&SegmentContainer=ts&AudioCodec=aac&Protocol=hls`;
   839→   839→      }
   840→   840→    } catch {
   841→   841→      // If playback info fails, fall back to direct stream URL
   842→   842→    }
   843→   843→
   844→   844→    // Fallback: try static stream directly
   845→   845→    return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${token}&UserId=${userId}`;
   846→   846→  }
