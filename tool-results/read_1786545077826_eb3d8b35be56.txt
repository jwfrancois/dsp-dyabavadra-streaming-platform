    28→function handlePlayEpisode(episode: PodcastEpisode) {
    29→  const podcastStore = usePodcastStore.getState();
    30→  const epState = podcastStore.episodeStates[episode.id];
    31→  const startPos = epState && !epState.completed ? (epState.resumePosition || 0) : 0;
    32→
    33→  const url = episode.audioUrl.startsWith('http')
    34→    ? `/api/proxy/podcast?url=${encodeURIComponent(episode.audioUrl)}`
    35→    : episode.audioUrl;
    36→
    37→  audioSetPlaybackSpeed(podcastStore.playbackSpeed);
    38→
    39→  usePlayerStore.setState({
    40→    audioUrl: url,
    41→    isPlaying: true,
    42→    playbackMode: 'podcast' as const,
    43→    isBuffering: true,
    44→  });
    45→
    46→  podcastStore.playEpisode(episode);
    47→
    48→  if (startPos > 0 && episode.duration > 0) {
    49→    const seekPercent = (startPos / episode.duration) * 100;
    50→    setTimeout(() => {
    51→      audioSeekTo(seekPercent);
    52→    }, 300);
    53→  }
    54→}
    55→
    56→function SpeedBadge() {
    57→  const { playbackSpeed, cyclePlaybackSpeed } = usePodcastStore();
    58→
    59→  const handleClick = () => {
    60→    cyclePlaybackSpeed();
    61→    setTimeout(() => {
    62→      const newSpeed = usePodcastStore.getState().playbackSpeed;
    63→      audioSetPlaybackSpeed(newSpeed);
    64→    }, 0);
    65→  };
    66→
    67→  return (
    68→    <Button
    69→      variant={playbackSpeed === 1.0 ? 'secondary' : 'default'}
    70→      size="sm"
    71→      className="h-7 text-xs font-mono min-w-[48px] justify-center"
    72→      onClick={handleClick}
    73→    >
    74→      {playbackSpeed}x
    75→    </Button>
    76→  );
    77→}

   457→          <Card className="bg-card border-border mb-6">
