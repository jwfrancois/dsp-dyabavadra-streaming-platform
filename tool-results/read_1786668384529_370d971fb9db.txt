    59→
    60→export function DSPConfigView() {
    61→  const engine = useDSPEngineStore();
    62→
    63→  const [selectedZoneId, setSelectedZoneId] = useState(zones[0].id);
    64→  const [dspConfig, setDspConfig] = useState<DSPConfig>(
    65→    zones[0].dspConfig ?? defaultDSPConfig,
    66→  );
    67→  const [expandedSections, setExpandedSections] = useState<Record<string, boolean>>({
    68→    eq: false,
    69→    upsampling: false,
    70→    volume: true,
    71→    clock: true,
    72→  });
    73→  const [volumeMode, setVolumeMode] = useState<'hardware' | 'dsp' | 'fixed'>(
    74→    zones[0].volumeMode ?? 'dsp',
    75→  );
    76→  const [maxVolume, setMaxVolume] = useState(zones[0].maxVolume ?? 85);
    77→  const [startupVolume, setStartupVolume] = useState(zones[0].startupVolume ?? 40);
    78→  const [rampTime, setRampTime] = useState(zones[0].syncOffsetMs ? 500 : 500);
    79→  const [clockMode, setClockMode] = useState<'auto' | 'master' | 'slave' | 'passthrough'>(
    80→    zones[0].clockMode ?? 'auto',
    81→  );
    82→
    83→  const selectedZone = zones.find(z => z.id === selectedZoneId);
    84→
    85→  // ─── Helpers ───
    86→
    87→  const toggleSection = (section: string) => {
    88→    setExpandedSections(prev => ({ ...prev, [section]: !prev[section] }));
    89→  };
    90→
    91→  const activeEQBands = (dspConfig.eq ?? []).filter(b => b.enabled).length;
    92→
    93→  const updateEQBand = (bandId: string, updates: Partial<EQBand>) => {
    94→    setDspConfig(prev => ({
    95→      ...prev,
    96→      eq: (prev.eq ?? []).map(b => (b.id === bandId ? { ...b, ...updates } : b)),
    97→    }));
    98→  };
    99→
   100→  const addEQBand = () => {
   101→    const newId = `eq-new-${Date.now()}`;
   102→    const newBand: EQBand = {
   103→      id: newId,
   104→      enabled: true,
   105→      type: 'parametric',
   106→      frequency: 1000,
   107→      gain: 0,
   108→      q: 1.0,
   109→      label: `Band ${(dspConfig.eq ?? []).length + 1}`,
   110→    };
   111→    setDspConfig(prev => ({ ...prev, eq: [...(prev.eq ?? []), newBand] }));
   112→  };
   113→
   114→  const removeEQBand = (bandId: string) => {
   115→    setDspConfig(prev => ({
   116→      ...prev,
   117→      eq: (prev.eq ?? []).filter(b => b.id !== bandId),
   118→    }));
   119→  };
   120→
   121→  const toggleModule = (key: keyof DSPConfig, nestedKey?: string) => {
   122→    setDspConfig(prev => {
   123→      const mod = prev[key];
   124→      if (!mod || typeof mod !== 'object' || Array.isArray(mod)) return prev;
   125→      return {
   126→        ...prev,
   127→        [key]: { ...(mod as Record<string, unknown>), enabled: !(mod as Record<string, unknown>).enabled },
   128→      };
   129→    });
   130→  };
   131→
   132→  const bandTypeColor = (type: string) => {
   133→    switch (type) {
   134→      case 'parametric': return 'bg-blue-500/20 text-blue-400';
   135→      case 'low-shelf': return 'bg-emerald-500/20 text-emerald-400';
   136→      case 'high-shelf': return 'bg-amber-500/20 text-amber-400';
   137→      case 'low-pass': return 'bg-purple-500/20 text-purple-400';
   138→      case 'high-pass': return 'bg-pink-500/20 text-pink-400';
   139→      case 'notch': return 'bg-red-500/20 text-red-400';
   140→      case 'band-pass': return 'bg-cyan-500/20 text-cyan-400';
   141→      default: return 'bg-surface/50 text-muted-foreground';
   142→    }
   143→  };
   144→
   145→  // ─── Decorative EQ curve SVG ───
   146→
   147→  const EQCurveSVG = () => {
   148→    const bands = dspConfig.eq ?? [];
   149→    const points = 48;
   150→    const w = 480;
   151→    const h = 100;
   152→    const mid = h / 2;
   153→
   154→    const computeGain = (freq: number) => {
   155→      let total = 0;
   156→      for (const band of bands) {
   157→        if (!band.enabled) continue;
   158→        const diff = Math.log2(freq / band.frequency);
   159→        const bw = band.q * diff;
   160→        const response = 1 / (1 + bw * bw);
   161→        if (band.type === 'high-shelf' && freq > band.frequency) {
   162→          total += band.gain * response;
   163→        } else if (band.type === 'low-shelf' && freq < band.frequency) {
   164→          total += band.gain * response;
   165→        } else if (band.type === 'parametric' || band.type === 'notch' || band.type === 'band-pass') {
   166→          total += band.gain * response;
   167→        }
   168→      }
   169→      return total;
   170→    };
   171→
   172→    const pathParts: string[] = [];
   173→    for (let i = 0; i <= points; i++) {
   174→      const t = i / points;
   175→      const freq = 20 * Math.pow(1000, t);
   176→      const gain = computeGain(freq);
   177→      const x = (i / points) * w;
   178→      const y = mid - (gain / 24) * (mid - 8);
   179→      pathParts.push(`${i === 0 ? 'M' : 'L'} ${x.toFixed(1)} ${y.toFixed(1)}`);
   180→    }
   181→
   182→    return (
   183→      <svg viewBox={`0 0 ${w} ${h}`} className="w-full h-24 opacity-40" preserveAspectRatio="none">
   184→        <defs>
   185→          <linearGradient id="eqGrad" x1="0" y1="0" x2="0" y2="1">
   186→            <stop offset="0%" stopColor="currentColor" stopOpacity="0.3" />
   187→            <stop offset="100%" stopColor="currentColor" stopOpacity="0" />
   188→          </linearGradient>
   189→        </defs>
   190→        {/* Grid lines */}
   191→        <line x1={0} y1={mid} x2={w} y2={mid} stroke="currentColor" strokeOpacity="0.15" strokeWidth="0.5" />
   192→        <line x1={0} y1={mid - (mid - 8)} x2={w} y2={mid - (mid - 8)} stroke="currentColor" strokeOpacity="0.08" strokeWidth="0.5" />
   193→        <line x1={0} y1={mid + (mid - 8)} x2={w} y2={mid + (mid - 8)} stroke="currentColor" strokeOpacity="0.08" strokeWidth="0.5" />
   194→        {/* Filled area */}
   195→        <path d={`${pathParts.join(' ')} L ${w} ${mid} L 0 ${mid} Z`} fill="url(#eqGrad)" className="text-primary" />
   196→        {/* Curve line */}
   197→        <path d={pathParts.join(' ')} fill="none" stroke="currentColor" strokeWidth="1.5" className="text-primary" />
   198→        {/* Frequency labels */}
   199→        <text x={0} y={h - 2} fill="currentColor" className="text-muted-foreground" fontSize="8" fontFamily="monospace">20</text>
   200→        <text x={w * 0.21} y={h - 2} fill="currentColor" className="text-muted-foreground" fontSize="8" fontFamily="monospace">100</text>
   201→        <text x={w * 0.43} y={h - 2} fill="currentColor" className="text-muted-foreground" fontSize="8" fontFamily="monospace">1k</text>
   202→        <text x={w * 0.64} y={h - 2} fill="currentColor" className="text-muted-foreground" fontSize="8" fontFamily="monospace">10k</text>
   203→        <text x={w * 0.88} y={h - 2} fill="currentColor" className="text-muted-foreground" fontSize="8" fontFamily="monospace">20k</text>
   204→      </svg>
   205→    );
   206→  };
   207→
   208→  // ─── Module Card renderer ───
   209→
   210→  type ModuleInfo = {
   211→    key: keyof DSPConfig;
   212→    icon: React.ReactNode;
   213→    label: string;
   214→    detail: string;
   215→    enabled: boolean;
   216→  };
   217→
   218→  const modules: ModuleInfo[] = [
   219→    {
   220→      key: 'eq',
   221→      icon: <Waves className="w-4 h-4" />,
   222→      label: 'Parametric EQ',
   223→      detail: `${activeEQBands} active band${activeEQBands !== 1 ? 's' : ''}`,
   224→      enabled: (dspConfig.eq ?? []).length > 0,
   225→    },
   226→    {
   227→      key: 'roomCorrection',
   228→      icon: <Crosshair className="w-4 h-4" />,
   229→      label: 'Room Correction',
   230→      detail: `${dspConfig.roomCorrection?.filterName ?? '—'} · ${formatSampleRate(dspConfig.roomCorrection?.samplerate ?? 48000)}`,
   231→      enabled: dspConfig.roomCorrection?.enabled ?? false,
   232→    },
   233→    {
   234→      key: 'upsampling',
   235→      icon: <RefreshCw className="w-4 h-4" />,
   236→      label: 'Upsampling',
   237→      detail: `${formatSampleRate(dspConfig.upsampling?.targetRate ?? 176400)} · ${dspConfig.upsampling?.filterType ?? 'minimum-phase'}`,
   238→      enabled: dspConfig.upsampling?.enabled ?? false,
   239→    },
   240→    {
   241→      key: 'headphoneCorrection',
   242→      icon: <Headphones className="w-4 h-4" />,
   243→      label: 'Headphone Correction',
   244→      detail: `${dspConfig.headphoneCorrection?.profileName ?? '—'} · ${dspConfig.headphoneCorrection?.manufacturer ?? '—'} ${dspConfig.headphoneCorrection?.model ?? ''}`,
   245→      enabled: dspConfig.headphoneCorrection?.enabled ?? false,
   246→    },
   247→    {
   248→      key: 'crossfeed',
   249→      icon: <MoveHorizontal className="w-4 h-4" />,
   250→      label: 'Crossfeed',
   251→      detail: `Preset: ${dspConfig.crossfeed?.preset ?? 'none'}`,
   252→      enabled: dspConfig.crossfeed?.enabled ?? false,
   253→    },
   254→    {
   255→      key: 'loudness',
   256→      icon: <Gauge className="w-4 h-4" />,
   257→      label: 'Loudness / ReplayGain',
   258→      detail: `${dspConfig.loudness?.targetLUFS ?? -16} LUFS · ${dspConfig.loudness?.method ?? 'ebu-r128'}`,
   259→      enabled: dspConfig.loudness?.enabled ?? false,
   260→    },
   261→    {
   262→      key: 'dither',
   263→      icon: <Cpu className="w-4 h-4" />,
   264→      label: 'Dither',
   265→      detail: dspConfig.dither?.type === 'none' ? 'Disabled' : `${dspConfig.dither?.type === 'tpdf' ? 'TPDF' : dspConfig.dither?.type === 'triangular' ? 'Triangular' : 'Noise-Shaped'}`,
   266→      enabled: dspConfig.dither?.enabled ?? false,
   267→    },
   268→  ];
   269→
   270→  // ─── Render ───
   271→
   272→  return (
   273→    <ScrollArea className="h-full">
   274→      <div className="p-6 max-w-5xl mx-auto space-y-6">
   275→        {/* Page Header */}
   276→        <div className="flex items-center gap-3 mb-2">
   277→          <Sliders className="w-6 h-6 text-primary" />
   278→          <h1 className="text-2xl font-bold">DSP Engine</h1>
   279→          <Badge variant="secondary" className="text-xs ml-2">Configuration</Badge>
   280→        </div>
   281→        <p className="text-muted-foreground text-sm -mt-3 mb-4">
   282→          Configure the digital signal processing chain for each zone.
   283→        </p>
   284→
   285→        {/* ─── 1. Zone Selector ─── */}
   286→        <Card className="bg-card border-border">
   287→          <CardContent className="p-4">
   288→            <div className="flex items-center gap-2 mb-3">
   289→              <MonitorSpeaker className="w-4 h-4 text-muted-foreground" />
   290→              <span className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">Zone</span>
   291→            </div>
   292→            <div className="flex gap-2 flex-wrap">
   293→              {zones.map(zone => (
   294→                <Button
   295→                  key={zone.id}
   296→                  variant="ghost"
   297→                  size="sm"
   298→                  className={
   299→                    zone.id === selectedZoneId
   300→                      ? 'bg-primary text-primary-foreground hover:bg-primary/90'
   301→                      : 'bg-surface hover:bg-surface/80 text-foreground'
   302→                  }
   303→                  onClick={() => {
   304→                    setSelectedZoneId(zone.id);
   305→                    setDspConfig(zone.dspConfig ?? defaultDSPConfig);
   306→                    setVolumeMode(zone.volumeMode ?? 'dsp');
   307→                    setMaxVolume(zone.maxVolume ?? 85);
   308→                    setStartupVolume(zone.startupVolume ?? 40);
   309→                    setClockMode(zone.clockMode ?? 'auto');
   310→                  }}
   311→                >
   312→                  {zone.name}
   313→                </Button>
   314→              ))}
   315→            </div>
   316→          </CardContent>
   317→        </Card>
   318→
   319→        {/* ─── 2. DSP Module Toggles ─── */}
   320→        <Card className="bg-card border-border">
   321→          <CardContent className="p-4">
   322→            <div className="flex items-center gap-2 mb-4">
   323→              <Zap className="w-4 h-4 text-muted-foreground" />
   324→              <span className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">Processing Modules</span>
   325→            </div>
   326→            <div className="grid gap-3">
   327→              {modules.map(mod => (
   328→                <div
   329→                  key={mod.key}
   330→                  className={`flex items-center justify-between p-3 rounded-lg bg-surface/50 border-l-2 transition-colors ${
   331→                    mod.enabled ? 'border-l-primary' : 'border-l-transparent'
   332→                  }`}
   333→                >
   334→                  <div className="flex items-center gap-3 min-w-0">
   335→                    <div className={`${mod.enabled ? 'text-primary' : 'text-muted-foreground'}`}>
   336→                      {mod.icon}
   337→                    </div>
   338→                    <div className="min-w-0">
   339→                      <div className="flex items-center gap-2">
   340→                        <span className="text-sm font-medium">{mod.label}</span>
   341→                        {mod.enabled && <CheckCircle2 className="w-3 h-3 text-signal-green" />}
   342→                      </div>
   343→                      <span className="text-xs text-muted-foreground truncate block">
   344→                        {mod.detail}
   345→                      </span>
   346→                    </div>
   347→                  </div>
   348→                  <Switch
   349→                    checked={mod.enabled}
   350→                    onCheckedChange={() => toggleModule(mod.key)}
   351→                  />
   352→                </div>
   353→              ))}
   354→            </div>
   355→          </CardContent>
   356→        </Card>
   357→
   358→        {/* ─── 3. Parametric EQ Editor ─── */}
   359→        <Card className="bg-card border-border">
   360→          <CardContent className="p-4">
   361→            <button
   362→              className="flex items-center justify-between w-full text-left"
   363→              onClick={() => toggleSection('eq')}
   364→            >
   365→              <div className="flex items-center gap-2">
   366→                <Waves className="w-4 h-4 text-muted-foreground" />
   367→                <span className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">Parametric EQ Editor</span>
   368→                <Badge variant="secondary" className="text-[10px]">{(dspConfig.eq ?? []).length} bands</Badge>
   369→              </div>
   370→              {expandedSections.eq ? (
   371→                <ChevronUp className="w-4 h-4 text-muted-foreground" />
   372→              ) : (
   373→                <ChevronDown className="w-4 h-4 text-muted-foreground" />
   374→              )}
   375→            </button>
   376→
   377→            {expandedSections.eq && (
   378→              <div className="mt-4 space-y-4">
   379→                {/* EQ Curve Visualization */}
   380→                <div className="rounded-lg bg-surface/50 p-3">
   381→                  <EQCurveSVG />
   382→                </div>
   383→
   384→                <Separator className="bg-border" />
   385→
   386→                {/* EQ Bands */}
   387→                <div className="space-y-3">
   388→                  {(dspConfig.eq ?? []).map((band, idx) => (
   389→                    <div
   390→                      key={band.id}
   391→                      className={`p-3 rounded-lg border border-border bg-surface/30 ${
   392→                        !band.enabled ? 'opacity-50' : ''
   393→                      }`}
   394→                    >
   395→                      <div className="flex items-center justify-between mb-2">
   396→                        <div className="flex items-center gap-2">
   397→                          <span className="text-xs font-medium text-foreground">{band.label ?? `Band ${idx + 1}`}</span>
   398→                          <Badge className={`text-[10px] ${bandTypeColor(band.type)}`} variant="secondary">
   399→                            {band.type}
   400→                          </Badge>
   401→                        </div>
   402→                        <div className="flex items-center gap-2">
   403→                          <Switch
   404→                            checked={band.enabled}
   405→                            onCheckedChange={checked => updateEQBand(band.id, { enabled: checked })}
   406→                          />
   407→                          <Button
   408→                            variant="ghost"
   409→                            size="icon"
   410→                            className="h-7 w-7 text-muted-foreground hover:text-signal-red"
   411→                            onClick={() => removeEQBand(band.id)}
   412→                          >
   413→                            <Trash2 className="w-3.5 h-3.5" />
   414→                          </Button>
   415→                        </div>
   416→                      </div>
   417→
   418→                      <div className="grid grid-cols-3 gap-3">
   419→                        {/* Frequency */}
   420→                        <div>
   421→                          <span className="text-[10px] uppercase tracking-wider text-muted-foreground block mb-1">Frequency</span>
   422→                          <span className="font-mono text-sm text-foreground">
   423→                            {band.frequency >= 1000
   424→                              ? `${(band.frequency / 1000).toFixed(1)} kHz`
   425→                              : `${band.frequency} Hz`}
   426→                          </span>
   427→                        </div>
   428→
   429→                        {/* Gain with +/- */}
   430→                        <div>
   431→                          <span className="text-[10px] uppercase tracking-wider text-muted-foreground block mb-1">Gain</span>
   432→                          <div className="flex items-center gap-1">
   433→                            <Button
   434→                              variant="outline"
   435→                              size="icon"
   436→                              className="h-6 w-6"
   437→                              onClick={() => updateEQBand(band.id, { gain: Math.max(-24, band.gain - 0.5) })}
   438→                            >
   439→                              <span className="text-xs">−</span>
   440→                            </Button>
   441→                            <span className={`font-mono text-sm min-w-[3rem] text-center ${
   442→                              band.gain > 0 ? 'text-signal-green' : band.gain < 0 ? 'text-signal-amber' : 'text-foreground'
   443→                            }`}>
   444→                              {band.gain > 0 ? '+' : ''}{band.gain.toFixed(1)} dB
   445→                            </span>
   446→                            <Button
   447→                              variant="outline"
   448→                              size="icon"
   449→                              className="h-6 w-6"
   450→                              onClick={() => updateEQBand(band.id, { gain: Math.min(24, band.gain + 0.5) })}
   451→                            >
   452→                              <span className="text-xs">+</span>
   453→                            </Button>
   454→                          </div>
   455→                        </div>
   456→
   457→                        {/* Q Value */}
   458→                        <div>
   459→                          <span className="text-[10px] uppercase tracking-wider text-muted-foreground block mb-1">Q</span>
   460→                          <span className="font-mono text-sm text-foreground">{band.q.toFixed(2)}</span>
   461→                        </div>
   462→                      </div>
   463→                    </div>
   464→                  ))}
   465→                </div>
   466→
   467→                <Button
   468→                  variant="outline"
   469→                  size="sm"
   470→                  className="w-full border-dashed"
   471→                  onClick={addEQBand}
   472→                >
   473→                  <Plus className="w-3.5 h-3.5 mr-2" />
   474→                  Add Band
   475→                </Button>
   476→              </div>
   477→            )}
   478→          </CardContent>
   479→        </Card>
   480→
   481→        {/* ─── 4. Upsampling Config ─── */}
   482→        <Card className="bg-card border-border">
   483→          <CardContent className="p-4">
   484→            <button
   485→              className="flex items-center justify-between w-full text-left"
   486→              onClick={() => toggleSection('upsampling')}
   487→            >
   488→              <div className="flex items-center gap-2">
   489→                <RefreshCw className="w-4 h-4 text-muted-foreground" />
   490→                <span className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">Upsampling Configuration</span>
   491→              </div>
   492→              {expandedSections.upsampling ? (
   493→                <ChevronUp className="w-4 h-4 text-muted-foreground" />
   494→              ) : (
