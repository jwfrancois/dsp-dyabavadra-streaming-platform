   109→   109→/** Map a Jellyfin Container string to a human-readable format name */
   110→   110→function getFormatName(container?: string): string {
   111→   111→  if (!container) return '';
   112→   112→  const lower = container.toLowerCase();
   113→   113→  const map: Record<string, string> = {
   114→   114→    flac: 'FLAC',
   115→   115→    mp3: 'MP3',
   116→   116→    wav: 'WAV',
   117→   117→    m4a: 'AAC',
   118→   118→    aac: 'AAC',
   119→   119→    ogg: 'OGG',
   120→   120→    opus: 'OPUS',
   121→   121→    wma: 'WMA',
   122→   122→    aiff: 'AIFF',
   123→   123→    dsd: 'DSD',
   124→   124→    dsf: 'DSD',
   125→   125→   dff: 'DSD',
   126→   126→  };
   127→   127→  return map[lower] || upperFirst(lower);
   128→   128→}
   129→   129→
   130→   130→function upperFirst(s: string): string {
   131→   131→  return s.charAt(0).toUpperCase() + s.slice(1);
   132→   132→}

   134→   134→/** Safely get an image URL from the Jellyfin client, returning an empty string on failure */
   135→   135→function safeGetImageUrl(
   136→   136→  itemId: string,
   137→   137→  maxWidth?: number,
   138→   138→  maxHeight?: number
   139→   139→): string {
   140→   140→  try {
   141→   141→    return jellyfinClient.getImageUrl(itemId, 'Primary', maxWidth, maxHeight);
   142→   142→  } catch {
   143→   143→    return '';
   144→   144→  }
   145→   145→}

   157→   157→/** Map a JellyfinItem (MusicAlbum) → JellyfinAlbum */
   158→   158→function mapAlbum(item: JellyfinItem): JellyfinAlbum {
   159→   159→  const artistObj = item.AlbumArtists?.[0];
   160→   160→  return {
   161→   161→    id: item.Id,
   162→   162→    name: item.Name,
   163→   163→    artistName: artistObj?.Name ?? item.AlbumArtist ?? 'Unknown Artist',
   164→   164→    artistId: artistObj?.Id ?? '',
   165→   165→    year: item.ProductionYear ?? 0,
   166→   166→    genre: item.Genres?.[0] ?? '',
   167→   167→    imageUrl: safeGetImageUrl(item.Id, 300, 300),
   168→   168→    duration: ticksToSeconds(item.RunTimeTicks),
   169→   169→    trackCount: item.ChildCount ?? 0,
   170→   170→    communityRating: item.CommunityRating ?? 0,
   171→   171→    isFavorite: item.UserData?.IsFavorite ?? false,
   172→   172→    overview: item.Overview,
   173→   173→  };
   174→   174→}

   176→   176→/** Map a JellyfinItem (Audio) → JellyfinTrack */
   177→   177→function mapTrack(item: JellyfinItem): JellyfinTrack {
   178→   178→  const artistObj = item.AlbumArtists?.[0];
   179→   179→  const albumArtist = artistObj?.Name ?? item.Artists?.[0] ?? 'Unknown Artist';
   180→   180→  const albumArtistId = artistObj?.Id ?? '';
   181→   181→
   182→   182→  return {
   183→   183→    id: item.Id,
   184→   184→    name: item.Name,
   185→   185→    albumId: item.AlbumId ?? item.ParentId ?? '',
   186→   186→    albumName: '', // populated from parent context when available
   187→   187→    artistName: albumArtist,
   188→   188→    artistId: albumArtistId,
   189→   189→    trackNumber: 0, // Jellyfin SortName is typically "disc-track name"; use index from list
   190→   190→    duration: ticksToSeconds(item.RunTimeTicks),
   191→   191→    format: getFormatName(item.Container),
   192→   192→    bitrate: item.BitRate ?? 0,
   193→   193→    sampleRate: item.SampleRate ?? 0,
   194→   194→    bitDepth: item.BitsPerSample ?? 0,
   195→   195→    channels: item.Channels ?? 0,
   196→   196→    imageUrl: safeGetImageUrl(item.AlbumId ?? item.ParentId ?? item.Id, 300, 300),
   197→   197→    isFavorite: item.UserData?.IsFavorite ?? false,
   198→   198→    playCount: item.UserData?.PlayCount ?? 0,
   199→   199→  };
   200→   200→}

   217→   217→/** Build a static Jellyfin stream URL for a given item ID (synchronous) */

   220→   220→  return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${config.accessToken}&UserId=${config.userId}`;

   252→   252→/** Fetch the total track count and update the store (fire-and-forget helper) */
   253→   253→async function refreshTotalTrackCount(): Promise<void> {
   254→   254→  try {
   255→   255→    const state = useJellyfinStore.getState();
   256→   256→    if (state.connectionStatus !== 'connected') return;
   257→   257→    const response = await jellyfinClient.getItems({
   258→   258→      includeItemTypes: ['Audio'],
   259→   259→      recursive: true,
   260→   260→      limit: 1,
   261→   261→    });
   262→   262→    useJellyfinStore.setState({ totalTracks: response.TotalRecordCount });
   263→   263→  } catch {
   264→   264→    // Non-critical; ignore
   265→   265→  }
   266→   266→}

   404→   404→        if (err.statusCode === 0) {
   405→   405→          message = 'Could not reach the Jellyfin server. Check the server URL and ensure the server is running and accessible.';

   407→   407→          message = 'Authentication failed. Please check your username and password.';

   409→   409→          message = 'Jellyfin server found but the API endpoint was not available. Ensure you are running Jellyfin 10.7+.';
   410→   410→        } else {
   411→   411→          message = err.message;
   412→   412→        }
   413→   413→      } else if (err instanceof Error) {
   414→   414→        message = err.message;
   415→   415→      } else {
   416→   416→        message = 'Connection failed';
   417→   417→      }
   418→   418→      set({ connectionStatus: 'error', error: message });
   419→   419→      return false;
   420→   420→    }

   963→   963→  search: async (query: string) => {
   964→   964→    const { connectionStatus } = get();
   965→   965→    if (connectionStatus !== 'connected') {
   966→   966→      return { artists: [], albums: [], tracks: [], playlists: [], podcasts: [], podcastEpisodes: [] };
   967→   967→    }
   968→   968→
   969→   969→    try {
   970→   970→      const response = await jellyfinClient.search(query);
   971→   971→
   972→   972→      const artists: JellyfinArtist[] = [];
   973→   973→      const albums: JellyfinAlbum[] = [];
   974→   974→      const tracks: JellyfinTrack[] = [];
   975→   975→      const playlists: JellyfinPlaylist[] = [];
   976→   976→      const podcasts: JellyfinPodcastShow[] = [];
   977→   977→      const podcastEpisodes: JellyfinPodcastEpisode[] = [];
   978→   978→
   979→   979→      for (const item of response.Items) {
   980→   980→        switch (item.Type) {
   981→   981→          case 'MusicArtist':
   982→   982→            artists.push(mapArtist(item));
   983→   983→            break;
   984→   984→          case 'MusicAlbum':
   985→   985→            albums.push(mapAlbum(item));
   986→   986→            break;
   987→   987→          case 'Audio':
   988→   988→            tracks.push(mapTrack(item));
   989→   989→            break;
   990→   990→          case 'Playlist':
   991→   991→            playlists.push(mapPlaylist(item));
