    25→export interface Work {
    26→  id: string;
    27→  title: string;
    28→  titleFull: string; // e.g., "Symphony No. 9 in D minor, Op. 125"
    29→  composerId: string;

    32→  genre: string; // "Symphony", "Concerto", "Sonata", "String Quartet", etc.

    34→  yearComposed?: number;
    35→  yearFirstPerformed?: number;
    36→  movements: Movement[];
    37→  duration: number; // total seconds
    38→  recordings: WorkRecording[];
    39→  description?: string;
    40→}

  1084→function levenshteinDistance(a: string, b: string): number {
  1085→  const matrix: number[][] = [];
  1086→  for (let i = 0; i <= b.length; i++) matrix[i] = [i];
  1087→  for (let j = 0; j <= a.length; j++) matrix[0][j] = j;
  1088→  for (let i = 1; i <= b.length; i++) {
  1089→    for (let j = 1; j <= a.length; j++) {
  1090→      if (b.charAt(i - 1) === a.charAt(j - 1)) {
  1091→        matrix[i][j] = matrix[i - 1][j - 1];
  1092→      } else {
  1093→        matrix[i][j] = Math.min(
  1094→          matrix[i - 1][j - 1] + 1,
  1095→          matrix[i][j - 1] + 1,
  1096→          matrix[i - 1][j] + 1
  1097→        );
  1098→      }
  1099→    }
  1100→  }
  1101→  return matrix[b.length][a.length];
  1102→}

  1104→function fuzzyMatch(query: string, target: string): number {
  1105→  const q = query.toLowerCase();
  1106→  const t = target.toLowerCase();
  1107→  if (t.includes(q)) return 100;
  1108→  if (q.includes(t)) return 80;
  1109→  // Check for all characters appearing in order
  1110→  let qi = 0;
  1111→  for (let ti = 0; ti < t.length && qi < q.length; ti++) {
  1112→    if (t[ti] === q[qi]) qi++;
  1113→  }
  1114→  if (qi === q.length) return 60;
  1115→  const dist = levenshteinDistance(q, t);
  1116→  const maxLen = Math.max(q.length, t.length);
  1117→  return Math.max(0, Math.round((1 - dist / maxLen) * 50));
  1118→}

  1122→export function fuzzySearch(query: string): FuzzySearchResult {

  1125→    credits: [], radioStations: [], genres: [],

  1180→  }

  1182→  // Search works

  1185→      fuzzyMatch(query, w.title),
  1186→      fuzzyMatch(query, w.titleFull),
  1187→      fuzzyMatch(query, w.composerName),
  1188→      fuzzyMatch(query, w.genre),
  1189→      fuzzyMatch(query, w.catalogNumber || '')
  1190→    );

  1226→  // Sort all by score descending

  1232→  results.credits.sort((a, b) => b.score - a.score);
  1233→  results.radioStations.sort((a, b) => b.score - a.score);
  1234→  results.genres.sort((a, b) => b.score - a.score);

  1255→export function generateRadio(seed: RadioSeed, count: number = 20): RadioStationGenerated {
  1256→  const allTracks = [...tracks];
  1257→  let pool: typeof tracks;
  1258→
  1259→  if (seed.type === 'track') {
  1260→    const track = allTracks.find(t => t.id === seed.id);
  1261→    if (track) {
  1262→      pool = allTracks.filter(t =>
  1263→        t.id !== track.id &&
  1264→        (t.genre === track.genre || t.artistId === track.artistId || t.composers.some(c => track.composers.includes(c)))
  1265→      );
  1266→    } else {
  1267→      pool = allTracks;
  1268→    }
  1269→  } else if (seed.type === 'artist') {
  1270→    pool = allTracks.filter(t => t.artistId !== seed.id);
  1271→    // Mix in similar artists
  1272→    const artist = artists.find(a => a.id === seed.id);
  1273→    if (artist) {
  1274→      const similarTracks = allTracks.filter(t => artist.similarArtists.includes(t.artistId));
  1275→      pool = [...pool, ...similarTracks];
  1276→    }
  1277→  } else if (seed.type === 'genre') {
  1278→    pool = allTracks.filter(t => t.genre === seed.name);
  1279→  } else {
  1280→    pool = allTracks;
  1281→  }
  1282→
  1283→  // Shuffle and take
  1284→  const shuffled = pool.sort(() => Math.random() - 0.5).slice(0, count);
  1285→  const trackIds = shuffled.map(t => t.id);
  1286→
  1287→  return {
  1288→    id: `radio-gen-${Date.now()}`,
  1289→    name: seed.type === 'genre' ? `${seed.name} Radio` :
  1290→          seed.type === 'artist' ? `${seed.name} Radio` :
  1291→          seed.type === 'playlist' ? `${seed.name} Mix` : `${seed.name} Mix`,
  1292→    seed,
  1293→    trackIds,
  1294→    description: `Algorithmic radio seeded from ${seed.name}. Blending local library tracks based on genre, artist similarity, and listening patterns.`,
  1295→  };
  1296→}
