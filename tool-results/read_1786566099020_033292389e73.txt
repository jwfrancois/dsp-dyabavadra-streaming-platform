    28→    28→function handlePlayEpisode(episode: PodcastEpisode) {
    29→    29→  const podcastStore = usePodcastStore.getState();
    30→    30→  const epState = podcastStore.episodeStates[episode.id];
    31→    31→  const startPos = epState && !epState.completed ? (epState.resumePosition || 0) : 0;
    32→    32→
    33→    33→  const url = episode.audioUrl.startsWith('http')
    34→    34→    ? `/api/proxy/podcast?url=${encodeURIComponent(episode.audioUrl)}`
    35→    35→    : episode.audioUrl;
    36→    36→
    37→    37→  audioSetPlaybackSpeed(podcastStore.playbackSpeed);
    38→    38→
    39→    39→  usePlayerStore.setState({
    40→    40→    audioUrl: url,
    41→    41→    isPlaying: true,
    42→    42→    playbackMode: 'podcast' as const,
    43→    43→    isBuffering: true,
    44→    44→  });
    45→    45→
    46→    46→  podcastStore.playEpisode(episode);
    47→    47→
    48→    48→  if (startPos > 0 && episode.duration > 0) {
    49→    49→    const seekPercent = (startPos / episode.duration) * 100;
    50→    50→    setTimeout(() => {
    51→    51→      audioSeekTo(seekPercent);
    52→    52→    }, 300);
    53→    53→  }
    54→    54→}
