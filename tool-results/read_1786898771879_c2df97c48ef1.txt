    21→    21→// ── Quality badge helper ──
    22→    22→function QualityBadge({ format, sampleRate, bitDepth }: { format: string; sampleRate: number; bitDepth: number }) {
    23→    23→  const isHiRes = sampleRate > 48000 || bitDepth > 16;
    24→    24→  const isLossless = ['FLAC', 'WAV', 'AIFF', 'ALAC', 'DSF', 'DFF', 'WavPack', 'APE', 'TAK'].includes(format.toUpperCase());
    25→    25→  const isDSD = ['DSF', 'DFF', 'DSD'].includes(format.toUpperCase());
    26→    26→
    27→    27→  if (isDSD) return <Badge className="text-[9px] bg-purple-500/20 text-purple-400 border-purple-500/30 h-4 px-1">DSD</Badge>;
    28→    28→  if (isHiRes && isLossless) return <Badge className="text-[9px] bg-primary/20 text-primary border-primary/30 h-4 px-1">Hi-Res</Badge>;
    29→    29→  if (isLossless) return <Badge className="text-[9px] bg-signal-green/20 text-signal-green border-signal-green/30 h-4 px-1">Lossless</Badge>;
    30→    30→  return null;
    31→    31→}

    33→    33→// ── Time-based greeting ──
    34→    34→function getGreeting(): { text: string; icon: React.ReactNode; emoji: string } {
    35→    35→  const hour = new Date().getHours();
    36→    36→  if (hour < 6) return { text: 'Night Owl Mode', icon: <Moon className="w-5 h-5" />, emoji: '' };
    37→    37→  if (hour < 12) return { text: 'Good Morning', icon: <Sunrise className="w-5 h-5" />, emoji: '' };
    38→    38→  if (hour < 14) return { text: 'Good Afternoon', icon: <Sun className="w-5 h-5" />, emoji: '' };
    39→    39→  if (hour < 18) return { text: 'Good Afternoon', icon: <Sun className="w-5 h-5" />, emoji: '' };
    40→    40→  if (hour < 22) return { text: 'Good Evening', icon: <Moon className="w-5 h-5" />, emoji: '' };
    41→    41→  return { text: 'Night Owl Mode', icon: <Moon className="w-5 h-5" />, emoji: '' };
    42→    42→}
