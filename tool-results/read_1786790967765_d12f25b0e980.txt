    83→function localTrackToTrack(lt: LocalTrack): Track {
    84→  return {
    85→    id: lt.id,
    86→    title: lt.title,
    87→    albumId: lt.album,
    88→    albumName: lt.album,
    89→    artistId: lt.artist,
    90→    artistName: lt.artist,
    91→    trackNumber: lt.trackNumber,
    92→    discNumber: lt.discNumber,
    93→    duration: lt.duration,
    94→    format: lt.format,
    95→    bitDepth: lt.bitDepth,
    96→    sampleRate: lt.sampleRate,
    97→    channels: lt.channels,
    98→    bitrate: lt.bitrate,
    99→    filePath: lt.filePath,
   100→    fileSize: lt.fileSize,
   101→    composers: lt.composer ? [lt.composer] : [],
   102→    performers: [],
   103→    genre: lt.genre,
   104→    loved: false,
   105→    playCount: 0,
   106→    source: 'local',
   107→    isAvailable: true,
   108→    blobUrl: (lt as any).blobUrl as string | undefined,
   109→  };
   110→}
   111→
   112→function playLocalTrack(lt: LocalTrack) {
   113→  const track = localTrackToTrack(lt);
   114→  usePlayerStore.getState().play(track);
   115→}
   116→
   117→function playLocalTracks(lts: LocalTrack[]) {
   118→  if (lts.length === 0) return;
   119→  const queue = lts.map(localTrackToTrack);
   120→  usePlayerStore.getState().setQueue(queue, 0);
   121→}

   309→
   310→function BrowseLocalPanel() {
   311→  const localStore = useLocalLibraryStore();
   312→  const [view, setView] = React.useState<BrowseLocalView>('artists');
   313→  const [searchQuery, setSearchQuery] = React.useState('');
   314→
   315→  // Breadcrumb navigation state
   316→  const [selectedArtist, setSelectedArtist] = React.useState<string | null>(null);
   317→  const [selectedAlbum, setSelectedAlbum] = React.useState<string | null>(null);
   318→
   319→  // Filter only browser-imported tracks (isLocal === true)
   320→  const allLocalTracks = localStore.tracks.filter((t: any) => t.isLocal);
   321→
   322→  // Derive artists / albums only from local tracks
   323→  const localArtists = React.useMemo(() => {
   324→    const set = new Set<string>();
   325→    for (const t of allLocalTracks) {
   326→      if (t.artist) set.add(t.artist);
   327→      if (t.albumArtist && t.albumArtist !== t.artist) set.add(t.albumArtist);
   328→    }
   329→    return [...set].sort();
   330→  }, [allLocalTracks]);
   331→
   332→  const localAlbums = React.useMemo(() => {
   333→    const map = new Map<string, { name: string; artist: string; albumArtist: string; trackCount: number; year: number; coverArt: string | null }>();
   334→    for (const t of allLocalTracks) {
   335→      const key = `${t.albumArtist || t.artist}|||${t.album}`;
   336→      const existing = map.get(key);
   337→      if (existing) {
   338→        existing.trackCount++;
   339→        if (t.year && (!existing.year || t.year > existing.year)) existing.year = t.year;
   340→        if (!existing.coverArt && t.coverArt) existing.coverArt = t.coverArt;
   341→      } else {
   342→        map.set(key, {
   343→          name: t.album,
   344→          artist: t.artist,
   345→          albumArtist: t.albumArtist || t.artist,
   346→          trackCount: 1,
   347→          year: t.year,
   348→          coverArt: t.coverArt,
   349→        });
   350→      }
   351→    }
   352→    return Array.from(map.values());
   353→  }, [allLocalTracks]);
   354→
   355→  const allArtists = localArtists;
   356→  const allAlbums = localAlbums;
   357→  const allTracks = allLocalTracks;
   358→
   359→  // Derived data based on navigation state
   360→  const breadcrumbs: BreadcrumbItem[] = [{ label: 'Artists', onClick: () => { setSelectedArtist(null); setSelectedAlbum(null); setView('artists'); } }];
   361→
   362→  let displayedArtists = allArtists;
   363→  let displayedAlbums = allAlbums;
   364→  let displayedTracks = allTracks;
   365→
   366→  // Local-only search helper
   367→  const searchLocalTracks = React.useCallback((query: string) => {
   368→    const q = query.toLowerCase();
   369→    return allLocalTracks.filter(
   370→      t =>
   371→        t.title.toLowerCase().includes(q) ||
   372→        t.artist.toLowerCase().includes(q) ||
   373→        t.album.toLowerCase().includes(q) ||
   374→        t.genre.toLowerCase().includes(q) ||
   375→        (t.composer && t.composer.toLowerCase().includes(q)) ||
   376→        (t.albumArtist && t.albumArtist.toLowerCase().includes(q))
   377→    );
   378→  }, [allLocalTracks]);
   379→
   380→  if (searchQuery.trim()) {
   381→    displayedTracks = searchLocalTracks(searchQuery.trim());
   382→    displayedAlbums = allAlbums.filter(a =>
   383→      displayedTracks.some(t => t.album === a.name && (t.artist === a.artist || t.albumArtist === a.albumArtist))
   384→    );
   385→    displayedArtists = allArtists.filter(a =>
   386→      displayedTracks.some(t => t.artist === a || t.albumArtist === a)
   387→    );
   388→  } else if (selectedArtist && selectedAlbum) {
   389→    // Artist > Album > Tracks
   390→    breadcrumbs.push(
   391→      { label: selectedArtist, onClick: () => { setSelectedAlbum(null); } },
   392→      { label: selectedAlbum, onClick: () => {} },
   393→    );
   394→    displayedTracks = allLocalTracks.filter(t => t.album === selectedAlbum && (t.artist === selectedArtist || t.albumArtist === selectedArtist));
   395→  } else if (selectedArtist) {
   396→    // Artist > Albums
   397→    breadcrumbs.push({ label: selectedArtist, onClick: () => {} });
   398→    displayedAlbums = allAlbums.filter(a => a.artist === selectedArtist || a.albumArtist === selectedArtist);
   399→    displayedTracks = allLocalTracks.filter(t => t.artist === selectedArtist || t.albumArtist === selectedArtist);
   400→  } else if (selectedAlbum) {
   401→    // Albums > Tracks
   402→    breadcrumbs.push({ label: selectedAlbum, onClick: () => {} });
   403→    displayedTracks = allLocalTracks.filter(t => t.album === selectedAlbum);
   404→  }
   405→
   406→  const showArtistsView = !selectedArtist && !selectedAlbum && view === 'artists' && !searchQuery.trim();
   407→  const showAlbumsView = (selectedArtist || view === 'albums') && !selectedAlbum && !searchQuery.trim();
   408→  const showTracksView = selectedAlbum || view === 'tracks' || searchQuery.trim();
   409→
   410→  return (
   411→    <div className="space-y-4">
   412→      {/* Search and View Switcher */}
   413→      <div className="flex items-center gap-3">
   414→        <div className="relative flex-1">
   415→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   416→          <Input
   417→            placeholder="Search local library..."
   418→            value={searchQuery}
   419→            onChange={e => { setSearchQuery(e.target.value); if (e.target.value) { setSelectedArtist(null); setSelectedAlbum(null); } }}
   420→            className="pl-9 h-9 text-sm"
   421→          />
   422→        </div>
   423→        <div className="flex items-center bg-surface rounded-lg p-0.5">
   424→          {([['artists', User, 'Artists'], ['albums', Disc3, 'Albums'], ['tracks', Music2, 'Tracks']] as const).map(([key, Icon, label]) => (
   425→            <Button
   426→              key={key}
   427→              variant={view === key ? 'default' : 'ghost'}
   428→              size="sm"
   429→              className="h-8 text-xs gap-1.5 rounded-md"
   430→              onClick={() => { setView(key as BrowseLocalView); setSelectedArtist(null); setSelectedAlbum(null); setSearchQuery(''); }}
   431→            >
   432→              <Icon className="w-3.5 h-3.5" /> {label}
   433→            </Button>
   434→          ))}
   435→        </div>
   436→      </div>
   437→
   438→      {/* Breadcrumb */}
   439→      {(selectedArtist || selectedAlbum) && !searchQuery.trim() && (
   440→        <div className="flex items-center gap-1 text-xs">
   441→          {breadcrumbs.map((crumb, i) => (
   442→            <React.Fragment key={i}>
   443→              {i > 0 && <ChevronRight className="w-3 h-3 text-muted-foreground" />}
   444→              <button
   445→                className={`${i === breadcrumbs.length - 1 ? 'text-foreground font-medium' : 'text-muted-foreground hover:text-foreground'} transition-colors`}
   446→                onClick={crumb.onClick}
   447→              >
   448→                {crumb.label}
   449→              </button>
   450→            </React.Fragment>
   451→          ))}
   452→        </div>
   453→      )}
   454→
   455→      {/* Empty State */}
   456→      {allTracks.length === 0 && (
   457→        <Card className="bg-card border-border">
   458→          <CardContent className="p-12 text-center">
   459→            <LibraryIcon className="w-12 h-12 mx-auto mb-4 text-muted-foreground/30" />
   460→            <h3 className="text-sm font-semibold mb-1">No Local Tracks</h3>
   461→            <p className="text-xs text-muted-foreground max-w-md mx-auto">
   462→              Scan your music folders to build your local library. Go to the &quot;Scan Folders&quot; tab to add directories and start scanning.
   463→            </p>
   464→          </CardContent>
   465→        </Card>
   466→      )}
   467→
   468→      {/* Artists Grid */}
   469→      {showArtistsView && allTracks.length > 0 && (
   470→        <div className="space-y-3">
   471→          <div className="flex items-center justify-between">
   472→            <h2 className="text-sm font-semibold">Artists ({displayedArtists.length})</h2>
   473→          </div>
   474→          {displayedArtists.length === 0 ? (
   475→            <p className="text-xs text-muted-foreground py-8 text-center">No artists found.</p>

   678→
   679→  const { tracks, isScanning, scanProgress, scanError, directories, lastScanTime } = localStore;
   680→
   681→  const handleScan = React.useCallback(async () => {
   682→    const dir = directoryInput.trim();
   683→    if (!dir) return;
   684→    localStore.addDirectory(dir);
   685→    setDirectoryInput('');

   742→              <Button

   745→                disabled={isScanning}

  1147→          await readEntry(entry);

  1519→      const res = await fetch('/api/library/mount', {

  1521→        body: JSON.stringify({ uncPath: unc, username: usernameInput.trim() || undefined, password: passwordInput || undefined }),
  1522→      });

  1525→        if (data.scanPath) { localStore.addDirectory(data.scanPath); localStore.startScan(data.scanPath); }

  1533→  const handleUnmount = async (shareId: string) => {

  1541→      const res = await fetch('/api/library/mount', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ uncPath: share.uncPath }) });
  1542→      const data = await res.json();

  1563→            </div>
  1564→          </div>
  1565→        </CardContent>
  1566→      </Card>
  1567→
  1568→      <Card className="bg-card border-border">
