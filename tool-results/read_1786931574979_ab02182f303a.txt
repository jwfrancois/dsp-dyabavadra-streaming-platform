   162→/** Convert Jellyfin ticks (100ns units) to seconds */
   163→export function ticksToSeconds(ticks: number | undefined): number {
   164→  if (ticks === undefined || ticks === null) return 0;
   165→  return Math.floor(ticks / TICKS_PER_SECOND);
   166→}
   167→
   168→/** Convert seconds to Jellyfin ticks */
   169→export function secondsToTicks(seconds: number): number {
   170→  return Math.floor(seconds * TICKS_PER_SECOND);
   171→}

   173→/** Strip trailing slash from a URL */
   174→function normalizeServerUrl(url: string): string {
   175→  return url.replace(/\/+$/, '');

   177→
   178→/** Build a query string from a record, filtering out undefined/null/empty values */
   179→function buildQueryString(params: Record<string, string | number | boolean | undefined | null | string[] | number[]>): string {
   180→  const parts: string[] = [];
   181→  for (const [key, value] of Object.entries(params)) {
   182→    if (value === undefined || value === null || value === '') continue;
   183→    if (Array.isArray(value)) {
   184→      if (value.length > 0) {
   185→        parts.push(`${encodeURIComponent(key)}=${value.map(v => encodeURIComponent(String(v))).join(',')}`);
   186→      }
   187→    } else {
   188→      parts.push(`${encodeURIComponent(key)}=${encodeURIComponent(String(value))}`);
   189→    }
   190→  }
   191→  return parts.length > 0 ? `?${parts.join('&')}` : '';
   192→}

   196→export class JellyfinClient {
   197→  private config: JellyfinConfig | null = null;
   198→  private storageAvailable = true;
   199→
   200→  constructor() {
   201→    // Check if localStorage is available (SSR-safe)
   202→    try {
   203→      const testKey = '__dsp_jellyfin_test__';
   204→      if (typeof window !== 'undefined') {
   205→        window.localStorage.setItem(testKey, '1');
   206→        window.localStorage.removeItem(testKey);
   207→      } else {
   208→        this.storageAvailable = false;
   209→      }
   210→    } catch {
   211→      this.storageAvailable = false;
   212→    }
   213→
   214→    // Auto-load persisted config
   215→    this.config = this.loadConfig();

   219→  // Persistence
   220→  // ═══════════════════════════════════════════════════════

   222→  private saveConfig(): void {
   223→    if (!this.config) return;
   224→    if (this.storageAvailable && typeof window !== 'undefined') {
   225→      try {
   226→        window.localStorage.setItem(STORAGE_KEY, JSON.stringify(this.config));
   227→      } catch {
   228→        // Storage full or unavailable — silently fail
   229→      }
   230→    }
   231→  }
   232→
   233→  private loadConfig(): JellyfinConfig | null {
   234→    if (!this.storageAvailable || typeof window === 'undefined') return null;
   235→    try {
   236→      const raw = window.localStorage.getItem(STORAGE_KEY);
   237→      if (!raw) return null;
   238→      const parsed = JSON.parse(raw) as JellyfinConfig;
   239→      // Basic validation
   240→      if (!parsed.serverUrl || !parsed.accessToken || !parsed.userId) return null;
   241→      return parsed;
   242→    } catch {
   243→      return null;
   244→    }
   245→  }

   247→  // ═══════════════════════════════════════════════════════
   248→  // Core HTTP
   249→  // ═══════════════════════════════════════════════════════
   250→
   251→  private getBaseUrl(): string {
   252→    if (!this.config) throw new JellyfinNotAuthenticatedError();
   253→    return normalizeServerUrl(this.config.serverUrl);
   254→  }
   255→
   256→  private getHeaders(): Record<string, string> {
   257→    if (!this.config) throw new JellyfinNotAuthenticatedError();
   258→    return {
   259→      'X-Emby-Token': this.config.accessToken,
   260→      'Content-Type': 'application/json',
   261→      'Accept': 'application/json',
   262→    };
   263→  }
   264→
   265→  private async request<T>(path: string, options: RequestInit = {}, skipAuth = false): Promise<T> {
   266→    const injectedBaseUrl = (options.headers as Record<string, string> | undefined)?.__baseUrl;
   267→    const baseUrl = skipAuth
   268→      ? normalizeServerUrl(injectedBaseUrl || '')
   269→      : this.getBaseUrl();
   270→    const url = `${baseUrl}${path}`;
   271→
   272→    // Build headers — extract __baseUrl before passing to fetch
   273→    const externalHeaders = { ...(options.headers as Record<string, string> | undefined) };
   274→    delete externalHeaders['__baseUrl'];
   275→
   276→    const headers: Record<string, string> = skipAuth
   277→      ? externalHeaders
   278→      : {
   279→          ...this.getHeaders(),
   280→          ...externalHeaders,
   281→        };
   282→
   283→    // Remove Content-Type for GET/DELETE if no body
   284→    if (!options.body && (options.method === 'GET' || options.method === 'DELETE')) {
   285→      delete headers['Content-Type'];
   286→    }
   287→
   288→    let response: Response;
   289→    try {
   290→      response = await fetch(url, {
   291→        ...options,
   292→        headers,
   293→      });
   294→    } catch (err) {
   295→      throw new JellyfinApiError(
   296→        `Network error: ${err instanceof Error ? err.message : 'Unknown error'}`,
   297→        0,
   298→        path,
   299→        err
   300→      );
   301→    }
   302→
   303→    if (!response.ok) {
   304→      let errorBody = '';
   305→      try {
   306→        errorBody = await response.text();
   307→      } catch {
   308→        // ignore
   309→      }
   310→      throw new JellyfinApiError(
   311→        `Jellyfin API error ${response.status}: ${response.statusText} — ${errorBody}`,
   312→        response.status,
   313→        path
   314→      );
   315→    }
   316→
   317→    // Some endpoints return empty responses (e.g., DELETE, favorite toggles)
   318→    const contentType = response.headers.get('content-type');
   319→    if (contentType && contentType.includes('application/json')) {
   320→      return response.json() as Promise<T>;
   321→    }
   322→
   323→    return undefined as unknown as T;
   324→  }

   330→  /**
   331→   * Connect to a Jellyfin server. Authenticates and fetches server info.
   332→   * Persists the configuration to localStorage.
   333→   */
   334→  async connect(serverUrl: string, username: string, password: string): Promise<JellyfinConfig> {
   335→    const baseUrl = normalizeServerUrl(serverUrl);
   336→
   337→    // Step 1: Authenticate (skip auth headers — we don't have a token yet)
   338→    const authResponse = await this.request<JellyfinAuthResponse>(
   339→      '/Users/AuthenticateByName',
   340→      {
   341→        method: 'POST',
   342→        body: JSON.stringify({ Username: username, Pw: password }),
   343→        headers: {
   344→          'Content-Type': 'application/json',
   345→          'Accept': 'application/json',
   346→          'Authorization': `MediaBrowser Client="DSP", Device="Browser", DeviceId="dsp-jellyfin-${typeof window !== 'undefined' ? window.navigator.userAgent.slice(0, 32).replace(/[^a-zA-Z0-9]/g, '') : 'ssr'}", Version="1.0.0"`,
   347→          '__baseUrl': baseUrl,
   348→        } as Record<string, string>,
   349→      },
   350→      true // skipAuth
   351→    );
   352→
   353→    const token = authResponse.AccessToken;
   354→    const userId = authResponse.User.Id;
   355→    const serverId = authResponse.ServerId;
   356→
   357→    // Step 2: Fetch server info
   358→    let serverName = 'Jellyfin';
   359→    let serverVersion = 'unknown';
   360→    try {
   361→      // Temporarily set config for the getSystemInfo call
   362→      this.config = {
   363→        serverUrl: baseUrl,
   364→        username,
   365→        accessToken: token,
   366→        userId,
   367→        serverId,
   368→        serverName,
   369→        serverVersion,
   370→        connectedAt: new Date().toISOString(),
   371→        lastHeartbeat: new Date().toISOString(),
   372→        musicLibraryId: '',
   373→      };
   374→      const sysInfo = await this.getSystemInfo();
   375→      serverName = sysInfo.ServerName;
   376→      serverVersion = sysInfo.Version;
   377→    } catch {
   378→      // If system info fails, continue with defaults
   379→    }
   380→
   381→    // Step 3: Auto-detect music library
   382→    let musicLibraryId = '';
   383→    try {
   384→      const views = await this.getViews();
   385→      const musicFolder = views.Items.find(
   386→        item => item.Type === 'CollectionFolder' && item.Name.toLowerCase().includes('music')
   387→      );
   388→      if (musicFolder) {
   389→        musicLibraryId = musicFolder.Id;
   390→      }
   391→    } catch {
   392→      // Continue without auto-detected library
   393→    }
   394→
   395→    // Step 4: Build and save config
   396→    const config: JellyfinConfig = {
   397→      serverUrl: baseUrl,
   398→      username,
   399→      accessToken: token,
   400→      userId,
   401→      serverId,
   402→      serverName,
   403→      serverVersion,
   404→      connectedAt: new Date().toISOString(),
   405→      lastHeartbeat: new Date().toISOString(),
   406→      musicLibraryId,
   407→    };
   408→
   409→    this.config = config;
   410→    this.saveConfig();
   411→
   412→    return config;
   413→  }

   415→  /**
   416→   * Disconnect from the Jellyfin server. Revokes the token and clears local state.
   417→   */
   418→  async disconnect(): Promise<void> {
   419→    if (this.config) {
   420→      try {
   421→        await this.request<void>(`/Users/${this.config.userId}/Tokens`, {
   422→          method: 'DELETE',
   423→        });
   424→      } catch {
   425→        // Best-effort — continue clearing local state
   426→      }
   427→    }
   428→
   429→    this.config = null;
   430→    if (this.storageAvailable && typeof window !== 'undefined') {
   431→      try {
   432→        window.localStorage.removeItem(STORAGE_KEY);
   433→      } catch {
   434→        // ignore
   435→      }
   436→    }
   437→  }

   790→  // ═══════════════════════════════════════════════════════

   792→  /**
   793→   * Get a playable audio stream URL for an item.
   794→   * Prefers direct play for best quality; falls back to transcoded stream.
   795→   */
   796→  async getPlaybackStreamUrl(
   797→    itemId: string,
   798→    maxBitrate: number = DEFAULT_MAX_BITRATE
   799→  ): Promise<string> {
   800→    const baseUrl = this.getBaseUrl();
   801→    const token = this.config!.accessToken;
   802→    const userId = this.config!.userId;
   803→
   804→    try {
   805→      // Request playback info to determine best playback method
   806→      const playbackInfo = await this.request<JellyfinPlaybackInfo>(
   807→        `/Items/${itemId}/PlaybackInfo`,
   808→        {
   809→          method: 'POST',
   810→          body: JSON.stringify({
   811→            UserId: userId,
   812→            MaxStreamingBitrate: maxBitrate,
   813→            AutoOpenLiveStreams: true,
   814→          }),
   815→        }
   816→      );
   817→
   818→      if (playbackInfo.MediaSources && playbackInfo.MediaSources.length > 0) {
   819→        const source = playbackInfo.MediaSources[0];
   820→
   821→        // Prefer direct play — best quality, no server transcoding
   822→        if (source.SupportsDirectPlay) {
   823→          return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${token}&UserId=${userId}`;
   824→        }
   825→
   826→        // Fallback to direct stream
   827→        if (source.SupportsDirectStream && source.DirectStreamUrl) {
   828→          return `${baseUrl}${source.DirectStreamUrl}`;
   829→        }
   830→
   831→        // Fallback to transcoded stream (HLS)
   832→        if (source.SupportsTranscoding && source.TranscodingUrl) {
   833→          return `${baseUrl}${source.TranscodingUrl}`;
   834→        }
   835→
   836→        // Last resort: construct a universal audio transcoding URL
   837→        const container = source.Container || 'mp3';
   838→        return `${baseUrl}/Audio/${itemId}/universal?api_key=${token}&UserId=${userId}&MaxStreamingBitrate=${maxBitrate}&Container=${container}&TranscodingMaxAudioChannels=2&SegmentContainer=ts&AudioCodec=aac&Protocol=hls`;
   839→      }
   840→    } catch {
   841→      // If playback info fails, fall back to direct stream URL
   842→    }
   843→
   844→    // Fallback: try static stream directly
   845→    return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${token}&UserId=${userId}`;
   846→  }

   948→        method: 'POST',

   951→          SupportedCommands: [
   952→            'Play',
   953→            'PlayNext',
   954→            'PlayPrevious',
   955→            'Stop',
   956→            'Seek',
   957→            'Mute',
   958→            'Unmute',
   959→            'ToggleMute',
   960→            'VolumeUp',
   961→            'VolumeDown',
   962→            'SetVolume',
   963→            'SetRepeatMode',
   964→            'SetShuffleMode',
   965→          ],
   966→          SupportsMediaControl: true,
   967→          SupportsPersistentIdentifier: true,
   968→        }),
   969→      });
   970→
   971→      this.config.lastHeartbeat = new Date().toISOString();
   972→      this.saveConfig();
   973→    } catch {
   974→      // Heartbeat failure is non-critical
   975→    }
   976→  }
