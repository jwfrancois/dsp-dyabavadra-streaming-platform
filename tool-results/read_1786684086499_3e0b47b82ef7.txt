    83→function localTrackToTrack(lt: LocalTrack): Track {
    84→  return {
    85→    id: lt.id,
    86→    title: lt.title,
    87→    albumId: lt.album,
    88→    albumName: lt.album,
    89→    artistId: lt.artist,
    90→    artistName: lt.artist,
    91→    trackNumber: lt.trackNumber,
    92→    discNumber: lt.discNumber,
    93→    duration: lt.duration,
    94→    format: lt.format,
    95→    bitDepth: lt.bitDepth,
    96→    sampleRate: lt.sampleRate,
    97→    channels: lt.channels,
    98→    bitrate: lt.bitrate,
    99→    filePath: lt.filePath,
   100→    fileSize: lt.fileSize,
   101→    composers: lt.composer ? [lt.composer] : [],
   102→    performers: [],
   103→    genre: lt.genre,
   104→    loved: false,
   105→    playCount: 0,
   106→    source: 'local',
   107→    isAvailable: true,
   108→  };
   109→}
   110→
   111→function playLocalTrack(lt: LocalTrack) {
   112→  const track = localTrackToTrack(lt);
   113→  usePlayerStore.getState().play(track);
   114→}
   115→
   116→function playLocalTracks(lts: LocalTrack[]) {
   117→  if (lts.length === 0) return;
   118→  const queue = lts.map(localTrackToTrack);
   119→  usePlayerStore.getState().setQueue(queue, 0);
   120→}

   286→
   287→function BrowseLocalPanel() {
   288→  const localStore = useLocalLibraryStore();
   289→  const [view, setView] = React.useState<BrowseLocalView>('artists');
   290→  const [searchQuery, setSearchQuery] = React.useState('');
   291→
   292→  // Breadcrumb navigation state
   293→  const [selectedArtist, setSelectedArtist] = React.useState<string | null>(null);
   294→  const [selectedAlbum, setSelectedAlbum] = React.useState<string | null>(null);
   295→
   296→  const allArtists = localStore.getArtists();
   297→  const allAlbums = localStore.getAlbums();
   298→  const allTracks = localStore.tracks;
   299→
   300→  // Derived data based on navigation state
   301→  const breadcrumbs: BreadcrumbItem[] = [{ label: 'Artists', onClick: () => { setSelectedArtist(null); setSelectedAlbum(null); setView('artists'); } }];
   302→
   303→  let displayedArtists = allArtists;
   304→  let displayedAlbums = allAlbums;
   305→  let displayedTracks = allTracks;
   306→
   307→  if (searchQuery.trim()) {
   308→    displayedTracks = localStore.searchTracks(searchQuery.trim());
   309→    displayedAlbums = allAlbums.filter(a =>
   310→      displayedTracks.some(t => t.album === a.name && t.artist === a.artist)
   311→    );
   312→    displayedArtists = allArtists.filter(a =>
   313→      displayedTracks.some(t => t.artist === a)
   314→    );
   315→  } else if (selectedArtist && selectedAlbum) {
   316→    // Artist > Album > Tracks
   317→    breadcrumbs.push(
   318→      { label: selectedArtist, onClick: () => { setSelectedAlbum(null); } },
   319→      { label: selectedAlbum, onClick: () => {} },
   320→    );
   321→    displayedTracks = localStore.getTracksByAlbum(selectedAlbum).filter(t => t.artist === selectedArtist);
   322→  } else if (selectedArtist) {
   323→    // Artist > Albums
   324→    breadcrumbs.push({ label: selectedArtist, onClick: () => {} });
   325→    displayedAlbums = allAlbums.filter(a => a.artist === selectedArtist);
   326→    displayedTracks = localStore.getTracksByArtist(selectedArtist);
   327→  } else if (selectedAlbum) {
   328→    // Albums > Tracks
   329→    breadcrumbs.push({ label: selectedAlbum, onClick: () => {} });
   330→    displayedTracks = localStore.getTracksByAlbum(selectedAlbum);
   331→  }
   332→
   333→  const showArtistsView = !selectedArtist && !selectedAlbum && view === 'artists' && !searchQuery.trim();
   334→  const showAlbumsView = (selectedArtist || view === 'albums') && !selectedAlbum && !searchQuery.trim();
   335→  const showTracksView = selectedAlbum || view === 'tracks' || searchQuery.trim();
   336→
   337→  return (
   338→    <div className="space-y-4">
   339→      {/* Search and View Switcher */}
   340→      <div className="flex items-center gap-3">
   341→        <div className="relative flex-1">
   342→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   343→          <Input
   344→            placeholder="Search local library..."
   345→            value={searchQuery}
   346→            onChange={e => { setSearchQuery(e.target.value); if (e.target.value) { setSelectedArtist(null); setSelectedAlbum(null); } }}
   347→            className="pl-9 h-9 text-sm"
   348→          />
   349→        </div>
   350→        <div className="flex items-center bg-surface rounded-lg p-0.5">
   351→          {([['artists', User, 'Artists'], ['albums', Disc3, 'Albums'], ['tracks', Music2, 'Tracks']] as const).map(([key, Icon, label]) => (
   352→            <Button
   353→              key={key}
   354→              variant={view === key ? 'default' : 'ghost'}
   355→              size="sm"
   356→              className="h-8 text-xs gap-1.5 rounded-md"
   357→              onClick={() => { setView(key as BrowseLocalView); setSelectedArtist(null); setSelectedAlbum(null); setSearchQuery(''); }}
   358→            >
   359→              <Icon className="w-3.5 h-3.5" /> {label}
   360→            </Button>
   361→          ))}
   362→        </div>
   363→      </div>
   364→
   365→      {/* Breadcrumb */}
   366→      {(selectedArtist || selectedAlbum) && !searchQuery.trim() && (
   367→        <div className="flex items-center gap-1 text-xs">
   368→          {breadcrumbs.map((crumb, i) => (
   369→            <React.Fragment key={i}>
   370→              {i > 0 && <ChevronRight className="w-3 h-3 text-muted-foreground" />}
   371→              <button
   372→                className={`${i === breadcrumbs.length - 1 ? 'text-foreground font-medium' : 'text-muted-foreground hover:text-foreground'} transition-colors`}
   373→                onClick={crumb.onClick}
   374→              >
   375→                {crumb.label}
   376→              </button>
   377→            </React.Fragment>
   378→          ))}
   379→        </div>
   380→      )}
   381→
   382→      {/* Empty State */}
   383→      {allTracks.length === 0 && (
   384→        <Card className="bg-card border-border">
   385→          <CardContent className="p-12 text-center">
   386→            <LibraryIcon className="w-12 h-12 mx-auto mb-4 text-muted-foreground/30" />
   387→            <h3 className="text-sm font-semibold mb-1">No Local Tracks</h3>
   388→            <p className="text-xs text-muted-foreground max-w-md mx-auto">
   389→              Scan your music folders to build your local library. Go to the &quot;Scan Folders&quot; tab to add directories and start scanning.
   390→            </p>
   391→          </CardContent>
   392→        </Card>
   393→      )}
   394→
   395→      {/* Artists Grid */}
   396→      {showArtistsView && allTracks.length > 0 && (
   397→        <div className="space-y-3">
   398→          <div className="flex items-center justify-between">
   399→            <h2 className="text-sm font-semibold">Artists ({displayedArtists.length})</h2>
   400→          </div>
   401→          {displayedArtists.length === 0 ? (
   402→            <p className="text-xs text-muted-foreground py-8 text-center">No artists found.</p>
   403→          ) : (
   404→            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-6 gap-3">
   405→              {displayedArtists.sort().map(artist => {
   406→                const artistAlbums = allAlbums.filter(a => a.artist === artist);
   407→                const artistTrackCount = artistAlbums.reduce((s, a) => s + a.trackCount, 0);
   408→                return (
   409→                  <Card
   410→                    key={artist}
   411→                    className="bg-card border-border hover:border-primary/30 cursor-pointer transition-all group"
   412→                    onClick={() => { setSelectedArtist(artist); setSelectedAlbum(null); setView('albums'); }}
   413→                  >
   414→                    <CardContent className="p-4 text-center">
   415→                      <div className="w-16 h-16 mx-auto rounded-full bg-gradient-to-br from-primary/20 to-primary/5 flex items-center justify-center mb-3 group-hover:from-primary/30 group-hover:to-primary/10 transition-all">
   416→                        <User className="w-7 h-7 text-primary/60" />
   417→                      </div>
   418→                      <p className="text-xs font-medium truncate">{artist}</p>
   419→                      <p className="text-[10px] text-muted-foreground mt-0.5">{artistAlbums.length} album{artistAlbums.length !== 1 ? 's' : ''} · {artistTrackCount} tracks</p>
   420→                    </CardContent>
   421→                  </Card>
   422→                );
   423→              })}
   424→            </div>
   425→          )}
   426→        </div>
   427→      )}
   428→
   429→      {/* Albums Grid */}
   430→      {showAlbumsView && allTracks.length > 0 && (
   431→        <div className="space-y-3">
   432→          <div className="flex items-center justify-between">
   433→            <h2 className="text-sm font-semibold">{selectedArtist ? `${selectedArtist} — ` : ''}Albums ({displayedAlbums.length})</h2>
   434→          </div>
   435→          {displayedAlbums.length === 0 ? (
   436→            <p className="text-xs text-muted-foreground py-8 text-center">No albums found.</p>
   437→          ) : (
   438→            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 gap-4">
   439→              {displayedAlbums.map(album => {
   440→                const albumTracks = localStore.getTracksByAlbum(album.name).filter(t => !selectedArtist || t.artist === selectedArtist);
   441→                const firstTrack = albumTracks[0];
   442→                return (
   443→                  <Card
   444→                    key={`${album.artist}-${album.name}`}
   445→                    className="bg-card border-border hover:border-primary/30 cursor-pointer transition-all group"
   446→                    onDoubleClick={() => playLocalTracks(albumTracks.sort((a, b) => a.trackNumber - b.trackNumber))}
   447→                    onClick={() => { setSelectedAlbum(album.name); if (selectedArtist) { /* keep artist */ } else { setSelectedArtist(album.artist); } }}
   448→                  >
   449→                    <CardContent className="p-3">
   450→                      {/* Cover Art */}
   451→                      <div className="relative aspect-square rounded-lg overflow-hidden mb-3">
   452→                        {firstTrack?.coverArt ? (
   453→                          <img
   454→                            src={`data:image/jpeg;base64,${firstTrack.coverArt}`}
   455→                            alt={album.name}
   456→                            className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-300"
   457→                          />
   458→                        ) : (
   459→                          <div className={`w-full h-full bg-gradient-to-br ${getCoverGradient(album.name + album.artist)} flex items-center justify-center`}>
   460→                            <Disc3 className="w-10 h-10 text-white/20" />
   461→                          </div>
   462→                        )}

   602→function ScanFoldersPanel() {
   603→  const localStore = useLocalLibraryStore();
   604→  const [directoryInput, setDirectoryInput] = React.useState('');
   605→
   606→  const { tracks, isScanning, scanProgress, scanError, directories, lastScanTime } = localStore;
   607→
   608→  const handleScan = React.useCallback(async () => {
   609→    const dir = directoryInput.trim();
   610→    if (!dir) return;
   611→    localStore.addDirectory(dir);
   612→    setDirectoryInput('');
   613→    await localStore.startScan(dir);
   614→  }, [directoryInput, localStore]);
   615→
   616→  const handleScanAll = React.useCallback(async () => {
   617→    await localStore.scanAllDirectories();
   618→  }, [localStore]);
   619→
   620→  // Compute scan stats
   621→  const totalDuration = tracks.reduce((s, t) => s + t.duration, 0);
   622→  const formatBreakdown = React.useMemo(() => {
   623→    const counts: Record<string, number> = {};
   624→    tracks.forEach(t => {
   625→      const fmt = t.format.toUpperCase();
   626→      counts[fmt] = (counts[fmt] || 0) + 1;
   627→    });
   628→    return Object.entries(counts).sort((a, b) => b[1] - a[1]);
   629→  }, [tracks]);
   630→
   631→  const avgSampleRate = tracks.length > 0
   632→    ? Math.round(tracks.reduce((s, t) => s + t.sampleRate, 0) / tracks.length)
   633→    : 0;
   634→  const avgBitDepth = tracks.length > 0
   635→    ? (tracks.reduce((s, t) => s + t.bitDepth, 0) / tracks.length).toFixed(1)
   636→    : '0';

   638→  return (

   640→      {/* Add Folder */}
   641→      <Card className="bg-card border-border">

   668→            {directories.length > 1 && (
   669→              <Button
   670→                variant="outline"
   671→                onClick={handleScanAll}
   672→                disabled={isScanning}
   673→                className="h-9 gap-1.5"
   674→              >
   675→                <RefreshCw className="w-4 h-4" />
   676→                Scan All ({directories.length})
   677→              </Button>

   692→              </p>
