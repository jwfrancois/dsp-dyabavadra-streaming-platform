   133→   133→
   134→   134→function countryCodeToFlag(code: string): string {
   135→   135→  if (!code || code.length !== 2) return '';
   136→   136→  const c = code.toUpperCase();
   137→   137→  const base = 0x1F1E6 - 65;
   138→   138→  const char1 = String.fromCodePoint(base + c.charCodeAt(0));
   139→   139→  const char2 = String.fromCodePoint(base + c.charCodeAt(1));
   140→   140→  return char1 + char2;
   141→   141→}

   144→   144→// SORT OPTIONS

   146→   146→
   147→   147→type SortOption = 'az' | 'za' | 'country' | 'genre';
   148→   148→
   149→   149→function sortStations(stations: RadioStation[], sort: SortOption): RadioStation[] {
   150→   150→  const sorted = [...stations];
   151→   151→  switch (sort) {
   152→   152→    case 'az':
   153→   153→      return sorted.sort((a, b) => a.name.localeCompare(b.name));
   154→   154→    case 'za':
   155→   155→      return sorted.sort((a, b) => b.name.localeCompare(a.name));
   156→   156→    case 'country':
   157→   157→      return sorted.sort((a, b) => a.country.localeCompare(b.country) || a.name.localeCompare(b.name));
   158→   158→    case 'genre':
   159→   159→      return sorted.sort((a, b) => a.genre.localeCompare(b.genre) || a.name.localeCompare(b.name));
   160→   160→    default:
   161→   161→      return sorted;
   162→   162→  }
   163→   163→}

   166→   166→// DEBOUNCED HOOK
   167→   167→// ═══════════════════════════════════════════════════════
   168→   168→
   169→   169→function useDebouncedValue<T>(value: T, delay: number): T {
   170→   170→  const [debounced, setDebounced] = useState(value);
   171→   171→  const timerRef = useRef<ReturnType<typeof setTimeout>>();
   172→   172→
   173→   173→  useEffect(() => {
   174→   174→    if (timerRef.current) clearTimeout(timerRef.current);
   175→   175→    timerRef.current = setTimeout(() => setDebounced(value), delay);
   176→   176→    return () => {
   177→   177→      if (timerRef.current) clearTimeout(timerRef.current);
   178→   178→    };
   179→   179→  }, [value, delay]);
   180→   180→
   181→   181→  return debounced;
   182→   182→}
