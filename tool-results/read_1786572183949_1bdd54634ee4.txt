    28→function handlePlayEpisode(episode: PodcastEpisode) {
    29→  // Delegate to the podcast store which handles all audio setup
    30→  usePodcastStore.getState().playEpisode(episode);
    31→}

    33→function SpeedBadge() {
    34→  const { playbackSpeed, cyclePlaybackSpeed } = usePodcastStore();
    35→
    36→  const handleClick = () => {
    37→    cyclePlaybackSpeed();
    38→    setTimeout(() => {
    39→      const newSpeed = usePodcastStore.getState().playbackSpeed;
    40→      audioSetPlaybackSpeed(newSpeed);
    41→    }, 0);
    42→  };
    43→
    44→  return (
    45→    <Button
    46→      variant={playbackSpeed === 1.0 ? 'secondary' : 'default'}
    47→      size="sm"
    48→      className="h-7 text-xs font-mono min-w-[48px] justify-center"
    49→      onClick={handleClick}
    50→    >
    51→      {playbackSpeed}x
    52→    </Button>
    53→  );
    54→}

    86→export function PodcastsView() {
    87→  const { navigate } = useUIStore();
    88→  const {
    89→    episodeStates, subscribedShowIds, toggleSubscribe,
    90→    markAllPlayed,
    91→    playbackSpeed, cyclePlaybackSpeed, skipSilence, toggleSkipSilence,
    92→    sleepTimerMinutes, setSleepTimer, toggleEpisodeDownload,
    93→    toggleEpisodeFavorite, currentEpisode, isPodcastMode,
    94→  } = usePodcastStore();
    95→  const { isPlaying, playbackMode } = usePlayerStore();
    96→
    97→  const [activeTab, setActiveTab] = useState('subscriptions');
    98→  const [discoverSearch, setDiscoverSearch] = useState('');
    99→  const [discoverGenre, setDiscoverGenre] = useState('all');
   100→  const [allEpisodesFilter, setAllEpisodesFilter] = useState('all');
   101→  const [showSleepTimer, setShowSleepTimer] = useState(false);
   102→  const [unsubscribeTarget, setUnsubscribeTarget] = useState<PodcastShow | null>(null);
   103→  const [expandedShowId, setExpandedShowId] = useState<string | null>(null);
   104→
   105→  // iTunes search state
   106→  const [iTunesResults, setITunesResults] = useState<Array<{
   107→    id: string; itunesId: number; title: string; author: string;
   108→    artworkUrl: string; genre: string; feedUrl: string;
   109→    episodeCount: number; description: string;
   110→  }>>([]);
   111→  const [iTunesLoading, setITunesLoading] = useState(false);
   112→  const [iTunesError, setITunesError] = useState<string | null>(null);
   113→  const [iTunesSearched, setITunesSearched] = useState(false);
   114→  const searchDebounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   115→
   116→  // iTunes search with debounce
   117→  const doITunesSearch = useCallback(async (query: string) => {
   118→    if (!query.trim()) {
   119→      setITunesResults([]);
   120→      setITunesSearched(false);
   121→      setITunesError(null);
   122→      return;
   123→    }
   124→    setITunesLoading(true);
   125→    setITunesError(null);
   126→    setITunesSearched(true);
   127→    try {
   128→      const res = await fetch(`/api/podcasts/search?q=${encodeURIComponent(query.trim())}&limit=25`);
   129→      if (!res.ok) throw new Error(`Search failed (${res.status})`);
   130→      const data = await res.json();
   131→      setITunesResults((data.results || []).map((r: Record<string, unknown>, idx: number) => ({
   132→        id: String(r.id || `result-${idx}`),
   133→        itunesId: Number(r.itunesId || 0),
   134→        title: String(r.title || 'Unknown'),
   135→        author: String(r.author || 'Unknown Author'),
   136→        artworkUrl: String(r.artworkUrl || r.artworkUrlLarge || ''),
   137→        genre: String(r.genre || 'Podcast'),
   138→        feedUrl: String(r.feedUrl || ''),
   139→        episodeCount: Number(r.episodeCount || 0),
   140→        description: String(r.description || ''),
   141→      })));
   142→    } catch (err) {
   143→      setITunesError(err instanceof Error ? err.message : 'Search failed');
   144→      setITunesResults([]);
   145→    } finally {
   146→      setITunesLoading(false);
   147→    }
   148→  }, []);
   149→
   150→  // Debounced search on input change
   151→  useEffect(() => {
   152→    if (searchDebounceRef.current) clearTimeout(searchDebounceRef.current);
   153→    searchDebounceRef.current = setTimeout(() => {
   154→      doITunesSearch(discoverSearch);
   155→    }, 400);
   156→    return () => {
   157→      if (searchDebounceRef.current) clearTimeout(searchDebounceRef.current);
   158→    };
   159→  }, [discoverSearch, doITunesSearch]);
   160→
   161→  const subscribedShows = podcastShows.filter(s => subscribedShowIds.includes(s.id));
   162→  const newEpisodes = getAllNewEpisodes();
   163→
   164→  const allSubscribedEpisodes = useMemo(() => {
   165→    return podcastEpisodes
   166→      .filter(ep => subscribedShowIds.includes(ep.showId))
   167→      .sort((a, b) => new Date(b.publishDate).getTime() - new Date(a.publishDate).getTime());
   168→  }, [subscribedShowIds]);
   169→
   170→  const filteredAllEpisodes = useMemo(() => {
   171→    return allSubscribedEpisodes.filter(ep => {
   172→      const st = episodeStates[ep.id];
   173→      if (!st) return allEpisodesFilter === 'all';
   174→      switch (allEpisodesFilter) {
   175→        case 'unplayed': return !st.isPlayed && !st.completed;
   176→        case 'in-progress': return !st.completed && st.resumePosition > 0;
   177→        case 'favorites': return st.favorite;
   178→        case 'downloaded': return st.isDownloaded;
   179→        default: return true;
   180→      }
   181→    });
   182→  }, [allSubscribedEpisodes, episodeStates, allEpisodesFilter]);
   183→
   184→  const downloadedEpisodes = useMemo(() => {
   185→    return podcastEpisodes.filter(ep => episodeStates[ep.id]?.isDownloaded);
   186→  }, [episodeStates]);
   187→
   188→  const allGenres = useMemo(() => {
   189→    const genres = new Set(podcastShows.map(s => s.genre));
   190→    return ['all', ...Array.from(genres)];
   191→  }, []);
   192→
   193→  const discoverShows = useMemo(() => {
   194→    return podcastShows.filter(s => {
   195→      const matchesSearch = discoverSearch.length === 0 ||
   196→        s.title.toLowerCase().includes(discoverSearch.toLowerCase()) ||
   197→        s.author.toLowerCase().includes(discoverSearch.toLowerCase());
   198→      const matchesGenre = discoverGenre === 'all' || s.genre === discoverGenre;
   199→      return matchesSearch && matchesGenre;
   200→    });
   201→  }, [discoverSearch, discoverGenre]);
   202→
   203→  const newAndNoteworthy = useMemo(() => {
   204→    return podcastShows.filter(s => getEpisodesByShow(s.id).length > 0).slice(0, 3);
   205→  }, []);
   206→
   207→  const trending = useMemo(() => {
   208→    return [...podcastShows].sort((a, b) => b.episodeCount - a.episodeCount).slice(0, 4);
   209→  }, []);
   210→
   211→  const isCurrentlyPlaying = (episodeId: string) => {
   212→    return isPodcastMode && currentEpisode?.id === episodeId && isPlaying;
   213→  };
   214→
   215→  const isCurrentEpisode = (episodeId: string) => {
   216→    return isPodcastMode && currentEpisode?.id === episodeId;
   217→  };
   218→
   219→  const handleShowContextMenu = (e: React.MouseEvent | React.TouchEvent, show: PodcastShow) => {
   220→    if (subscribedShowIds.includes(show.id)) {
   221→      e.preventDefault();
   222→      e.stopPropagation();
   223→      setUnsubscribeTarget(show);
   224→    }
   225→  };
   226→
   227→  const confirmUnsubscribe = () => {
   228→    if (unsubscribeTarget) {
   229→      toggleSubscribe(unsubscribeTarget.id);
   230→      setUnsubscribeTarget(null);
   231→    }
   232→  };
   233→
   234→  const renderEpisodeRow = (ep: PodcastEpisode, size: string, showActions: boolean) => {
   235→    const show = podcastShows.find(s => s.id === ep.showId);
   236→    const state = episodeStates[ep.id];
   237→    const active = isCurrentEpisode(ep.id);
   238→    const playing = isCurrentlyPlaying(ep.id);
   239→    const unplayed = state && !state.isPlayed && !state.completed;
   240→    const played = state?.completed;
   241→    const progress = state && !state.completed && state.resumePosition > 0
   242→      ? (state.resumePosition / ep.duration) * 100
   243→      : 0;
   244→    const artSize = size === 'sm' ? 'w-10 h-10' : 'w-12 h-12';
   245→
   246→    return (
   247→      <div
   248→        key={ep.id}
   249→        className={[
   250→          'flex items-center gap-3 p-3 rounded-lg transition-colors group cursor-pointer',
   251→          active ? 'bg-primary/10 border border-primary/30' : 'hover:bg-accent/30',
   252→          played ? 'opacity-60' : '',
   253→        ].join(' ')}
   254→        onClick={() => handlePlayEpisode(ep)}
   255→      >
   256→        <div className={`relative flex-shrink-0 ${artSize} rounded bg-gradient-to-br ${getCoverGradient(ep.showId)}`}>
   257→          {unplayed && (
   258→            <span className="absolute -top-1 -left-1 w-2.5 h-2.5 rounded-full bg-blue-500 border-2 border-background" />
   259→          )}
   260→          {(playing || (active && !isPlaying)) && (
   261→            <div className="absolute inset-0 flex items-center justify-center bg-black/40 rounded">
   262→              {playing ? (
   263→                <Pause className="w-4 h-4 text-white" />
   264→              ) : (
   265→                <Play className="w-4 h-4 text-white ml-0.5" />
   266→              )}
   267→            </div>
   268→          )}
   269→          {progress > 0 && (
   270→            <div className="absolute bottom-0 left-0 right-0 h-0.5 bg-white/20 rounded-b">
   271→              <div className="h-full bg-primary rounded-b" style={{ width: `${progress}%` }} />
   272→            </div>
   273→          )}
   274→        </div>
   275→
   276→        <div className="flex-1 min-w-0">
   277→          <p className={`text-sm font-medium truncate ${active ? 'text-primary' : ''}`}>{ep.title}</p>
   278→          <p className="text-xs text-muted-foreground truncate">
   279→            {show?.title}{' \u00b7 '}{formatDate(ep.publishDate)}
   280→          </p>
   281→        </div>
   282→
   283→        <span className="text-xs text-muted-foreground hidden sm:inline flex-shrink-0">
   284→          {formatEpisodeDuration(ep.duration)}
   285→        </span>
   286→
   287→        {progress > 0 && (
   288→          <span className="text-[10px] text-muted-foreground flex-shrink-0">
   289→            {Math.round(progress)}%
   290→          </span>
   291→        )}
   292→
   293→        {active && playbackMode === 'podcast' && <SpeedBadge />}
   294→
   295→        {showActions && (
   296→          <div className="flex items-center gap-1 flex-shrink-0 opacity-0 group-hover:opacity-100 transition-opacity">
   297→            <Button
   298→              variant="ghost"
   299→              size="icon"
   300→              className={`h-7 w-7 ${state?.favorite ? 'text-red-400' : ''}`}
   301→              onClick={(e) => { e.stopPropagation(); toggleEpisodeFavorite(ep.id); }}
   302→            >
   303→              <Heart className={`w-3.5 h-3.5 ${state?.favorite ? 'fill-current' : ''}`} />
   304→            </Button>
   305→            <Button
   306→              variant="ghost"
   307→              size="icon"
   308→              className={`h-7 w-7 ${state?.isDownloaded ? 'text-primary' : ''}`}
   309→              onClick={(e) => { e.stopPropagation(); toggleEpisodeDownload(ep.id); }}
   310→            >
   311→              <Download className="w-3.5 h-3.5" />
   312→            </Button>
   313→          </div>
   314→        )}
   315→      </div>
   316→    );
   317→  };
   318→
   319→  const renderShowCard = (show: PodcastShow) => {
   320→    const isSub = subscribedShowIds.includes(show.id);
   321→    const episodes = getEpisodesByShow(show.id);
   322→    const unplayed = episodes.filter(ep => {
   323→      const st = episodeStates[ep.id];
   324→      return st && !st.isPlayed && !st.completed;
   325→    });
   326→    const latestEp = episodes[0];
   327→    const latestState = latestEp ? episodeStates[latestEp.id] : null;
   328→
   329→    return (
   330→      <Card
   331→        key={show.id}
   332→        className="bg-card border-border hover:border-muted-foreground/20 cursor-pointer transition-all group"
   333→        onClick={() => navigate('podcast-detail', { showId: show.id })}
   334→        onContextMenu={(e) => handleShowContextMenu(e, show)}
   335→      >
   336→        <CardContent className="p-4">
   337→          <div className="flex gap-4">
   338→            <div className={`w-20 h-20 rounded-lg bg-gradient-to-br ${getCoverGradient(show.id)} flex-shrink-0 relative`}>
   339→              {unplayed.length > 0 && (
   340→                <span className="absolute -top-1.5 -right-1.5 w-5 h-5 rounded-full bg-signal-red text-[10px] font-bold text-white flex items-center justify-center">
   341→                  {unplayed.length > 9 ? '9+' : unplayed.length}
   342→                </span>
   343→              )}
   344→            </div>
   345→
   346→            <div className="flex-1 min-w-0">
   347→              <h3 className="text-sm font-semibold truncate">{show.title}</h3>
   348→              <p className="text-xs text-muted-foreground">{show.author}</p>
   349→              <p className="text-xs text-muted-foreground mt-1 line-clamp-2">{show.description}</p>
   350→              <div className="flex items-center gap-3 mt-2">
   351→                <Badge variant="outline" className="text-[10px]">{show.genre}</Badge>
   352→                <span className="text-[11px] text-muted-foreground">{show.episodeCount} episodes</span>
   353→                {latestEp && (
   354→                  <span className="text-[11px] text-muted-foreground flex items-center gap-1">
   355→                    <Rss className="w-3 h-3" />
   356→                    {formatDate(latestEp.publishDate)}
   357→                  </span>
   358→                )}
   359→              </div>
   360→            </div>
   361→
   362→            <div className="flex flex-col items-end gap-2 flex-shrink-0">
   363→              <Button
   364→                variant={isSub ? 'secondary' : 'default'}
   365→                size="sm"
   366→                className="h-8 gap-1.5"
   367→                onClick={(e) => {
   368→                  e.stopPropagation();
   369→                  if (isSub) { setUnsubscribeTarget(show); } else { toggleSubscribe(show.id); }
   370→                }}
   371→              >
   372→                {isSub ? <Check className="w-3.5 h-3.5" /> : <Plus className="w-3.5 h-3.5" />}
   373→                {isSub ? 'Subscribed' : 'Subscribe'}
   374→              </Button>
   375→
   376→              {latestEp && latestState && !latestState.completed && (
   377→                <Button
   378→                  variant="ghost"
   379→                  size="sm"
   380→                  className="h-8 gap-1.5 text-xs"
   381→                  onClick={(e) => { e.stopPropagation(); handlePlayEpisode(latestEp); }}
   382→                >
   383→                  <Play className="w-3.5 h-3.5" />
   384→                  {latestState.resumePosition > 0 ? 'Resume' : 'Play Latest'}
   385→                </Button>
   386→              )}
   387→
   388→              {latestEp && latestState && latestState.resumePosition > 0 && !latestState.completed && (
   389→                <div className="text-[10px] text-muted-foreground">
   390→                  {Math.round((latestState.resumePosition / latestEp.duration) * 100)}% listened
   391→                </div>
   392→              )}
   393→            </div>
   394→          </div>
   395→        </CardContent>
   396→      </Card>
   397→    );
   398→  };
   399→
   400→  const renderDiscoverShowCard = (show: PodcastShow) => {
   401→    const isSub = subscribedShowIds.includes(show.id);
   402→    const isExpanded = expandedShowId === show.id;
   403→    const episodes = getEpisodesByShow(show.id);
   404→    return (
   405→      <Card
   406→        key={show.id}
   407→        className="bg-card border-border hover:border-muted-foreground/20 transition-all"
   408→      >
   409→        <CardContent className="p-3">
   410→          <div className="flex gap-3 cursor-pointer" onClick={() => setExpandedShowId(isExpanded ? null : show.id)}>
   411→            <div className={`w-16 h-16 rounded-lg bg-gradient-to-br ${getCoverGradient(show.id)} flex-shrink-0`} />
   412→            <div className="flex-1 min-w-0">
   413→              <p className="text-sm font-medium truncate">{show.title}</p>
   414→              <p className="text-xs text-muted-foreground truncate">{show.author}</p>
   415→              <div className="flex items-center gap-2 mt-1">
   416→                <Badge variant="outline" className="text-[10px]">{show.genre}</Badge>
   417→                <span className="text-[10px] text-muted-foreground">{show.episodeCount} eps</span>
   418→                <span className="text-[10px] text-muted-foreground">{episodes.length} available</span>
   419→              </div>
   420→              <p className="text-[11px] text-muted-foreground mt-1 line-clamp-2">{show.description}</p>
   421→            </div>
   422→            <div className="flex flex-col items-end gap-2 flex-shrink-0">
   423→              <Button
   424→                variant={isSub ? 'secondary' : 'default'}
   425→                size="sm"
   426→                className="h-8 gap-1.5"
   427→                onClick={(e) => { e.stopPropagation(); toggleSubscribe(show.id); }}
   428→              >
   429→                {isSub ? <Check className="w-3.5 h-3.5" /> : <Plus className="w-3.5 h-3.5" />}
   430→                {isSub ? 'Subscribed' : 'Subscribe'}
   431→              </Button>
   432→              <Button
   433→                variant="ghost"
   434→                size="sm"
   435→                className="h-7 gap-1 text-xs"
   436→                onClick={(e) => { e.stopPropagation(); if (episodes.length > 0) handlePlayEpisode(episodes[0]); }}
   437→              >
   438→                <Play className="w-3 h-3" />
   439→                Play Latest
   440→              </Button>
   441→              <Button
   442→                variant="ghost"
   443→                size="sm"
   444→                className="h-7 text-xs text-muted-foreground"
   445→                onClick={(e) => { e.stopPropagation(); setExpandedShowId(isExpanded ? null : show.id); }}
   446→              >
   447→                {isExpanded ? 'Hide' : `${episodes.length} episodes`}
   448→              </Button>
   449→            </div>
   450→          </div>
   451→
   452→          {isExpanded && episodes.length > 0 && (
   453→            <div className="mt-3 ml-[76px] space-y-1 border-l-2 border-primary/20 pl-3">
   454→              <p className="text-xs font-semibold text-muted-foreground mb-2">Episodes</p>
   455→              {episodes.map(ep => renderEpisodeRow(ep, 'sm', false))}
   456→            </div>
   457→          )}
   458→        </CardContent>
   459→      </Card>
   460→    );
   461→  };
   462→
   463→  return (
   464→    <>
   465→      {unsubscribeTarget && (
   466→        <UnsubscribeConfirm
   467→          show={unsubscribeTarget}
   468→          onConfirm={confirmUnsubscribe}
   469→          onCancel={() => setUnsubscribeTarget(null)}
   470→        />
   471→      )}
   472→
   473→      <ScrollArea className="h-full">
   474→        <div className="p-6 max-w-5xl mx-auto">
   475→          <div className="flex items-center justify-between mb-6">
   476→            <h1 className="text-2xl font-bold flex items-center gap-2">
   477→              <Podcast className="w-6 h-6 text-primary" /> Podcasts
   478→            </h1>
   479→            <div className="flex items-center gap-2">
   480→              {newEpisodes.length > 0 && (
   481→                <Badge variant="outline" className="text-xs text-signal-red border-signal-red/30">
   482→                  <Circle className="w-2 h-2 mr-1 fill-signal-red text-signal-red" />
   483→                  {newEpisodes.length} new
   484→                </Badge>
   485→              )}
   486→              <Badge variant="secondary" className="text-xs">{subscribedShows.length} subscribed</Badge>
   487→            </div>
   488→          </div>
   489→
   490→          <Card className="bg-card border-border mb-6">
   491→            <CardContent className="p-3">
   492→              <div className="flex items-center gap-4 flex-wrap">
   493→                <div className="flex items-center gap-2">
   494→                  <span className="text-[11px] text-muted-foreground">Speed:</span>
   495→                  <SpeedBadge />
   496→                </div>
   497→                <div className="flex items-center gap-2">
   498→                  <Button
   499→                    variant={skipSilence ? 'secondary' : 'ghost'}
   500→                    size="sm"
   501→                    className="h-7 gap-1.5 text-xs"
