  1245→export function getSignalPath(trackId: string, zoneId: string): SignalPathStep[] {
  1246→  const track = getTrackById(trackId);
  1247→  const zone = zones.find(z => z.id === zoneId);
  1248→  if (!track || !zone) return [];
  1249→
  1250→  const steps: SignalPathStep[] = [];
  1251→  let currentFormat = track.format;
  1252→  let currentSR = track.sampleRate;
  1253→  let currentBD = track.bitDepth;
  1254→  let isDSD = track.format === 'DSF' || track.format === 'DFF';
  1255→
  1256→  // Source step
  1257→  steps.push({
  1258→    label: 'Source',
  1259→    format: currentFormat,
  1260→    sampleRate: currentSR,
  1261→    bitDepth: currentBD,
  1262→    isBitPerfect: true,
  1263→    isDSD,
  1264→    dsdMode: isDSD ? 'native' : undefined,
  1265→    processingDetail: `Local: ${track.filePath}`,
  1266→    latencyMs: 0,
  1267→  });
  1268→
  1269→  if (zone.dspConfig) {
  1270→    // Room Correction (convolution)
  1271→    if (zone.dspConfig.roomCorrection?.enabled) {
  1272→      steps.push({
  1273→        label: 'Room Correction',
  1274→        format: 'PCM',
  1275→        sampleRate: currentSR,
  1276→        bitDepth: currentBD,
  1277→        isBitPerfect: false,
  1278→        filterType: 'linear-phase',
  1279→        processingDetail: `Filter: ${zone.dspConfig.roomCorrection.filterName}`,
  1280→        latencyMs: 5.2,
  1281→      });
  1282→    }
  1283→
  1284→    // Parametric EQ
  1285→    const activeBands = zone.dspConfig.eq?.filter(b => b.enabled) || [];
  1286→    if (activeBands.length > 0) {
  1287→      steps.push({
  1288→        label: 'Parametric EQ',
  1289→        format: 'PCM',
  1290→        sampleRate: currentSR,
  1291→        bitDepth: currentBD,
  1292→        isBitPerfect: false,
  1293→        processingDetail: `${activeBands.length} active band${activeBands.length > 1 ? 's' : ''}`,
  1294→        latencyMs: 0.1,
  1295→      });
  1296→    }
  1297→
  1298→    // Headphone Correction
  1299→    if (zone.dspConfig.headphoneCorrection?.enabled) {
  1300→      steps.push({
  1301→        label: `Headphone: ${zone.dspConfig.headphoneCorrection.profileName}`,
  1302→        format: 'PCM',
  1303→        sampleRate: currentSR,
  1304→        bitDepth: currentBD,
  1305→        isBitPerfect: false,
  1306→        processingDetail: `${zone.dspConfig.headphoneCorrection.manufacturer} ${zone.dspConfig.headphoneCorrection.model}`,
  1307→        latencyMs: 0.3,
  1308→      });
  1309→    }
  1310→
  1311→    // Crossfeed
  1312→    if (zone.dspConfig.crossfeed?.enabled && zone.dspConfig.crossfeed.preset !== 'none') {
  1313→      steps.push({
  1314→        label: `Crossfeed (${zone.dspConfig.crossfeed.preset})`,
  1315→        format: 'PCM',
  1316→        sampleRate: currentSR,
  1317→        bitDepth: currentBD,
  1318→        isBitPerfect: false,
  1319→        processingDetail: `Preset: ${zone.dspConfig.crossfeed.preset}`,
  1320→        latencyMs: 0.1,
  1321→      });
  1322→    }
  1323→
  1324→    // Loudness / ReplayGain
  1325→    if (zone.dspConfig.loudness?.enabled) {
  1326→      steps.push({
  1327→        label: `Loudness (${zone.dspConfig.loudness.method})`,
  1328→        format: 'PCM',
  1329→        sampleRate: currentSR,
  1330→        bitDepth: currentBD,
  1331→        isBitPerfect: false,
  1332→        processingDetail: `Target: ${zone.dspConfig.loudness.targetLUFS} LUFS`,
  1333→        latencyMs: 0,
  1334→      });
  1335→    }
  1336→
  1337→    // Upsampling
  1338→    if (zone.dspConfig.upsampling?.enabled && currentSR <= zone.dspConfig.upsampling.maxSrcRate) {
  1339→      const wasBitPerfect = steps.every(s => s.isBitPerfect);
  1340→      currentSR = zone.dspConfig.upsampling.targetRate / 1000;
  1341→      currentBD = zone.dspConfig.upsampling.targetBitDepth;
  1342→      isDSD = false;
  1343→      currentFormat = 'PCM';
  1344→      steps.push({
  1345→        label: 'Upsampling',
  1346→        format: 'PCM',
  1347→        sampleRate: currentSR,
  1348→        bitDepth: currentBD,
  1349→        isBitPerfect: false,
  1350→        filterType: zone.dspConfig.upsampling.filterType,
  1351→        processingDetail: `${formatSampleRate(zone.dspConfig.upsampling.targetRate / 1000)} / ${currentBD}-bit`,
  1352→        latencyMs: 2.8,
  1353→      });
  1354→    }
  1355→  }
  1356→
  1357→  // Volume Control
  1358→  const isVolBitPerfect = zone.volumeMode !== 'dsp';
  1359→  steps.push({
  1360→    label: 'Volume Control',
  1361→    format: isDSD ? (track.format) : 'PCM',
  1362→    sampleRate: currentSR,
  1363→    bitDepth: currentBD,
  1364→    isBitPerfect: isVolBitPerfect,
  1365→    processingDetail: zone.volumeMode === 'hardware' ? 'Hardware (DAC)' : zone.volumeMode === 'dsp' ? 'DSP attenuator + TPDF dither' : 'Fixed output',
  1366→    ditherType: zone.volumeMode === 'dsp' ? 'tpdf' : 'none',
  1367→    latencyMs: 0,
  1368→  });
  1369→
  1370→  // Format conversion if needed for output
  1371→  const endpoint = zone.endpoints[0];
  1372→  const needsDownsample = currentSR > (endpoint?.maxSampleRate || zone.sampleRate);
  1373→  const needsBDReduce = currentBD > (endpoint?.maxBitDepth || zone.bitDepth);
  1374→  const needsDoP = isDSD && endpoint?.supportsDoP && !endpoint?.supportsDSD;
  1375→
  1376→  if (needsDoP) {
  1377→    steps.push({
  1378→      label: 'DSD → DoP Conversion',
  1379→      format: 'PCM (DoP)',
  1380→      sampleRate: currentSR,
  1381→      bitDepth: currentBD,
  1382→      isBitPerfect: false,
  1383→      isDSD: true,
  1384→      dsdMode: 'DoP',
  1385→      processingDetail: `DSD over PCM for ${endpoint?.name}`,
  1386→      latencyMs: 0.1,
  1387→    });
  1388→    isDSD = false;
  1389→    currentFormat = 'PCM';
  1390→  }
  1391→
  1392→  if (needsDownsample || needsBDReduce) {
  1393→    steps.push({
  1394→      label: 'Format Conversion',
  1395→      format: currentFormat,
  1396→      sampleRate: needsDownsample ? (endpoint?.maxSampleRate || zone.sampleRate) : currentSR,
  1397→      bitDepth: needsBDReduce ? (endpoint?.maxBitDepth || zone.bitDepth) : currentBD,
  1398→      isBitPerfect: false,
  1399→      filterType: 'linear-phase',
  1400→      processingDetail: `Target: ${formatSampleRate(needsDownsample ? (endpoint?.maxSampleRate || zone.sampleRate) : currentSR)}`,
  1401→      latencyMs: 1.5,
  1402→    });
  1403→  }
  1404→
  1405→  // Network transport
  1406→  steps.push({
  1407→    label: 'Network Transport',
  1408→    format: currentFormat,
  1409→    sampleRate: needsDownsample ? (endpoint?.maxSampleRate || zone.sampleRate) : currentSR,
  1410→    bitDepth: needsBDReduce ? (endpoint?.maxBitDepth || zone.bitDepth) : currentBD,
  1411→    isBitPerfect: true,
  1412→    processingDetail: `DSP Audio Protocol · ${(endpoint?.latencyMs || 5).toFixed(1)}ms latency`,
  1413→    latencyMs: endpoint?.latencyMs || 5,
  1414→  });
  1415→
  1416→  // Output endpoint / DAC
  1417→  steps.push({
  1418→    label: `DAC: ${endpoint?.dac || 'Unknown'}`,
  1419→    format: needsDoP ? 'DoP' : currentFormat,
  1420→    sampleRate: needsDownsample ? (endpoint?.maxSampleRate || zone.sampleRate) : currentSR,
  1421→    bitDepth: needsBDReduce ? (endpoint?.maxBitDepth || zone.bitDepth) : currentBD,
  1422→    isBitPerfect: !needsDownsample && !needsBDReduce && isVolBitPerfect,
  1423→    processingDetail: endpoint?.clockSource ? `Clock: ${endpoint.clockSource}` : undefined,
  1424→    latencyMs: 0.01,
  1425→  });
  1426→
  1427→  return steps;
  1428→}

  1430→export function getDetailedSignalPath(trackId: string, zoneId: string): {
  1431→  steps: SignalPathStep[];
  1432→  overallBitPerfect: boolean;
  1433→  sourceFormat: string;
  1434→  outputFormat: string;
  1435→  totalLatencyMs: number;
  1436→  dspChainCount: number;
  1437→  zoneInfo: Zone | undefined;
  1438→  trackInfo: Track | undefined;
  1439→} {
  1440→  const steps = getSignalPath(trackId, zoneId);
  1441→  const track = getTrackById(trackId);
  1442→  const zone = zones.find(z => z.id === zoneId);
  1443→  const dspSteps = steps.filter(s => !s.isBitPerfect);
  1444→  const totalLatency = steps.reduce((sum, s) => sum + (s.latencyMs || 0), 0);
  1445→
  1446→  return {
  1447→    steps,
  1448→    overallBitPerfect: steps.every(s => s.isBitPerfect),
  1449→    sourceFormat: track ? `${track.format} ${formatSampleRate(track.sampleRate)} ${track.bitDepth}-bit` : '—',
  1450→    outputFormat: steps.length > 0 ? `${steps[steps.length - 1].format} ${formatSampleRate(steps[steps.length - 1].sampleRate)} ${steps[steps.length - 1].bitDepth}-bit` : '—',
  1451→    totalLatencyMs: totalLatency,
  1452→    dspChainCount: dspSteps.length,
  1453→    zoneInfo: zone,
  1454→    trackInfo: track,
  1455→  };
  1456→}
