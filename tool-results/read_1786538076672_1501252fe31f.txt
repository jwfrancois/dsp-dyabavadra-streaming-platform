   176→                  {[15, 30, 45, 60, 90].map(min => (
   177→                    <Button
   178→                      key={min}
   179→                      variant={sleepTimerMinutes === min ? 'default' : 'outline'}
   180→                      size="sm"
   181→                      className="h-7 text-xs px-2"
   182→                      onClick={() => { setSleepTimer(min); setShowSleepTimer(false); }}

   204→            <TabsTrigger value="in-progress" className="text-xs gap-1.5">

   207→                <Badge variant="secondary" className="h-4 min-w-[16px] px-1 text-[10px]">{inProgress.length}</Badge>

   209→            </TabsTrigger>
   210→            <TabsTrigger value="discover" className="text-xs gap-1.5">
