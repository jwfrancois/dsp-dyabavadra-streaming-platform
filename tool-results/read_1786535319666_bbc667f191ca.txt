  1063→  1063→export function formatDuration(seconds: number): string {
  1064→  1064→  const h = Math.floor(seconds / 3600);
  1065→  1065→  const m = Math.floor((seconds % 3600) / 60);
  1066→  1066→  const s = seconds % 60;
  1067→  1067→  if (h > 0) {
  1068→  1068→    return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
  1069→  1069→  }
  1070→  1070→  return `${m}:${s.toString().padStart(2, '0')}`;
  1071→  1071→}

  1073→  1073→export function formatFileSize(bytes: number): string {

  1075→  1075→  if (bytes >= 1048576) return `${(bytes / 1048576).toFixed(1)} MB`;
  1076→  1076→  if (bytes >= 1024) return `${(bytes / 1024).toFixed(1)} KB`;
  1077→  1077→  return `${bytes} B`;
  1078→  1078→}

  1121→  1121→export function getSignalPath(trackId: string, zoneId: string): SignalPathStep[] {
  1122→  1122→  const track = getTrackById(trackId);
  1123→  1123→  const zone = zones.find(z => z.id === zoneId);
  1124→  1124→  if (!track || !zone) return [];
  1125→  1125→
  1126→  1126→  const steps: SignalPathStep[] = [];
  1127→  1127→
  1128→  1128→  // Source step
  1129→  1129→  steps.push({
  1130→  1130→    label: 'Source',
  1131→  1131→    format: track.format,
  1132→  1132→    sampleRate: track.sampleRate,
  1133→  1133→    bitDepth: track.bitDepth,
  1134→  1134→    isBitPerfect: true,
  1135→  1135→  });
  1136→  1136→
  1137→  1137→  // DSP steps
  1138→  1138→  if (zone.dspEnabled && zone.dspChain && zone.dspChain.length > 0) {
  1139→  1139→    for (const dsp of zone.dspChain) {
  1140→  1140→      steps.push({
  1141→  1141→        label: dsp,
  1142→  1142→        format: 'PCM',
  1143→  1143→        sampleRate: track.sampleRate,
  1144→  1144→        bitDepth: track.bitDepth,
  1145→  1145→        isBitPerfect: false,
  1146→  1146→      });
  1147→  1147→    }
  1148→  1148→  }
  1149→  1149→
  1150→  1150→  // Volume control
  1151→  1151→  steps.push({
  1152→  1152→    label: 'Volume Control',
  1153→  1153→    format: 'PCM',
  1154→  1154→    sampleRate: track.sampleRate,
  1155→  1155→    bitDepth: track.bitDepth,
  1156→  1156→    isBitPerfect: zone.volume === 100,
  1157→  1157→  });
  1158→  1158→
  1159→  1159→  // Output step
  1160→  1160→  const endpointNeedsDownsample = track.sampleRate > zone.sampleRate;
  1161→  1161→  steps.push({
  1162→  1162→    label: `Output (${zone.endpoints[0]?.name || 'Unknown'})`,
  1163→  1163→    format: zone.outputFormat,
  1164→  1164→    sampleRate: endpointNeedsDownsample ? zone.sampleRate : track.sampleRate,
  1165→  1165→    bitDepth: endpointNeedsDownsample ? zone.bitDepth : track.bitDepth,
  1166→  1166→    isBitPerfect: !endpointNeedsDownsample && zone.volume === 100 && (!zone.dspEnabled || !zone.dspChain?.length),
  1167→  1167→  });
  1168→  1168→
  1169→  1169→  return steps;
  1170→  1170→}
