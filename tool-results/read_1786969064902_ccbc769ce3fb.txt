   165→/** Convert Jellyfin ticks (100ns units) to seconds */
   166→export function ticksToSeconds(ticks: number | undefined): number {
   167→  if (ticks === undefined || ticks === null) return 0;
   168→  return Math.floor(ticks / TICKS_PER_SECOND);
   169→}
   170→
   171→/** Convert seconds to Jellyfin ticks */
   172→export function secondsToTicks(seconds: number): number {
   173→  return Math.floor(seconds * TICKS_PER_SECOND);
   174→}

   176→/** Strip trailing slash from a URL */
   177→function normalizeServerUrl(url: string): string {
   178→  return url.replace(/\/+$/, '');

   181→/** Build a query string from a record, filtering out undefined/null/empty values */
   182→function buildQueryString(params: Record<string, string | number | boolean | undefined | null | string[] | number[]>): string {
   183→  const parts: string[] = [];
   184→  for (const [key, value] of Object.entries(params)) {
   185→    if (value === undefined || value === null || value === '') continue;
   186→    if (Array.isArray(value)) {
   187→      if (value.length > 0) {
   188→        parts.push(`${encodeURIComponent(key)}=${value.map(v => encodeURIComponent(String(v))).join(',')}`);
   189→      }
   190→    } else {
   191→      parts.push(`${encodeURIComponent(key)}=${encodeURIComponent(String(value))}`);
   192→    }
   193→  }
   194→  return parts.length > 0 ? `?${parts.join('&')}` : '';
   195→}

   225→  private saveConfig(): void {
   226→    if (!this.config) return;
   227→    if (this.storageAvailable && typeof window !== 'undefined') {
   228→      try {
   229→        window.localStorage.setItem(STORAGE_KEY, JSON.stringify(this.config));
   230→      } catch {
   231→        // Storage full or unavailable — silently fail
   232→      }
   233→    }

   235→
   236→  private loadConfig(): JellyfinConfig | null {
   237→    if (!this.storageAvailable || typeof window === 'undefined') return null;
   238→    try {
   239→      const raw = window.localStorage.getItem(STORAGE_KEY);
   240→      if (!raw) return null;
   241→      const parsed = JSON.parse(raw) as JellyfinConfig;
   242→      // Basic validation
   243→      if (!parsed.serverUrl || !parsed.accessToken || !parsed.userId) return null;
   244→      return parsed;
   245→    } catch {
   246→      return null;
   247→    }
   248→  }

   259→  /**
   260→   * Make an API request through the Next.js server-side proxy.
   261→   * This avoids browser CORS issues when connecting to self-hosted Jellyfin servers.
   262→   * The proxy at /api/jellyfin/[...path] forwards requests server-to-server.
   263→   */
   264→  private async request<T>(path: string, options: RequestInit = {}, skipAuth = false): Promise<T> {
   265→    // Extract injected metadata from headers (used during connect flow)
   266→    const injectedHeaders = (options.headers as Record<string, string> | undefined);
   267→    const injectedBaseUrl = injectedHeaders?.__baseUrl;
   268→    const injectedAuth = injectedHeaders?.__authorization;
   269→
   270→    // Determine the target Jellyfin server URL
   271→    const serverUrl = skipAuth
   272→      ? normalizeServerUrl(injectedBaseUrl || '')
   273→      : this.getBaseUrl();
   274→
   275→    if (!serverUrl) {
   276→      throw new JellyfinApiError(
   277→        'No Jellyfin server URL configured. Please connect first.',
   278→        0,
   279→        path
   280→      );
   281→    }
   282→
   283→    // Build the proxy URL: /api/jellyfin/Users/AuthenticateByName
   284→    const proxyUrl = `/api/jellyfin${path}`;
   285→
   286→    // Build headers for the proxy — the proxy reads these to forward correctly
   287→    const proxyHeaders: Record<string, string> = {
   288→      'Content-Type': 'application/json',
   289→      'Accept': 'application/json',
   290→      'X-Jellyfin-Server-Url': serverUrl,
   291→    };
   292→
   293→    // Tell the proxy to skip forwarding the auth token for login endpoints
   294→    if (skipAuth) {
   295→      proxyHeaders['X-Jellyfin-Skip-Auth'] = 'true';
   296→    } else if (this.config) {
   297→      // Forward the auth token for authenticated requests
   298→      proxyHeaders['X-Jellyfin-Token'] = this.config.accessToken;
   299→    }
   300→
   301→    // Forward the MediaBrowser authorization header if present (for login)
   302→    if (injectedAuth) {
   303→      proxyHeaders['X-Jellyfin-Authorization'] = injectedAuth;
   304→    }
   305→
   306→    // Remove Content-Type for GET/DELETE if no body
   307→    if (!options.body && (options.method === 'GET' || options.method === 'DELETE')) {
   308→      delete proxyHeaders['Content-Type'];
   309→    }
   310→
   311→    let response: Response;
   312→    try {
   313→      response = await fetch(proxyUrl, {
   314→        method: options.method || 'GET',
   315→        headers: proxyHeaders,
   316→        body: options.body || undefined,
   317→      });
   318→    } catch (err) {
   319→      throw new JellyfinApiError(
   320→        `Network error: ${err instanceof Error ? err.message : 'Unknown error'}`,
   321→        0,
   322→        path,
   323→        err
   324→      );
   325→    }
   326→
   327→    if (!response.ok) {
   328→      let errorBody = '';
   329→      try {
   330→        errorBody = await response.text();
   331→      } catch {
   332→        // ignore
   333→      }
   334→      throw new JellyfinApiError(
   335→        `Jellyfin API error ${response.status}: ${response.statusText} — ${errorBody}`,
   336→        response.status,
   337→        path
   338→      );
   339→    }
   340→
   341→    // Some endpoints return empty responses (e.g., DELETE, favorite toggles)
   342→    const contentType = response.headers.get('content-type');
   343→    if (contentType && contentType.includes('application/json')) {
   344→      return response.json() as Promise<T>;
   345→    }
   346→
   347→    return undefined as unknown as T;
   348→  }

   354→  /**
   355→   * Connect to a Jellyfin server. Authenticates and fetches server info.
   356→   * Persists the configuration to localStorage.
   357→   */
   358→  async connect(serverUrl: string, username: string, password: string): Promise<JellyfinConfig> {
   359→    const baseUrl = normalizeServerUrl(serverUrl);
   360→
   361→    // Step 1: Authenticate (skip auth headers — we don't have a token yet)
   362→    const deviceId = typeof window !== 'undefined'
   363→      ? window.navigator.userAgent.slice(0, 32).replace(/[^a-zA-Z0-9]/g, '')
   364→      : 'ssr';
   365→    const mediaBrowserAuth = `MediaBrowser Client="DSP", Device="Browser", DeviceId="dsp-jellyfin-${deviceId}", Version="1.0.0"`;
   366→
   367→    const authResponse = await this.request<JellyfinAuthResponse>(
   368→      '/Users/AuthenticateByName',
   369→      {
   370→        method: 'POST',
   371→        body: JSON.stringify({ Username: username, Pw: password }),
   372→        headers: {
   373→          '__baseUrl': baseUrl,
   374→          '__authorization': mediaBrowserAuth,
   375→        } as Record<string, string>,
   376→      },
   377→      true // skipAuth
   378→    );
   379→
   380→    const token = authResponse.AccessToken;
   381→    const userId = authResponse.User.Id;
   382→    const serverId = authResponse.ServerId;
   383→
   384→    // Step 2: Fetch server info
   385→    let serverName = 'Jellyfin';
   386→    let serverVersion = 'unknown';
   387→    try {
   388→      // Temporarily set config for the getSystemInfo call
   389→      this.config = {
   390→        serverUrl: baseUrl,
   391→        username,
   392→        accessToken: token,
   393→        userId,
   394→        serverId,
   395→        serverName,
   396→        serverVersion,
   397→        connectedAt: new Date().toISOString(),
   398→        lastHeartbeat: new Date().toISOString(),
   399→        musicLibraryId: '',
   400→        podcastLibraryId: '',
   401→      };
   402→      const sysInfo = await this.getSystemInfo();
   403→      serverName = sysInfo.ServerName;
   404→      serverVersion = sysInfo.Version;
   405→    } catch {
   406→      // If system info fails, continue with defaults
   407→    }
   408→
   409→    // Step 3: Auto-detect music and podcast libraries
   410→    // Jellyfin's /Library/VirtualFolders returns an ARRAY of folder objects.
   411→    // Each has CollectionType ("music", "podcasts", etc.) and ItemId (the view ID for ParentId queries).
   412→    let musicLibraryId = '';
   413→    let podcastLibraryId = '';
   414→    try {
   415→      const vfData = await this.request<Array<{ Name: string; CollectionType?: string; ItemId?: string }> | { Items?: Array<{ Name: string; CollectionType?: string; ItemId?: string }> }>('/Library/VirtualFolders');
   416→      // The API returns either a plain array or { Items: [...] } depending on Jellyfin version
   417→      const folders = Array.isArray(vfData) ? vfData : (vfData.Items ?? []);
   418→      for (const folder of folders) {
   419→        const ct = (folder.CollectionType || '').toLowerCase();
   420→        if (ct === 'music' && !musicLibraryId) {
   421→          musicLibraryId = folder.ItemId || '';
   422→        }
   423→        if ((ct === 'podcasts' || ct === 'podcast') && !podcastLibraryId) {
   424→          podcastLibraryId = folder.ItemId || '';
   425→        }
   426→      }
   427→    } catch {
   428→ // VirtualFolders may fail; fall through to Views-based detection
   429→    }
   430→    // Fallback: use /Users/{userId}/Views to find libraries by name
   431→    if (!musicLibraryId || !podcastLibraryId) {
   432→      try {
   433→        const views = await this.getViews();
   434→        for (const item of views.Items) {
   435→          if (item.Type !== 'CollectionFolder') continue;
   436→          const name = item.Name.toLowerCase();
   437→          if (!musicLibraryId && name.includes('music') && !name.includes('podcast')) {
   438→            musicLibraryId = item.Id;
   439→          }
   440→          if (!podcastLibraryId && (name.includes('podcast'))) {
   441→            podcastLibraryId = item.Id;
   442→          }
   443→        }
   444→      } catch {
   445→        // Continue without auto-detected library
   446→      }
   447→    }
   448→
   449→    console.log(`[Jellyfin] Library detection: musicLibraryId=${musicLibraryId || '(none)'}, podcastLibraryId=${podcastLibraryId || '(none)'}`);
   450→
   451→    // Step 4: Build and save config
   452→    const config: JellyfinConfig = {
   453→      serverUrl: baseUrl,
   454→      username,
   455→      accessToken: token,
   456→      userId,
   457→      serverId,
   458→      serverName,
   459→      serverVersion,
   460→      connectedAt: new Date().toISOString(),
   461→      lastHeartbeat: new Date().toISOString(),
   462→      musicLibraryId,
   463→      podcastLibraryId,
   464→    };
   465→
   466→    this.config = config;
   467→    this.saveConfig();
   468→
   469→    return config;
   470→  }

   472→  /**
   473→   * Disconnect from the Jellyfin server. Revokes the token and clears local state.
   474→   */
   475→  async disconnect(): Promise<void> {
   476→    if (this.config) {
   477→      try {
   478→        await this.request<void>(`/Users/${this.config.userId}/Tokens`, {
   479→          method: 'DELETE',
   480→        });
   481→      } catch {
   482→        // Best-effort — continue clearing local state
   483→      }
   484→    }
   485→
   486→    this.config = null;
   487→    if (this.storageAvailable && typeof window !== 'undefined') {
   488→      try {
   489→        window.localStorage.removeItem(STORAGE_KEY);
   490→      } catch {
   491→        // ignore
   492→      }
   493→    }
   494→  }

   511→  /**

   516→  async refreshLibraryIds(): Promise<void> {
   517→    if (!this.config) throw new JellyfinNotAuthenticatedError();
   518→
   519→    try {
   520→      const vfData = await this.request<Array<{ Name: string; CollectionType?: string; ItemId?: string }> | { Items?: Array<{ Name: string; CollectionType?: string; ItemId?: string }> }>('/Library/VirtualFolders');
   521→      const folders = Array.isArray(vfData) ? vfData : (vfData.Items ?? []);
   522→      for (const folder of folders) {
   523→        const ct = (folder.CollectionType || '').toLowerCase();
   524→        if (ct === 'music' && !this.config.musicLibraryId) {
   525→          this.config.musicLibraryId = folder.ItemId || '';
   526→        }
   527→        if ((ct === 'podcasts' || ct === 'podcast') && !this.config.podcastLibraryId) {
   528→          this.config.podcastLibraryId = folder.ItemId || '';
   529→        }
   530→      }
   531→      // Fallback to Views-based matching
   532→      if (!this.config.musicLibraryId || !this.config.podcastLibraryId) {
   533→        const views = await this.getViews();
   534→        for (const item of views.Items) {
   535→          if (item.Type !== 'CollectionFolder') continue;
   536→          const name = item.Name.toLowerCase();
   537→          if (!this.config.musicLibraryId && name.includes('music') && !name.includes('podcast')) {
   538→            this.config.musicLibraryId = item.Id;
   539→          }
   540→          if (!this.config.podcastLibraryId && name.includes('podcast')) {
   541→            this.config.podcastLibraryId = item.Id;
   542→          }
   543→        }
   544→      }
   545→      this.saveConfig();
   546→    } catch {
   547→      // Non-critical — queries will still work globally
   548→    }
   549→  }

   569→      username: 'API Key User',
   570→      accessToken: apiKey,
   571→      userId: '',
   572→      serverId: '',

   608→    } catch {
   609→      // Continue with defaults
   610→    }
   611→
   612→    // Auto-detect libraries
   613→    let musicLibraryId = '';
   614→    let podcastLibraryId = '';
   615→    try {
   616→      const vfData = await this.request<Array<{ Name: string; CollectionType?: string; ItemId?: string }> | { Items?: Array<{ Name: string; CollectionType?: string; ItemId?: string }> }>('/Library/VirtualFolders');
   617→      const folders = Array.isArray(vfData) ? vfData : (vfData.Items ?? []);
   618→      for (const folder of folders) {
   619→        const ct = (folder.CollectionType || '').toLowerCase();
   620→        if (ct === 'music' && !musicLibraryId) {
   621→          musicLibraryId = folder.ItemId || '';
   622→        }
   623→        if ((ct === 'podcasts' || ct === 'podcast') && !podcastLibraryId) {
   624→          podcastLibraryId = folder.ItemId || '';
   625→        }
   626→      }
   627→      if (!musicLibraryId || !podcastLibraryId) {
   628→        const views = await this.getViews();
   629→        for (const item of views.Items) {
   630→          if (item.Type !== 'CollectionFolder') continue;
   631→          const name = item.Name.toLowerCase();
   632→          if (!musicLibraryId && name.includes('music') && !name.includes('podcast')) {
   633→            musicLibraryId = item.Id;
   634→          }
   635→          if (!podcastLibraryId && name.includes('podcast')) {
   636→            podcastLibraryId = item.Id;
   637→          }
   638→        }
   639→      }
   640→    } catch {
   641→      // Continue without auto-detected library
   642→    }
   643→
   644→    const config: JellyfinConfig = {
   645→      serverUrl: baseUrl,
   646→      username: userInfo.Name || 'API Key User',
   647→      accessToken: token,
   648→      userId,
   649→      serverId,
   650→      serverName,
   651→      serverVersion,
   652→      connectedAt: new Date().toISOString(),
   653→      lastHeartbeat: new Date().toISOString(),
   654→      musicLibraryId,
   655→      podcastLibraryId,
   656→    };
   657→
   658→    this.config = config;
   659→    this.saveConfig();
   660→    return config;
   661→  }

   960→  /**
   961→   * Get podcast shows from the server. Combines multiple discovery strategies
   962→   * and deduplicates by ID to ensure ALL podcasts are found.
   963→   * Supports pagination via startIndex/limit.
   964→   */
   965→  async getPodcasts(params?: { limit?: number; startIndex?: number }): Promise<JellyfinItemsResponse> {
   966→    const limit = params?.limit ?? 200;
   967→    const startIndex = params?.startIndex ?? 0;
   968→    console.log(`[Jellyfin] getPodcasts called. podcastLibraryId=${this.config?.podcastLibraryId || '(none)'}, limit=${limit}, startIndex=${startIndex}`);
   969→
   970→    const seenIds = new Set<string>();
   971→    let allItems: JellyfinItem[] = [];
   972→
   973→    // Strategy 1: Query ALL items in the podcast library (no type filter)
   974→    // This catches Series, BoxSet, Folder, CollectionFolder — anything the server uses
   975→    if (this.config?.podcastLibraryId) {
   976→      try {
   977→        // 1a: Get top-level containers (non-recursive) — these are the show-level items
   978→        const qs1 = buildQueryString({
   979→          UserId: this.config!.userId,
   980→          ParentId: this.config.podcastLibraryId,
   981→          SortBy: 'SortName',
   982→          SortOrder: 'Ascending',
   983→          Recursive: false,
   984→          Fields: [
   985→            'PrimaryImageAspectRatio', 'BasicSyncInfo', 'Genres',
   986→            'Overview', 'Tags', 'DateCreated', 'ChildCount',
   987→          ],
   988→          Limit: 1000, // get all top-level items
   989→        });
   990→        const topLevel = await this.request<JellyfinItemsResponse>(
   991→          `/Users/${this.config!.userId}/Items${qs1}`
   992→        );
   993→        for (const item of topLevel.Items) {
   994→          if (!seenIds.has(item.Id)) {
   995→            seenIds.add(item.Id);
   996→            allItems.push(item);
   997→          }
   998→        }
   999→        console.log(`[Jellyfin] getPodcasts strategy 1a (top-level): ${topLevel.Items.length} items (types: ${[...new Set(topLevel.Items.map(i => i.Type))].join(', ')})`);
  1000→
  1001→        // 1b: Also get Series items recursively (some servers nest podcasts deeper)
  1002→        const qs1b = buildQueryString({
  1003→          UserId: this.config!.userId,
  1004→          ParentId: this.config.podcastLibraryId,
  1005→          IncludeItemTypes: ['Series'],
  1006→          SortBy: 'SortName',
  1007→          SortOrder: 'Ascending',
  1008→          Recursive: true,
  1009→          Fields: [
  1010→            'PrimaryImageAspectRatio', 'BasicSyncInfo', 'Genres',
  1011→            'Overview', 'Tags', 'DateCreated', 'ChildCount',
  1012→          ],
  1013→          Limit: 1000,
  1014→        });
  1015→        const seriesItems = await this.request<JellyfinItemsResponse>(
  1016→          `/Users/${this.config!.userId}/Items${qs1b}`
  1017→        );
  1018→        for (const item of seriesItems.Items) {
  1019→          if (!seenIds.has(item.Id)) {
  1020→            seenIds.add(item.Id);
  1021→            allItems.push(item);
  1022→          }
  1023→        }
  1024→        console.log(`[Jellyfin] getPodcasts strategy 1b (Series recursive): ${seriesItems.Items.length} new items`);
  1025→      } catch {
  1026→        // Continue to next strategies
  1027→      }
  1028→    }
  1029→
  1030→    // Strategy 2: Global search for Series (in case podcasts aren't in the detected library)
  1031→    if (allItems.length === 0) {
  1032→      try {
  1033→        const qs2 = buildQueryString({
  1034→          UserId: this.config!.userId,
  1035→          IncludeItemTypes: ['Series'],
  1036→          SortBy: 'SortName',
  1037→          SortOrder: 'Ascending',
  1038→          Recursive: true,
  1039→          Fields: [
  1040→            'PrimaryImageAspectRatio', 'BasicSyncInfo', 'Genres',
  1041→            'Overview', 'Tags', 'DateCreated', 'ChildCount',
  1042→          ],
  1043→          Limit: 1000,
  1044→        });
  1045→        const globalSeries = await this.request<JellyfinItemsResponse>(
  1046→          `/Users/${this.config!.userId}/Items${qs2}`
  1047→        );
  1048→        for (const item of globalSeries.Items) {
  1049→          if (!seenIds.has(item.Id)) {
  1050→            seenIds.add(item.Id);
  1051→            allItems.push(item);
  1052→          }
  1053→        }
  1054→        console.log(`[Jellyfin] getPodcasts strategy 2 (global Series): ${globalSeries.Items.length} items`);
  1055→      } catch {
  1056→        // Continue
  1057→      }
  1058→    }
  1059→
  1060→    // Strategy 3: If still empty, synthesize shows from episodes grouped by ParentId
  1061→    if (allItems.length === 0 && this.config?.podcastLibraryId) {
  1062→      try {
  1063→        const qs3 = buildQueryString({
  1064→          UserId: this.config!.userId,
  1065→          ParentId: this.config.podcastLibraryId,
  1066→          IncludeItemTypes: ['Audio', 'Episode'],
  1067→          SortBy: 'SortName',
  1068→          SortOrder: 'Ascending',
  1069→          Recursive: true,
  1070→          Fields: [
  1071→            'PrimaryImageAspectRatio', 'BasicSyncInfo', 'Genres',
  1072→            'Overview', 'DateCreated',
  1073→          ],
  1074→          Limit: 1000,
  1075→        });
  1076→        const allEpisodes = await this.request<JellyfinItemsResponse>(
  1077→          `/Users/${this.config!.userId}/Items${qs3}`
  1078→        );
  1079→        if (allEpisodes.TotalRecordCount > 0) {
  1080→          const showMap = new Map<string, { name: string; count: number; genres: string[]; imageUrl: string; dateCreated: string }>();
  1081→          for (const ep of allEpisodes.Items) {
  1082→            const pid = ep.ParentId || ep.AlbumId || '__unknown__';
  1083→            if (!showMap.has(pid)) {
  1084→              showMap.set(pid, {
  1085→                name: ep.Album || ep.AlbumArtists?.[0]?.Name || (pid === '__unknown__' ? 'Unknown Podcast' : 'Unknown Show'),
  1086→                count: 0,
  1087→                genres: ep.Genres ?? [],
  1088→                imageUrl: ep.ImageTags?.Primary ? ep.Id : '',
  1089→                dateCreated: ep.DateCreated || '',
  1090→              });
  1091→            }
  1092→            const entry = showMap.get(pid)!;
  1093→            entry.count++;
  1094→            if (!entry.imageUrl && ep.ImageTags?.Primary) entry.imageUrl = ep.Id;
  1095→            if (pid !== '__unknown__' && entry.name === 'Unknown Show') {
  1096→              entry.name = ep.Artists?.[0] || ep.AlbumArtists?.[0]?.Name || ep.Album || entry.name;
  1097→            }
  1098→          }
  1099→          for (const [id, show] of showMap) {
  1100→            if (id === '__unknown__') continue;
  1101→            allItems.push({
  1102→              Id: id,
  1103→              Name: show.name,
  1104→              Type: 'Series',
  1105→              ChildCount: show.count,
  1106→              Genres: show.genres,
  1107→              ImageTags: show.imageUrl ? { Primary: 'synthetic' } : undefined,
  1108→              DateCreated: show.dateCreated,
  1109→            });
  1110→          }
  1111→          console.log(`[Jellyfin] getPodcasts strategy 3 (synthesize): ${allItems.length} shows from ${allEpisodes.TotalRecordCount} episodes`);
  1112→        }
  1113→      } catch {
  1114→        // Fall through
  1115→      }
  1116→    }
  1117→
  1118→    // Filter out Audio items that may have slipped in (we only want show-level containers)
  1119→    const showItems = allItems.filter(i => i.Type !== 'Audio' && i.Type !== 'Episode');
  1120→
  1121→    // Sort by name
  1122→    showItems.sort((a, b) => (a.SortName || a.Name).localeCompare(b.SortName || b.Name));
  1123→
  1124→    console.log(`[Jellyfin] getPodcasts: total ${showItems.length} unique shows`);
  1125→
  1126→    // Apply pagination
  1127→    const total = showItems.length;
  1128→    const paged = showItems.slice(startIndex, startIndex + limit);
  1129→
  1130→    return { Items: paged, TotalRecordCount: total, StartIndex: startIndex };
  1131→  }

  1207→  /**
  1208→   * Get a playable audio stream URL for an item.
  1209→   * Prefers direct play for best quality; falls back to transcoded stream.
  1210→   */
  1211→  async getPlaybackStreamUrl(
  1212→    itemId: string,
  1213→    maxBitrate: number = DEFAULT_MAX_BITRATE
  1214→  ): Promise<string> {
  1215→    const baseUrl = this.getBaseUrl();
  1216→    const token = this.config!.accessToken;
  1217→    const userId = this.config!.userId;
  1218→
  1219→    try {
  1220→      // Request playback info to determine best playback method
  1221→      const playbackInfo = await this.request<JellyfinPlaybackInfo>(
  1222→        `/Items/${itemId}/PlaybackInfo`,
  1223→        {
  1224→          method: 'POST',
  1225→          body: JSON.stringify({
  1226→            UserId: userId,
  1227→            MaxStreamingBitrate: maxBitrate,
  1228→            AutoOpenLiveStreams: true,
  1229→          }),
  1230→        }
  1231→      );
  1232→
  1233→      if (playbackInfo.MediaSources && playbackInfo.MediaSources.length > 0) {
  1234→        const source = playbackInfo.MediaSources[0];
  1235→
  1236→        // Prefer direct play — best quality, no server transcoding
  1237→        if (source.SupportsDirectPlay) {
  1238→          return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${token}&UserId=${userId}`;
  1239→        }
  1240→
  1241→        // Fallback to direct stream
  1242→        if (source.SupportsDirectStream && source.DirectStreamUrl) {
  1243→          return `${baseUrl}${source.DirectStreamUrl}`;
  1244→        }
  1245→
  1246→        // Fallback to transcoded stream (HLS)
  1247→        if (source.SupportsTranscoding && source.TranscodingUrl) {
  1248→          return `${baseUrl}${source.TranscodingUrl}`;
  1249→        }
  1250→
  1251→        // Last resort: construct a universal audio transcoding URL
  1252→        const container = source.Container || 'mp3';
  1253→        return `${baseUrl}/Audio/${itemId}/universal?api_key=${token}&UserId=${userId}&MaxStreamingBitrate=${maxBitrate}&Container=${container}&TranscodingMaxAudioChannels=2&SegmentContainer=ts&AudioCodec=aac&Protocol=hls`;
  1254→      }
  1255→    } catch {
  1256→      // If playback info fails, fall back to direct stream URL
  1257→    }
  1258→
  1259→    // Fallback: try static stream directly
  1260→    return `${baseUrl}/Audio/${itemId}/stream?static=true&api_key=${token}&UserId=${userId}`;
  1261→  }
