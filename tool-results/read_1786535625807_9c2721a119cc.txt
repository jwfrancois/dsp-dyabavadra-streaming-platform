    21→// ─── Helpers ───
    22→
    23→function formatBytes(bytes: number): string {
    24→  if (bytes >= 1099511627776) return (bytes / 1099511627776).toFixed(1) + ' TB';
    25→  if (bytes >= 1073741824) return (bytes / 1073741824).toFixed(1) + ' GB';
    26→  if (bytes >= 1048576) return (bytes / 1048576).toFixed(1) + ' MB';
    27→  return bytes.toLocaleString() + ' B';
    28→}
    29→
    30→function formatUptime(s: number): string {
    31→  return Math.floor(s / 86400) + 'd ' + Math.floor((s % 86400) / 3600) + 'h ' + Math.floor((s % 3600) / 60) + 'm';
    32→}
    33→
    34→function usageColor(pct: number): string {
    35→  if (pct < 60) return 'bg-signal-green';
    36→  if (pct < 80) return 'bg-signal-amber';
    37→  return 'bg-signal-red';
    38→}
    39→
    40→function UsageBar({ value, label }: { value: number; label?: string }) {
    41→  const pct = Math.min(Math.max(value, 0), 100);
    42→  return (
    43→    <div className="flex items-center gap-3">
    44→      <div className="flex-1 h-1.5 rounded-full bg-white/5 overflow-hidden">
    45→        <div className={`h-full rounded-full transition-all ${usageColor(pct)}`} style={{ width: `${pct}%` }} />
    46→      </div>
    47→      <span className="font-mono text-xs text-muted-foreground w-12 text-right">
    48→        {label ?? `${Math.round(pct)}%`}
    49→      </span>
    50→    </div>
    51→  );
    52→}

    82→
    83→function formatTimeAgo(iso?: string): string {
    84→  if (!iso) return '—';
    85→  const diff = Date.now() - new Date(iso).getTime();
    86→  const mins = Math.floor(diff / 60000);
    87→  if (mins < 1) return 'just now';
    88→  if (mins < 60) return `${mins}m ago`;
    89→  const hrs = Math.floor(mins / 60);
    90→  if (hrs < 24) return `${hrs}h ago`;
    91→  return `${Math.floor(hrs / 24)}d ago`;
    92→}
