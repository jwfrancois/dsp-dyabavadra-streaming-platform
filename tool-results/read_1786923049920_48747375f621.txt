    27→// ── Quality badge helper ──
    28→function QualityBadge({ format, sampleRate, bitDepth }: { format: string; sampleRate: number; bitDepth: number }) {
    29→  const isHiRes = sampleRate > 48000 || bitDepth > 16;
    30→  const isLossless = ['FLAC', 'WAV', 'AIFF', 'ALAC', 'DSF', 'DFF', 'WavPack', 'APE', 'TAK'].includes(format.toUpperCase());
    31→  const isDSD = ['DSF', 'DFF', 'DSD'].includes(format.toUpperCase());
    32→
    33→  if (isDSD) return <Badge className="text-[9px] bg-purple-500/20 text-purple-400 border-purple-500/30 h-4 px-1">DSD</Badge>;
    34→  if (isHiRes && isLossless) return <Badge className="text-[9px] bg-primary/20 text-primary border-primary/30 h-4 px-1">Hi-Res</Badge>;
    35→  if (isLossless) return <Badge className="text-[9px] bg-signal-green/20 text-signal-green border-signal-green/30 h-4 px-1">Lossless</Badge>;
    36→  return null;
    37→}

    39→// ── Time-based greeting ──
    40→function getGreeting(): { text: string; icon: React.ReactNode; emoji: string } {
    41→  const hour = new Date().getHours();
    42→  if (hour < 6) return { text: 'Night Owl Mode', icon: <Moon className="w-5 h-5" />, emoji: '' };
    43→  if (hour < 12) return { text: 'Good Morning', icon: <Sunrise className="w-5 h-5" />, emoji: '' };
    44→  if (hour < 14) return { text: 'Good Afternoon', icon: <Sun className="w-5 h-5" />, emoji: '' };
    45→  if (hour < 18) return { text: 'Good Afternoon', icon: <Sun className="w-5 h-5" />, emoji: '' };
    46→  if (hour < 22) return { text: 'Good Evening', icon: <Moon className="w-5 h-5" />, emoji: '' };
    47→  return { text: 'Night Owl Mode', icon: <Moon className="w-5 h-5" />, emoji: '' };

    63→// ── Time ago helper ──
    64→function timeAgo(isoString: string): string {
    65→  const diffMin = Math.floor((Date.now() - new Date(isoString).getTime()) / 60000);
    66→  if (diffMin < 1) return 'Just now';
    67→  if (diffMin < 60) return `${diffMin}m ago`;
    68→  const diffHr = Math.floor(diffMin / 60);
    69→  if (diffHr < 24) return `${diffHr}h ago`;
    70→  return `${Math.floor(diffHr / 24)}d ago`;
    71→}

    89→function AnalogClock({ className }: { className?: string }) {
    90→  const [now, setNow] = useState(() => new Date());
    91→
    92→  useEffect(() => {
    93→    const timer = setInterval(() => setNow(new Date()), 1000);
    94→    return () => clearInterval(timer);
    95→  }, []);
    96→
    97→  const hours = now.getHours() % 12;
    98→  const minutes = now.getMinutes();
    99→  const seconds = now.getSeconds();
   100→
   101→  const hourAngle = (hours * 30) + (minutes * 0.5);
   102→  const minuteAngle = minutes * 6;
   103→  const secondAngle = seconds * 6;
   104→
   105→  const timeStr = now.toLocaleTimeString('en-US', {
   106→    hour: '2-digit', minute: '2-digit', hour12: true
   107→  });
   108→  const dateStr = now.toLocaleDateString('en-US', {
   109→    weekday: 'short', month: 'short', day: 'numeric'
   110→  });
   111→
   112→  const size = 180;
   113→  const center = size / 2;
   114→  const radius = center - 8;
   115→
   116→  return (
   117→    <div className={cn('flex flex-col items-center gap-2', className)}>
   118→      <svg viewBox={`0 0 ${size} ${size}`} className="w-full max-w-[200px] drop-shadow-2xl">
   119→        <defs>
   120→          <radialGradient id="clockGlow" cx="50%" cy="50%" r="50%">
   121→            <stop offset="80%" stopColor="transparent" />
   122→            <stop offset="100%" stopColor="hsl(var(--primary) / 0.08)" />
   123→          </radialGradient>
   124→          <linearGradient id="clockFace" x1="0%" y1="0%" x2="100%" y2="100%">
   125→            <stop offset="0%" stopColor="hsl(var(--card))" />
   126→            <stop offset="100%" stopColor="hsl(var(--card) / 0.8)" />
   127→          </linearGradient>
   128→          <filter id="handShadow">
   129→            <feDropShadow dx="0" dy="1" stdDeviation="1" floodOpacity="0.3" />
   130→          </filter>
   131→        </defs>
   132→
   133→        <circle cx={center} cy={center} r={radius + 6} fill="url(#clockGlow)" />
   134→        <circle cx={center} cy={center} r={radius} fill="url(#clockFace)" stroke="hsl(var(--border))" strokeWidth="1.5" />
   135→        <circle cx={center} cy={center} r={radius - 4} fill="none" stroke="hsl(var(--border) / 0.3)" strokeWidth="0.5" />
   136→
   137→        {Array.from({ length: 60 }).map((_, i) => {
   138→          const angle = (i * 6) * (Math.PI / 180);
   139→          const isHour = i % 5 === 0;
   140→          const outerR = radius - 6;
   141→          const innerR = isHour ? radius - 18 : radius - 10;
   142→          return (
   143→            <line key={`tick-${i}`}
   144→              x1={center + Math.sin(angle) * innerR}
   145→              y1={center - Math.cos(angle) * innerR}
   146→              x2={center + Math.sin(angle) * outerR}
   147→              y2={center - Math.cos(angle) * outerR}
   148→              stroke={isHour ? 'hsl(var(--foreground) / 0.7)' : 'hsl(var(--muted-foreground) / 0.25)'}
   149→              strokeWidth={isHour ? 2 : 0.8}
   150→              strokeLinecap="round"
   151→            />
   152→          );
   153→        })}
   154→
   155→        {Array.from({ length: 12 }).map((_, i) => {
   156→          const num = i === 0 ? 12 : i;
   157→          const angle = (i * 30) * (Math.PI / 180);
   158→          const numR = radius - 28;
   159→          return (
   160→            <text key={`num-${i}`}
   161→              x={center + Math.sin(angle) * numR}
   162→              y={center - Math.cos(angle) * numR + 4}
   163→              textAnchor="middle"
   164→              fill="hsl(var(--foreground) / 0.5)"
   165→              fontSize="11"
   166→              fontWeight="500"
   167→              fontFamily="var(--font-geist-sans), system-ui, sans-serif"
   168→            >
   169→              {num}
   170→            </text>
   171→          );
   172→        })}
   173→
   174→        <line x1={center} y1={center}
   175→          x2={center + Math.sin((hourAngle) * Math.PI / 180) * (radius * 0.45)}
   176→          y2={center - Math.cos((hourAngle) * Math.PI / 180) * (radius * 0.45)}
   177→          stroke="hsl(var(--foreground))" strokeWidth="3.5" strokeLinecap="round" filter="url(#handShadow)" />
   178→
   179→        <line x1={center} y1={center}
   180→          x2={center + Math.sin((minuteAngle) * Math.PI / 180) * (radius * 0.65)}
   181→          y2={center - Math.cos((minuteAngle) * Math.PI / 180) * (radius * 0.65)}
   182→          stroke="hsl(var(--foreground))" strokeWidth="2.5" strokeLinecap="round" filter="url(#handShadow)" />
   183→
   184→        <g style={{ transition: 'transform 0.2s cubic-bezier(0.4, 2.08, 0.55, 0.44)', transformOrigin: `${center}px ${center}px` }}>
   185→          <line
   186→            x1={center - Math.sin((secondAngle) * Math.PI / 180) * 15}
   187→            y1={center + Math.cos((secondAngle) * Math.PI / 180) * 15}
   188→            x2={center + Math.sin((secondAngle) * Math.PI / 180) * (radius * 0.72)}
   189→            y2={center - Math.cos((secondAngle) * Math.PI / 180) * (radius * 0.72)}
   190→            stroke="hsl(var(--primary))" strokeWidth="1.2" strokeLinecap="round"
   191→          />
   192→        </g>
   193→
   194→        <circle cx={center} cy={center} r="4" fill="hsl(var(--card))" stroke="hsl(var(--primary))" strokeWidth="1.5" />
   195→        <circle cx={center} cy={center} r="1.5" fill="hsl(var(--primary))" />
   196→      </svg>
   197→
   198→      <div className="text-center">
   199→        <p className="text-sm font-medium tabular-nums" style={{ fontVariantNumeric: 'tabular-nums' }}>{timeStr}</p>
   200→        <p className="text-[11px] text-muted-foreground">{dateStr}</p>
   201→      </div>
   202→    </div>
   203→  );
   204→}

   210→function InteractiveCalendar({ className }: { className?: string }) {
   211→  const today = new Date();
   212→  const [viewDate, setViewDate] = useState(() => new Date());
   213→  const [selectedDate, setSelectedDate] = useState<Date | null>(null);
   214→
   215→  const year = viewDate.getFullYear();
   216→  const month = viewDate.getMonth();
   217→
   218→  const monthName = viewDate.toLocaleDateString('en-US', { month: 'long', year: 'numeric' });
   219→
   220→  const firstDayOfMonth = new Date(year, month, 1);
   221→  const lastDayOfMonth = new Date(year, month + 1, 0);
   222→  const daysInMonth = lastDayOfMonth.getDate();
   223→  const startingDay = firstDayOfMonth.getDay();
   224→
   225→  const days = useMemo(() => {
   226→    const result: (number | null)[] = [];
   227→    for (let i = 0; i < startingDay; i++) result.push(null);
   228→    for (let d = 1; d <= daysInMonth; d++) result.push(d);
   229→    return result;
   230→  }, [startingDay, daysInMonth]);
   231→
   232→  const prevMonth = useCallback(() => {
   233→    setViewDate(d => new Date(d.getFullYear(), d.getMonth() - 1, 1));
   234→  }, []);
   235→
   236→  const nextMonth = useCallback(() => {
   237→    setViewDate(d => new Date(d.getFullYear(), d.getMonth() + 1, 1));
   238→  }, []);
   239→
   240→  const isToday = (day: number) =>
   241→    day === today.getDate() && month === today.getMonth() && year === today.getFullYear();
   242→
   243→  const isSelected = (day: number) =>
   244→    selectedDate &&
   245→    day === selectedDate.getDate() &&
   246→    month === selectedDate.getMonth() &&
   247→    year === selectedDate.getFullYear();
   248→
   249→  const weekDays = ['Su', 'Mo', 'Tu', 'We', 'Th', 'Fr', 'Sa'];
   250→
   251→  return (
   252→    <div className={cn('rounded-xl border border-border bg-card overflow-hidden', className)}>
   253→      <div className="flex items-center justify-between px-4 py-3 bg-gradient-to-r from-primary/5 to-accent/5 border-b border-border/50">
   254→        <button onClick={prevMonth} className="p-1 rounded-md hover:bg-accent/50 transition-colors text-muted-foreground hover:text-foreground">
   255→          <ChevronLeft className="w-4 h-4" />
   256→        </button>
   257→        <h3 className="text-sm font-semibold tracking-wide">{monthName}</h3>
   258→        <button onClick={nextMonth} className="p-1 rounded-md hover:bg-accent/50 transition-colors text-muted-foreground hover:text-foreground">
   259→          <ChevronRight className="w-4 h-4" />
   260→        </button>
   261→      </div>
   262→
   263→      <div className="grid grid-cols-7 border-b border-border/30">
   264→        {weekDays.map(d => (
   265→          <div key={d} className="py-2 text-center text-[10px] font-medium text-muted-foreground uppercase tracking-wider">
   266→            {d}
   267→          </div>
   268→        ))}
   269→      </div>
   270→
   271→      <div className="grid grid-cols-7">
   272→        {days.map((day, idx) => {
   273→          if (day === null) return <div key={`empty-${idx}`} className="h-9" />;
   274→          const todayFlag = isToday(day);
   275→          const selectedFlag = isSelected(day);
   276→          return (
   277→            <button key={day}
   278→              onClick={() => setSelectedDate(new Date(year, month, day))}
   279→              className={cn(
   280→                'h-9 flex items-center justify-center text-xs font-medium transition-all duration-200 relative',
   281→                'hover:bg-accent/40 rounded-md m-0.5',
   282→                todayFlag && !selectedFlag && 'text-primary font-bold',
   283→                selectedFlag && 'bg-primary text-primary-foreground rounded-md shadow-sm shadow-primary/20',
   284→                !todayFlag && !selectedFlag && 'text-foreground/70'
   285→              )}
   286→            >
   287→              {day}
   288→              {todayFlag && !selectedFlag && (
   289→                <span className="absolute bottom-0.5 left-1/2 -translate-x-1/2 w-1 h-1 rounded-full bg-primary" />
   290→              )}
   291→            </button>
   292→          );
   293→        })}
   294→      </div>
   295→
   296→      {selectedDate && (
   297→        <div className="px-4 py-2.5 border-t border-border/30 bg-primary/[0.03]">
   298→          <p className="text-xs text-muted-foreground">
   299→            Selected:{' '}
   300→            <span className="text-foreground font-medium">
   301→              {selectedDate.toLocaleDateString('en-US', { weekday: 'long', month: 'long', day: 'numeric', year: 'numeric' })}
   302→            </span>
   303→          </p>
   304→        </div>
   305→      )}
   306→    </div>
   307→  );
   308→}

   356→  // ── Recently Played from history ──

   380→  }, [activeProfile, localTracks]);

   390→      }

   408→      if (!artistAlbums[t.artist]) artistAlbums[t.artist] = new Set();
   409→      artistAlbums[t.artist].add(t.album);
   410→    });
   411→    return Object.entries(artistAlbums)
   412→      .map(([name, albums]) => ({
   413→        name,
   414→        trackCount: localTracks.filter(t => t.artist === name).length,
   415→        albumCount: albums.size,
   416→      }))
   417→      .sort((a, b) => b.trackCount - a.trackCount)
   418→      .slice(0, 6);
   419→  }, [localTracks]);

   421→  const playAlbum = (albumName: string, artistName: string) => {
   422→    const albumTracks = localTracks.filter(t => t.album === albumName && t.artist === artistName).sort((a, b) => a.trackNumber - b.trackNumber);
   423→    if (albumTracks.length > 0) {
   424→      const trackObjs = albumTracks.map(t => ({
   425→        id: t.id, title: t.title, albumId: t.album, albumName: t.album, artistId: t.artist, artistName: t.artist,
   426→        trackNumber: t.trackNumber, discNumber: t.discNumber, duration: t.duration, format: t.format,
   427→        bitDepth: t.bitDepth, sampleRate: t.sampleRate, channels: t.channels, bitrate: t.bitrate,
   428→        filePath: t.filePath, fileSize: t.fileSize, composers: [t.composer], performers: [],
   429→        genre: t.genre, loved: false, playCount: 0, source: 'local' as const, isAvailable: true,
   430→      }));
   431→      setQueue(trackObjs, 0);
   432→    }
   433→  };
   434→
   435→  const playTrack = (trackId: string) => {
   436→    const t = localTracks.find(tr => tr.id === trackId);
   437→    if (t) {
   438→      play({
   439→        id: t.id, title: t.title, albumId: t.album, albumName: t.album, artistId: t.artist, artistName: t.artist,
   440→        trackNumber: t.trackNumber, discNumber: t.discNumber, duration: t.duration, format: t.format,
   441→        bitDepth: t.bitDepth, sampleRate: t.sampleRate, channels: t.channels, bitrate: t.bitrate,
   442→        filePath: t.filePath, fileSize: t.fileSize, composers: [t.composer], performers: [],
   443→        genre: t.genre, loved: false, playCount: 0, source: 'local' as const, isAvailable: true,
   444→      });
   445→    }
   446→  };
   447→
   448→  // Shuffle play all tracks in a genre
   449→  const playGenreShuffled = (genre: string) => {
   450→    const genreTracks = genreMap[genre];
