    39→    }, 0);
    40→  };

    44→      variant={playbackSpeed === 1.0 ? 'secondary' : 'default'}
    45→      size="sm"

    84→export function PodcastsView() {
    85→  const { navigate } = useUIStore();
    86→  const {
    87→    episodeStates, subscribedShowIds, toggleSubscribe,
    88→    markAllPlayed,
    89→    playbackSpeed, cyclePlaybackSpeed, skipSilence, toggleSkipSilence,
    90→    sleepTimerMinutes, setSleepTimer, toggleEpisodeDownload,
    91→    toggleEpisodeFavorite, currentEpisode, isPodcastMode,
    92→  } = usePodcastStore();
    93→  const { isPlaying, playbackMode } = usePlayerStore();
    94→
    95→  const [activeTab, setActiveTab] = useState('subscriptions');
    96→  const [discoverSearch, setDiscoverSearch] = useState('');
    97→  const [discoverGenre, setDiscoverGenre] = useState('all');
    98→  const [allEpisodesFilter, setAllEpisodesFilter] = useState('all');
    99→  const [showSleepTimer, setShowSleepTimer] = useState(false);
   100→  const [unsubscribeTarget, setUnsubscribeTarget] = useState<PodcastShow | null>(null);
   101→  const [expandedShowId, setExpandedShowId] = useState<string | null>(null);
   102→
   103→  // iTunes search state
   104→  const [iTunesResults, setITunesResults] = useState<Array<{
   105→    id: string; itunesId: number; title: string; author: string;
   106→    artworkUrl: string; genre: string; feedUrl: string;
   107→    episodeCount: number; description: string;
   108→  }>>([]);
   109→  const [iTunesLoading, setITunesLoading] = useState(false);
   110→  const [iTunesError, setITunesError] = useState<string | null>(null);
   111→  const [iTunesSearched, setITunesSearched] = useState(false);
   112→  const searchDebounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   113→
   114→  // iTunes search with debounce
   115→  const doITunesSearch = useCallback(async (query: string) => {
   116→    if (!query.trim()) {
   117→      setITunesResults([]);
   118→      setITunesSearched(false);
   119→      setITunesError(null);
   120→      return;
   121→    }
   122→    setITunesLoading(true);
   123→    setITunesError(null);
   124→    setITunesSearched(true);
   125→    try {
   126→      const res = await fetch(`/api/podcasts/search?q=${encodeURIComponent(query.trim())}&limit=25`);
   127→      if (!res.ok) throw new Error(`Search failed (${res.status})`);
   128→      const data = await res.json();
   129→      setITunesResults((data.results || []).map((r: Record<string, unknown>, idx: number) => ({
   130→        id: String(r.id || `result-${idx}`),
   131→        itunesId: Number(r.itunesId || 0),
   132→        title: String(r.title || 'Unknown'),
   133→        author: String(r.author || 'Unknown Author'),
   134→        artworkUrl: String(r.artworkUrl || r.artworkUrlLarge || ''),
   135→        genre: String(r.genre || 'Podcast'),
   136→        feedUrl: String(r.feedUrl || ''),
   137→        episodeCount: Number(r.episodeCount || 0),
   138→        description: String(r.description || ''),
   139→      })));
   140→    } catch (err) {
   141→      setITunesError(err instanceof Error ? err.message : 'Search failed');
   142→      setITunesResults([]);
   143→    } finally {
   144→      setITunesLoading(false);
   145→    }
   146→  }, []);
   147→
   148→  // Debounced search on input change
   149→  useEffect(() => {
   150→    if (searchDebounceRef.current) clearTimeout(searchDebounceRef.current);
   151→    searchDebounceRef.current = setTimeout(() => {
   152→      doITunesSearch(discoverSearch);
   153→    }, 400);
   154→    return () => {
   155→      if (searchDebounceRef.current) clearTimeout(searchDebounceRef.current);
   156→    };
   157→  }, [discoverSearch, doITunesSearch]);
   158→
   159→  const discoveredShows = usePodcastStore(s => s.discoveredShows);
   160→  const discoveredEpisodes = usePodcastStore(s => s.discoveredEpisodes);
   161→
   162→  // Build subscription list from discovered (iTunes) shows only
   163→  const subscribedShows = useMemo(() => {
   164→    return subscribedShowIds
   165→      .map(id => discoveredShows[id])
   166→      .filter(Boolean);
   167→  }, [subscribedShowIds, discoveredShows]);
   168→
   169→  const newEpisodes = usePodcastStore(s => {
   170→    let count = 0;
   171→    const st = s.episodeStates;
   172→    for (const showId of s.subscribedShowIds) {
   173→      const eps = s.discoveredEpisodes[showId] || [];
   174→      for (const ep of eps) {
   175→        const epState = st[ep.id];
   176→        if (epState && !epState.isPlayed && !epState.completed) count++;
   177→      }
   178→    }
   179→    return count;
   180→  });
   181→
   182→  const allSubscribedEpisodes = useMemo(() => {
   183→    // Include discovered episodes for subscribed shows only
   184→    const discovered: PodcastEpisode[] = [];
   185→    for (const showId of subscribedShowIds) {
   186→      const eps = discoveredEpisodes[showId] || [];
   187→      discovered.push(...eps);
   188→    }
   189→    return discovered
   190→      .sort((a, b) => new Date(b.publishDate).getTime() - new Date(a.publishDate).getTime());
   191→  }, [subscribedShowIds, discoveredEpisodes]);
   192→
   193→  const filteredAllEpisodes = useMemo(() => {
   194→    return allSubscribedEpisodes.filter(ep => {
   195→      const st = episodeStates[ep.id];
   196→      if (!st) return allEpisodesFilter === 'all';
   197→      switch (allEpisodesFilter) {
   198→        case 'unplayed': return !st.isPlayed && !st.completed;
   199→        case 'in-progress': return !st.completed && st.resumePosition > 0;
   200→        case 'favorites': return st.favorite;
   201→        case 'downloaded': return st.isDownloaded;
   202→        default: return true;
   203→      }
   204→    });
   205→  }, [allSubscribedEpisodes, episodeStates, allEpisodesFilter]);
   206→
   207→  const downloadedEpisodes = useMemo(() => {
   208→    const allEps: PodcastEpisode[] = [];
   209→    for (const showId of subscribedShowIds) {
   210→      const eps = discoveredEpisodes[showId] || [];
   211→      allEps.push(...eps);
   212→    }
   213→    return allEps.filter(ep => episodeStates[ep.id]?.isDownloaded);
   214→  }, [subscribedShowIds, discoveredEpisodes, episodeStates]);
   215→
   216→  const allGenres = useMemo(() => {
   217→    const genres = new Set<string>();
   218→    for (const show of Object.values(discoveredShows)) {
   219→      if (show.genre) genres.add(show.genre);
   220→    }
   221→    for (const r of iTunesResults) {
   222→      if (r.genre) genres.add(r.genre);
   223→    }
   224→    return ['all', ...Array.from(genres)];
   225→  }, [discoveredShows, iTunesResults]);
   226→
   227→  const discoverShows: PodcastShow[] = []; // Mock shows removed; use iTunes search results instead
   228→
   229→  const newAndNoteworthy: PodcastShow[] = []; // Mock shows removed
   230→
   231→  const trending: PodcastShow[] = []; // Mock shows removed
   232→
   233→  const isCurrentlyPlaying = (episodeId: string) => {
   234→    return isPodcastMode && currentEpisode?.id === episodeId && isPlaying;
   235→  };
   236→
   237→  const isCurrentEpisode = (episodeId: string) => {
   238→    return isPodcastMode && currentEpisode?.id === episodeId;
   239→  };
   240→
   241→  const handleShowContextMenu = (e: React.MouseEvent | React.TouchEvent, show: PodcastShow) => {
   242→    if (subscribedShowIds.includes(show.id)) {
   243→      e.preventDefault();
   244→      e.stopPropagation();

   247→  };
   248→

   252→      setUnsubscribeTarget(null);
   253→    }
   254→  };
   255→
   256→  const renderEpisodeRow = (ep: PodcastEpisode, size: string, showActions: boolean) => {
   257→    const show = discoveredShows[ep.showId] || null;
   258→    const state = episodeStates[ep.id];
   259→    const active = isCurrentEpisode(ep.id);
   260→    const playing = isCurrentlyPlaying(ep.id);
   261→    const unplayed = state && !state.isPlayed && !state.completed;
   262→    const played = state?.completed;
   263→    const progress = state && !state.completed && state.resumePosition > 0
   264→      ? (state.resumePosition / ep.duration) * 100
   265→      : 0;
   266→    const artSize = size === 'sm' ? 'w-10 h-10' : 'w-12 h-12';
   267→
   268→    return (
   269→      <div
   270→        key={ep.id}
   271→        className={[
   272→          'flex items-center gap-3 p-3 rounded-lg transition-colors group cursor-pointer',
   273→          active ? 'bg-primary/10 border border-primary/30' : 'hover:bg-accent/30',
   274→          played ? 'opacity-60' : '',
   275→        ].join(' ')}
   276→        onClick={() => handlePlayEpisode(ep)}
   277→      >
   278→        <div className={`relative flex-shrink-0 ${artSize} rounded bg-gradient-to-br ${getCoverGradient(ep.showId)}`}>
   279→          {unplayed && (
   280→            <span className="absolute -top-1 -left-1 w-2.5 h-2.5 rounded-full bg-blue-500 border-2 border-background" />
   281→          )}
   282→          {(playing || (active && !isPlaying)) && (
   283→            <div className="absolute inset-0 flex items-center justify-center bg-black/40 rounded">
   284→              {playing ? (
   285→                <Pause className="w-4 h-4 text-white" />
   286→              ) : (
   287→                <Play className="w-4 h-4 text-white ml-0.5" />
   288→              )}
   289→            </div>
   290→          )}
   291→          {progress > 0 && (
   292→            <div className="absolute bottom-0 left-0 right-0 h-0.5 bg-white/20 rounded-b">
   293→              <div className="h-full bg-primary rounded-b" style={{ width: `${progress}%` }} />
   294→            </div>
   295→          )}
   296→        </div>
   297→
   298→        <div className="flex-1 min-w-0">
   299→          <p className={`text-sm font-medium truncate ${active ? 'text-primary' : ''}`}>{ep.title}</p>
   300→          <p className="text-xs text-muted-foreground truncate">
   301→            {show?.title}{' \u00b7 '}{formatDate(ep.publishDate)}
   302→          </p>
   303→        </div>
   304→
   305→        <span className="text-xs text-muted-foreground hidden sm:inline flex-shrink-0">
   306→          {formatEpisodeDuration(ep.duration)}
   307→        </span>
   308→
   309→        {progress > 0 && (
   310→          <span className="text-[10px] text-muted-foreground flex-shrink-0">
   311→            {Math.round(progress)}%
   312→          </span>
   313→        )}
   314→
   315→        {active && playbackMode === 'podcast' && <SpeedBadge />}
   316→
   317→        {showActions && (
   318→          <div className="flex items-center gap-1 flex-shrink-0 opacity-0 group-hover:opacity-100 transition-opacity">
   319→            <Button
   320→              variant="ghost"
   321→              size="icon"
   322→              className={`h-7 w-7 ${state?.favorite ? 'text-red-400' : ''}`}
   323→              onClick={(e) => { e.stopPropagation(); toggleEpisodeFavorite(ep.id); }}
   324→            >
   325→              <Heart className={`w-3.5 h-3.5 ${state?.favorite ? 'fill-current' : ''}`} />
   326→            </Button>
   327→            <Button
   328→              variant="ghost"
   329→              size="icon"
   330→              className={`h-7 w-7 ${state?.isDownloaded ? 'text-primary' : ''}`}
   331→              onClick={(e) => { e.stopPropagation(); toggleEpisodeDownload(ep.id); }}
   332→            >
   333→              <Download className="w-3.5 h-3.5" />
   334→            </Button>
   335→          </div>
   336→        )}
   337→      </div>
   338→    );
   339→  };
   340→
   341→  const renderShowCard = (show: PodcastShow) => {
   342→    const isSub = subscribedShowIds.includes(show.id);
   343→    const isDiscovered = !!discoveredShows[show.id];
   344→    // Get episodes from discovered RSS feeds
   345→    const episodes = isDiscovered
   346→      ? (discoveredEpisodes[show.id] || []).sort((a, b) => new Date(b.publishDate).getTime() - new Date(a.publishDate).getTime())
   347→      : [];
   348→    const unplayed = episodes.filter(ep => {
   349→      const st = episodeStates[ep.id];
   350→      return st && !st.isPlayed && !st.completed;
   351→    });
   352→    const latestEp = episodes[0];
   353→    const latestState = latestEp ? episodeStates[latestEp.id] : null;
   354→    const hasArtworkUrl = show.artworkUrl && show.artworkUrl.startsWith('http');
   355→
   356→    return (
   357→      <Card
   358→        key={show.id}
   359→        className="bg-card border-border hover:border-muted-foreground/20 cursor-pointer transition-all group"
   360→        onClick={() => {
   361→          // Ensure discovered show is set before navigating
   362→          if (isDiscovered) {
   363→            usePodcastStore.getState().setDiscoveredShow(show);
   364→          }
   365→          navigate('podcast-detail', { showId: show.id });
   366→        }}
   367→        onContextMenu={(e) => handleShowContextMenu(e, show)}
   368→      >
   369→        <CardContent className="p-4">
   370→          <div className="flex gap-4">
   371→            <div className={`w-20 h-20 rounded-lg ${hasArtworkUrl ? '' : `bg-gradient-to-br ${getCoverGradient(show.id)}`} flex-shrink-0 relative overflow-hidden`}>
   372→              {hasArtworkUrl ? (
   373→                <img src={show.artworkUrl} alt={show.title} className="w-full h-full object-cover" />
   374→              ) : (
   375→                <div className="absolute inset-0 bg-[radial-gradient(circle_at_30%_30%,rgba(255,255,255,0.1),transparent_70%)]" />
   376→              )}
   377→              {unplayed.length > 0 && (
   378→                <span className="absolute -top-1.5 -right-1.5 w-5 h-5 rounded-full bg-signal-red text-[10px] font-bold text-white flex items-center justify-center z-10">
   379→                  {unplayed.length > 9 ? '9+' : unplayed.length}
   380→                </span>
   381→              )}
   382→            </div>
   383→
   384→            <div className="flex-1 min-w-0">
   385→              <h3 className="text-sm font-semibold truncate">{show.title}</h3>
   386→              <p className="text-xs text-muted-foreground">{show.author}</p>
   387→              <p className="text-xs text-muted-foreground mt-1 line-clamp-2">{show.description}</p>
   388→              <div className="flex items-center gap-3 mt-2">
   389→                <Badge variant="outline" className="text-[10px]">{show.genre}</Badge>
   390→                <span className="text-[11px] text-muted-foreground">{isDiscovered ? episodes.length : show.episodeCount} episodes</span>
   391→                {latestEp && (
   392→                  <span className="text-[11px] text-muted-foreground flex items-center gap-1">
   393→                    <Rss className="w-3 h-3" />
   394→                    {formatDate(latestEp.publishDate)}
   395→                  </span>
   396→                )}
   397→              </div>
   398→            </div>
   399→
   400→            <div className="flex flex-col items-end gap-2 flex-shrink-0">
   401→              <Button
   402→                variant={isSub ? 'secondary' : 'default'}
   403→                size="sm"
   404→                className="h-8 gap-1.5"

   408→                }}

   410→                {isSub ? <Check className="w-3.5 h-3.5" /> : <Plus className="w-3.5 h-3.5" />}
   411→                {isSub ? 'Subscribed' : 'Subscribe'}
   412→              </Button>
   413→
   414→              {latestEp && latestState && !latestState.completed && (
   415→                <Button
   416→                  variant="ghost"
   417→                  size="sm"
   418→                  className="h-8 gap-1.5 text-xs"
   419→                  onClick={(e) => { e.stopPropagation(); handlePlayEpisode(latestEp); }}
   420→                >
   421→                  <Play className="w-3.5 h-3.5" />
   422→                  {latestState.resumePosition > 0 ? 'Resume' : 'Play Latest'}
   423→                </Button>
   424→              )}
   425→
   426→              {latestEp && latestState && latestState.resumePosition > 0 && !latestState.completed && (
   427→                <div className="text-[10px] text-muted-foreground">
   428→                  {Math.round((latestState.resumePosition / latestEp.duration) * 100)}% listened
   429→                </div>
   430→              )}
   431→            </div>
   432→          </div>
   433→        </CardContent>
   434→      </Card>
   435→    );
   436→  };
   437→
   438→  const renderDiscoverShowCard = (show: PodcastShow) => {
   439→    const isSub = subscribedShowIds.includes(show.id);
   440→    const isExpanded = expandedShowId === show.id;
   441→    const episodes = discoveredEpisodes[show.id] || [];
   442→    return (
   443→      <Card
   444→        key={show.id}
   445→        className="bg-card border-border hover:border-muted-foreground/20 transition-all"
   446→      >
   447→        <CardContent className="p-3">

   449→            <div className={`w-16 h-16 rounded-lg bg-gradient-to-br ${getCoverGradient(show.id)} flex-shrink-0`} />
   450→            <div className="flex-1 min-w-0">
   451→              <p className="text-sm font-medium truncate">{show.title}</p>
   452→              <p className="text-xs text-muted-foreground truncate">{show.author}</p>
   453→              <div className="flex items-center gap-2 mt-1">
   454→                <Badge variant="outline" className="text-[10px]">{show.genre}</Badge>
   455→                <span className="text-[10px] text-muted-foreground">{show.episodeCount} eps</span>
   456→                <span className="text-[10px] text-muted-foreground">{episodes.length} available</span>
   457→              </div>
   458→              <p className="text-[11px] text-muted-foreground mt-1 line-clamp-2">{show.description}</p>
   459→            </div>
   460→            <div className="flex flex-col items-end gap-2 flex-shrink-0">
   461→              <Button
   462→                variant={isSub ? 'secondary' : 'default'}
   463→                size="sm"
   464→                className="h-8 gap-1.5"
   465→                onClick={(e) => { e.stopPropagation(); toggleSubscribe(show.id); }}
   466→              >
   467→                {isSub ? <Check className="w-3.5 h-3.5" /> : <Plus className="w-3.5 h-3.5" />}
   468→                {isSub ? 'Subscribed' : 'Subscribe'}
   469→              </Button>
   470→              <Button
   471→                variant="ghost"
   472→                size="sm"
   473→                className="h-7 gap-1 text-xs"
   474→                onClick={(e) => { e.stopPropagation(); if (episodes.length > 0) handlePlayEpisode(episodes[0]); }}
   475→              >
   476→                <Play className="w-3 h-3" />
   477→                Play Latest
   478→              </Button>
   479→              <Button
   480→                variant="ghost"
   481→                size="sm"
   482→                className="h-7 text-xs text-muted-foreground"
   483→                onClick={(e) => { e.stopPropagation(); setExpandedShowId(isExpanded ? null : show.id); }}

   487→            </div>
   488→          </div>
   489→
   490→          {isExpanded && episodes.length > 0 && (
   491→            <div className="mt-3 ml-[76px] space-y-1 border-l-2 border-primary/20 pl-3">

   495→          )}
   496→        </CardContent>
   497→      </Card>
   498→    );
   499→  };
   500→
   501→  return (
   502→    <>
