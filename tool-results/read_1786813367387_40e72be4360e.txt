  1508→interface NetworkShareInfo {

  1511→  server: string;
  1512→  shareName: string;
  1513→  subPath: string;
  1514→  mountPoint: string;

  1516→  mounted: boolean;

  1521→function NetworkSharesPanel() {
  1522→  const localStore = useLocalLibraryStore();
  1523→  const [shares, setShares] = React.useState<NetworkShareInfo[]>([]);
  1524→  const [uncInput, setUncInput] = React.useState('');
  1525→  const [usernameInput, setUsernameInput] = React.useState('');
  1526→  const [passwordInput, setPasswordInput] = React.useState('');
  1527→  const [mounting, setMounting] = React.useState<string | null>(null);
  1528→  const [loading, setLoading] = React.useState(true);
  1529→  const [error, setError] = React.useState<string | null>(null);
  1530→
  1531→  React.useEffect(() => { fetchShares(); }, []);
  1532→
  1533→  const fetchShares = async () => {
  1534→    setLoading(true);
  1535→    try {
  1536→      const res = await fetch('/api/library/mount');
  1537→      const data = await res.json();
  1538→      if (data.success) { setShares(data.shares || []); setError(null); }
  1539→      else { setError(data.error || 'Failed to load network shares'); }
  1540→    } catch { setError('Failed to connect to server'); }
  1541→    setLoading(false);
  1542→  };
  1543→
  1544→  const handleMount = async () => {
  1545→    const unc = uncInput.trim();
  1546→    if (!unc) return;
  1547→    setMounting(unc); setError(null);
  1548→    try {
  1549→      const res = await fetch('/api/library/mount', {
  1550→        method: 'POST', headers: { 'Content-Type': 'application/json' },
  1551→        body: JSON.stringify({ uncPath: unc, username: usernameInput.trim() || undefined, password: passwordInput || undefined }),
  1552→      });
  1553→      const data = await res.json();
  1554→      if (data.success && data.share) {
  1555→        if (data.scanPath) { localStore.addDirectory(data.scanPath); localStore.startScan(data.scanPath); }
  1556→        setUncInput(''); setUsernameInput(''); setPasswordInput('');
  1557→        await fetchShares();
  1558→      } else { setError(data.error || 'Mount failed'); }
  1559→    } catch { setError('Failed to mount share'); }
  1560→    setMounting(null);
  1561→  };
  1562→
  1563→  const handleUnmount = async (shareId: string) => {
  1564→    try { await fetch(`/api/library/mount?id=${encodeURIComponent(shareId)}`, { method: 'DELETE' }); await fetchShares(); }
  1565→    catch { setError('Failed to unmount'); }
  1566→  };
  1567→
  1568→  const handleRemount = async (share: NetworkShareInfo) => {
  1569→    setMounting(share.id); setError(null);
  1570→    try {
  1571→      const res = await fetch('/api/library/mount', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ uncPath: share.uncPath }) });
  1572→      const data = await res.json();
  1573→      if (data.success) { if (data.scanPath) localStore.addDirectory(data.scanPath); await fetchShares(); }
  1574→      else { setError(data.error || 'Remount failed'); }
  1575→    } catch { setError('Failed to remount'); }
  1576→    setMounting(null);
  1577→  };
  1578→
  1579→  const handleScanMounted = (share: NetworkShareInfo) => {
  1580→    const scanPath = share.mountPoint + (share.subPath ? `/${share.subPath}` : '');
  1581→    localStore.addDirectory(scanPath); localStore.startScan(scanPath);
  1582→  };

  1584→  return (

  1586→      <Card className="bg-card border-border">

  1592→              <p>Mount SMB/CIFS network shares (TrueNAS, Synology, Windows shares, etc.) to scan and play music from network-attached storage.</p>
  1593→            </div>
  1594→          </div>
  1595→        </CardContent>
  1596→      </Card>
  1597→
  1598→      <Card className="bg-card border-border">
  1599→        <CardHeader className="pb-3"><CardTitle className="text-sm flex items-center gap-2"><Globe className="w-4 h-4 text-primary" /> Add Network Share</CardTitle></CardHeader>
  1600→        <CardContent className="space-y-4">
  1601→          <div className="space-y-2">
  1602→            <label className="text-xs font-medium text-muted-foreground">Network Path (UNC or SMB URL)</label>
  1603→            <Input placeholder="\\10.0.0.80\iguey\Media\Music" value={uncInput} onChange={e => setUncInput(e.target.value)} className="h-9 text-sm font-mono" disabled={!!mounting} />
  1604→          </div>
  1605→          <div className="grid grid-cols-2 gap-3">
  1606→            <div className="space-y-2">
  1607→              <label className="text-xs font-medium text-muted-foreground">Username (optional)</label>
