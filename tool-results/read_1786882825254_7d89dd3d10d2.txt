     1→'use client';
     2→
     3→import React, { useMemo } from 'react';
     4→import { useUIStore } from '@/store/ui';
     5→import { usePlayerStore } from '@/store/player';
     6→import { usePodcastStore } from '@/store/podcast';
     7→import { useLocalLibraryStore } from '@/store/local-library';
     8→import { formatEpisodeDuration } from '@/lib/podcast-data';
     9→import { formatDuration, formatSampleRate, formatFileSize, getCoverGradient } from '@/lib/data';
    10→import { Card, CardContent } from '@/components/ui/card';
    11→import { Button } from '@/components/ui/button';
    12→import { Badge } from '@/components/ui/badge';
    13→import { ScrollArea } from '@/components/ui/scroll-area';
    14→import { Separator } from '@/components/ui/separator';
    15→import { Slider } from '@/components/ui/slider';
    16→import { useLyrics, useArtistBio, useArtistImage, useAlbumInfo } from '@/lib/use-music-metadata';
    17→import {
    18→  Play, Pause, SkipForward, SkipBack, Shuffle, Repeat, Repeat1,
    19→  Volume2, VolumeX, Volume1, Heart, ListMusic, Share2,
    20→  ArrowLeft, Gauge, AlertCircle, Music,
    21→  Radio, Disc3, ExternalLink, Image as ImageIcon,
    22→  User, Mic2, Clock, Waves,
    23→} from 'lucide-react';
    24→import { useProfilesStore } from '@/store/profiles';
    25→
    26→export function NowPlayingView() {
    27→  const {
    28→    isPlaying, currentTrack, queue, queueIndex, activeZoneId,
    29→    progress, currentTime, duration, volume, isShuffle, repeatMode, isMuted,
    30→    togglePlay, next, previous, seek, setVolume, toggleMute,
    31→    toggleShuffle, toggleRepeat, setActiveZone, playbackMode,
    32→  } = usePlayerStore();
    33→  const { navigate } = useUIStore();
    34→  const { isPodcastMode, currentEpisode, playbackSpeed, cyclePlaybackSpeed } = usePodcastStore();
    35→  const toggleLoveTrack = useProfilesStore(s => s.toggleLoveTrack);
    36→  const isTrackLoved = useProfilesStore(s => s.isTrackLoved);
    37→  const localTracks = useLocalLibraryStore(s => s.tracks);
    38→
    39→  // Fetch rich metadata for the current track
    40→  const { data: lyricsData, loading: lyricsLoading } = useLyrics(currentTrack?.artistName || '', currentTrack?.title || '');
    41→  const { data: artistBio, loading: bioLoading } = useArtistBio(currentTrack?.artistName || '');
    42→  const { data: artistImage, loading: artistImageLoading } = useArtistImage(currentTrack?.artistName || '');
    43→  const { data: albumInfoData, loading: albumInfoLoading } = useAlbumInfo(currentTrack?.artistName || '', currentTrack?.albumName || '');
    44→
    45→  // Look up cover art from local library for the current track
    46→  const trackCoverArt = useMemo(() => {
    47→    if (!currentTrack) return null;
    48→    // Check if currentTrack has blobUrl or if we can find cover from local library
    49→    const localTrack = localTracks.find(t => t.id === currentTrack.id);
    50→    if (localTrack?.coverArt) return localTrack.coverArt;
    51→    // Check other tracks in the same album for cover art
    52→    const albumTracks = localTracks.filter(t => t.album === currentTrack.albumName && (t.artist === currentTrack.artistName || t.albumArtist === currentTrack.artistName));
    53→    for (const at of albumTracks) {
    54→      if (at.coverArt) return at.coverArt;
    55→    }
    56→    return null;
    57→  }, [currentTrack, localTracks]);
    58→
    59→  const activeZone = activeZoneId ? {
    60→    id: activeZoneId,
    61→    name: activeZoneId === 'zone-1' ? 'Main Listening Room' : activeZoneId === 'zone-2' ? 'Study' : `Zone ${activeZoneId}`,
    62→    endpoints: [{ dac: 'ESS Sabre ES9038Q2M' }],
    63→    dspEnabled: false,
    64→    dspChain: [],
    65→  } : null;
    66→
    67→  // ── Dynamic background gradient derived from the cover art ──
    68→  const bgGradient = useMemo(() => {
    69→    if (!currentTrack) return 'from-zinc-950 via-zinc-900 to-zinc-950';
    70→    return `from-zinc-950 via-zinc-900 to-zinc-950`;
    71→  }, [currentTrack]);
    72→
    73→  // Podcast mode rendering
    74→  if (isPodcastMode && currentEpisode) {
    75→    const show = currentEpisode ? { title: currentEpisode.showId.replace(/-/g, ' ').replace(/\b\w/g, (l: string) => l.toUpperCase()) } : null;
    76→    const dur = duration > 0 ? duration : currentEpisode.duration;
    77→    const prog = dur > 0 ? (currentTime / dur) * 100 : 0;
    78→
    79→    return (
    80→      <ScrollArea className="h-full">
    81→        <div className="max-w-5xl mx-auto p-6">
    82→          <Button variant="ghost" size="sm" className="mb-4 text-muted-foreground" onClick={() => navigate('home')}>
    83→            <ArrowLeft className="w-4 h-4 mr-1" /> Back
    84→          </Button>
    85→
    86→          <div className="flex flex-col md:flex-row gap-8">
    87→            {/* Cover Art */}
    88→            <div className="flex-shrink-0">
    89→              <div className={`w-56 h-56 rounded-xl bg-gradient-to-br ${getCoverGradient(currentEpisode.showId)} shadow-lg`} />
    90→            </div>
    91→
    92→            {/* Track Info */}
    93→            <div className="flex-1 min-w-0">
    94→              <p className="text-xs text-muted-foreground uppercase tracking-wider mb-1">Podcast Episode</p>
    95→              <h1 className="text-2xl font-bold mb-1">{currentEpisode.title}</h1>
    96→              <p className="text-sm text-muted-foreground mb-4">{show?.title}</p>
    97→
    98→              <div className="flex items-center gap-3 mb-4">
    99→                <Badge variant="outline" className="text-xs">{currentEpisode.format}</Badge>
   100→                <Badge variant="outline" className="text-xs">{formatEpisodeDuration(currentEpisode.duration)}</Badge>
   101→                <Button
   102→                  variant="secondary" size="sm" className="h-7 text-xs font-mono"
   103→                  onClick={cyclePlaybackSpeed}
   104→                >
   105→                  {playbackSpeed}x
   106→                </Button>
   107→              </div>
   108→
   109→              <div className="flex items-center gap-2 w-full max-w-lg">
   110→                <span className="text-xs text-muted-foreground w-12 text-right tabular-nums">{formatDuration(currentTime)}</span>
   111→                <Slider value={[prog]} min={0} max={100} step={0.1} onValueChange={(v) => seek(v[0])} className="flex-1" />
   112→                <span className="text-xs text-muted-foreground w-16 tabular-nums">{formatEpisodeDuration(currentEpisode.duration)}</span>
   113→              </div>
   114→            </div>
   115→          </div>
   116→
   117→          <Separator className="my-8" />
   118→
   119→          {/* Controls */}
   120→          <div className="flex items-center justify-center gap-4">
   121→            <Button variant="ghost" size="icon" className="h-10 w-10 text-muted-foreground" onClick={previous}>
   122→              <SkipBack className="w-5 h-5" />
   123→            </Button>
   124→            <Button variant="default" size="icon" className="h-12 w-12 rounded-full" onClick={togglePlay}>
   125→              {isPlaying ? <Pause className="w-6 h-6" /> : <Play className="w-6 h-6 ml-0.5" />}
   126→            </Button>
   127→            <Button variant="ghost" size="icon" className="h-10 w-10 text-muted-foreground" onClick={next}>
   128→              <SkipForward className="w-5 h-5" />
   129→            </Button>
   130→          </div>
   131→
   132→          <Separator className="my-8" />
   133→
   134→          {/* Volume & Zone */}
   135→          <div className="flex items-center justify-center gap-4">
   136→            <Button variant="ghost" size="icon" className="h-8 w-8" onClick={toggleMute}>
   137→              {isMuted || volume === 0 ? <VolumeX className="w-4 h-4 text-muted-foreground" /> : volume < 50 ? <Volume1 className="w-4 h-4 text-muted-foreground" /> : <Volume2 className="w-4 h-4 text-muted-foreground" />}
   138→            </Button>
   139→            <Slider value={[isMuted ? 0 : volume]} min={0} max={100} step={1} onValueChange={(v) => setVolume(v[0])} className="w-40" />
   140→            {activeZone && (
   141→              <Button variant="ghost" size="sm" className="h-7 gap-1 text-xs text-muted-foreground" onClick={() => navigate('zones')}>
   142→                <Gauge className="w-3.5 h-3.5" /> {activeZone.name}
   143→              </Button>
   144→            )}
   145→          </div>
   146→        </div>
   147→      </ScrollArea>
   148→    );
   149→  }
   150→
   151→  if (!currentTrack) {
   152→    return (
   153→      <div className="flex items-center justify-center h-full text-muted-foreground">
   154→        <div className="text-center">
   155→          <ListMusic className="w-16 h-16 mx-auto mb-4 opacity-30" />
   156→          <p className="text-lg">Nothing playing</p>
   157→          <p className="text-sm">Select a track to start listening</p>
   158→        </div>
   159→      </div>
   160→    );
   161→  }
   162→
   163→  return (
   164→    <div className={`h-full bg-gradient-to-b ${bgGradient}`}>
   165→      <ScrollArea className="h-full">
   166→        <div className="max-w-6xl mx-auto p-6 pb-32">
   167→          {/* Back button */}
   168→          <Button variant="ghost" size="sm" className="mb-6 text-muted-foreground hover:text-foreground" onClick={() => navigate('home')}>
   169→            <ArrowLeft className="w-4 h-4 mr-2" /> Back
   170→          </Button>
   171→
   172→          {/* ── HERO: Large Album Cover + Track Info ── */}
   173→          <div className="flex flex-col md:flex-row gap-8 mb-10">
   174→            {/* Album Cover — Large */}
   175→            <div className="flex-shrink-0 self-center">
   176→              <div className={`w-72 h-72 lg:w-80 lg:h-80 rounded-2xl bg-gradient-to-br ${getCoverGradient(currentTrack.id)} shadow-2xl relative overflow-hidden group`}>
   177→                {/* Show real cover art if available */}
   178→                {trackCoverArt ? (
   179→                  <img
   180→                    src={trackCoverArt}
   181→                    alt={currentTrack.albumName}
   182→                    className="w-full h-full object-cover"
   183→                  />
   184→                ) : null}
   185→                {/* Shine overlay */}
   186→                <div className="absolute inset-0 bg-[radial-gradient(circle_at_30%_30%,rgba(255,255,255,0.12),transparent_70%)] pointer-events-none" />
   187→                {/* Vinyl peek effect */}
   188→                <div className="absolute -bottom-2 -right-2 w-48 h-48 rounded-full bg-zinc-900 border border-zinc-800 shadow-xl transition-transform duration-700 group-hover:translate-x-4 opacity-0 group-hover:opacity-80">
   189→                  <div className="absolute inset-[20%] rounded-full border border-zinc-700/40" />
   190→                  <div className="absolute inset-[35%] rounded-full border border-zinc-700/30" />
   191→                  <div className="absolute inset-[48%] rounded-full bg-zinc-800 flex items-center justify-center">
   192→                    <div className="w-2 h-2 rounded-full bg-zinc-600" />
   193→                  </div>
   194→                </div>
   195→              </div>
   196→            </div>
   197→
   198→            {/* Track Info + Controls — stacked vertically */}
   199→            <div className="flex-1 min-w-0 flex flex-col justify-center">
   200→              <div className="text-center md:text-left w-full max-w-md md:max-w-none">
   201→                {/* Format badges */}
   202→                <div className="flex items-center gap-2 mb-3 justify-center md:justify-start">
   203→                  <Badge variant="outline" className="text-[10px] font-mono bg-primary/5 border-primary/20">
   204→                    <Waves className="w-3 h-3 mr-1" /> {currentTrack.format}
   205→                  </Badge>
   206→                  <Badge variant="outline" className="text-[10px] font-mono">
   207→                    {formatSampleRate(currentTrack.sampleRate)}
   208→                  </Badge>
   209→                  <Badge variant="outline" className="text-[10px] font-mono">
   210→                    {currentTrack.bitDepth}-bit
   211→                  </Badge>
   212→                  {currentTrack.channels === 1 && <Badge variant="outline" className="text-[10px]">Mono</Badge>}
   213→                </div>
   214→
   215→                {/* Title */}
   216→                <h1 className="text-3xl lg:text-4xl font-bold mb-2 leading-tight">{currentTrack.title}</h1>
   217→
   218→                {/* Artist — clickable */}
   219→                <p
   220→                  className="text-lg text-muted-foreground hover:text-primary transition-colors cursor-pointer inline-flex items-center gap-1.5"
   221→                  onClick={() => navigate('artist-detail', { artistId: currentTrack.artistId })}
   222→                >
   223→                  {currentTrack.artistName}
   224→                </p>
   225→
   226→                {/* Album — clickable */}
   227→                <p
   228→                  className="text-sm text-muted-foreground hover:text-foreground transition-colors cursor-pointer inline-flex items-center gap-1"
   229→                  onClick={() => navigate('album-detail', { albumId: currentTrack.albumId })}
   230→                >
   231→                  <Disc3 className="w-3.5 h-3.5" /> {currentTrack.albumName}
   232→                  {currentTrack.trackNumber > 0 && <span className="ml-1 text-muted-foreground/60">· Track {currentTrack.trackNumber}</span>}
   233→                </p>
   234→
   235→                {/* Performer credits */}
   236→                {currentTrack.performers.length > 0 && (
   237→                  <div className="mt-3 flex flex-wrap gap-1.5 justify-center md:justify-start">
   238→                    {currentTrack.performers.map((p, i) => (
   239→                      <Badge key={i} variant="outline" className="text-[10px]">
   240→                        <Mic2 className="w-2.5 h-2.5 mr-0.5" /> {p.name} — {p.instrument || p.role}
   241→                      </Badge>
   242→                    ))}
   243→                  </div>
   244→                )}
   245→
   246→                {/* Composer */}
   247→                {currentTrack.composers.length > 0 && (
   248→                  <p className="text-xs text-muted-foreground/70 mt-1">
   249→                    Composer: {currentTrack.composers.join(', ')}
   250→                  </p>
   251→                )}
   252→              </div>
   253→
   254→              {/* Full Controls */}
   255→              <div className="w-full max-w-md md:max-w-lg mt-8 space-y-4 self-center md:self-start">
   256→                {/* Progress */}
   257→                <div className="space-y-1">
   258→                  <Slider
   259→                    value={[progress]}
   260→                    min={0}
   261→                    max={100}
   262→                    step={0.1}
   263→                    onValueChange={(v) => seek(v[0])}
   264→                    className="w-full cursor-pointer [&_[role=slider]]:h-4 [&_[role=slider]]:w-4"
   265→                  />
   266→                  <div className="flex justify-between text-xs text-muted-foreground tabular-nums">
   267→                    <span>{formatDuration(currentTime)}</span>
   268→                    <span>{formatDuration(currentTrack.duration)}</span>
   269→                  </div>
   270→                </div>
   271→
   272→                {/* Transport Buttons */}
   273→                <div className="flex items-center justify-center gap-4">
   274→                  <Button variant="ghost" size="icon" className={`h-10 w-10 ${isShuffle ? 'text-primary' : 'text-muted-foreground'}`} onClick={toggleShuffle}>
   275→                    <Shuffle className="w-5 h-5" />
   276→                  </Button>
   277→                  <Button variant="ghost" size="icon" className="h-10 w-10 text-muted-foreground" onClick={previous}>
   278→                    <SkipBack className="w-5 h-5" />
   279→                  </Button>
   280→                  <Button variant="default" size="icon" className="h-14 w-14 rounded-full shadow-xl shadow-primary/20" onClick={togglePlay}>
   281→                    {isPlaying ? <Pause className="w-7 h-7" /> : <Play className="w-7 h-7 ml-1" />}
   282→                  </Button>
   283→                  <Button variant="ghost" size="icon" className="h-10 w-10 text-muted-foreground" onClick={next}>
   284→                    <SkipForward className="w-5 h-5" />
   285→                  </Button>
   286→                  <Button variant="ghost" size="icon" className={`h-10 w-10 ${repeatMode !== 'off' ? 'text-primary' : 'text-muted-foreground'}`} onClick={toggleRepeat}>
   287→                    {repeatMode === 'one' ? <Repeat1 className="w-5 h-5" /> : <Repeat className="w-5 h-5" />}
   288→                  </Button>
   289→                </div>
   290→
   291→                {/* Volume */}
   292→                <div className="flex items-center gap-3 justify-center">
   293→                  <Button variant="ghost" size="icon" className="h-8 w-8" onClick={toggleMute}>
   294→                    {isMuted || volume === 0 ? <VolumeX className="w-4 h-4 text-muted-foreground" /> : volume < 50 ? <Volume1 className="w-4 h-4 text-muted-foreground" /> : <Volume2 className="w-4 h-4 text-muted-foreground" />}
   295→                  </Button>
   296→                  <Slider value={[isMuted ? 0 : volume]} min={0} max={100} step={1} onValueChange={(v) => setVolume(v[0])} className="w-40" />
   297→                  <span className="text-xs text-muted-foreground tabular-nums w-8 text-right">{volume}%</span>
   298→                </div>
   299→
   300→                {/* Action Buttons */}
   301→                <div className="flex items-center justify-center gap-2 mt-1">
   302→                  <Button variant="ghost" size="sm" className={`h-8 gap-1.5 ${isTrackLoved(currentTrack.id) ? 'text-red-500' : 'text-muted-foreground'}`} onClick={() => toggleLoveTrack(currentTrack.id)}>
   303→                    <Heart className={`w-4 h-4 ${isTrackLoved(currentTrack.id) ? 'fill-red-500 text-red-500' : ''}`} />
   304→                    {isTrackLoved(currentTrack.id) ? 'Loved' : 'Love'}
   305→                  </Button>
   306→                  <Button variant="ghost" size="sm" className="h-8 gap-1.5 text-muted-foreground" onClick={() => navigate('radio')}>
   307→                    <Radio className="w-4 h-4" /> Radio
   308→                  </Button>
   309→                  <Button variant="ghost" size="sm" className="h-8 gap-1.5 text-muted-foreground" onClick={() => {
   310→                    const info = `${currentTrack.title} — ${currentTrack.artistName} (${currentTrack.albumName})`;
   311→                    navigator.clipboard.writeText(info).catch(() => {});
   312→                  }}>
   313→                    <Share2 className="w-4 h-4" /> Share
   314→                  </Button>
   315→                </div>
   316→              </div>
   317→            </div>
   318→          </div>
   319→
   320→          {/* ── BOTTOM PANELS ── */}
   321→          <div className="grid md:grid-cols-[1fr_1fr] gap-6">
   322→            {/* Left Column */}
   323→            <div className="space-y-6">
   324→              {/* Artist Info Card */}
   325→              <Card className="bg-card/80 backdrop-blur border-border overflow-hidden">
   326→                <CardContent className="p-0">
   327→                  <div className="flex items-center gap-4 p-4 cursor-pointer hover:bg-accent/20 transition-colors" onClick={() => navigate('artist-detail', { artistId: currentTrack.artistId })}>
   328→                    {/* Artist avatar */}
   329→                    <div className="w-16 h-16 rounded-full bg-gradient-to-br from-primary/30 to-primary/5 flex items-center justify-center flex-shrink-0 ring-1 ring-primary/20">
   330→                      {artistImageLoading ? (
   331→                        <div className="w-4 h-4 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   332→                      ) : artistImage?.imageUrl ? (
   333→                        <div className="w-16 h-16 rounded-full bg-muted overflow-hidden">
   334→                          <div className="w-full h-full bg-gradient-to-br from-primary/20 to-primary/5 flex items-center justify-center">
   335→                            <User className="w-8 h-8 text-primary/40" />
   336→                          </div>
   337→                        </div>
   338→                      ) : (
   339→                        <User className="w-8 h-8 text-primary/40" />
   340→                      )}
   341→                    </div>
   342→                    <div className="flex-1 min-w-0">
   343→                      <p className="text-[10px] text-muted-foreground uppercase tracking-wider mb-0.5">Now Playing Artist</p>
   344→                      <h3 className="text-base font-semibold truncate">{currentTrack.artistName}</h3>
   345→                      <p className="text-xs text-muted-foreground">View full artist profile</p>
   346→                    </div>
   347→                    <ArrowLeft className="w-4 h-4 text-muted-foreground rotate-180" />
   348→                  </div>
   349→                  {/* Quick bio */}
   350→                  {artistBio && !bioLoading && artistBio.summaries.length > 0 && (
   351→                    <div className="px-4 pb-4">
   352→                      <p className="text-xs text-muted-foreground leading-relaxed line-clamp-3">
   353→                        {artistBio.summaries[0].snippet}
   354→                      </p>
   355→                      <a href={artistBio.summaries[0].url} target="_blank" rel="noopener noreferrer" className="text-[10px] text-primary hover:underline inline-flex items-center gap-0.5 mt-1">
   356→                        Read full bio on {artistBio.summaries[0].source} <ExternalLink className="w-2.5 h-2.5" />
   357→                      </a>
   358→                    </div>
   359→                  )}
   360→                  {bioLoading && (
   361→                    <div className="px-4 pb-3 flex items-center gap-2 text-xs text-muted-foreground">
   362→                      <div className="w-3 h-3 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   363→                      Loading artist info...
   364→                    </div>
   365→                  )}
   366→                </CardContent>
   367→              </Card>
   368→
   369→              {/* Album Info Card */}
   370→              <Card className="bg-card/80 backdrop-blur border-border overflow-hidden">
   371→                <CardContent className="p-0">
   372→                  <div className="flex items-center gap-4 p-4 cursor-pointer hover:bg-accent/20 transition-colors" onClick={() => navigate('album-detail', { albumId: currentTrack.albumId })}>
   373→                    {/* Album thumbnail */}
   374→                    <div className={`w-16 h-16 rounded-lg bg-gradient-to-br ${getCoverGradient(currentTrack.albumId)} shadow-md flex-shrink-0 overflow-hidden`}>
   375→                      {trackCoverArt && <img src={trackCoverArt} alt="" className="w-full h-full object-cover" />}
   376→                    </div>
   377→                    <div className="flex-1 min-w-0">
   378→                      <p className="text-[10px] text-muted-foreground uppercase tracking-wider mb-0.5">From the Album</p>
   379→                      <h3 className="text-base font-semibold truncate">{currentTrack.albumName}</h3>
   380→                      <p className="text-xs text-muted-foreground">{currentTrack.artistName} · {currentTrack.format} {formatSampleRate(currentTrack.sampleRate)}/{currentTrack.bitDepth}-bit</p>
   381→                    </div>
   382→                    <ArrowLeft className="w-4 h-4 text-muted-foreground rotate-180" />
   383→                  </div>
   384→                  {/* Quick album description */}
   385→                  {albumInfoData && !albumInfoLoading && albumInfoData.summaries.length > 0 && (
   386→                    <div className="px-4 pb-4">
   387→                      <p className="text-xs text-muted-foreground leading-relaxed line-clamp-3">
   388→                        {albumInfoData.summaries[0].snippet}
   389→                      </p>
   390→                      <a href={albumInfoData.summaries[0].url} target="_blank" rel="noopener noreferrer" className="text-[10px] text-primary hover:underline inline-flex items-center gap-0.5 mt-1">
   391→                        Read more on {albumInfoData.summaries[0].source} <ExternalLink className="w-2.5 h-2.5" />
   392→                      </a>
   393→                    </div>
   394→                  )}
   395→                  {albumInfoLoading && (
   396→                    <div className="px-4 pb-3 flex items-center gap-2 text-xs text-muted-foreground">
   397→                      <div className="w-3 h-3 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   398→                      Loading album info...
   399→                    </div>
   400→                  )}
   401→                </CardContent>
   402→              </Card>
   403→
   404→              {/* Lyrics Card */}
