    83→    83→function localTrackToTrack(lt: LocalTrack): Track {

    95→    95→    bitDepth: lt.bitDepth,

    99→    99→    filePath: lt.filePath,
   100→   100→    fileSize: lt.fileSize,

   293→   293→
   294→   294→function BrowseLocalPanel() {
   295→   295→  const localStore = useLocalLibraryStore();
   296→   296→  const [view, setView] = React.useState<BrowseLocalView>('artists');
   297→   297→  const [searchQuery, setSearchQuery] = React.useState('');
   298→   298→
   299→   299→  // Breadcrumb navigation state
   300→   300→  const [selectedArtist, setSelectedArtist] = React.useState<string | null>(null);
   301→   301→  const [selectedAlbum, setSelectedAlbum] = React.useState<string | null>(null);
   302→   302→
   303→   303→  const allArtists = localStore.getArtists();
   304→   304→  const allAlbums = localStore.getAlbums();
   305→   305→  const allTracks = localStore.tracks;
   306→   306→
   307→   307→  // Derived data based on navigation state
   308→   308→  const breadcrumbs: BreadcrumbItem[] = [{ label: 'Artists', onClick: () => { setSelectedArtist(null); setSelectedAlbum(null); setView('artists'); } }];
   309→   309→
   310→   310→  let displayedArtists = allArtists;
   311→   311→  let displayedAlbums = allAlbums;
   312→   312→  let displayedTracks = allTracks;
   313→   313→
   314→   314→  if (searchQuery.trim()) {
   315→   315→    displayedTracks = localStore.searchTracks(searchQuery.trim());
   316→   316→    displayedAlbums = allAlbums.filter(a =>
   317→   317→      displayedTracks.some(t => t.album === a.name && t.artist === a.artist)
   318→   318→    );
   319→   319→    displayedArtists = allArtists.filter(a =>
   320→   320→      displayedTracks.some(t => t.artist === a)
   321→   321→    );
   322→   322→  } else if (selectedArtist && selectedAlbum) {
   323→   323→    // Artist > Album > Tracks
   324→   324→    breadcrumbs.push(
   325→   325→      { label: selectedArtist, onClick: () => { setSelectedAlbum(null); } },
   326→   326→      { label: selectedAlbum, onClick: () => {} },
   327→   327→    );
   328→   328→    displayedTracks = localStore.getTracksByAlbum(selectedAlbum).filter(t => t.artist === selectedArtist);
   329→   329→  } else if (selectedArtist) {
   330→   330→    // Artist > Albums
   331→   331→    breadcrumbs.push({ label: selectedArtist, onClick: () => {} });
   332→   332→    displayedAlbums = allAlbums.filter(a => a.artist === selectedArtist);
   333→   333→    displayedTracks = localStore.getTracksByArtist(selectedArtist);
   334→   334→  } else if (selectedAlbum) {
   335→   335→    // Albums > Tracks
   336→   336→    breadcrumbs.push({ label: selectedAlbum, onClick: () => {} });
   337→   337→    displayedTracks = localStore.getTracksByAlbum(selectedAlbum);
   338→   338→  }
   339→   339→
   340→   340→  const showArtistsView = !selectedArtist && !selectedAlbum && view === 'artists' && !searchQuery.trim();
   341→   341→  const showAlbumsView = (selectedArtist || view === 'albums') && !selectedAlbum && !searchQuery.trim();
   342→   342→  const showTracksView = selectedAlbum || view === 'tracks' || searchQuery.trim();
   343→   343→
   344→   344→  return (
   345→   345→    <div className="space-y-4">
   346→   346→      {/* Search and View Switcher */}
   347→   347→      <div className="flex items-center gap-3">
   348→   348→        <div className="relative flex-1">
   349→   349→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   350→   350→          <Input
   351→   351→            placeholder="Search local library..."
   352→   352→            value={searchQuery}
   353→   353→            onChange={e => { setSearchQuery(e.target.value); if (e.target.value) { setSelectedArtist(null); setSelectedAlbum(null); } }}
   354→   354→            className="pl-9 h-9 text-sm"
   355→   355→          />
   356→   356→        </div>
   357→   357→        <div className="flex items-center bg-surface rounded-lg p-0.5">
   358→   358→          {([['artists', User, 'Artists'], ['albums', Disc3, 'Albums'], ['tracks', Music2, 'Tracks']] as const).map(([key, Icon, label]) => (
   359→   359→            <Button
   360→   360→              key={key}
   361→   361→              variant={view === key ? 'default' : 'ghost'}
   362→   362→              size="sm"
   363→   363→              className="h-8 text-xs gap-1.5 rounded-md"
   364→   364→              onClick={() => { setView(key as BrowseLocalView); setSelectedArtist(null); setSelectedAlbum(null); setSearchQuery(''); }}
   365→   365→            >
   366→   366→              <Icon className="w-3.5 h-3.5" /> {label}
   367→   367→            </Button>
   368→   368→          ))}
   369→   369→        </div>
   370→   370→      </div>
   371→   371→
   372→   372→      {/* Breadcrumb */}
   373→   373→      {(selectedArtist || selectedAlbum) && !searchQuery.trim() && (
   374→   374→        <div className="flex items-center gap-1 text-xs">
   375→   375→          {breadcrumbs.map((crumb, i) => (
   376→   376→            <React.Fragment key={i}>
   377→   377→              {i > 0 && <ChevronRight className="w-3 h-3 text-muted-foreground" />}
   378→   378→              <button
   379→   379→                className={`${i === breadcrumbs.length - 1 ? 'text-foreground font-medium' : 'text-muted-foreground hover:text-foreground'} transition-colors`}
   380→   380→                onClick={crumb.onClick}
   381→   381→              >
   382→   382→                {crumb.label}
   383→   383→              </button>
   384→   384→            </React.Fragment>
   385→   385→          ))}
   386→   386→        </div>
   387→   387→      )}
   388→   388→
   389→   389→      {/* Empty State */}
   390→   390→      {allTracks.length === 0 && (
   391→   391→        <Card className="bg-card border-border">
   392→   392→          <CardContent className="p-12 text-center">
   393→   393→            <LibraryIcon className="w-12 h-12 mx-auto mb-4 text-muted-foreground/30" />
   394→   394→            <h3 className="text-sm font-semibold mb-1">No Local Tracks</h3>
   395→   395→            <p className="text-xs text-muted-foreground max-w-md mx-auto">
   396→   396→              Scan your music folders to build your local library. Go to the &quot;Scan Folders&quot; tab to add directories and start scanning.
   397→   397→            </p>
   398→   398→          </CardContent>
   399→   399→        </Card>
   400→   400→      )}
   401→   401→
   402→   402→      {/* Artists Grid */}
   403→   403→      {showArtistsView && allTracks.length > 0 && (
   404→   404→        <div className="space-y-3">
   405→   405→          <div className="flex items-center justify-between">
   406→   406→            <h2 className="text-sm font-semibold">Artists ({displayedArtists.length})</h2>
