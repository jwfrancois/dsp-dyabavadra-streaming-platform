   333→function ScannerPanel({ store }: { store: ReturnType<typeof useLibraryStore> }) {
   334→  const scan = useLibraryStore(s => s.scan);
   335→  const isRunning = scan.status === 'running';
   336→
   337→  const phaseLabels: Record<string, string> = {
   338→    idle: 'Idle',
   339→    discovering: 'Discovering files...',
   340→    'reading-tags': 'Reading embedded tags...',
   341→    fingerprinting: 'Acoustic fingerprinting...',
   342→    deduplicating: 'Checking for duplicates...',
   343→    finalizing: 'Finalizing index...',
   344→  };
   345→
   346→  return (
   347→    <div className="space-y-6">
   348→      {/* Scan Controls */}
   349→      <Card className="bg-card border-border">
   350→        <CardHeader className="pb-3">
   351→          <div className="flex items-center justify-between">
   352→            <CardTitle className="text-sm flex items-center gap-2">
   353→              <ScanSearch className="w-4 h-4 text-primary" /> Library Scanner
   354→            </CardTitle>
   355→            <div className="flex items-center gap-2">
   356→              {isRunning ? (
   357→                <Button variant="destructive" size="sm" className="h-7 text-xs">
   358→                  <Pause className="w-3 h-3 mr-1" /> Pause Scan
   359→                </Button>
   360→              ) : (
   361→                <Button size="sm" className="h-7 text-xs" onClick={() => store.triggerScan()}>
   362→                  <RefreshCw className={`w-3 h-3 mr-1 ${isRunning ? 'animate-spin' : ''}`} />
   363→                  {scan.status === 'completed' ? 'Rescan All' : 'Start Scan'}
   364→                </Button>
   365→              )}
   366→            </div>
   367→          </div>
   368→        </CardHeader>
   369→        <CardContent className="space-y-4">
   370→          {/* Progress */}
   371→          <div>
   372→            <div className="flex justify-between text-xs mb-1">
   373→              <span className="text-muted-foreground">{phaseLabels[scan.phase] || scan.phase}</span>
   374→              <span className="font-mono">{Math.round(scan.progress)}%</span>
   375→            </div>
   376→            <Progress value={scan.progress} className="h-2" />
   377→          </div>
   378→
   379→          {/* Scan Stats */}
   380→          <div className="grid grid-cols-2 md:grid-cols-5 gap-3">
   381→            <div className="p-2 rounded bg-surface/50">
   382→              <p className="text-xs text-muted-foreground">Total Files</p>
   383→              <p className="text-sm font-bold">{scan.totalFiles.toLocaleString()}</p>
   384→            </div>
   385→            <div className="p-2 rounded bg-surface/50">
   386→              <p className="text-xs text-muted-foreground">Processed</p>
   387→              <p className="text-sm font-bold">{scan.processedFiles.toLocaleString()}</p>
   388→            </div>
   389→            <div className="p-2 rounded bg-signal-green/10">
   390→              <p className="text-xs text-signal-green">New</p>
   391→              <p className="text-sm font-bold text-signal-green">+{scan.newFiles}</p>
   392→            </div>
   393→            <div className="p-2 rounded bg-signal-amber/10">
   394→              <p className="text-xs text-signal-amber">Updated</p>
   395→              <p className="text-sm font-bold text-signal-amber">~{scan.updatedFiles}</p>
   396→            </div>
   397→            <div className="p-2 rounded bg-signal-red/10">
   398→              <p className="text-xs text-signal-red">Removed</p>
   399→              <p className="text-sm font-bold text-signal-red">-{scan.removedFiles}</p>
   400→            </div>
   401→          </div>
   402→
   403→          {/* Last scan info */}
   404→          {scan.status === 'completed' && scan.completedAt && (
   405→            <div className="flex items-center gap-4 text-xs text-muted-foreground">
   406→              <span>Started: {new Date(scan.startedAt).toLocaleString()}</span>
   407→              <span>Completed: {new Date(scan.completedAt).toLocaleString()}</span>
   408→              <span>Duration: {formatScanDuration(Math.floor((new Date(scan.completedAt).getTime() - new Date(scan.startedAt).getTime()) / 1000))}</span>
   409→            </div>
   410→          )}
   411→        </CardContent>
   412→      </Card>
   413→
   414→      {/* Scan Errors */}
   415→      {scan.errors.length > 0 && (
   416→        <Card className="bg-card border-signal-amber/30">
   417→          <CardHeader className="pb-3">
   418→            <CardTitle className="text-sm flex items-center gap-2">
   419→              <AlertTriangle className="w-4 h-4 text-signal-amber" />
   420→              Scan Errors ({scan.errors.length})
   421→            </CardTitle>
   422→          </CardHeader>
   423→          <CardContent>
   424→            <div className="space-y-2">
   425→              {scan.errors.map((err, i) => (
   426→                <div key={i} className="flex items-start gap-2 p-2 rounded bg-surface/30">
   427→                  {err.severity === 'error' ? <XCircle className="w-4 h-4 text-signal-red flex-shrink-0 mt-0.5" /> : <AlertTriangle className="w-4 h-4 text-signal-amber flex-shrink-0 mt-0.5" />}
   428→                  <div className="min-w-0">
   429→                    <p className="text-xs font-mono text-muted-foreground truncate">{err.file}</p>
   430→                    <p className="text-xs">{err.message}</p>
   431→                  </div>
   432→                </div>
   433→              ))}
   434→            </div>
   435→          </CardContent>
   436→        </Card>
   437→      )}
   438→
   439→      {/* Scan Settings */}
   440→      <Card className="bg-card border-border">
   441→        <CardHeader className="pb-3">
   442→          <CardTitle className="text-sm">Scan Settings</CardTitle>
   443→        </CardHeader>
   444→        <CardContent className="space-y-3">
   445→          <div className="flex items-center justify-between">
